package com.irurueta.authserver.serviceclient;

import java.io.PrintStream;
import lombok.RequiredArgsConstructor;

/**
 * Writes the lines a person (or a test) reads from the {@code client-credentials} and {@code token-exchange}
 * runners. Lines always end with {@code \n}, never with the platform's line separator.
 */
@RequiredArgsConstructor
public class CliOutput {

    /** The stream. */
    private final PrintStream stream;

    /**
     * Writes one line and flushes it.
     *
     * @param text the line, without terminator.
     */
    public void line(final String text) {
        stream.print(text);
        stream.print('\n');
        stream.flush();
    }
}
