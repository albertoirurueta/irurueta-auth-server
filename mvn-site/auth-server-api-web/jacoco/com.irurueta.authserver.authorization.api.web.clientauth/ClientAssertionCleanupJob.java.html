package com.irurueta.authserver.authorization.api.web.clientauth;

import com.irurueta.authserver.authorization.application.clientauth.PurgeClientAssertionsUseCase;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

/**
 * Scheduled entry point that deletes the identifiers of the client assertions that already expired. The schedule
 * comes from {@code authserver.client-assertion.cleanup.cron} (default every hour, at minute 30). Scheduling itself is
 * enabled once for every feature by {@code SchedulingConfiguration}.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class ClientAssertionCleanupJob {

    /** Deletes the expired identifiers. */
    private final PurgeClientAssertionsUseCase purgeClientAssertions;

    /**
     * Deletes the expired identifiers.
     */
    @Scheduled(cron = "${authserver.client-assertion.cleanup.cron:0 30 * * * *}")
    public void run() {
        log.debug("Deleted {} expired client assertion identifiers", purgeClientAssertions.execute());
    }
}
