42  /**
43   * Runtime OpenAPI documents of the authorization server, served by springdoc. The contracts under
44   * {@code apis/auth-server/} are the source of truth: the operations are described by the interfaces generated from
45   * them, and this class only adds what springdoc cannot read from those (the info blocks and the security schemes).
46   * Two groups are published: {@value #API_GROUP} ({@code /api/**}, the REST API of
47   * {@code rest-server/auth-server-api-v1.yaml}) and {@value #PAGES_GROUP} (the HTML pages of
48   * {@code web-pages/auth-server-pages-v1.yaml}, at {@value #PAGES_DOCS_PATH}). {@code OpenApiDriftIT} fails when a
49   * document served differs from its contract. The OAuth 2.1 / OpenID Connect protocol endpoints are defined by the
50   * specifications and documented by the discovery document, so they are not published here.
51   */
52  @Configuration(proxyBeanMethods = false)
53  public class OpenApiConfig {
54  
55      /** Path of the OpenAPI document. */
56      public static final String API_DOCS_PATH = "/v3/api-docs";
57  
58      /** Name of the group of the REST API. */
59      public static final String API_GROUP = "api";
60  
61      /** Name of the group of the HTML pages. */
62      public static final String PAGES_GROUP = "pages";
63  
64      /** Path of the OpenAPI document of the HTML pages. */
65      public static final String PAGES_DOCS_PATH = API_DOCS_PATH + "/" + PAGES_GROUP;
66  
67      /** Name of the bearer-token security scheme the REST contract's operations refer to. */
68      public static final String BEARER_SCHEME = "bearer";
69  
70      /** Name of the session-cookie security scheme the pages contract's operations refer to. */
71      public static final String SESSION_SCHEME = "session";
72  
73      /** Name of the session cookie (see {@code LogoutChainCustomizer}, which deletes it). */
74      static final String SESSION_COOKIE = "JSESSIONID";
75  
76      /**
77       * Describes the API.
78       *
79       * @return the OpenAPI info block.
80       */
81      @Bean
82      public OpenAPI authServerOpenApi() {
83          return new OpenAPI()
84                  .info(new Info()
85                          .title("Irurueta Auth Server API")
86                          .description("REST endpoints of the example OAuth 2.1 / OpenID Connect authorization server.")
87                          .version("1.0.0")
88                          .license(license()))
89                  .components(new Components().addSecuritySchemes(BEARER_SCHEME, new SecurityScheme()
90                          .type(SecurityScheme.Type.HTTP)
91                          .scheme("bearer")
92                          .bearerFormat("JWT")));
93      }
94  
95      /**
96       * Groups the application's own API paths, leaving out the protocol endpoints and the HTML pages.
97       *
98       * @return the group of API paths.
99       */
100     @Bean
101     public GroupedOpenApi apiGroup() {
102         return GroupedOpenApi.builder()
103                 .group(API_GROUP)
104                 .pathsToMatch("/api/**")
105                 .build();
106     }
107 
108     // tag::pages-openapi-group[]
109     /**
110      * Groups the HTML pages of the contract {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml}, with their
111      * own info block and the session-cookie scheme instead of the REST API's.
112      *
113      * @return the group of the HTML pages.
114      */
115     @Bean
116     public GroupedOpenApi pagesGroup() {
117         return GroupedOpenApi.builder()
118                 .group(PAGES_GROUP)
119                 .pathsToMatch(LoginPageApi.PATH_LOGIN_PAGE, SecondFactorPageApi.PATH_SECOND_FACTOR_PAGE,
120                         RecoveryCodesPageApi.PATH_RECOVERY_CODES_PAGE, ConsentPageApi.PATH_CONSENT_PAGE,
121                         LogoutPageApi.PATH_LOGOUT_PAGE, PasskeysPageApi.PATH_PASSKEYS_PAGE,
122                         PhonePageApi.PATH_PHONE_PAGE, PhonePageApi.PATH_VERIFY_PHONE_PAGE,
123                         PhonePageApi.PATH_REMOVE_PHONE_PAGE,
124                         IdentitiesPageApi.PATH_IDENTITIES_PAGE, IdentitiesPageApi.PATH_LINK_IDENTITY_PAGE,
125                         OneTimeTokenPageApi.PATH_OTT_CODE_PAGE, OneTimeTokenPageApi.PATH_OTT_CONFIRM_PAGE,
126                         DeviceVerificationPageApi.PATH_DEVICE_VERIFICATION_PAGE,
127                         DevSmsInboxPageApi.PATH_DEV_SMS_INBOX_PAGE)
128                 .addOpenApiCustomizer(openApi -> {
129                     final var components = Objects.requireNonNullElseGet(openApi.getComponents(), Components::new);
130                     components.setSecuritySchemes(new LinkedHashMap<>(Map.of(SESSION_SCHEME, new SecurityScheme()
131                             .type(SecurityScheme.Type.APIKEY)
132                             .in(SecurityScheme.In.COOKIE)
133                             .name(SESSION_COOKIE))));
134                     openApi.info(new Info()
135                             .title("Irurueta Auth Server Pages")
136                             .description("Server-rendered HTML pages of the example OAuth 2.1 / OpenID Connect "
137                                     + "authorization server.")
138                             .version("1.0.0")
139                             .license(license()))
140                             .components(components);
141                 })
142                 .build();
143     }
144     // end::pages-openapi-group[]
145 
146     /**
147      * Describes the license of the documents.
148      *
149      * @return the license.
150      */
151     private static License license() {
152         return new License().name("Apache License 2.0").url("https://www.apache.org/licenses/LICENSE-2.0.txt");
153     }
154 }
