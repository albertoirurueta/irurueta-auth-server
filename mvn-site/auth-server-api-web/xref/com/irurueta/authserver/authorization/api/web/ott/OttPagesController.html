16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
19  import com.irurueta.authserver.authorization.api.web.pages.v1.generated.api.OneTimeTokenPageApi;
20  import com.irurueta.authserver.authorization.domain.ott.OttChannel;
21  import com.irurueta.authserver.authorization.domain.ott.OttPurpose;
22  import jakarta.servlet.http.HttpServletRequest;
23  import java.util.HashMap;
24  import java.util.Locale;
25  import java.util.Map;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.http.CacheControl;
28  import org.springframework.http.ResponseEntity;
29  import org.springframework.security.authentication.LockedException;
30  import org.springframework.security.web.WebAttributes;
31  import org.springframework.stereotype.Controller;
32  
33  /**
34   * Serves the two pages of the one-time token flow, as the contract {@code apis/auth-server/web-pages/
35   * auth-server-pages-v1.yaml} describes them through the generated {@link OneTimeTokenPageApi}: the page where the code
36   * is entered (or that says to check the inbox when a magic link was requested) and the page a magic link opens, whose
37   * button posts the token. What the first one shows comes from the pending token of the session (see
38   * {@link PendingOtt}), so it is the same page whatever identifier was typed.
39   */
40  @Controller
41  @RequiredArgsConstructor
42  public class OttPagesController implements OneTimeTokenPageApi {
43  
44      /** Path of the page where the code is entered, taken from the contract. */
45      public static final String OTT_CODE_PATH = PATH_OTT_CODE_PAGE;
46  
47      /** Name of the template of the page where the code is entered. */
48      static final String CODE_VIEW_NAME = "ott-code";
49  
50      /** Name of the template of the confirmation page. */
51      static final String CONFIRM_VIEW_NAME = "ott-confirm";
52  
53      /** Name of the model attribute that tells whether this browser has asked for a token. */
54      static final String PENDING_ATTRIBUTE = "pending";
55  
56      /** Name of the model attribute that holds the kind of token, {@code code} or {@code link}. */
57      static final String KIND_ATTRIBUTE = "kind";
58  
59      /** Name of the model attribute that holds the channel, {@code email} or {@code sms}. */
60      static final String CHANNEL_ATTRIBUTE = "channel";
61  
62      /** Name of the model attribute that holds the purpose, {@code login} or {@code second-factor}. */
63      static final String PURPOSE_ATTRIBUTE = "purpose";
64  
65      /** Name of the model attribute that holds the path the code form posts to. */
66      static final String POST_PATH_ATTRIBUTE = "postPath";
67  
68      /** Name of the model attribute that holds the name of the form field of the code. */
69      static final String CODE_FIELD_ATTRIBUTE = "codeField";
70  
71      /** Name of the model attribute that holds the path a new token is requested at. */
72      static final String GENERATE_PATH_ATTRIBUTE = "generatePath";
73  
74      /** Name of the model attribute that holds the key of the message of a failed attempt. */
75      static final String ERROR_MESSAGE_ATTRIBUTE = "errorMessageKey";
76  
77      /** Name of the model attribute that holds the token of the magic link. */
78      static final String TOKEN_ATTRIBUTE = "token";
79  
80      /** Key of the message shown when a code is wrong. */
81      static final String GENERIC_ERROR = "ott.error.generic";
82  
83      /** Key of the message shown when the second factor is locked. */
84      static final String LOCKED_ERROR = "ott.error.locked";
85  
86      /** Value of the purpose attribute of a sign-in. */
87      static final String LOGIN_VALUE = "login";
88  
89      /** Value of the kind attribute of a numeric code. */
90      static final String CODE_VALUE = "code";
91  
92      /** Name of the header that tells browsers to send no referrer. */
93      static final String REFERRER_POLICY = "Referrer-Policy";
94  
95      /** Renders the pages. */
96      private final PageRenderer pageRenderer;
97  
98      /** The current request, injected as a thread-bound proxy, whose session holds the pending token. */
99      private final HttpServletRequest request;
100 
101     /**
102      * Renders the page where the code is entered.
103      *
104      * @param error present (even if empty) when the previous attempt failed.
105      * @return the page.
106      */
107     @Override
108     public ResponseEntity<String> ottCodePage(final String error) {
109         final var pending = PendingOtt.of(request);
110         final var secondFactor = pending.map(PendingOtt::purpose).filter(OttPurpose.SECOND_FACTOR::equals).isPresent();
111         final var model = new HashMap<String, Object>();
112         model.put(PENDING_ATTRIBUTE, pending.isPresent());
113         model.put(KIND_ATTRIBUTE, pending.map(token -> lowerCase(token.kind().name())).orElse(CODE_VALUE));
114         model.put(CHANNEL_ATTRIBUTE, pending.map(token -> lowerCase(token.channel().name()))
115                 .orElse(lowerCase(OttChannel.EMAIL.name())));
116         model.put(PURPOSE_ATTRIBUTE, secondFactor ? OttGenerationParameters.SECOND_FACTOR_VALUE : LOGIN_VALUE);
117         model.put(POST_PATH_ATTRIBUTE, secondFactor ? OttSecondFactorFilter.SECOND_FACTOR_OTT_PATH : OTT_CODE_PATH);
118         model.put(CODE_FIELD_ATTRIBUTE, secondFactor ? OttSecondFactorFilter.CODE_PARAMETER
119                 : OttAuthenticationConverter.TOKEN_PARAMETER);
120         model.put(GENERATE_PATH_ATTRIBUTE, OttRateLimitFilter.GENERATE_PATH);
121         if (error != null) {
122             model.put(ERROR_MESSAGE_ATTRIBUTE, lastFailureWasLockout() ? LOCKED_ERROR : GENERIC_ERROR);
123         }
124         return pageRenderer.render(CODE_VIEW_NAME, model);
125     }
126 
127     /**
128      * Renders the page a magic link opens. It is never cached and a browser sends no referrer from it, because the
129      * token is in its address.
130      *
131      * @param token the token of the link, which is only posted back, never checked here.
132      * @return the confirmation page.
133      */
134     @Override
135     public ResponseEntity<String> ottConfirmPage(final String token) {
136         final var page = pageRenderer.render(CONFIRM_VIEW_NAME, Map.of(
137                 TOKEN_ATTRIBUTE, token,
138                 POST_PATH_ATTRIBUTE, OTT_CODE_PATH,
139                 CODE_FIELD_ATTRIBUTE, OttAuthenticationConverter.TOKEN_PARAMETER));
140         return ResponseEntity.status(page.getStatusCode())
141                 .headers(page.getHeaders())
142                 .cacheControl(CacheControl.noStore())
143                 .header(REFERRER_POLICY, "no-referrer")
144                 .body(page.getBody());
145     }
146 
147     /**
148      * Tells whether the previous attempt failed because the second factor is locked, which Spring Security keeps in
149      * the session.
150      *
151      * @return {@code true} when the session holds a {@link LockedException} as the last failure.
152      */
153     private boolean lastFailureWasLockout() {
154         final var session = request.getSession(false);
155         final var failure = session == null ? null : session.getAttribute(WebAttributes.AUTHENTICATION_EXCEPTION);
156         return failure instanceof LockedException;
157     }
158 
159     /**
160      * Lower-cases a name, whatever the platform's locale is.
161      *
162      * @param name the name.
163      * @return the name in lower case.
164      */
165     private static String lowerCase(final String name) {
166         return name.toLowerCase(Locale.ROOT);
167     }
168 }
