16  package com.irurueta.authserver.authorization.api.web.x509;
17  
18  import lombok.RequiredArgsConstructor;
19  import org.apache.catalina.connector.Connector;
20  import org.apache.coyote.http11.AbstractHttp11Protocol;
21  import org.apache.coyote.http11.Http11NioProtocol;
22  import org.apache.tomcat.util.net.SSLHostConfig;
23  import org.apache.tomcat.util.net.SSLHostConfigCertificate;
24  import org.springframework.boot.tomcat.servlet.TomcatServletWebServerFactory;
25  import org.springframework.boot.web.server.WebServerFactoryCustomizer;
26  
27  // tag::x509-tls-connector[]
28  /**
29   * Adds the dedicated TLS port that requests a client certificate, when {@code authserver.x509.port} is set. The main
30   * port is left as it is, so the rest of the application never asks a browser for a certificate. The connector
31   * <em>requests</em> a certificate without requiring one and without validating it against a CA
32   * ({@code optionalNoCA}): the TLS handshake proves that the client holds the private key, and what identifies the
33   * client is the fingerprint of the certificate, which has to be bound to an account. The servlet container leaves the
34   * certificate in the request attribute that the X.509 filter reads. The connector's trust manager
35   * ({@link AnyClientCertificateTrustManager}) advertises no certificate authority: without it the handshake lists the
36   * authorities of the JVM's default trust store, and a client offers only a certificate issued by a listed authority,
37   * so the certificate of a private authority would never be presented.
38   */
39  @RequiredArgsConstructor
40  public class X509TlsConnectorCustomizer implements WebServerFactoryCustomizer<TomcatServletWebServerFactory> {
41  
42      /** Value of the certificate verification that asks for a certificate but does not validate it. */
43      static final String CERTIFICATE_VERIFICATION = "optionalNoCA";
44  
45      /** Scheme of the connector. */
46      private static final String SCHEME = "https";
47  
48      /** Settings of the sign-in with a certificate. */
49      private final X509Properties properties;
50  
51      /** Finds the key store with the server certificate. */
52      private final X509TlsKeyStoreResolver keyStoreResolver;
53  
54      /**
55       * Adds the connector to the server, when the port is configured.
56       *
57       * @param factory the factory of the embedded Tomcat.
58       */
59      @Override
60      public void customize(final TomcatServletWebServerFactory factory) {
61          if (properties.tlsEnabled()) {
62              factory.addAdditionalConnectors(connector(keyStoreResolver.resolve()));
63          }
64      }
65  
66      /**
67       * Builds the connector.
68       *
69       * @param keyStore the key store with the server certificate.
70       * @return the connector.
71       */
72      Connector connector(final X509TlsKeyStore keyStore) {
73          final var connector = new Connector(Http11NioProtocol.class.getName());
74          connector.setScheme(SCHEME);
75          connector.setSecure(true);
76          connector.setPort(properties.port());
77          final var protocol = (AbstractHttp11Protocol<?>) connector.getProtocolHandler();
78          protocol.setSSLEnabled(true);
79          final var host = new SSLHostConfig();
80          host.setCertificateVerification(CERTIFICATE_VERIFICATION);
81          host.setTrustManagerClassName(AnyClientCertificateTrustManager.class.getName());
82          final var certificate = new SSLHostConfigCertificate(host, SSLHostConfigCertificate.Type.UNDEFINED);
83          certificate.setCertificateKeystoreFile(keyStore.location().toString());
84          certificate.setCertificateKeystorePassword(keyStore.password());
85          certificate.setCertificateKeystoreType(keyStore.type());
86          host.addCertificate(certificate);
87          protocol.addSslHostConfig(host);
88          return connector;
89      }
90  }
