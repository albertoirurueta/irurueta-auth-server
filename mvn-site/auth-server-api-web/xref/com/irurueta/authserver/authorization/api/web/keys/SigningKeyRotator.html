17  package com.irurueta.authserver.authorization.api.web.keys;
18  
19  import com.irurueta.authserver.authorization.application.keys.RotateSigningKeysUseCase;
20  import com.irurueta.authserver.authorization.application.keys.SigningKeysRotatedEvent;
21  import java.time.Clock;
22  import lombok.RequiredArgsConstructor;
23  import org.springframework.context.ApplicationEventPublisher;
24  import org.springframework.stereotype.Component;
25  
26  /**
27   * Runs the key lifecycle and announces it, so every caller (the endpoint, the scheduled job and the startup
28   * bootstrap) gets the same cache refresh without depending on the adapter that holds the caches.
29   */
30  @Component
31  @RequiredArgsConstructor
32  public class SigningKeyRotator {
33  
34      /** Applies the lifecycle transitions that are due. */
35      private final RotateSigningKeysUseCase rotateSigningKeysUseCase;
36  
37      /** Publishes the rotation event. */
38      private final ApplicationEventPublisher eventPublisher;
39  
40      /** Source of the current time. */
41      private final Clock clock;
42  
43      /**
44       * Applies the lifecycle transitions that are due and publishes a {@link SigningKeysRotatedEvent}: the path of
45       * the start-up runner and the scheduled job.
46       */
47      public void rotate() {
48          rotateSigningKeysUseCase.execute();
49          published();
50      }
51  
52      /**
53       * Replaces the active key now, whatever its age, and publishes a {@link SigningKeysRotatedEvent}: the path of
54       * the administration endpoint.
55       */
56      public void rotateNow() {
57          rotateSigningKeysUseCase.executeForced();
58          published();
59      }
60  
61      /**
62       * Publishes the rotation event, so the cached key sets are refreshed.
63       */
64      private void published() {
65          eventPublisher.publishEvent(new SigningKeysRotatedEvent(clock.instant()));
66      }
67  }
