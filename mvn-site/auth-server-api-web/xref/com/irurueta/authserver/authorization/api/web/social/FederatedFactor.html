17  package com.irurueta.authserver.authorization.api.web.social;
18  
19  import com.irurueta.authserver.authorization.domain.mfa.AuthenticationFactor;
20  import com.irurueta.authserver.authorization.domain.mfa.FactorStrength;
21  import org.springframework.context.annotation.Bean;
22  import org.springframework.context.annotation.Configuration;
23  
24  /**
25   * Registers the sign-in with an external provider as an authentication factor, in two flavours that are never held
26   * together by one session:
27   * <ul>
28   *   <li>{@link #AUTHORITY} ({@code fed}, a {@link FactorStrength#PRIMARY} factor): the normal case. The provider
29   *   proved the user, but the server does not know how, so the normal second-factor flow applies;</li>
30   *   <li>{@link #STRONG_AUTHORITY} ({@code fed}, {@link FactorStrength#PRIMARY_STRONG}): granted instead of the first
31   *   only when the provider is marked {@code trust-mfa} <em>and</em> its upstream {@code amr} contains {@code mfa}, so
32   *   the sign-in satisfies the multi-factor policy on its own, like a passkey. Unlike a passkey it is also
33   *   {@link AuthenticationFactor#multiFactor() multi-factor alone}: the provider proved several factors, so the
34   *   session reaches the multi-factor {@code acr} and passes a step-up without a local second factor.</li>
35   * </ul>
36   * Both report the same {@code amr} value, so the tokens read {@code fed} either way (plus {@code mfa} and the upstream
37   * values for the strong one, see {@link IdpClaimsContributor}).
38   */
39  @Configuration(proxyBeanMethods = false)
40  public class FederatedFactor {
41  
42      /** Authority of a federated sign-in, a primary factor. */
43      public static final String AUTHORITY = "FACTOR_FEDERATED";
44  
45      /** Authority of a federated sign-in whose provider proved several factors and is trusted for it. */
46      public static final String STRONG_AUTHORITY = "FACTOR_FEDERATED_MFA";
47  
48      /** RFC 8176 {@code amr} value of a federated sign-in. */
49      public static final String AMR_VALUE = "fed";
50  
51      // tag::social-factor[]
52      /**
53       * Declares the federated factor.
54       *
55       * @return the factor.
56       */
57      @Bean
58      public AuthenticationFactor federatedAuthenticationFactor() {
59          return new AuthenticationFactor("federated", AMR_VALUE, AUTHORITY, FactorStrength.PRIMARY);
60      }
61  
62      /**
63       * Declares the federated factor of a trusted provider that proved several factors.
64       *
65       * @return the factor.
66       */
67      @Bean
68      public AuthenticationFactor federatedStrongAuthenticationFactor() {
69          return new AuthenticationFactor("federated-mfa", AMR_VALUE, STRONG_AUTHORITY, FactorStrength.PRIMARY_STRONG,
70                  true);
71      }
72      // end::social-factor[]
73  }
