18  import com.irurueta.authserver.authorization.api.web.login.LoginChainCustomizer;
19  import lombok.RequiredArgsConstructor;
20  import org.springframework.http.HttpMethod;
21  import org.springframework.security.authentication.AuthenticationManager;
22  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
23  import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
24  import org.springframework.security.core.userdetails.AuthenticationUserDetailsService;
25  import org.springframework.security.web.authentication.preauth.PreAuthenticatedAuthenticationToken;
26  import org.springframework.security.web.authentication.preauth.x509.X509AuthenticationFilter;
27  import org.springframework.security.web.authentication.preauth.x509.X509PrincipalExtractor;
28  import org.springframework.security.web.authentication.session.SessionAuthenticationStrategy;
29  import org.springframework.security.web.servlet.util.matcher.PathPatternRequestMatcher;
30  import org.springframework.security.web.util.matcher.RequestMatcher;

32  // tag::x509-chain[]
33  /**
34   * Adds the sign-in with a client certificate to the login/UI chain through {@code http.x509(...)}. The filter only
35   * looks at {@code POST /login/x509}, the form of {@link X509LoginMethodFragment} on the login page, which carries the
36   * CSRF token like every other form of the chain; the CSRF filter runs before the X.509 one, so a cross-site page can
37   * never make a browser sign in with its certificate. The form posts to the TLS port that requests a certificate (or
38   * to the proxy that forwards one); every other request of the chain is untouched, and {@code GET /login/x509} is not
39   * a sign-in at all (it is not opened, so it ends with the chain's catch-all rule). A certificate that is bound to an
40   * account signs it in and resumes the interrupted request, anything else goes back to the login page. The path is
41   * open to everybody, because it is where the certificate is checked. When a proxy may forward the certificate,
42   * {@link ForwardedCertificateFilter} runs first. Once the chain is configured, the success handler gets the chain's
43   * session authentication strategy, the one the form login uses.
44   */
45  @RequiredArgsConstructor
46  public class X509ChainCustomizer implements LoginChainCustomizer {
47  
48      /** Order of the customizer: after the social login one. */
49      public static final int ORDER = 6;
50  
51      /** Path of the sign-in with a certificate, which is served by the security filter. */
52      public static final String LOGIN_PATH = "/login/x509";
53  
54      /** Where a failed sign-in goes. */
55      static final String FAILURE_URL = "/login?error";
56  
57      /** The filter that authenticates the certificate. */
58      private final X509LoginFilter filter;
59  
60      /** Takes the principal of the certificate. */
61      private final X509PrincipalExtractor principalExtractor;
62  
63      /** Loads the account of the certificate. */
64      private final AuthenticationUserDetailsService<PreAuthenticatedAuthenticationToken> userDetailsService;
65  
66      /** The filter that reads a forwarded certificate, or {@code null} when the transport is off. */
67      private final ForwardedCertificateFilter forwardedFilter;
68  
69      /** Completes the sign-in. */
70      private final X509AuthenticationSuccessHandler successHandler;
71  
72      /**
73       * Customizes the login/UI chain.
74       *
75       * @param http security builder of the chain.
76       */
77      @Override
78      public void customize(final HttpSecurity http) {
79          final var loginRequest = PathPatternRequestMatcher.withDefaults().matcher(HttpMethod.POST, LOGIN_PATH);
80          http.authorizeHttpRequests(authorize -> authorize.requestMatchers(loginRequest).permitAll());
81          prepare(filter, loginRequest, http);
82          if (forwardedFilter != null) {
83              http.addFilterBefore(forwardedFilter, X509AuthenticationFilter.class);
84          }
85          http.x509(x509 -> x509
86                  .x509AuthenticationFilter(filter)
87                  .x509PrincipalExtractor(principalExtractor)
88                  .authenticationUserDetailsService(userDetailsService));
89          http.with(new SessionStrategyConfigurer(successHandler));
90      }
91  
92      /**
93       * Sets up the filter, which the {@code x509} configurer takes as it is.
94       *
95       * @param loginFilter  the filter.
96       * @param loginRequest matcher of the sign-in requests.
97       * @param http         security builder of the chain, whose authentication manager the filter uses once built.
98       */
99      private void prepare(final X509LoginFilter loginFilter, final RequestMatcher loginRequest,
100             final HttpSecurity http) {
101         loginFilter.setPrincipalExtractor(principalExtractor);
102         loginFilter.setRequiresAuthenticationRequestMatcher(loginRequest);
103         loginFilter.setMfaEnabled(true);
104         loginFilter.setAuthenticationSuccessHandler(successHandler);
105         loginFilter.setAuthenticationFailureHandler(
106                 (request, response, exception) -> response.sendRedirect(request.getContextPath() + FAILURE_URL));
107         // the chain's manager only exists once the chain is built, so it is read when a request arrives
108         loginFilter.setAuthenticationManager(authentication ->
109                 http.getSharedObject(AuthenticationManager.class)
110                         .authenticate(authentication));
111     }
112 
113     /**
114      * Hands the session authentication strategy of the chain to the success handler. The strategy is a shared object
115      * of the chain that only exists once every configurer has been initialised (the CSRF configurer adds its
116      * {@code CsrfAuthenticationStrategy} to the session management one), so it is read in {@code configure}, as the
117      * form login does for its own filter. Package-private so its test can drive it.
118      */
119     @RequiredArgsConstructor
120     static final class SessionStrategyConfigurer
121             extends AbstractHttpConfigurer<SessionStrategyConfigurer, HttpSecurity> {
122 
123         /** The handler that gets the strategy. */
124         private final X509AuthenticationSuccessHandler handler;
125 
126         /**
127          * Sets the shared strategy, when the chain has one, on the handler.
128          *
129          * @param http security builder of the chain.
130          */
131         @Override
132         public void configure(final HttpSecurity http) {
133             final var strategy = http.getSharedObject(SessionAuthenticationStrategy.class);
134             if (strategy != null) {
135                 handler.setSessionAuthenticationStrategy(strategy);
136             }
137         }
138     }
139 }
