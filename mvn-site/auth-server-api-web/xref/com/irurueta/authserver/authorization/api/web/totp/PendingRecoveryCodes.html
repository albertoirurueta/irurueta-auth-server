16  package com.irurueta.authserver.authorization.api.web.totp;
17  
18  import jakarta.servlet.http.HttpServletRequest;
19  import java.util.ArrayList;
20  import java.util.List;
21  import org.springframework.stereotype.Component;
22  
23  /**
24   * Keeps, in the session, the recovery codes generated when an authenticator app is enrolled during the sign-in, from
25   * the post that confirms it until the page that shows them once ({@link RecoveryCodesPageController}), which takes
26   * them out of the session.
27   */
28  @Component
29  public class PendingRecoveryCodes {
30  
31      /** Name of the session attribute that holds the codes. */
32      static final String ATTRIBUTE = PendingRecoveryCodes.class.getName() + ".CODES";
33  
34      /**
35       * Stores the codes in the session of a request.
36       *
37       * @param request the request.
38       * @param codes   the codes in clear text.
39       */
40      public void save(final HttpServletRequest request, final List<String> codes) {
41          request.getSession().setAttribute(ATTRIBUTE, new ArrayList<>(codes));
42      }
43  
44      /**
45       * Tells whether the session of a request holds codes not shown yet.
46       *
47       * @param request the request.
48       * @return {@code true} when there are codes waiting to be shown.
49       */
50      public boolean isPending(final HttpServletRequest request) {
51          final var session = request.getSession(false);
52          return session != null && session.getAttribute(ATTRIBUTE) != null;
53      }
54  
55      /**
56       * Takes the codes out of the session of a request, so they are shown only once.
57       *
58       * @param request the request.
59       * @return the codes; empty when there are none.
60       */
61      public List<String> take(final HttpServletRequest request) {
62          final var session = request.getSession(false);
63          if (session == null) {
64              return List.of();
65          }
66          final var stored = session.getAttribute(ATTRIBUTE);
67          session.removeAttribute(ATTRIBUTE);
68          if (!(stored instanceof List<?> codes)) {
69              return List.of();
70          }
71          return codes.stream().map(String::valueOf).toList();
72      }
73  }
