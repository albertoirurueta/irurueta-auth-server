17  package com.irurueta.authserver.authorization.api.web.social;
18  
19  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
20  import java.util.Objects;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.security.oauth2.client.oidc.userinfo.OidcUserRequest;
23  import org.springframework.security.oauth2.client.userinfo.OAuth2UserService;
24  import org.springframework.security.oauth2.core.OAuth2AuthenticationException;
25  import org.springframework.security.oauth2.core.OAuth2Error;
26  import org.springframework.security.oauth2.core.OAuth2ErrorCodes;
27  import org.springframework.security.oauth2.core.oidc.user.OidcUser;
28  
29  // tag::social-oidc-user-service[]
30  /**
31   * The user service of the OpenID Connect providers (Google, Microsoft, Apple): the framework's service loads the user
32   * as it always does (the ID token, and the UserInfo endpoint where the provider has one), then the account is resolved
33   * (found, linked by verified e-mail or provisioned, and refused when it may not sign in) and the principal becomes the
34   * local one, named after the account.
35   */
36  @RequiredArgsConstructor
37  public class FederatedOidcUserService implements OAuth2UserService<OidcUserRequest, OidcUser> {
38  
39      /** Message of the failure when the framework's service answers no user. */
40      private static final String MISSING_USER = "The provider's user is missing";
41  
42      /** Message of the failure when the user has no subject. */
43      private static final String MISSING_SUBJECT = "The provider's subject is missing";
44  
45      /** The framework's service, which loads the user the provider describes. */
46      private final OAuth2UserService<OidcUserRequest, OidcUser> delegate;
47  
48      /** Resolves the local account. */
49      private final FederatedAccountResolver accountResolver;
50  
51      /** Builds the local principal. */
52      private final FederatedUserMapper userMapper;
53  
54      /**
55       * Loads the user of a sign-in.
56       *
57       * @param request the request, with the registration, the access token and the ID token.
58       * @return the local principal.
59       * @throws OAuth2AuthenticationException if the provider is unknown or the sign-in is refused.
60       */
61      @Override
62      public OidcUser loadUser(final OidcUserRequest request) throws OAuth2AuthenticationException {
63          final var provider = IdentityProvider.fromId(request.getClientRegistration().getRegistrationId())
64                  .orElseThrow(() -> new OAuth2AuthenticationException(
65                          new OAuth2Error(OAuth2ErrorCodes.INVALID_REQUEST, "Unknown provider", null)));
66          final var upstream = Objects.requireNonNull(delegate.loadUser(request), MISSING_USER);
67          final var subject = Objects.requireNonNull(upstream.getSubject(), MISSING_SUBJECT);
68          final var account = accountResolver.resolve(
69                  provider, subject, upstream.getEmail(), userMapper.emailVerified(upstream));
70          return userMapper.toLocalUser(upstream, account.getUsername(), account.getAuthorities());
71      }
72  }
