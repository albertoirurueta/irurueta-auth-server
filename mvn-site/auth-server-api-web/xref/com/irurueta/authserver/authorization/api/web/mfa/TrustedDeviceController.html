17  package com.irurueta.authserver.authorization.api.web.mfa;
18  
19  import com.irurueta.authserver.authorization.api.web.v1.generated.api.TrustedDevicesApi;
20  import com.irurueta.authserver.authorization.api.web.v1.generated.model.TrustedDeviceResponse;
21  import com.irurueta.authserver.authorization.application.mfa.ListTrustedDevicesUseCase;
22  import com.irurueta.authserver.authorization.application.mfa.RevokeTrustedDeviceUseCase;
23  import java.util.List;
24  import java.util.UUID;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.http.ResponseEntity;
27  import org.springframework.web.bind.annotation.RestController;
28  
29  /**
30   * Lists and revokes the trusted devices of the authenticated account. The paths, status codes and bodies are those
31   * of the contract {@code apis/auth-server/rest-server/auth-server-api-v1.yaml}, through the generated
32   * {@link TrustedDevicesApi}; failures are answered by {@link MfaExceptionHandler}. Revoking a device needs a recent
33   * multi-factor sign-in ({@link StepUpGuard}).
34   */
35  @RestController
36  @RequiredArgsConstructor
37  public class TrustedDeviceController implements TrustedDevicesApi {
38  
39      /** Resolves the account of the request. */
40      private final CurrentAccountResolver currentAccountResolver;
41  
42      /** Lists the devices of an account. */
43      private final ListTrustedDevicesUseCase listTrustedDevicesUseCase;
44  
45      /** Revokes a device. */
46      private final RevokeTrustedDeviceUseCase revokeTrustedDeviceUseCase;
47  
48      /** Requires a recent multi-factor sign-in before a device is revoked. */
49      private final StepUpGuard stepUpGuard;
50  
51      /** Maps the devices onto the contract's DTOs. */
52      private final TrustedDeviceDtoMapper trustedDeviceDtoMapper;
53  
54      /**
55       * Lists the devices that are still trusted.
56       *
57       * @return {@code 200 OK} with the devices of the authenticated account.
58       */
59      @Override
60      public ResponseEntity<List<TrustedDeviceResponse>> listTrustedDevices() {
61          final var account = currentAccountResolver.resolve();
62          return ResponseEntity.ok(trustedDeviceDtoMapper.toResponses(listTrustedDevicesUseCase.execute(account.id())));
63      }
64  
65      /**
66       * Revokes a device of the authenticated account.
67       *
68       * @param id identifier of the device.
69       * @return {@code 204 No Content}.
70       * @throws TrustedDeviceNotFoundException          if the account has no such device.
71       * @throws InsufficientUserAuthenticationException if the token does not show a recent multi-factor sign-in.
72       */
73      @Override
74      public ResponseEntity<Void> revokeTrustedDevice(final UUID id) {
75          final var account = currentAccountResolver.resolve();
76          stepUpGuard.requireRecentMultiFactor(account.id());
77          if (!revokeTrustedDeviceUseCase.execute(account.id(), account.username(), id)) {
78              throw new TrustedDeviceNotFoundException(id);
79          }
80          return ResponseEntity.noContent().build();
81      }
82  }
