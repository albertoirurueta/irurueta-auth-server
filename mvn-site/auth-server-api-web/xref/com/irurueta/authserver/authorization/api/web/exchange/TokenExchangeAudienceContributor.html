35  // tag::exchange-claims[]
36  /**
37   * Honours the {@code audience} parameter of a token exchange request, which the framework parses but ignores. The
38   * audiences of an exchanged access token are those the exchanging client may have (its
39   * {@value IssuedTokenClaims#AUDIENCES_SETTING} setting); when the request names audiences, only the ones among them
40   * are kept. An audience outside the list of the client is simply not granted, and when none of the requested ones is
41   * allowed the request is refused with {@value #INVALID_TARGET} (RFC 8693, section 2.2.2), so a narrowed request never
42   * comes back with every audience of the client. Tokens of any other grant type are not touched.
43   */
44  @Component
45  @Order(TokenExchangeAudienceContributor.ORDER)
46  public class TokenExchangeAudienceContributor implements JwtClaimsContributor {
47  
48      /** Order of the contributor: after the multi-factor ({@code 100}) and identity provider ({@code 110}) ones. */
49      public static final int ORDER = 120;
50  
51      /** Error code of an unacceptable target (RFC 8693, section 2.2.2), which the framework has no constant for. */
52      public static final String INVALID_TARGET = "invalid_target";
53  
54      @Override
55      public void contribute(final OAuth2TokenContext context, final Map<String, Object> claims) {
56          if (!OAuth2TokenType.ACCESS_TOKEN.equals(context.getTokenType())
57                  || !AuthorizationGrantType.TOKEN_EXCHANGE.equals(context.getAuthorizationGrantType())
58                  || !(context.getAuthorizationGrant() instanceof OAuth2TokenExchangeAuthenticationToken exchange)
59                  || exchange.getAudiences().isEmpty()) {
60              return;
61          }
62          final var allowed = context.getRegisteredClient().getClientSettings()
63                  .<Object>getSetting(IssuedTokenClaims.AUDIENCES_SETTING) instanceof List<?> audiences
64                  ? audiences : List.of();
65          // a mutable list on purpose: the authorization store only deserializes allow-listed types
66          final var granted = allowed.stream().map(String::valueOf).filter(exchange.getAudiences()::contains)
67                  .collect(Collectors.toCollection(ArrayList::new));
68          if (granted.isEmpty()) {
69              throw new OAuth2AuthenticationException(new OAuth2Error(INVALID_TARGET,
70                      "None of the requested audiences is allowed for the client", null));
71          }
72          claims.put(OAuth2TokenClaimNames.AUD, granted);
73      }
74  }
