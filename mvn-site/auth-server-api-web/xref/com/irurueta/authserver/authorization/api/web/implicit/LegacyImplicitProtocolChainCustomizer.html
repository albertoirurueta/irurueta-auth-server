16  package com.irurueta.authserver.authorization.api.web.implicit;
17  
18  import com.irurueta.authserver.authorization.api.web.ProtocolChainCustomizer;
19  import lombok.RequiredArgsConstructor;
20  import org.springframework.security.authentication.AuthenticationProvider;
21  import org.springframework.security.config.annotation.web.configurers.oauth2.server.authorization.OAuth2AuthorizationServerConfigurer;
22  import org.springframework.security.web.authentication.AuthenticationConverter;
23  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
24  
25  // tag::implicit-chain[]
26  /**
27   * Adds the legacy implicit flow to the authorization endpoint: its converter first, its provider first and its
28   * response handler in place of the framework's (which it delegates to for an authorization code).
29   */
30  @RequiredArgsConstructor
31  public class LegacyImplicitProtocolChainCustomizer implements ProtocolChainCustomizer {
32  
33      /** Order of the customizer, after the client authentication one ({@code 3}). */
34      public static final int ORDER = 4;
35  
36      /** Reads a request with {@code response_type=token}. */
37      private final AuthenticationConverter converter;
38  
39      /** Grants the request. */
40      private final AuthenticationProvider provider;
41  
42      /** Redirects with the token in the fragment. */
43      private final AuthenticationSuccessHandler responseHandler;
44  
45      /**
46       * Customizes the authorization endpoint.
47       *
48       * @param server configurer of the authorization server.
49       */
50      @Override
51      public void customize(final OAuth2AuthorizationServerConfigurer server) {
52          server.authorizationEndpoint(endpoint -> endpoint
53                  .authorizationRequestConverters(converters -> converters.addFirst(converter))
54                  .authenticationProviders(providers -> providers.addFirst(provider))
55                  .authorizationResponseHandler(responseHandler));
56      }
57  }
58  // end::implicit-chain[]
