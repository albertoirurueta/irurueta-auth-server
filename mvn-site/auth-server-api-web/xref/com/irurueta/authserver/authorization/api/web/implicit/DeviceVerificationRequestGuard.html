35  // tag::device-guard[]
36  /**
37   * Guards the device verification endpoint. It is installed first among the converters of the endpoint and never
38   * converts anything: it either lets the request go on (returning {@code null}) or refuses it. Every request that
39   * carries a user code (the code typed on the verification page, a link that carries it, and the consent decision) is
40   * counted against the limit on the attempts, and the multi-factor policy of the account and of the client must be
41   * satisfied, as the authorization endpoint requires before it issues a code: a user who owes a second factor is sent
42   * to the second-factor page and resumes afterwards. The framework's own MFA customizer covers only
43   * {@code /oauth2/authorize}.
44   *
45   * <p>The client is the one the device authorization stored for the user code was issued to, because the code typed on
46   * the page (or carried by the link) is the only thing the request names: so a client that requires multi-factor
47   * authentication is enforced even when it skips the consent page. Only when no authorization holds the code (an
48   * unknown or expired one, which the endpoint refuses anyway) is the {@code client_id} of the consent decision used.
49   */
50  @RequiredArgsConstructor
51  public class DeviceVerificationRequestGuard implements AuthenticationConverter {
52  
53      /** Type of the token the user code is stored as. */
54      static final OAuth2TokenType USER_CODE_TOKEN_TYPE = new OAuth2TokenType(OAuth2ParameterNames.USER_CODE);
55  
56      /** Number of letters and digits of a user code. */
57      private static final int USER_CODE_LENGTH = 8;
58  
59      /** Position of the dash the framework stores a user code with. */
60      private static final int USER_CODE_GROUP = USER_CODE_LENGTH / 2;
61  
62      /** Everything a user may type around the letters and digits of a code (dashes, spaces). */
63      private static final Pattern NOT_ALPHANUMERIC = Pattern.compile("[^A-Z\\d]+");
64  
65      /** Limits the attempts. */
66      private final DeviceVerificationRateLimiter rateLimiter;
67  
68      /** Applies the multi-factor policy. */
69      private final MfaAuthorizationManager authorizationManager;
70  
71      /** Finds the device authorization of a user code. */
72      private final OAuth2AuthorizationService authorizationService;
73  
74      /** Finds the client of a device authorization. */
75      private final RegisteredClientRepository registeredClientRepository;
76  
77      /**
78       * Checks the request.
79       *
80       * @param request the request.
81       * @return always {@code null}: this converter only vetoes requests.
82       * @throws DeviceVerificationRateLimitedException if the client address or the user made too many attempts.
83       * @throws SecondFactorRequiredException          if the user still owes a second factor.
84       */
85      @Override
86      public Authentication convert(final HttpServletRequest request) {
87          final var user = SecurityContextHolder.getContext().getAuthentication();
88          if (!MfaAuthorizationManager.isUser(user)) {
89              return null;
90          }
91          rateLimiter.check(request.getRemoteAddr(), user.getName());
92          final var clientRequires = authorizationManager.clientRequiresMfa(clientIdOf(request));
93          if (!authorizationManager.evaluate(user, clientRequires).isGranted()) {
94              throw new SecondFactorRequiredException();
95          }
96          return null;
97      }
98  
99      /**
100      * Finds the client a request is about: the client of the device authorization stored for its user code, or the
101      * {@code client_id} parameter when no authorization holds that code.
102      *
103      * @param request the request.
104      * @return the client id, or {@code null} when neither tells it.
105      */
106     private String clientIdOf(final HttpServletRequest request) {
107         return normalize(request.getParameter(OAuth2ParameterNames.USER_CODE))
108                 .map(userCode -> authorizationService.findByToken(userCode, USER_CODE_TOKEN_TYPE))
109                 .map(authorization -> registeredClientRepository.findById(authorization.getRegisteredClientId()))
110                 .map(RegisteredClient::getClientId)
111                 .orElseGet(() -> request.getParameter(OAuth2ParameterNames.CLIENT_ID));
112     }
113 
114     /**
115      * Normalises a user code as the framework stores it: upper case letters and digits in two groups of four,
116      * separated by a dash.
117      *
118      * @param userCode the code as typed, may be {@code null}.
119      * @return the stored form, or empty when the value cannot be a user code.
120      */
121     private static Optional<String> normalize(final String userCode) {
122         if (!StringUtils.hasText(userCode)) {
123             return Optional.empty();
124         }
125         final var compact = NOT_ALPHANUMERIC.matcher(userCode.toUpperCase(Locale.ROOT)).replaceAll("");
126         if (compact.length() != USER_CODE_LENGTH) {
127             return Optional.empty();
128         }
129         return Optional.of(compact.substring(0, USER_CODE_GROUP) + "-" + compact.substring(USER_CODE_GROUP));
130     }
131 }
