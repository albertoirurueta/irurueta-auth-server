16  package com.irurueta.authserver.authorization.api.web.implicit;
17  
18  import jakarta.servlet.ServletException;
19  import jakarta.servlet.http.HttpServletRequest;
20  import jakarta.servlet.http.HttpServletResponse;
21  import java.io.IOException;
22  import java.nio.charset.StandardCharsets;
23  import java.time.Duration;
24  import java.util.ArrayList;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.http.HttpHeaders;
27  import org.springframework.security.core.Authentication;
28  import org.springframework.security.oauth2.core.OAuth2AccessToken;
29  import org.springframework.security.oauth2.core.endpoint.OAuth2ParameterNames;
30  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
31  import org.springframework.util.StringUtils;
32  import org.springframework.web.util.UriUtils;
33  
34  // tag::implicit-response-handler[]
35  /**
36   * Answers the authorization endpoint: for the two results of the legacy implicit flow it redirects with {@code 302} to
37   * the registered redirect URI and carries the token, or the error, in its fragment (so it never reaches a server log
38   * as a query would); any other result is the framework's authorization code response and goes to the delegate. The
39   * response is never cached and the browser sends no referrer, so the fragment does not leak through the next request.
40   */
41  @RequiredArgsConstructor
42  public class LegacyImplicitResponseHandler implements AuthenticationSuccessHandler {
43  
44      /** Value of {@code Cache-Control} of the response. */
45      static final String NO_STORE = "no-store";
46  
47      /** Value of {@code Referrer-Policy} of the response. */
48      static final String NO_REFERRER = "no-referrer";
49  
50      /** Name of the {@code Referrer-Policy} header. */
51      private static final String REFERRER_POLICY = "Referrer-Policy";
52  
53      /** Token type reported to the client. */
54      private static final String BEARER = "Bearer";
55  
56      /** Answers the results that are not the legacy flow's. */
57      private final AuthenticationSuccessHandler delegate;
58  
59      /**
60       * Answers a result.
61       *
62       * @param request        the request.
63       * @param response       the response.
64       * @param authentication the result of the authorization request.
65       * @throws IOException      if the response cannot be written.
66       * @throws ServletException if the delegate fails.
67       */
68      @Override
69      public void onAuthenticationSuccess(final HttpServletRequest request, final HttpServletResponse response,
70              final Authentication authentication) throws IOException, ServletException {
71          if (authentication instanceof LegacyImplicitAccessTokenToken granted) {
72              redirect(response, granted.getRedirectUri(), grantedFragment(granted));
73          } else if (authentication instanceof LegacyImplicitErrorToken refused) {
74              redirect(response, refused.getRedirectUri(), errorFragment(refused));
75          } else {
76              delegate.onAuthenticationSuccess(request, response, authentication);
77          }
78      }
79  
80      /**
81       * Builds the fragment of a granted request.
82       *
83       * @param granted the result.
84       * @return the encoded fragment.
85       */
86      private static String grantedFragment(final LegacyImplicitAccessTokenToken granted) {
87          final var token = granted.getAccessToken();
88          final var parameters = new ArrayList<String>();
89          parameters.add(parameter(OAuth2ParameterNames.ACCESS_TOKEN, token.getTokenValue()));
90          parameters.add(parameter(OAuth2ParameterNames.TOKEN_TYPE, BEARER));
91          parameters.add(parameter(OAuth2ParameterNames.EXPIRES_IN, Long.toString(expiresIn(token))));
92          parameters.add(parameter(OAuth2ParameterNames.SCOPE, String.join(" ", granted.getScopes())));
93          addState(parameters, granted.getState());
94          return String.join("&", parameters);
95      }
96  
97      /**
98       * Builds the fragment of a refusal.
99       *
100      * @param refused the result.
101      * @return the encoded fragment.
102      */
103     private static String errorFragment(final LegacyImplicitErrorToken refused) {
104         final var parameters = new ArrayList<String>();
105         parameters.add(parameter(OAuth2ParameterNames.ERROR, refused.getError()));
106         if (StringUtils.hasText(refused.getDescription())) {
107             parameters.add(parameter(OAuth2ParameterNames.ERROR_DESCRIPTION, refused.getDescription()));
108         }
109         addState(parameters, refused.getState());
110         return String.join("&", parameters);
111     }
112 
113     /**
114      * Adds the state, when the request had one.
115      *
116      * @param parameters the encoded parameters.
117      * @param state      the state, may be {@code null}.
118      */
119     private static void addState(final ArrayList<String> parameters, final String state) {
120         if (StringUtils.hasText(state)) {
121             parameters.add(parameter(OAuth2ParameterNames.STATE, state));
122         }
123     }
124 
125     /**
126      * Encodes one parameter.
127      *
128      * @param name  name of the parameter.
129      * @param value value of the parameter.
130      * @return {@code name=value}, with the value encoded.
131      */
132     private static String parameter(final String name, final String value) {
133         return name + "=" + UriUtils.encodeQueryParam(value, StandardCharsets.UTF_8);
134     }
135 
136     /**
137      * Computes the lifetime of a token in seconds.
138      *
139      * @param token the token.
140      * @return the seconds between its issue and its expiry.
141      */
142     private static long expiresIn(final OAuth2AccessToken token) {
143         return Duration.between(token.getIssuedAt(), token.getExpiresAt()).getSeconds();
144     }
145 
146     /**
147      * Sends the redirect, with the headers that keep the fragment private.
148      *
149      * @param response    the response.
150      * @param redirectUri the registered redirect URI.
151      * @param fragment    the encoded fragment.
152      * @throws IOException if the response cannot be written.
153      */
154     private static void redirect(final HttpServletResponse response, final String redirectUri, final String fragment)
155             throws IOException {
156         response.setHeader(HttpHeaders.CACHE_CONTROL, NO_STORE);
157         response.setHeader(REFERRER_POLICY, NO_REFERRER);
158         response.sendRedirect(redirectUri + "#" + fragment);
159     }
160 }
