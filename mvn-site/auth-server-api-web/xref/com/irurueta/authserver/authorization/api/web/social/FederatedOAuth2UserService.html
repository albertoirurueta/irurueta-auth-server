32  /**
33   * The user service of the providers that are plain OAuth 2.0 (GitHub), which have no ID token: the framework's service
34   * reads the profile from the provider's API, the verified primary e-mail address is read from {@code /user/emails}
35   * (through the domain port {@link VerifiedEmailLookup}, implemented by the GitHub adapter of
36   * {@code infrastructure/github} with a client generated from the contract of that GitHub operation), the account is
37   * resolved (found, linked by verified e-mail or provisioned, and refused when it may not sign in) and the principal
38   * becomes the local one, named after the account. The stable subject is GitHub's numeric id, never the login name or
39   * the e-mail address, which the user can change.
40   */
41  @RequiredArgsConstructor
42  public class FederatedOAuth2UserService implements OAuth2UserService<OAuth2UserRequest, OAuth2User> {
43  
44      /** Attributes of GitHub's profile that are kept in the session: scalars only. */
45      static final List<String> KEPT_ATTRIBUTES = List.of("id", "login", "name", "avatar_url");
46  
47      /** Message of the failure when the framework's service answers no user. */
48      private static final String MISSING_USER = "The provider's user is missing";
49  
50      /** The framework's service, which reads the profile. */
51      private final OAuth2UserService<OAuth2UserRequest, OAuth2User> delegate;
52  
53      /** Reads the verified primary e-mail address. */
54      private final VerifiedEmailLookup verifiedEmailLookup;
55  
56      /** Resolves the local account. */
57      private final FederatedAccountResolver accountResolver;
58  
59      /** Builds the local principal. */
60      private final FederatedUserMapper userMapper;
61  
62      /**
63       * Loads the user of a sign-in.
64       *
65       * @param request the request, with the registration and the access token.
66       * @return the local principal.
67       * @throws OAuth2AuthenticationException if the provider is not GitHub or the sign-in is refused.
68       */
69      @Override
70      public OAuth2User loadUser(final OAuth2UserRequest request) throws OAuth2AuthenticationException {
71          final var registration = request.getClientRegistration();
72          if (!IdentityProvider.GITHUB.id().equals(registration.getRegistrationId())) {
73              throw new OAuth2AuthenticationException(
74                      new OAuth2Error(OAuth2ErrorCodes.INVALID_REQUEST, "Unknown provider", null));
75          }
76          final var upstream = Objects.requireNonNull(delegate.loadUser(request), MISSING_USER);
77          final var email = verifiedEmailLookup.verifiedPrimaryEmail(
78                  registration.getProviderDetails().getUserInfoEndpoint().getUri(),
79                  request.getAccessToken().getTokenValue());
80          final var account = accountResolver.resolve(
81                  IdentityProvider.GITHUB, upstream.getName(), email.orElse(null), email.isPresent());
82          return userMapper.toLocalUser(
83                  upstream, KEPT_ATTRIBUTES, email.orElse(null), account.getUsername(), account.getAuthorities());
84      }
85  }
