17  package com.irurueta.authserver.authorization.api.web.clients;
18  
19  import com.nimbusds.jose.jwk.JWK;
20  import com.nimbusds.jose.jwk.JWKSet;
21  import com.nimbusds.jose.util.JSONObjectUtils;
22  import java.text.ParseException;
23  import java.util.Map;
24  import org.springframework.stereotype.Component;
25  
26  /**
27   * Parses and validates the public JWK set a client registers for {@code private_key_jwt} and self-signed mutual TLS
28   * authentication, and converts it between the JSON object of the contract and the JSON text the domain keeps.
29   */
30  @Component
31  public class ClientJwksParser {
32  
33      // tag::clients-auth-settings[]
34      /**
35       * Validates a JWK set sent by an administrator and returns its JSON text. Only public keys are accepted: a key
36       * with private material (including a symmetric key) is refused.
37       *
38       * @param jwks the JWK set as a JSON object, or {@code null}.
39       * @return the JWK set as JSON text, or {@code null} when none was given.
40       * @throws InvalidClientJwksException if the object is not a JWK set, has no keys or has a private key.
41       */
42      public String toJson(final Map<String, Object> jwks) {
43          if (jwks == null) {
44              return null;
45          }
46          final JWKSet parsed;
47          try {
48              parsed = JWKSet.parse(JSONObjectUtils.toJSONString(jwks));
49          } catch (ParseException | IllegalArgumentException e) {
50              throw new InvalidClientJwksException("jwks is not a valid JWK set", e);
51          }
52          if (parsed.getKeys().isEmpty()) {
53              throw new InvalidClientJwksException("jwks must contain at least one key");
54          }
55          if (parsed.getKeys().stream().anyMatch(JWK::isPrivate)) {
56              throw new InvalidClientJwksException("jwks must contain public keys only");
57          }
58          return parsed.toString();
59      }
60      // end::clients-auth-settings[]
61  
62      /**
63       * Converts the JSON text of a registered JWK set to the JSON object of the contract.
64       *
65       * @param json the JWK set as JSON text, or {@code null}.
66       * @return the JWK set as an object, or {@code null} when there is none.
67       * @throws InvalidClientJwksException if the stored text is not a JWK set.
68       */
69      public Map<String, Object> toObject(final String json) {
70          if (json == null) {
71              return null;
72          }
73          try {
74              return JWKSet.parse(json).toJSONObject(true);
75          } catch (ParseException e) {
76              throw new InvalidClientJwksException("the registered jwks is not a valid JWK set", e);
77          }
78      }
79  }
