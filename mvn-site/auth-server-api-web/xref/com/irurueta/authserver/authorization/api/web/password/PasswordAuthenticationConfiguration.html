17  package com.irurueta.authserver.authorization.api.web.password;
18  
19  import org.springframework.beans.factory.ObjectProvider;
20  import org.springframework.context.annotation.Bean;
21  import org.springframework.context.annotation.Configuration;
22  import org.springframework.security.authentication.AccountStatusUserDetailsChecker;
23  import org.springframework.security.authentication.dao.DaoAuthenticationProvider;
24  import org.springframework.security.authentication.password.CompromisedPasswordChecker;
25  import org.springframework.security.core.userdetails.UserDetailsChecker;
26  import org.springframework.security.core.userdetails.UserDetailsPasswordService;
27  import org.springframework.security.core.userdetails.UserDetailsService;
28  import org.springframework.security.crypto.password.PasswordEncoder;
29  
30  /**
31   * Declares the authentication provider of the password-based sign-in, instead of the one Spring Security builds
32   * from the {@link UserDetailsService} bean, to close a user enumeration.
33   *
34   * <p>The provider Spring Security builds checks the account status (locked, disabled, expired) <em>before</em> the
35   * password, so a wrong password for, say, a disabled account answers "disabled": the page would reveal that the
36   * account exists to anyone who types its username. This provider checks nothing before the password and the account
37   * status after it ({@link AccountStatusUserDetailsChecker}), so a wrong password always gets the same generic answer,
38   * known user or not, and a status message only appears to whoever proved they know the password.
39   *
40   * <p>Everything else is kept as the auto-built provider had it: the {@link PasswordEncoder}, the
41   * {@link UserDetailsPasswordService} (which stores the stronger hash after a sign-in with an outdated encoding), a
42   * {@link CompromisedPasswordChecker} when one bean of that type exists, and the hiding of unknown users behind a bad
43   * credentials error. Being the only {@code AuthenticationProvider} bean, it is the one the global authentication
44   * manager uses.
45   */
46  @Configuration(proxyBeanMethods = false)
47  public class PasswordAuthenticationConfiguration {
48  
49      /**
50       * Checks nothing before the password: the account status is only checked, and reported, after it.
51       */
52      static final UserDetailsChecker NO_PRE_AUTHENTICATION_CHECKS = user -> {
53          // the account status is checked after the password, by the post-authentication checks
54      };
55  
56      // tag::password-authentication-provider[]
57      /**
58       * Creates the authentication provider of the username and password sign-in.
59       *
60       * @param userDetailsService         loads the account by username.
61       * @param userDetailsPasswordService stores the upgraded hash after a sign-in with an outdated encoding.
62       * @param passwordEncoder            verifies the password, and tells when its hash needs upgrading.
63       * @param compromisedPasswordChecker the compromised-password checker of Spring Security, when one is declared.
64       * @return the provider, which checks the account status only after a correct password.
65       */
66      @Bean
67      public DaoAuthenticationProvider passwordAuthenticationProvider(final UserDetailsService userDetailsService,
68              final UserDetailsPasswordService userDetailsPasswordService, final PasswordEncoder passwordEncoder,
69              final ObjectProvider<CompromisedPasswordChecker> compromisedPasswordChecker) {
70          final var provider = new DaoAuthenticationProvider(userDetailsService);
71          provider.setPasswordEncoder(passwordEncoder);
72          provider.setUserDetailsPasswordService(userDetailsPasswordService);
73          compromisedPasswordChecker.ifUnique(provider::setCompromisedPasswordChecker);
74          provider.setHideUserNotFoundExceptions(true);
75          provider.setPreAuthenticationChecks(NO_PRE_AUTHENTICATION_CHECKS);
76          provider.setPostAuthenticationChecks(new AccountStatusUserDetailsChecker());
77          return provider;
78      }
79      // end::password-authentication-provider[]
80  }
