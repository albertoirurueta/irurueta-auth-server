16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.domain.ott.OttChannel;
19  import com.irurueta.authserver.authorization.domain.ott.OttKind;
20  import com.irurueta.authserver.authorization.domain.ott.OttPurpose;
21  import jakarta.servlet.http.HttpServletRequest;
22  import java.util.Locale;
23  import org.springframework.security.authentication.AnonymousAuthenticationToken;
24  import org.springframework.security.core.Authentication;
25  
26  /**
27   * What a request to {@code POST /ott/generate} asks for, read from its form fields: through which channel, a numeric
28   * code or a magic link, and for a sign-in or as a second factor. Every unknown or missing value falls back to the
29   * default (an e-mail code for a sign-in), so a malformed request is answered like any other.
30   *
31   * @param channel channel the token is delivered through.
32   * @param kind    whether the token is a numeric code or a magic link.
33   * @param purpose whether the token signs the user in or is a second factor of a signed-in user.
34   */
35  public record OttGenerationParameters(OttChannel channel, OttKind kind, OttPurpose purpose) {
36  
37      /** Name of the field that holds the identifier (the user name) of a sign-in. */
38      public static final String USERNAME_PARAMETER = "username";
39  
40      /** Name of the field that holds the channel, {@code email} (default) or {@code sms}. */
41      public static final String CHANNEL_PARAMETER = "channel";
42  
43      /** Name of the field that holds the kind of token, {@code code} (default) or {@code link}. */
44      public static final String KIND_PARAMETER = "kind";
45  
46      /** Name of the field that holds the purpose, {@code login} (default) or {@code second-factor}. */
47      public static final String PURPOSE_PARAMETER = "purpose";
48  
49      /** Value of the channel field that selects SMS. */
50      public static final String SMS_VALUE = "sms";
51  
52      /** Value of the kind field that selects a magic link. */
53      public static final String LINK_VALUE = "link";
54  
55      /** Value of the purpose field that selects the second factor. */
56      public static final String SECOND_FACTOR_VALUE = "second-factor";
57  
58      /**
59       * Reads the parameters of a request. A magic link is only sent by e-mail, so an SMS request is always a code, and
60       * a second factor is only possible for a user who is signed in: for anybody else the request is a sign-in.
61       *
62       * @param request the request.
63       * @param current the authentication of the request, or {@code null}.
64       * @return the parameters.
65       */
66      public static OttGenerationParameters from(final HttpServletRequest request, final Authentication current) {
67          final var sms = SMS_VALUE.equals(lowerCase(request.getParameter(CHANNEL_PARAMETER)));
68          final var link = !sms && LINK_VALUE.equals(lowerCase(request.getParameter(KIND_PARAMETER)));
69          final var secondFactor = SECOND_FACTOR_VALUE.equals(lowerCase(request.getParameter(PURPOSE_PARAMETER)))
70                  && isSignedIn(current);
71          return new OttGenerationParameters(sms ? OttChannel.SMS : OttChannel.EMAIL,
72                  link ? OttKind.LINK : OttKind.CODE, secondFactor ? OttPurpose.SECOND_FACTOR : OttPurpose.LOGIN);
73      }
74  
75      /**
76       * Tells whether an authentication is a signed-in user.
77       *
78       * @param authentication the authentication, or {@code null}.
79       * @return {@code true} when it is authenticated and not anonymous.
80       */
81      static boolean isSignedIn(final Authentication authentication) {
82          return authentication != null && authentication.isAuthenticated()
83                  && !(authentication instanceof AnonymousAuthenticationToken);
84      }
85  
86      /**
87       * Lower-cases a value, whatever the platform's locale is.
88       *
89       * @param value the value, or {@code null}.
90       * @return the value in lower case; the empty text for {@code null}.
91       */
92      private static String lowerCase(final String value) {
93          return value == null ? "" : value.trim().toLowerCase(Locale.ROOT);
94      }
95  }
