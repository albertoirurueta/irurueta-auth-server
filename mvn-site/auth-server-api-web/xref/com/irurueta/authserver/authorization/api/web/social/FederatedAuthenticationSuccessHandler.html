21  import jakarta.servlet.http.HttpServletRequest;
22  import jakarta.servlet.http.HttpServletResponse;

24  import lombok.RequiredArgsConstructor;
25  import org.springframework.security.core.Authentication;
26  import org.springframework.security.core.context.SecurityContextHolder;
27  import org.springframework.security.core.context.SecurityContextHolderStrategy;
28  import org.springframework.security.oauth2.client.authentication.OAuth2AuthenticationToken;
29  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
30  import org.springframework.security.web.context.SecurityContextRepository;

32  // tag::social-success-handler[]
33  /**
34   * Success handler of the sign-in with an external provider. The framework's authentication carries no factor
35   * authority, so this handler grants the federated factor through {@link FactorGrantor} (the strong one when the
36   * provider is trusted, see {@link FederatedFactorSelector}), makes the result the current authentication, saves the
37   * security context again (the filter saved the first one before calling the handler), and then lets the usual handler
38   * resume the saved request, for example the {@code /oauth2/authorize} that sent the user to the login page.
39   *
40   * <p>When the answer of the provider completed a link ({@link PendingIdentityLinks#linkedBy}), nobody signs in: the
41   * authentication the session had before is put back and saved, no factor is granted, and the browser returns to the
42   * identities page with the status {@code linked}.
43   */
44  @RequiredArgsConstructor
45  public class FederatedAuthenticationSuccessHandler implements AuthenticationSuccessHandler {
46  
47      /** Query of the identities page after a link. */
48      static final String LINKED_STATUS = "?status=linked";
49  
50      /** Adds the factor authority. */
51      private final FactorGrantor factorGrantor;
52  
53      /** Chooses the factor. */
54      private final FederatedFactorSelector factorSelector;
55  
56      /** Saves the security context, the same repository the sign-in filter uses. */
57      private final SecurityContextRepository securityContextRepository;
58  
59      /** The handler that resumes the saved request. */
60      private final AuthenticationSuccessHandler delegate;
61  
62      /** The link markers of the explicit link flow. */
63      private final PendingIdentityLinks pendingIdentityLinks;
64  
65      /** Holder of the security context of the thread. */
66      private final SecurityContextHolderStrategy holderStrategy = SecurityContextHolder.getContextHolderStrategy();
67  
68      /**
69       * Grants the factor and continues.
70       *
71       * @param request        the request.
72       * @param response       the response.
73       * @param authentication the authentication of the sign-in.
74       * @throws IOException if the redirect fails.
75       * @throws jakarta.servlet.ServletException if the redirect fails.
76       */
77      @Override
78      public void onAuthenticationSuccess(final HttpServletRequest request, final HttpServletResponse response,
79              final Authentication authentication) throws IOException, jakarta.servlet.ServletException {
80          final var linkedBy = pendingIdentityLinks.linkedBy(request);
81          if (linkedBy.isPresent()) {
82              save(linkedBy.get(), request, response);
83              response.sendRedirect(request.getContextPath() + IdentitiesPageApi.PATH_IDENTITIES_PAGE + LINKED_STATUS);
84              return;
85          }
86          if (authentication instanceof final OAuth2AuthenticationToken token) {
87              final var granted = factorGrantor.grant(token, factorSelector.authorityOf(token));
88              save(granted, request, response);
89              delegate.onAuthenticationSuccess(request, response, granted);
90              return;
91          }
92          delegate.onAuthenticationSuccess(request, response, authentication);
93      }
94  
95      /**
96       * Makes an authentication the current one and saves it in the session.
97       *
98       * @param authentication the authentication.
99       * @param request        the request.
100      * @param response       the response.
101      */
102     private void save(final Authentication authentication, final HttpServletRequest request,
103                       final HttpServletResponse response) {
104         final var context = holderStrategy.createEmptyContext();
105         context.setAuthentication(authentication);
106         holderStrategy.setContext(context);
107         securityContextRepository.saveContext(context, request, response);
108     }
109 }
