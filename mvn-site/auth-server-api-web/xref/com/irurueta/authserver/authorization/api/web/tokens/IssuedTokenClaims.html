35  // tag::tokens-claims[]
36  /**
37   * The claims this server adds to the tokens it issues, whatever their format, so the self-contained (JWT) access
38   * tokens and the reference (opaque) ones carry the same claims: {@link RolesTenantAudienceTokenCustomizer} applies it
39   * to the JWTs and {@link ReferenceTokenClaimsCustomizer} to the claims stored for a reference token (which the
40   * introspection endpoint returns).
41   *
42   * <p>An access token gets {@code roles}, {@code tenant} and {@code aud}: the roles are the authorities of the user,
43   * the tenant comes from the user account, the audiences from the {@value #AUDIENCES_SETTING} custom setting of the
44   * client. Tokens of the client credentials grant have no user, so they get no roles and no tenant. An ID token gets
45   * none of them. Then every {@link JwtClaimsContributor}, in order, adds its own claims to the access and ID tokens
46   * (it can tell them apart with {@link OAuth2TokenContext#getTokenType()}); no other token is touched.
47   */
48  @Component
49  @RequiredArgsConstructor
50  public class IssuedTokenClaims {
51  
52      /**
53       * Name of the custom client setting that holds the audiences of the access tokens. It is the same name the
54       * persistence adapter writes (that module cannot be referenced from here).
55       */
56      public static final String AUDIENCES_SETTING = "settings.client.audiences";
57  
58      /** Name of the claim that holds the roles of the user. */
59      public static final String ROLES_CLAIM = "roles";
60  
61      /** Name of the claim that holds the tenant of the user. */
62      public static final String TENANT_CLAIM = "tenant";
63  
64      /** Type of the OpenID Connect ID token. */
65      static final OAuth2TokenType ID_TOKEN = new OAuth2TokenType(OidcParameterNames.ID_TOKEN);
66  
67      /** Prefix of the authorities that record how a user authenticated, such as {@code FACTOR_PASSWORD}. */
68      private static final String FACTOR_PREFIX = "FACTOR_";
69  
70      /** Finds the account of the user. */
71      private final FindUserAccountUseCase findUserAccountUseCase;
72  
73      /** Contributors of additional claims, in order. */
74      private final List<JwtClaimsContributor> contributors;
75  
76      /**
77       * Adds the claims of the token being issued.
78       *
79       * @param context the token context, with the token type, the principal, the client and the grant type.
80       * @param claims  the mutable claims of the token being built.
81       */
82      public void apply(final OAuth2TokenContext context, final Map<String, Object> claims) {
83          if (OAuth2TokenType.ACCESS_TOKEN.equals(context.getTokenType())) {
84              addAccessTokenClaims(context, claims);
85          } else if (!ID_TOKEN.equals(context.getTokenType())) {
86              return;
87          }
88          contributors.forEach(contributor -> contributor.contribute(context, claims));
89      }
90  
91      /**
92       * Adds the roles, the tenant and the audiences of an access token.
93       *
94       * @param context the token context.
95       * @param claims  the mutable claims of the token being built.
96       */
97      private void addAccessTokenClaims(final OAuth2TokenContext context, final Map<String, Object> claims) {
98          if (context.getAuthorizationGrantType() != null
99                  && !AuthorizationGrantType.CLIENT_CREDENTIALS.equals(context.getAuthorizationGrantType())) {
100             final Authentication principal =
101                     Objects.requireNonNull(context.getPrincipal(), "a user grant has a principal");
102             // factor authorities (FACTOR_PASSWORD, ...) record how the user signed in; they are not roles. The
103             // list is a mutable one on purpose: the authorization store only deserializes allow-listed types, and
104             // an immutable list copy (Stream.toList) is not one of them.
105             claims.put(ROLES_CLAIM, principal.getAuthorities().stream()
106                     .map(GrantedAuthority::getAuthority)
107                     .filter(authority -> !authority.startsWith(FACTOR_PREFIX)).sorted()
108                     .collect(Collectors.toCollection(ArrayList::new)));
109             findUserAccountUseCase.execute(principal.getName())
110                     .ifPresent(account -> claims.put(TENANT_CLAIM, account.tenant()));
111         }
112         if (context.getRegisteredClient().getClientSettings().getSetting(AUDIENCES_SETTING)
113                 instanceof List<?> audiences && !audiences.isEmpty()) {
114             claims.put(OAuth2TokenClaimNames.AUD, audiences.stream().map(String::valueOf)
115                     .collect(Collectors.toCollection(ArrayList::new)));
116         }
117     }
118 }
