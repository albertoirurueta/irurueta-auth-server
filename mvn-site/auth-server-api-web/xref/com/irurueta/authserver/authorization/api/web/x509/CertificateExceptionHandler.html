16  package com.irurueta.authserver.authorization.api.web.x509;
17  
18  import com.irurueta.authserver.authorization.api.web.mfa.MfaExceptionHandler;
19  import com.irurueta.authserver.authorization.api.web.social.SignInMethodExceptionHandler;
20  import com.irurueta.authserver.authorization.api.web.v1.generated.api.CertificatesApi;
21  import com.irurueta.authserver.authorization.domain.x509.CertificateAlreadyBoundException;
22  import com.irurueta.authserver.authorization.domain.x509.CertificateNotFoundException;
23  import org.springframework.http.HttpStatus;
24  import org.springframework.http.ProblemDetail;
25  import org.springframework.http.ResponseEntity;
26  import org.springframework.web.bind.annotation.ExceptionHandler;
27  import org.springframework.web.bind.annotation.RestControllerAdvice;
28  
29  /**
30   * Answers the failures of the certificates API as {@code application/problem+json}. The last-sign-in-method and step-up
31   * answers are the shared ones of {@link SignInMethodExceptionHandler}, so a client handles them as for the identities.
32   */
33  @RestControllerAdvice(assignableTypes = CertificatesApi.class)
34  public class CertificateExceptionHandler extends SignInMethodExceptionHandler {
35  
36      /** Base of the problem type URNs. */
37      private static final String TYPE_BASE = "urn:problem:certificate:";
38  
39      /**
40       * Creates the handler.
41       *
42       * @param mfaExceptionHandler builds the step-up challenge.
43       */
44      public CertificateExceptionHandler(final MfaExceptionHandler mfaExceptionHandler) {
45          super(TYPE_BASE, mfaExceptionHandler);
46      }
47  
48      /**
49       * Answers a value that is not a usable certificate.
50       *
51       * @param exception the failure.
52       * @return the {@code 400} problem detail.
53       */
54      @ExceptionHandler(InvalidCertificateException.class)
55      public ResponseEntity<ProblemDetail> handleInvalidCertificate(final InvalidCertificateException exception) {
56          return respond(HttpStatus.BAD_REQUEST, "invalid-certificate", "Invalid certificate", exception.getMessage());
57      }
58  
59      /**
60       * Answers a certificate that is bound to another account.
61       *
62       * @param exception the failure.
63       * @return the {@code 409} problem detail.
64       */
65      @ExceptionHandler(CertificateAlreadyBoundException.class)
66      public ResponseEntity<ProblemDetail> handleAlreadyBound(final CertificateAlreadyBoundException exception) {
67          return respond(HttpStatus.CONFLICT, "certificate-already-bound", "Certificate already bound",
68                  exception.getMessage());
69      }
70  
71      /**
72       * Answers a certificate the account does not have (another account's certificate is answered the same way).
73       *
74       * @param exception the failure.
75       * @return the {@code 404} problem detail.
76       */
77      @ExceptionHandler(CertificateNotFoundException.class)
78      public ResponseEntity<ProblemDetail> handleNotFound(final CertificateNotFoundException exception) {
79          return respond(HttpStatus.NOT_FOUND, "certificate-not-found", "Certificate not found", exception.getMessage());
80      }
81  }
