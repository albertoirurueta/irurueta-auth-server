16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.api.web.ProblemJsonExceptionHandler;
19  import com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException;
20  import com.irurueta.authserver.authorization.api.web.mfa.MfaExceptionHandler;
21  import com.irurueta.authserver.authorization.api.web.v1.generated.api.PhoneApi;
22  import com.irurueta.authserver.authorization.application.sms.InvalidPhoneNumberException;
23  import com.irurueta.authserver.authorization.application.sms.SmsCountryNotAllowedException;
24  import java.net.URI;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.http.HttpHeaders;
27  import org.springframework.http.HttpStatus;
28  import org.springframework.http.ProblemDetail;
29  import org.springframework.http.ResponseEntity;
30  import org.springframework.web.bind.annotation.ExceptionHandler;
31  import org.springframework.web.bind.annotation.RestControllerAdvice;
32  
33  /**
34   * Translates the failures of the phone endpoints to RFC 9457 problem details, answered as
35   * {@code application/problem+json}. It is scoped to the generated {@link PhoneApi}, so it covers its controller
36   * without depending on the package it lives in. Request validation failures are handled by the parent class. A
37   * missing step-up is answered as the multi-factor handler does. The messages never say which numbers are registered.
38   */
39  @RestControllerAdvice(assignableTypes = PhoneApi.class)
40  @RequiredArgsConstructor
41  public class OttExceptionHandler extends ProblemJsonExceptionHandler {
42  
43      /** Base of the problem type URNs. */
44      private static final String TYPE_BASE = "urn:problem:phone:";
45  
46      /** Answers a missing step-up. */
47      private final MfaExceptionHandler mfaExceptionHandler;
48  
49      /**
50       * Answers an account without a verified phone.
51       *
52       * @param exception the failure.
53       * @return the {@code 404} problem detail.
54       */
55      @ExceptionHandler(PhoneNotFoundException.class)
56      public ResponseEntity<ProblemDetail> handleNotFound(final PhoneNotFoundException exception) {
57          return respond(HttpStatus.NOT_FOUND, "phone-not-found", "Phone not found", exception.getMessage());
58      }
59  
60      /**
61       * Answers a number that is not a valid phone number.
62       *
63       * @param exception the failure.
64       * @return the {@code 400} problem detail.
65       */
66      @ExceptionHandler(InvalidPhoneNumberException.class)
67      public ResponseEntity<ProblemDetail> handleInvalidPhone(final InvalidPhoneNumberException exception) {
68          return respond(HttpStatus.BAD_REQUEST, "invalid-phone", "Invalid phone number", exception.getMessage());
69      }
70  
71      /**
72       * Answers a number of a country SMS messages are not sent to.
73       *
74       * @param exception the failure.
75       * @return the {@code 400} problem detail.
76       */
77      @ExceptionHandler(SmsCountryNotAllowedException.class)
78      public ResponseEntity<ProblemDetail> handleCountryNotAllowed(final SmsCountryNotAllowedException exception) {
79          return respond(HttpStatus.BAD_REQUEST, "country-not-allowed", "Country not allowed", exception.getMessage());
80      }
81  
82      /**
83       * Answers a wrong, expired or missing verification code.
84       *
85       * @param exception the failure.
86       * @return the {@code 400} problem detail.
87       */
88      @ExceptionHandler(InvalidPhoneCodeException.class)
89      public ResponseEntity<ProblemDetail> handleInvalidCode(final InvalidPhoneCodeException exception) {
90          return respond(HttpStatus.BAD_REQUEST, "invalid-code", "Invalid code", exception.getMessage());
91      }
92  
93      /**
94       * Answers a client or an account that asked or tried too many times, telling when to retry.
95       *
96       * @param exception the failure.
97       * @return the {@code 429} problem detail, with a {@code Retry-After} header in seconds.
98       */
99      @ExceptionHandler(OttRateLimitExceededException.class)
100     public ResponseEntity<ProblemDetail> handleRateLimited(final OttRateLimitExceededException exception) {
101         final var problem = ProblemDetail.forStatusAndDetail(HttpStatus.TOO_MANY_REQUESTS, exception.getMessage());
102         problem.setType(URI.create(TYPE_BASE + "rate-limited"));
103         problem.setTitle("Too many requests");
104         final var headers = new HttpHeaders();
105         headers.set(HttpHeaders.RETRY_AFTER, Long.toString(Math.max(1, exception.getRetryAfter().toSeconds())));
106         return problemResponse(problem, headers);
107     }
108 
109     /**
110      * Answers a token whose sign-in is not recent or strong enough to remove the phone.
111      *
112      * @param exception the failure.
113      * @return the {@code 401} problem detail with the {@code WWW-Authenticate} challenge.
114      */
115     @ExceptionHandler(InsufficientUserAuthenticationException.class)
116     public ResponseEntity<ProblemDetail> handleInsufficientUserAuthentication(
117             final InsufficientUserAuthenticationException exception) {
118         return mfaExceptionHandler.handleInsufficientUserAuthentication(exception);
119     }
120 
121     /**
122      * Builds a problem response.
123      *
124      * @param status HTTP status.
125      * @param type   last segment of the problem type.
126      * @param title  short summary.
127      * @param detail explanation.
128      * @return the response.
129      */
130     private static ResponseEntity<ProblemDetail> respond(
131             final HttpStatus status, final String type, final String title, final String detail) {
132         final var problem = ProblemDetail.forStatusAndDetail(status, detail);
133         problem.setType(URI.create(TYPE_BASE + type));
134         problem.setTitle(title);
135         return problemResponse(problem);
136     }
137 }
