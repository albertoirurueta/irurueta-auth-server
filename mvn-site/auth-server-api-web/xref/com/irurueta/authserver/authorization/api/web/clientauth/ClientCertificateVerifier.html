42  // tag::clientauth-certificate-verifier[]
43  /**
44   * Verifies the certificate a client presented on the TLS port (RFC 8705), replacing the framework's verifier, which
45   * has no trust anchors of its own.
46   * <ul>
47   * <li>{@code tls_client_auth}: the chain must validate (PKIX path validation) against the configured trust anchors,
48   * and the subject DN of the leaf must equal the one registered for the client. Revocation checking is disabled: it
49   * needs a CRL or OCSP endpoint reachable from the server, which this example does not assume. No configured anchor
50   * means no client is accepted.</li>
51   * <li>{@code self_signed_tls_client_auth}: the certificate must be self-signed and its public key one of the keys of
52   * the client's registered JWK set; otherwise the decision is the framework's own (which compares it with the keys of
53   * the client's {@code jwkSetUrl}), whose provider is {@code defaultProvider}.</li>
54   * </ul>
55   * Every refusal is {@code invalid_client}.
56   */
57  @RequiredArgsConstructor
58  public class ClientCertificateVerifier implements Consumer<OAuth2ClientAuthenticationContext> {
59  
60      /** Type of the certification path. */
61      private static final String CERTIFICATE_TYPE = "X.509";
62  
63      /** Algorithm of the path validation. */
64      private static final String PKIX = "PKIX";
65  
66      /** The trusted certificate authorities. */
67      private final MtlsTrustAnchors trustAnchors;
68  
69      /** A default-configured provider of the framework, asked about a self-signed certificate with no registered key. */
70      private final AuthenticationProvider defaultProvider;
71  
72      /**
73       * Verifies the certificate of the client.
74       *
75       * @param context the client and its authentication.
76       * @throws OAuth2AuthenticationException {@code invalid_client} if the certificate is not accepted.
77       */
78      @Override
79      public void accept(final OAuth2ClientAuthenticationContext context) {
80          final OAuth2ClientAuthenticationToken authentication = context.getAuthentication();
81          final var method = authentication.getClientAuthenticationMethod();
82          final var chain = chainOf(authentication);
83          if (ClientAuthenticationMethod.TLS_CLIENT_AUTH.equals(method)) {
84              verifyCaIssued(context, chain);
85          } else if (ClientAuthenticationMethod.SELF_SIGNED_TLS_CLIENT_AUTH.equals(method)) {
86              verifySelfSigned(context, authentication, chain);
87          } else {
88              throw invalidClient();
89          }
90      }
91  
92      /**
93       * Verifies a certificate issued by a trusted authority.
94       *
95       * @param context the client and its authentication.
96       * @param chain   the presented chain, the leaf first.
97       */
98      private void verifyCaIssued(final OAuth2ClientAuthenticationContext context, final X509Certificate[] chain) {
99          if (trustAnchors.anchors().isEmpty()) {
100             throw invalidClient();
101         }
102         final var leaf = chain[0];
103         // the leaf is the certificate whose key the TLS handshake proved and whose DN is compared below: it must be
104         // the first certificate of the validated path, never an anchor left out of it
105         if (isAnchor(leaf)) {
106             throw invalidClient();
107         }
108         final var path = Arrays.stream(chain).filter(certificate -> !isAnchor(certificate)).toList();
109         try {
110             final var parameters = new PKIXParameters(trustAnchors.anchors());
111             parameters.setRevocationEnabled(false);
112             CertPathValidator.getInstance(PKIX).validate(
113                     CertificateFactory.getInstance(CERTIFICATE_TYPE).generateCertPath(path), parameters);
114         } catch (GeneralSecurityException e) {
115             throw invalidClient();
116         }
117         final var expected = context.getRegisteredClient().getClientSettings().getX509CertificateSubjectDN();
118         if (!StringUtils.hasText(expected) || !expected.equals(leaf.getSubjectX500Principal().getName())) {
119             throw invalidClient();
120         }
121     }
122 
123     /**
124      * Verifies a self-signed certificate.
125      *
126      * @param context        the client and its authentication.
127      * @param authentication the client authentication.
128      * @param chain          the presented chain, the leaf first.
129      */
130     private void verifySelfSigned(final OAuth2ClientAuthenticationContext context,
131                                   final OAuth2ClientAuthenticationToken authentication,
132                                   final X509Certificate[] chain) {
133         final var leaf = chain[0];
134         if (leaf.getIssuerX500Principal().equals(leaf.getSubjectX500Principal())
135                 && matchesRegisteredKey(context, leaf.getPublicKey())) {
136             return;
137         }
138         // the result is not needed: the provider throws invalid_client when the certificate does not match
139         defaultProvider.authenticate(authentication);
140     }
141 
142     /**
143      * Tells whether a key is one of the registered JWK set of the client.
144      *
145      * @param context the client and its authentication.
146      * @param key     the public key of the certificate.
147      * @return {@code true} when a registered key equals it.
148      */
149     private boolean matchesRegisteredKey(final OAuth2ClientAuthenticationContext context, final PublicKey key) {
150         final var jwks = context.getRegisteredClient().getClientSettings()
151                 .<Object>getSetting(RegisteredJwksClientAssertionDecoderFactory.JWKS_SETTING);
152         if (!(jwks instanceof String json) || !StringUtils.hasText(json)) {
153             return false;
154         }
155         try {
156             for (final JWK jwk : JWKSet.parse(json).getKeys()) {
157                 if (jwk instanceof AsymmetricJWK asymmetric && Arrays.equals(asymmetric.toPublicKey().getEncoded(),
158                         key.getEncoded())) {
159                     return true;
160                 }
161             }
162             return false;
163         } catch (ParseException | JOSEException e) {
164             return false;
165         }
166     }
167 
168     /**
169      * Tells whether a certificate is itself a trust anchor, which a certification path must not contain.
170      *
171      * @param certificate the certificate.
172      * @return {@code true} when an anchor holds exactly that certificate.
173      */
174     private boolean isAnchor(final X509Certificate certificate) {
175         return trustAnchors.anchors().stream().anyMatch(anchor -> certificate.equals(anchor.getTrustedCert()));
176     }
177 
178     /**
179      * Reads the presented chain.
180      *
181      * @param authentication the client authentication.
182      * @return the chain, the leaf first.
183      */
184     private static X509Certificate[] chainOf(final Authentication authentication) {
185         if (authentication.getCredentials() instanceof X509Certificate[] chain && chain.length > 0) {
186             return chain;
187         }
188         throw invalidClient();
189     }
190 
191     /**
192      * Builds the refusal.
193      *
194      * @return the {@code invalid_client} exception.
195      */
196     private static OAuth2AuthenticationException invalidClient() {
197         return new OAuth2AuthenticationException(new OAuth2Error(OAuth2ErrorCodes.INVALID_CLIENT));
198     }
199 }
