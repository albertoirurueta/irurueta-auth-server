52  /**
53   * Security filter chain of the REST API under {@code /api/**}: stateless, bearer tokens issued by this server and
54   * validated by a decoder of its own (never the framework-wide one, which also serves the UserInfo endpoint). That
55   * decoder only accepts a token that this server signed, whose issuer is this server, whose timestamps are valid,
56   * whose {@code aud} lists the API audience ({@code authserver.api.audience}, default {@code auth-server}) and that
57   * does not carry the {@code purpose} claim of the account tokens (an e-mail verification JWT is never a bearer
58   * token). Authentication and access failures are answered as described by RFC 6750
59   * ({@code WWW-Authenticate: Bearer ...}).
60   */
61  @Configuration(proxyBeanMethods = false)
62  public class ApiSecurityConfig {
63  
64      /** Path pattern of the REST API. */
65      public static final String API_PATTERN = "/api/**";
66  
67      /** Path pattern of the administration API. */
68      public static final String ADMIN_PATTERN = "/api/admin/**";
69  
70      /** Scope that grants access to the administration API; a role alone never does. */
71      public static final String ADMIN_SCOPE_AUTHORITY = "SCOPE_admin";
72  
73      /** Property that holds the audience a token must carry to call the API. */
74      public static final String API_AUDIENCE_PROPERTY = "authserver.api.audience";
75  
76      /** Prefix of the authorities derived from roles. */
77      private static final String ROLE_PREFIX = "ROLE_";
78  
79      /** Name of the claim that holds the roles. */
80      private static final String ROLES_CLAIM = "roles";
81  
82      // tag::api-security[]
83      /**
84       * Creates the chain of the REST API.
85       *
86       * @param http      security builder.
87       * @param jwkSource the published keys of this server.
88       * @param issuer    issuer of this server ({@code spring.security.oauth2.authorizationserver.issuer}).
89       * @param audience  audience the API requires ({@code authserver.api.audience}).
90       * @return the API filter chain.
91       * @throws Exception if the chain cannot be built.
92       */
93      @Bean
94      @Order(0)
95      public SecurityFilterChain apiSecurityFilterChain(
96              final HttpSecurity http, final JWKSource<SecurityContext> jwkSource,
97              @Value("${spring.security.oauth2.authorizationserver.issuer}") final String issuer,
98              @Value("${" + API_AUDIENCE_PROPERTY + ":" + DefaultClientDefinitions.DEFAULT_API_AUDIENCE + "}")
99              final String audience) throws Exception {
100         http
101                 .securityMatcher(API_PATTERN)
102                 .csrf(AbstractHttpConfigurer::disable)
103                 .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
104                 .authorizeHttpRequests(authorize -> authorize
105                         .requestMatchers(HttpMethod.POST, "/api/account/register", "/api/account/password/reset",
106                                 "/api/account/password/reset-request")
107                         .permitAll()
108                         .requestMatchers(HttpMethod.GET, "/api/account/verify").permitAll()
109                         .requestMatchers(ADMIN_PATTERN).hasAuthority(ADMIN_SCOPE_AUTHORITY)
110                         .anyRequest().authenticated())
111                 .oauth2ResourceServer(resource -> resource
112                         .jwt(jwt -> jwt.decoder(apiJwtDecoder(jwkSource, issuer, audience))
113                                 .jwtAuthenticationConverter(apiJwtAuthenticationConverter())));
114         return http.build();
115     }
116 
117     /**
118      * Creates the decoder of the API's bearer tokens: signature against the published keys, issuer, timestamps,
119      * audience, and no {@code purpose} claim.
120      *
121      * @param jwkSource the published keys of this server.
122      * @param issuer    issuer every token must have.
123      * @param audience  audience every token must list.
124      * @return the decoder.
125      */
126     static JwtDecoder apiJwtDecoder(final JWKSource<SecurityContext> jwkSource, final String issuer,
127                                     final String audience) {
128         final var decoder = NimbusJwtDecoder.withJwkSource(jwkSource).jwsAlgorithm(SignatureAlgorithm.RS256).build();
129         decoder.setJwtValidator(new DelegatingOAuth2TokenValidator<>(List.<OAuth2TokenValidator<Jwt>>of(
130                 JwtValidators.createDefaultWithIssuer(issuer),
131                 new JwtClaimValidator<List<String>>(JwtClaimNames.AUD,
132                         audiences -> audiences != null && audiences.contains(audience)),
133                 ApiSecurityConfig::rejectPurposeBoundToken)));
134         return decoder;
135     }
136 
137     /**
138      * Rejects a purpose-bound account token (such as an e-mail verification JWT): it is signed by this server but
139      * never meant as a bearer token.
140      *
141      * @param jwt the token.
142      * @return a failure when the token carries the {@code purpose} claim, a success otherwise.
143      */
144     private static OAuth2TokenValidatorResult rejectPurposeBoundToken(final Jwt jwt) {
145         if (jwt.hasClaim(SignedTokenService.PURPOSE_CLAIM)) {
146             return OAuth2TokenValidatorResult.failure(new OAuth2Error(OAuth2ErrorCodes.INVALID_TOKEN,
147                     "A purpose-bound account token is not a bearer token", null));
148         }
149         return OAuth2TokenValidatorResult.success();
150     }
151 
152     /**
153      * Creates the converter that maps the {@code scope} claim to {@code SCOPE_*} authorities and the {@code roles}
154      * claim to {@code ROLE_*} authorities.
155      *
156      * @return the converter.
157      */
158     JwtAuthenticationConverter apiJwtAuthenticationConverter() {
159         final var scopes = new JwtGrantedAuthoritiesConverter();
160         final var converter = new JwtAuthenticationConverter();
161         converter.setJwtGrantedAuthoritiesConverter(jwt -> {
162             final var authorities = new ArrayList<GrantedAuthority>(scopes.convert(jwt));
163             authorities.addAll(roleAuthorities(jwt));
164             return authorities;
165         });
166         return converter;
167     }
168     // end::api-security[]
169 
170     /**
171      * Maps the roles of a token to authorities, adding the {@code ROLE_} prefix when it is missing.
172      *
173      * @param jwt token to read.
174      * @return the role authorities, empty when the token has no roles.
175      */
176     private static Collection<GrantedAuthority> roleAuthorities(final Jwt jwt) {
177         final var roles = jwt.getClaimAsStringList(ROLES_CLAIM);
178         if (roles == null) {
179             return List.of();
180         }
181         return roles.stream()
182                 .map(role -> role.startsWith(ROLE_PREFIX) ? role : ROLE_PREFIX + role)
183                 .<GrantedAuthority>map(SimpleGrantedAuthority::new)
184                 .toList();
185     }
186 }
