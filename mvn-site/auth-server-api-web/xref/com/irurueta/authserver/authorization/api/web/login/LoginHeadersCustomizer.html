16  package com.irurueta.authserver.authorization.api.web.login;
17  
18  import lombok.RequiredArgsConstructor;
19  import org.springframework.core.annotation.Order;
20  import org.springframework.security.config.Customizer;
21  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
22  import org.springframework.security.web.header.writers.ReferrerPolicyHeaderWriter.ReferrerPolicy;
23  import org.springframework.stereotype.Component;
24  
25  // tag::login-headers[]
26  /**
27   * Adds the security headers of the server-rendered pages: the content security policy that
28   * {@link LoginContentSecurityPolicy} builds (same-origin resources only, no framing, and the origin of an absolute
29   * branding logo URL for images), {@code X-Content-Type-Options} and {@code Referrer-Policy}.
30   *
31   * <p>The policy deliberately has no {@code form-action} directive: browsers also apply it to the redirects that follow
32   * a form submission, so {@code form-action 'self'} would block the redirect from the login or consent form back to the
33   * client application's {@code redirect_uri}, which is on another origin.
34   */
35  @Component
36  @Order(0)
37  @RequiredArgsConstructor
38  public class LoginHeadersCustomizer implements LoginChainCustomizer {
39  
40      /** Builds the content security policy from the branding. */
41      private final LoginContentSecurityPolicy contentSecurityPolicy;
42  
43      @Override
44      public void customize(final HttpSecurity http) {
45          final var policy = contentSecurityPolicy.policy();
46          http.headers(headers -> {
47              headers.contentSecurityPolicy(csp -> csp.policyDirectives(policy));
48              headers.contentTypeOptions(Customizer.withDefaults());
49              headers.referrerPolicy(referrer -> referrer.policy(ReferrerPolicy.NO_REFERRER));
50          });
51      }
52  }
53  // end::login-headers[]
