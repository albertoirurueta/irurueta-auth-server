16  package com.irurueta.authserver.authorization.api.web.x509;
17  
18  import java.util.Locale;
19  import org.springframework.boot.cloud.CloudPlatform;
20  import org.springframework.core.env.Environment;
21  
22  // tag::x509-forwarded-headers-guard[]
23  /**
24   * Refuses to start when a certificate may be forwarded in a header while the server rewrites the address of its
25   * clients from the forwarded headers.
26   *
27   * <p>{@link ForwardedCertificateFilter} trusts a forwarded certificate only from the addresses of
28   * {@code authserver.x509.trusted-proxies}, and it reads the address with {@code getRemoteAddr()}, which is the address
29   * left once Spring Boot's forwarded-header handling has run. With {@code server.forward-headers-strategy=framework},
30   * Spring's {@code ForwardedHeaderFilter} replaces it with the first value of {@code X-Forwarded-For} (or
31   * {@code Forwarded}), which any client can send: naming a trusted proxy there would make a forged certificate header
32   * look trusted. With {@code native}, Tomcat's {@code RemoteIpValve} replaces it with the client the proxies report,
33   * so the address of the proxy itself is never seen and the check is made against an address that came from a header.
34   * The same happens without the property on a cloud platform that Spring Boot detects, where {@code native} is the
35   * default. Only with {@code none} is the address the direct network peer, which no header can change, so that is the
36   * one combination allowed. The proxy is still free to send {@code X-Forwarded-*} headers; they are just not used.
37   */
38  public final class X509ForwardedHeadersGuard {
39  
40      /** Property of Spring Boot that selects how the forwarded headers are handled. */
41      static final String STRATEGY_PROPERTY = "server.forward-headers-strategy";
42  
43      /** Value of the strategy that leaves the address of the client alone. */
44      private static final String NONE = "none";
45  
46      /** Value of the strategy of the servlet container. */
47      private static final String NATIVE = "native";
48  
49      /** Value of the strategy of Spring's filter. */
50      private static final String FRAMEWORK = "framework";
51  
52      /** Environment of the application, with the forwarded-header strategy. */
53      private final Environment environment;
54  
55      /**
56       * Checks the configuration.
57       *
58       * @param properties  settings of the sign-in with a certificate.
59       * @param environment environment of the application, with the forwarded-header strategy.
60       * @throws IllegalStateException if the forwarded certificate header is enabled and the strategy (configured or
61       *                               deduced from the cloud platform) rewrites the address of the client.
62       */
63      public X509ForwardedHeadersGuard(final X509Properties properties, final Environment environment) {
64          this.environment = environment;
65          if (properties.forwardedEnabled() && rewritesClientAddress()) {
66              throw new IllegalStateException("authserver.x509.forwarded-header needs " + STRATEGY_PROPERTY + "=" + NONE
67                      + ": with any other strategy the address checked against authserver.x509.trusted-proxies comes "
68                      + "from a forwarded header that a client can choose");
69          }
70      }
71  
72      /**
73       * Tells whether the server replaces the address of its clients with one taken from the forwarded headers.
74       *
75       * @return {@code true} for the {@code native} and {@code framework} strategies, and without a strategy when the
76       *         detected cloud platform uses the forwarded headers.
77       */
78      public boolean rewritesClientAddress() {
79          final var strategy = environment.getProperty(STRATEGY_PROPERTY, "").strip().toLowerCase(Locale.ROOT);
80          return switch (strategy) {
81              case NATIVE, FRAMEWORK -> true;
82              case NONE -> false;
83              default -> {
84                  final var platform = CloudPlatform.getActive(environment);
85                  yield platform != null && platform.isUsingForwardHeaders();
86              }
87          };
88      }
89  }
90  // end::x509-forwarded-headers-guard[]
