16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.api.web.mfa.CurrentAccountResolver;
19  import com.irurueta.authserver.authorization.api.web.mfa.StepUpGuard;
20  import com.irurueta.authserver.authorization.api.web.v1.generated.api.PhoneApi;
21  import com.irurueta.authserver.authorization.api.web.v1.generated.model.PhoneEnrolmentResponse;
22  import com.irurueta.authserver.authorization.api.web.v1.generated.model.PhoneResponse;
23  import com.irurueta.authserver.authorization.api.web.v1.generated.model.StartPhoneEnrolmentRequest;
24  import com.irurueta.authserver.authorization.api.web.v1.generated.model.VerifyPhoneRequest;
25  import com.irurueta.authserver.authorization.application.sms.ConfirmPhoneUseCase;
26  import com.irurueta.authserver.authorization.application.sms.FindUserPhoneUseCase;
27  import com.irurueta.authserver.authorization.application.sms.RemovePhoneUseCase;
28  import com.irurueta.authserver.authorization.application.sms.StartPhoneEnrolmentUseCase;
29  import jakarta.servlet.http.HttpServletRequest;
30  import lombok.RequiredArgsConstructor;
31  import org.springframework.http.CacheControl;
32  import org.springframework.http.HttpStatus;
33  import org.springframework.http.ResponseEntity;
34  import org.springframework.web.bind.annotation.RestController;
35  
36  /**
37   * Shows, enrols, verifies and removes the phone number of the authenticated account, the number the SMS codes go to.
38   * The paths, status codes and bodies are those of the contract {@code apis/auth-server/rest-server/
39   * auth-server-api-v1.yaml}, through the generated {@link PhoneApi}; failures are answered by
40   * {@link OttExceptionHandler}. Starting an enrolment and verifying it spend SMS budget and guess a code, so both are
41   * limited per client address and per account ({@link OttRateLimiter}). Starting an enrolment (which may replace the
42   * verified phone, a second factor) and removing the phone need a recent multi-factor sign-in while the account has a
43   * confirmed second factor ({@link StepUpGuard}).
44   */
45  @RestController
46  @RequiredArgsConstructor
47  public class PhoneController implements PhoneApi {
48  
49      /** Prefix of the rate-limit key of an account, which cannot clash with a user name typed in a sign-in form. */
50      static final String ACCOUNT_KEY_PREFIX = "phone:";
51  
52      /** Resolves the account of the request. */
53      private final CurrentAccountResolver currentAccountResolver;
54  
55      /** Finds the verified phone of the account. */
56      private final FindUserPhoneUseCase findUserPhoneUseCase;
57  
58      /** Starts an enrolment. */
59      private final StartPhoneEnrolmentUseCase startPhoneEnrolmentUseCase;
60  
61      /** Confirms an enrolment. */
62      private final ConfirmPhoneUseCase confirmPhoneUseCase;
63  
64      /** Removes the phone. */
65      private final RemovePhoneUseCase removePhoneUseCase;
66  
67      /** Requires a recent multi-factor sign-in before the phone is replaced or removed. */
68      private final StepUpGuard stepUpGuard;
69  
70      /** Limits the requests. */
71      private final OttRateLimiter rateLimiter;
72  
73      /** Maps the results onto the contract's DTOs. */
74      private final PhoneDtoMapper phoneDtoMapper;
75  
76      /** The current request, injected as a thread-bound proxy, which tells the address of the client. */
77      private final HttpServletRequest request;
78  
79      /**
80       * Gets the verified phone, masked.
81       *
82       * @return {@code 200 OK} with the phone.
83       * @throws PhoneNotFoundException if the account has no verified phone.
84       */
85      @Override
86      public ResponseEntity<PhoneResponse> getPhone() {
87          final var account = currentAccountResolver.resolve();
88          final var phone = findUserPhoneUseCase.execute(account.id()).orElseThrow(PhoneNotFoundException::new);
89          return ResponseEntity.ok().cacheControl(CacheControl.noStore()).body(phoneDtoMapper.toResponse(phone));
90      }
91  
92      /**
93       * Starts the enrolment of a phone: a verification code is sent by SMS.
94       *
95       * @param startPhoneEnrolmentRequest the number.
96       * @return {@code 202 Accepted} with how long the code stays valid.
97       * @throws com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException if the account
98       *         has a confirmed second factor and the token does not show a recent multi-factor sign-in.
99       * @throws OttRateLimitExceededException if the client or the account asked too many times.
100      */
101     @Override
102     public ResponseEntity<PhoneEnrolmentResponse> startPhoneEnrolment(
103             final StartPhoneEnrolmentRequest startPhoneEnrolmentRequest) {
104         final var account = currentAccountResolver.resolve();
105         stepUpGuard.requireRecentMultiFactor(account.id());
106         rateLimiter.checkGeneration(request.getRemoteAddr(), ACCOUNT_KEY_PREFIX + account.id());
107         final var started = startPhoneEnrolmentUseCase.execute(account.id(), startPhoneEnrolmentRequest.phone());
108         return ResponseEntity.status(HttpStatus.ACCEPTED).body(phoneDtoMapper.toResponse(started));
109     }
110 
111     /**
112      * Confirms the phone with the code received by SMS.
113      *
114      * @param verifyPhoneRequest the code.
115      * @return {@code 200 OK} with the verified phone.
116      * @throws InvalidPhoneCodeException     if the code is not valid.
117      * @throws OttRateLimitExceededException if the client or the account tried too many times.
118      */
119     @Override
120     public ResponseEntity<PhoneResponse> verifyPhone(final VerifyPhoneRequest verifyPhoneRequest) {
121         final var account = currentAccountResolver.resolve();
122         rateLimiter.checkVerification(request.getRemoteAddr(), ACCOUNT_KEY_PREFIX + account.id());
123         if (!confirmPhoneUseCase.execute(account.id(), account.username(), verifyPhoneRequest.code())) {
124             throw new InvalidPhoneCodeException();
125         }
126         final var phone = findUserPhoneUseCase.execute(account.id()).orElseThrow(PhoneNotFoundException::new);
127         return ResponseEntity.ok().cacheControl(CacheControl.noStore()).body(phoneDtoMapper.toResponse(phone));
128     }
129 
130     /**
131      * Removes the phone and any pending verification.
132      *
133      * @return {@code 204 No Content}.
134      * @throws com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException if the token
135      *         does not show a recent multi-factor sign-in.
136      */
137     @Override
138     public ResponseEntity<Void> removePhone() {
139         final var account = currentAccountResolver.resolve();
140         stepUpGuard.requireRecentMultiFactor(account.id());
141         removePhoneUseCase.execute(account.id(), account.username());
142         return ResponseEntity.noContent().build();
143     }
144 }
