19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertFalse;
21  import static org.junit.jupiter.api.Assertions.assertNull;
22  import static org.junit.jupiter.api.Assertions.assertSame;
23  import static org.junit.jupiter.api.Assertions.assertTrue;
24  import static org.mockito.ArgumentMatchers.anyMap;
25  import static org.mockito.ArgumentMatchers.eq;
26  import static org.mockito.Mockito.verify;
27  import static org.mockito.Mockito.verifyNoInteractions;
28  import static org.mockito.Mockito.when;
29  
30  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
31  import java.util.List;
32  import java.util.Map;
33  import org.junit.jupiter.api.BeforeEach;
34  import org.junit.jupiter.api.Test;
35  import org.junit.jupiter.api.extension.ExtendWith;
36  import org.mockito.ArgumentCaptor;
37  import org.mockito.Mock;
38  import org.mockito.junit.jupiter.MockitoExtension;
39  import org.springframework.http.ResponseEntity;
40  import org.springframework.mock.web.MockHttpServletRequest;
41  import org.springframework.security.authentication.BadCredentialsException;
42  import org.springframework.security.authentication.LockedException;
43  import org.springframework.security.web.WebAttributes;
44  
45  /**
46   * Unit tests of {@link LoginController}.
47   */
48  @ExtendWith(MockitoExtension.class)
49  class LoginControllerTest {
50  
51      /** Order of the fragment that is rendered first. */
52      private static final int FIRST_ORDER = 10;
53  
54      /** Order of the fragment that is rendered second. */
55      private static final int SECOND_ORDER = 20;
56  
57      /** Maps authentication exceptions to message keys. */
58      @Mock
59      private LoginErrorMessages errorMessages;
60  
61      /** Renders the page. */
62      @Mock
63      private PageRenderer pageRenderer;
64  
65      /** Fragment that is rendered first. */
66      @Mock
67      private LoginMethodFragment first;
68  
69      /** Fragment that is rendered second. */
70      @Mock
71      private LoginMethodFragment second;
72  
73      /** The current request. */
74      private MockHttpServletRequest request;
75  
76      /**
77       * Creates the request of each test.
78       */
79      @BeforeEach
80      void setUp() {
81          request = new MockHttpServletRequest();
82      }
83  
84      /**
85       * Verifies that the model holds an empty list when no authentication method contributes a fragment, and that
86       * the rendered page is answered.
87       */
88      @Test
89      void loginPage_whenThereAreNoFragments_putsAnEmptyList() {
90          final var page = ResponseEntity.ok("<html></html>");
91          when(pageRenderer.render(eq(LoginController.VIEW_NAME), anyMap())).thenReturn(page);
92  
93          final var response = controller(List.of()).loginPage(null, null);
94  
95          final var model = renderedModel();
96          assertSame(page, response);
97          assertEquals(LoginController.LOGIN_PATH, model.get(LoginController.LOGIN_PATH_ATTRIBUTE));
98          assertTrue(((List<?>) model.get(LoginController.FRAGMENTS_ATTRIBUTE)).isEmpty());
99      }
100 
101     /**
102      * Verifies that the fragments are sorted by their order.
103      */
104     @Test
105     void loginPage_whenThereAreTwoFragments_putsThemSortedByOrder() {
106         when(first.order()).thenReturn(FIRST_ORDER);
107         when(second.order()).thenReturn(SECOND_ORDER);
108 
109         controller(List.of(second, first)).loginPage(null, null);
110 
111         assertEquals(List.of(first, second), renderedModel().get(LoginController.FRAGMENTS_ATTRIBUTE));
112     }
113 
114     /**
115      * Verifies that a failed attempt puts the message key of the exception kept in the session in the model.
116      */
117     @Test
118     void loginPage_whenPreviousAttemptFailed_putsTheMessageKeyOfTheException() {
119         final var exception = new LockedException("locked");
120         request.getSession().setAttribute(WebAttributes.AUTHENTICATION_EXCEPTION, exception);
121         when(errorMessages.keyOf(exception)).thenReturn(LoginErrorMessages.LOCKED);
122 
123         controller(List.of()).loginPage("", null);
124 
125         final var model = renderedModel();
126         assertEquals(LoginErrorMessages.LOCKED, model.get(LoginController.ERROR_MESSAGE_ATTRIBUTE));
127         assertFalse((Boolean) model.get(LoginController.LOGGED_OUT_ATTRIBUTE));
128     }
129 
130     /**
131      * Verifies that a failed attempt without a session still shows an error.
132      */
133     @Test
134     void loginPage_whenErrorHasNoSession_putsTheKeyOfNoException() {
135         when(errorMessages.keyOf(null)).thenReturn(LoginErrorMessages.GENERIC);
136 
137         controller(List.of()).loginPage("", null);
138 
139         assertEquals(LoginErrorMessages.GENERIC, renderedModel().get(LoginController.ERROR_MESSAGE_ATTRIBUTE));
140     }
141 
142     /**
143      * Verifies that a session attribute that is not an authentication exception is ignored.
144      */
145     @Test
146     void loginPage_whenSessionHoldsSomethingElse_putsTheKeyOfNoException() {
147         request.getSession().setAttribute(WebAttributes.AUTHENTICATION_EXCEPTION, "not an exception");
148         when(errorMessages.keyOf(null)).thenReturn(LoginErrorMessages.GENERIC);
149 
150         controller(List.of()).loginPage("", null);
151 
152         assertEquals(LoginErrorMessages.GENERIC, renderedModel().get(LoginController.ERROR_MESSAGE_ATTRIBUTE));
153     }
154 
155     /**
156      * Verifies that a stale exception in the session is ignored when the page is not an error page.
157      */
158     @Test
159     void loginPage_whenThereIsNoErrorParameter_ignoresTheStoredException() {
160         request.getSession().setAttribute(WebAttributes.AUTHENTICATION_EXCEPTION, new BadCredentialsException("x"));
161 
162         controller(List.of()).loginPage(null, null);
163 
164         assertNull(renderedModel().get(LoginController.ERROR_MESSAGE_ATTRIBUTE));
165         verifyNoInteractions(errorMessages);
166     }
167 
168     /**
169      * Verifies that the logout parameter is reflected in the model.
170      */
171     @Test
172     void loginPage_whenUserHasSignedOut_flagsIt() {
173         controller(List.of()).loginPage(null, "");
174 
175         assertTrue((Boolean) renderedModel().get(LoginController.LOGGED_OUT_ATTRIBUTE));
176     }
177 
178     /**
179      * Creates the controller under test.
180      *
181      * @param fragments the fragments of the login methods.
182      * @return the controller.
183      */
184     private LoginController controller(final List<LoginMethodFragment> fragments) {
185         return new LoginController(fragments, errorMessages, pageRenderer, request);
186     }
187 
188     /**
189      * Gets the model the login template was rendered with.
190      *
191      * @return the model.
192      */
193     private Map<String, ?> renderedModel() {
194         final var captor = ArgumentCaptor.<Map<String, ?>>captor();
195         verify(pageRenderer).render(eq(LoginController.VIEW_NAME), captor.capture());
196         return captor.getValue();
197     }
198 }
