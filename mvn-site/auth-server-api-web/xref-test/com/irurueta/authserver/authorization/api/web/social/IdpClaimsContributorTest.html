52  /**
53   * Unit tests of {@link IdpClaimsContributor}, alone and after the real {@link MfaClaimsContributor} over a real
54   * {@link FactorTimeline}, which is how the tokens are built.
55   */
56  class IdpClaimsContributorTest {
57  
58      /** Authority of the TOTP factor. */
59      private static final String TOTP_AUTHORITY = "FACTOR_TOTP";
60  
61      /** The instant the federated sign-in happened. */
62      private static final Instant SIGN_IN = Instant.parse("2026-01-01T10:00:00Z");
63  
64      /** The instant the second factor was presented. */
65      private static final Instant SECOND = Instant.parse("2026-01-01T10:01:00Z");
66  
67      /** Contributor under test. */
68      private final IdpClaimsContributor contributor = new IdpClaimsContributor(new FederatedUserMapper());
69  
70      /** Contributor of the multi-factor claims, which runs before. */
71      private MfaClaimsContributor mfaContributor;
72  
73      /**
74       * Creates the multi-factor contributor over a registry with the federated, password and TOTP factors.
75       */
76      @BeforeEach
77      void setUp() {
78          final var declarations = new FederatedFactor();
79          final var registry = new AuthenticationFactorRegistry(List.of(
80                  new PasswordFactor().passwordAuthenticationFactor(),
81                  new AuthenticationFactor("totp", "otp", TOTP_AUTHORITY, FactorStrength.SECOND),
82                  declarations.federatedAuthenticationFactor(), declarations.federatedStrongAuthenticationFactor()));
83          mfaContributor = new MfaClaimsContributor(new FactorTimeline(registry, new MfaPolicy()));
84      }
85  
86      /**
87       * Verifies that an ordinary federated sign-in gets the provider and is a single factor: the upstream methods are
88       * not copied, however they read.
89       */
90      @Test
91      void contribute_whenSignInIsOrdinary_addsTheProviderAndNoUpstreamMethods() {
92          final var claims = new HashMap<String, Object>();
93          final var context = context(token(List.of("pwd", "mfa"), factor(FederatedFactor.AUTHORITY, SIGN_IN)));
94  
95          mfaContributor.contribute(context, claims);
96          contributor.contribute(context, claims);
97  
98          assertEquals("google", claims.get(IdpClaimsContributor.IDP_CLAIM));
99          assertEquals(List.of("fed"), claims.get(IdTokenClaimNames.AMR));
100         assertEquals(AuthenticationContextClass.PASSWORD, claims.get(IdTokenClaimNames.ACR));
101     }
102 
103     /**
104      * Verifies that an ordinary federated sign-in followed by a second factor reads {@code fed}, {@code otp} and
105      * {@code mfa}, each once.
106      */
107     @Test
108     void contribute_whenOrdinarySignInIsFollowedBySecondFactor_listsEachValueOnce() {
109         final var claims = new HashMap<String, Object>();
110         final var context = context(token(List.of("pwd"), factor(FederatedFactor.AUTHORITY, SIGN_IN),
111                 factor(TOTP_AUTHORITY, SECOND)));
112 
113         mfaContributor.contribute(context, claims);
114         contributor.contribute(context, claims);
115 
116         assertEquals(List.of("fed", "otp", "mfa"), claims.get(IdTokenClaimNames.AMR));
117         assertEquals(AuthenticationContextClass.MULTI_FACTOR, claims.get(IdTokenClaimNames.ACR));
118     }
119 
120     /**
121      * Verifies that a strong sign-in, multi-factor alone, gets {@code mfa} and the multi-factor class from the
122      * multi-factor contributor, and that the upstream methods are then added once each (the {@code fed} and
123      * {@code mfa} the provider repeated, too), in a mutable list.
124      */
125     @Test
126     void contribute_whenSignInIsStrong_mergesTheUpstreamMethodsOnceAndReportsMultiFactor() {
127         final var claims = new HashMap<String, Object>();
128         final var context = context(token(List.of("pwd", "mfa", "pwd", "fed"),
129                 factor(FederatedFactor.STRONG_AUTHORITY, SIGN_IN)));
130 
131         mfaContributor.contribute(context, claims);
132         contributor.contribute(context, claims);
133 
134         assertEquals(List.of("fed", "mfa", "pwd"), claims.get(IdTokenClaimNames.AMR));
135         assertInstanceOf(ArrayList.class, claims.get(IdTokenClaimNames.AMR));
136         assertEquals(AuthenticationContextClass.MULTI_FACTOR, claims.get(IdTokenClaimNames.ACR));
137         assertEquals("google", claims.get(IdpClaimsContributor.IDP_CLAIM));
138     }
139 
140     /**
141      * Verifies that a strong sign-in whose claims have no {@code amr} yet still gets one, and that this contributor
142      * never sets {@code acr} itself: the multi-factor contributor owns it.
143      */
144     @Test
145     void contribute_whenSignInIsStrongAndClaimsHaveNoAmr_createsItWithoutAcr() {
146         final var claims = new HashMap<String, Object>();
147         final var context = context(token(List.of("mfa"), factor(FederatedFactor.STRONG_AUTHORITY, SIGN_IN)));
148 
149         contributor.contribute(context, claims);
150 
151         assertEquals(List.of("mfa"), claims.get(IdTokenClaimNames.AMR));
152         assertFalse(claims.containsKey(IdTokenClaimNames.ACR));
153     }
154 
155     /**
156      * Verifies that a token of another kind of sign-in gets nothing.
157      */
158     @Test
159     void contribute_whenPrincipalIsNotAFederatedSignIn_addsNothing() {
160         final var claims = new HashMap<String, Object>();
161 
162         contributor.contribute(context(new TestingAuthenticationToken("bob", "secret", "ROLE_USER")), claims);
163         contributor.contribute(mock(OAuth2TokenContext.class), claims);
164 
165         assertTrue(claims.isEmpty());
166         assertFalse(claims.containsKey(IdpClaimsContributor.IDP_CLAIM));
167     }
168 
169     /**
170      * Verifies the position of the contributor: after the multi-factor one it extends.
171      */
172     @Test
173     void order_whenRead_isAfterTheMultiFactorContributor() {
174         assertEquals(IdpClaimsContributor.ORDER, IdpClaimsContributor.class.getAnnotation(Order.class).value());
175         assertTrue(IdpClaimsContributor.ORDER > MfaClaimsContributor.ORDER);
176     }
177 
178     /**
179      * Builds the authentication of a federated sign-in.
180      *
181      * @param amr       upstream methods.
182      * @param authority factor authorities of the session.
183      * @return the authentication.
184      */
185     private static OAuth2AuthenticationToken token(final List<String> amr, final GrantedAuthority... authority) {
186         final var user = new DefaultOAuth2User(List.of(new SimpleGrantedAuthority("ROLE_USER")),
187                 Map.of("amr", new ArrayList<>(amr), "local_username", "alice"), "local_username");
188         final var authorities = new ArrayList<GrantedAuthority>(user.getAuthorities());
189         authorities.addAll(List.of(authority));
190         return new OAuth2AuthenticationToken(user, authorities, "google");
191     }
192 
193     /**
194      * Builds a factor authority.
195      *
196      * @param authority authority of the factor.
197      * @param issuedAt  instant it was granted.
198      * @return the authority.
199      */
200     private static FactorGrantedAuthority factor(final String authority, final Instant issuedAt) {
201         return FactorGrantedAuthority.withAuthority(authority).issuedAt(issuedAt).build();
202     }
203 
204     /**
205      * Builds a token context of a principal.
206      *
207      * @param principal the principal.
208      * @return the context.
209      */
210     private static OAuth2TokenContext context(final Authentication principal) {
211         final var context = mock(OAuth2TokenContext.class);
212         when(context.getPrincipal()).thenReturn(principal);
213         return context;
214     }
