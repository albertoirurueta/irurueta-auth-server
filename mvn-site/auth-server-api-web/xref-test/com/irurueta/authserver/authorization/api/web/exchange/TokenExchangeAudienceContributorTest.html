45  /**
46   * Unit tests of {@link TokenExchangeAudienceContributor}.
47   */
48  class TokenExchangeAudienceContributorTest {
49  
50      /** Audience of the resource server. */
51      private static final String RESOURCE_SERVER = "resource-server";
52  
53      /** Audience of the API of the server. */
54      private static final String API = "auth-server";
55  
56      /** Audience that the client may not have. */
57      private static final String FOREIGN = "somebody-else";
58  
59      /** Token type URN of an access token. */
60      private static final String ACCESS_TOKEN_TYPE = "urn:ietf:params:oauth:token-type:access_token";
61  
62      /** Expected order of the contributor. */
63      private static final int EXPECTED_ORDER = 120;
64  
65      /** Subject under test. */
66      private final TokenExchangeAudienceContributor contributor = new TokenExchangeAudienceContributor();
67  
68      /**
69       * The order places the contributor after the multi-factor and identity provider ones.
70       */
71      @Test
72      void order_always_is120() {
73          assertEquals(EXPECTED_ORDER, TokenExchangeAudienceContributor.ORDER);
74          assertEquals(TokenExchangeAudienceContributor.ORDER,
75                  TokenExchangeAudienceContributor.class.getAnnotation(Order.class).value());
76      }
77  
78      /**
79       * The exchange grant constant is the RFC 8693 URN.
80       */
81      @Test
82      void tokenExchangeGrant_always_isTheGrantTypeUrn() {
83          assertEquals(AuthorizationGrantType.TOKEN_EXCHANGE.getValue(), DefaultClientDefinitions.TOKEN_EXCHANGE_GRANT);
84      }
85  
86      /**
87       * Only the requested audiences the client may have are kept.
88       */
89      @Test
90      void contribute_whenRequestedAudiencesAreAllowed_keepsOnlyThoseRequested() {
91          final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
92  
93          contributor.contribute(contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.ACCESS_TOKEN,
94                  Set.of(RESOURCE_SERVER, FOREIGN)), claims);
95  
96          assertEquals(List.of(RESOURCE_SERVER), claims.get(OAuth2TokenClaimNames.AUD));
97      }
98  
99      /**
100      * A request whose audiences the client may not have is refused with invalid_target and the claims stay untouched.
101      */
102     @Test
103     void contribute_whenNoRequestedAudienceIsAllowed_throwsInvalidTarget() {
104         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
105         final var context = contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.ACCESS_TOKEN,
106                 Set.of(FOREIGN));
107 
108         final var error = assertThrows(OAuth2AuthenticationException.class,
109                 () -> contributor.contribute(context, claims));
110 
111         assertEquals(TokenExchangeAudienceContributor.INVALID_TARGET, error.getError().getErrorCode());
112         assertEquals(List.of(RESOURCE_SERVER, API), claims.get(OAuth2TokenClaimNames.AUD));
113     }
114 
115     /**
116      * When only some requested audiences are allowed, the allowed ones are granted and the others dropped.
117      */
118     @Test
119     void contribute_whenSomeRequestedAudiencesAreAllowed_grantsOnlyThose() {
120         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
121 
122         contributor.contribute(contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.ACCESS_TOKEN,
123                 new LinkedHashSet<>(List.of(API, FOREIGN))), claims);
124 
125         assertEquals(List.of(API), claims.get(OAuth2TokenClaimNames.AUD));
126     }
127 
128     /**
129      * A request with no audience leaves the audiences of the client.
130      */
131     @Test
132     void contribute_whenNoAudienceIsRequested_keepsTheAudiencesOfTheClient() {
133         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
134 
135         contributor.contribute(contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.ACCESS_TOKEN,
136                 Set.of()), claims);
137 
138         assertEquals(List.of(RESOURCE_SERVER, API), claims.get(OAuth2TokenClaimNames.AUD));
139     }
140 
141     /**
142      * Another grant type is not touched.
143      */
144     @Test
145     void contribute_whenGrantIsNotATokenExchange_doesNothing() {
146         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
147 
148         contributor.contribute(contextOf(AuthorizationGrantType.CLIENT_CREDENTIALS, OAuth2TokenType.ACCESS_TOKEN,
149                 Set.of(RESOURCE_SERVER)), claims);
150 
151         assertEquals(List.of(RESOURCE_SERVER, API), claims.get(OAuth2TokenClaimNames.AUD));
152     }
153 
154     /**
155      * A token that is not an access token is not touched.
156      */
157     @Test
158     void contribute_whenTokenIsNotAnAccessToken_doesNothing() {
159         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
160 
161         contributor.contribute(contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.REFRESH_TOKEN,
162                 Set.of(RESOURCE_SERVER)), claims);
163 
164         assertEquals(List.of(RESOURCE_SERVER, API), claims.get(OAuth2TokenClaimNames.AUD));
165     }
166 
167     /**
168      * A client without the audiences setting may have no requested audience: the request is refused.
169      */
170     @Test
171     void contribute_whenClientHasNoAudiencesSetting_throwsInvalidTarget() {
172         final var claims = new HashMap<String, Object>();
173         final var client = clientBuilder().build();
174         final var context = OAuth2TokenClaimsContext.with(OAuth2TokenClaimsSet.builder().subject("user"))
175                 .registeredClient(client).principal(new TestingAuthenticationToken("user", "n"))
176                 .authorizationGrantType(AuthorizationGrantType.TOKEN_EXCHANGE)
177                 .authorizationGrant(exchange(client, Set.of(RESOURCE_SERVER)))
178                 .tokenType(OAuth2TokenType.ACCESS_TOKEN).build();
179 
180         final var error = assertThrows(OAuth2AuthenticationException.class,
181                 () -> contributor.contribute(context, claims));
182 
183         assertEquals(TokenExchangeAudienceContributor.INVALID_TARGET, error.getError().getErrorCode());
184         assertFalse(claims.containsKey(OAuth2TokenClaimNames.AUD));
185     }
186 
187     /**
188      * The claim is a mutable list, as the authorization store only deserializes allow-listed types.
189      */
190     @Test
191     void contribute_whenAudiencesAreGranted_putsAMutableList() {
192         final var claims = claimsWithAudiences(RESOURCE_SERVER, API);
193 
194         contributor.contribute(contextOf(AuthorizationGrantType.TOKEN_EXCHANGE, OAuth2TokenType.ACCESS_TOKEN,
195                 Set.of(RESOURCE_SERVER)), claims);
196 
197         final var audiences = claims.get(OAuth2TokenClaimNames.AUD);
198         assertNotNull(audiences);
199         assertEquals(ArrayList.class, audiences.getClass());
200     }
201 
202     /**
203      * Builds the claims of an access token that already carry the audiences of the client.
204      *
205      * @param audiences audiences of the client.
206      * @return the mutable claims.
207      */
208     private static HashMap<String, Object> claimsWithAudiences(final String... audiences) {
209         final var claims = new HashMap<String, Object>();
210         claims.put(OAuth2TokenClaimNames.AUD, new ArrayList<>(List.of(audiences)));
211         return claims;
212     }
213 
214     /**
215      * Builds a client that may exchange tokens and may have the resource server and API audiences.
216      *
217      * @return the builder.
218      */
219     private static RegisteredClient.Builder clientBuilder() {
220         return RegisteredClient.withId(UUID.randomUUID().toString()).clientId("client")
221                 .authorizationGrantType(AuthorizationGrantType.TOKEN_EXCHANGE)
222                 .redirectUri("https://client.example.com/callback");
223     }
224 
225     /**
226      * Builds a token exchange request that asks for the given audiences.
227      *
228      * @param client    the exchanging client.
229      * @param audiences requested audiences.
230      * @return the request.
231      */
232     private static OAuth2TokenExchangeAuthenticationToken exchange(final RegisteredClient client,
233                                                                    final Set<String> audiences) {
234         return new OAuth2TokenExchangeAuthenticationToken(ACCESS_TOKEN_TYPE, "subject", ACCESS_TOKEN_TYPE,
235                 new TestingAuthenticationToken(client.getClientId(), "n"), null, null, null,
236                 new LinkedHashSet<>(audiences), null, null);
237     }
238 
239     /**
240      * Builds the context of a token of an exchanging client that may have the resource server and API audiences.
241      *
242      * @param grantType grant type of the context.
243      * @param tokenType type of the token.
244      * @param requested audiences requested in the exchange.
245      * @return the context.
246      */
247     private static OAuth2TokenClaimsContext contextOf(final AuthorizationGrantType grantType,
248                                                       final OAuth2TokenType tokenType, final Set<String> requested) {
249         final var client = clientBuilder().clientSettings(ClientSettings.builder()
250                 .setting(IssuedTokenClaims.AUDIENCES_SETTING, List.of(RESOURCE_SERVER, API)).build()).build();
251         return OAuth2TokenClaimsContext.with(OAuth2TokenClaimsSet.builder().subject("user"))
252                 .registeredClient(client).principal(new TestingAuthenticationToken("user", "n"))
253                 .authorizationGrantType(grantType).authorizationGrant(exchange(client, requested))
254                 .tokenType(tokenType).build();
255     }
