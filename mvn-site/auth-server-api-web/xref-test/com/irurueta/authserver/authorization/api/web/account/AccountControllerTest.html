117     /**
118      * Checks that registration is accepted and delegated.
119      *
120      * @throws Exception if the request fails.
121      */
122     @Test
123     void register_whenRequestIsValid_answersAcceptedAndDelegates() throws Exception {
124         mockMvc.perform(post("/api/account/register").contentType(MediaType.APPLICATION_JSON)
125                         .content(json("username", USERNAME, "email", EMAIL, "password", secret)))
126                 .andExpect(status().isAccepted())
127                 .andExpect(content().string(""));
128 
129         verify(accountRateLimiter).check(CLIENT_ADDRESS, EMAIL);
130         verify(registerAccountUseCase).execute(USERNAME, EMAIL, secret);
131     }

133     /**
134      * Checks that a registration over the limit is answered {@code 429} with a {@code Retry-After} header and never
135      * reaches the use case.
136      *
137      * @throws Exception if the request fails.
138      */
139     @Test
140     void register_whenRateLimitIsExceeded_answersTooManyRequests() throws Exception {
141         doThrow(new AccountRateLimitExceededException(Duration.ofSeconds(RETRY_SECONDS)))
142                 .when(accountRateLimiter).check(CLIENT_ADDRESS, EMAIL);
143 
144         mockMvc.perform(post("/api/account/register").contentType(MediaType.APPLICATION_JSON)
145                         .content(json("username", USERNAME, "email", EMAIL, "password", secret)))
146                 .andExpect(status().isTooManyRequests())
147                 .andExpect(header().string(HttpHeaders.RETRY_AFTER, Long.toString(RETRY_SECONDS)))
148                 .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_PROBLEM_JSON));
149 
150         verifyNoInteractions(registerAccountUseCase);
151     }

365     /**
366      * Builds a JSON object from key and value pairs.
367      *
368      * @param pairs alternating keys and values.
369      * @return the JSON text.
370      */
371     private static String json(final String... pairs) {
372         final var builder = new StringBuilder("{");
373         for (var i = 0; i < pairs.length; i += 2) {
374             if (i > 0) {
375                 builder.append(',');
376             }
377             builder.append('"').append(pairs[i]).append("\":\"").append(pairs[i + 1]).append('"');
378         }
379         return builder.append('}').toString();
380     }
