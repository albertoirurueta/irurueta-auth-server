21  import static org.junit.jupiter.api.Assertions.assertTrue;

25  import com.irurueta.authserver.authorization.application.ott.OttVerificationResult;

38  import org.junit.jupiter.api.extension.ExtendWith;

40  import org.mockito.junit.jupiter.MockitoExtension;
41  import org.springframework.security.authentication.BadCredentialsException;

43  import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
44  import org.springframework.security.core.Authentication;
45  import org.springframework.security.core.GrantedAuthority;
46  import org.springframework.security.core.authority.FactorGrantedAuthority;

48  /** Unit tests of {@link OttSecondFactorAuthenticationProvider}. */
49  @ExtendWith(MockitoExtension.class)
50  class OttSecondFactorAuthenticationProviderTest {
51  
52      /** The code. */
53      private static final String CODE = "123456";
54  
55      /** Every channel. */
56      private static final Set<OttChannel> ALL = EnumSet.allOf(OttChannel.class);
57  
58      /** Verifies the code. */
59      @Mock
60      private VerifyOneTimeCodeUseCase verifyOneTimeCodeUseCase;
61  
62      /** Decides which channels may be the second factor. */
63      @Mock
64      private OttSecondFactorChannels secondFactorChannels;
65  
66      /** Provider under test. */
67      private OttSecondFactorAuthenticationProvider provider;
68  
69      /** Creates the provider over a real grantor. */
70      @BeforeEach
71      void setUp() {
72          provider = new OttSecondFactorAuthenticationProvider(verifyOneTimeCodeUseCase,
73                  new FactorGrantor(Clock.fixed(Instant.EPOCH, ZoneOffset.UTC)), secondFactorChannels);
74      }
75  
76      /** A correct e-mail code adds the e-mail second factor to what the user holds. */
77      @Test
78      void authenticate_whenAnEmailCodeIsCorrect_addsTheEmailSecondFactor() {
79          final var primary = primary("FACTOR_PASSWORD");
80          when(secondFactorChannels.allowed(primary)).thenReturn(ALL);
81          when(verifyOneTimeCodeUseCase.execute("binding", "alice", CODE, OttPurpose.SECOND_FACTOR, ALL))
82                  .thenReturn(OttVerificationResult.success(OttChannel.EMAIL));
83  
84          final var result = provider.authenticate(new OttSecondFactorToken(primary, CODE, "binding"));
85  
86          assertEquals(Set.of("FACTOR_PASSWORD", "FACTOR_OTT_EMAIL_2FA"), authorities(result));
87      }
88  
89      /** A correct SMS code adds the SMS second factor. */
90      @Test
91      void authenticate_whenAnSmsCodeIsCorrect_addsTheSmsSecondFactor() {
92          final var primary = primary("FACTOR_PASSWORD");
93          when(secondFactorChannels.allowed(primary)).thenReturn(ALL);
94          when(verifyOneTimeCodeUseCase.execute("binding", "alice", CODE, OttPurpose.SECOND_FACTOR, ALL))
95                  .thenReturn(OttVerificationResult.success(OttChannel.SMS));
96  
97          final var result = provider.authenticate(new OttSecondFactorToken(primary, CODE, "binding"));
98  
99          assertEquals(Set.of("FACTOR_PASSWORD", "FACTOR_OTT_SMS_2FA"), authorities(result));
100     }
101 
102     /** A wrong code is refused. */
103     @Test
104     void authenticate_whenTheCodeIsWrong_throwsBadCredentialsException() {
105         final var primary = primary("FACTOR_PASSWORD");
106         when(secondFactorChannels.allowed(primary)).thenReturn(ALL);
107         when(verifyOneTimeCodeUseCase.execute("binding", "alice", CODE, OttPurpose.SECOND_FACTOR, ALL))
108                 .thenReturn(OttVerificationResult.failure());
109         final var request = new OttSecondFactorToken(primary, CODE, "binding");
110 
111         assertThrows(BadCredentialsException.class, () -> provider.authenticate(request));
112     }
113 
114     /** A locked second factor is refused as locked, even for a correct code. */
115     @Test
116     void authenticate_whenTheSecondFactorIsLocked_throwsLockedException() {
117         final var primary = primary("FACTOR_PASSWORD");
118         when(secondFactorChannels.allowed(primary)).thenReturn(ALL);
119         when(verifyOneTimeCodeUseCase.execute("binding", "alice", CODE, OttPurpose.SECOND_FACTOR, ALL))
120                 .thenReturn(OttVerificationResult.locked());
121         final var request = new OttSecondFactorToken(primary, CODE, "binding");
122 
123         assertThrows(LockedException.class, () -> provider.authenticate(request));
124     }
125 
126     /**
127      * Only the allowed channels are handed to the verification, before the code is compared: a code of another channel
128      * (here the e-mail that signed the user in) is a failure counted by the use case, and refused.
129      */
130     @Test
131     void authenticate_whenTheChannelIsNotAllowed_passesTheAllowedChannelsAndThrowsBadCredentialsException() {
132         final var primary = primary("FACTOR_OTT");
133         final var allowed = EnumSet.of(OttChannel.SMS);
134         when(secondFactorChannels.allowed(primary)).thenReturn(allowed);
135         when(verifyOneTimeCodeUseCase.execute("binding", "alice", CODE, OttPurpose.SECOND_FACTOR, allowed))
136                 .thenReturn(OttVerificationResult.failure());
137         final var request = new OttSecondFactorToken(primary, CODE, "binding");
138 
139         assertThrows(BadCredentialsException.class, () -> provider.authenticate(request));
140     }
141 
142     /** Only second-factor requests are supported. */
143     @Test
144     void supports_whenGivenARequestType_isTrueForSecondFactorRequestsOnly() {
145         assertTrue(provider.supports(OttSecondFactorToken.class));
146         assertFalse(provider.supports(UsernamePasswordAuthenticationToken.class));
147     }
148 
149     /**
150      * Builds the authentication the user already holds.
151      *
152      * @param factor the factor authority it holds.
153      * @return the authentication.
154      */
155     private static Authentication primary(final String factor) {
156         return UsernamePasswordAuthenticationToken.authenticated("alice", "x",
157                 List.of(FactorGrantedAuthority.fromAuthority(factor)));
158     }
159 
160     /**
161      * Lists the authorities of an authentication by name.
162      *
163      * @param authentication the authentication.
164      * @return their names.
165      */
166     private static Set<String> authorities(final Authentication authentication) {
167         return authentication.getAuthorities().stream().map(GrantedAuthority::getAuthority)
168                 .collect(Collectors.toSet());
169     }
170 }
