60  /**
61   * Unit tests of {@link CertificateController}, with a standalone MockMvc.
62   */
63  @ExtendWith(MockitoExtension.class)
64  class CertificateControllerTest {
65  
66      /** Username used by the tests. */
67      private static final String USERNAME = "alice";
68  
69      /** Path of the collection. */
70      private static final String PATH = "/api/account/certificates";
71  
72      /** PEM text sent by the tests; the reader is mocked, so it is never parsed. */
73      private static final String PEM = "-----BEGIN CERTIFICATE-----\nAAAA\n-----END CERTIFICATE-----\n";
74  
75      /** Body of a bind request. */
76      private static final String BODY = "{\"certificate\":\"" + PEM.replace("\n", "\\n") + "\"}";
77  
78      /** Fingerprint used by the tests. */
79      private static final String FINGERPRINT = "f".repeat(CertificateBinding.FINGERPRINT_LENGTH);
80  
81      /** Instant the certificate was bound. */
82      private static final Instant CREATED_AT = Instant.parse("2026-01-01T10:00:00Z");
83  
84      /** Maximum age of the step-up, in minutes. */
85      private static final int STEP_UP_MINUTES = 5;
86  
87      /** Account of the requests. */
88      private final CurrentAccount account = new CurrentAccount(UUID.randomUUID(), USERNAME);
89  
90      /** Identifier of the binding the tests use. */
91      private final UUID bindingId = UUID.randomUUID();
92  
93      /** Resolves the account of the request. */
94      @Mock
95      private CurrentAccountResolver currentAccountResolver;
96  
97      /** Lists the certificates. */
98      @Mock
99      private ListCertificatesUseCase listCertificatesUseCase;
100 
101     /** Binds a certificate. */
102     @Mock
103     private BindCertificateUseCase bindCertificateUseCase;
104 
105     /** Unbinds a certificate. */
106     @Mock
107     private UnbindCertificateUseCase unbindCertificateUseCase;
108 
109     /** Requires a recent multi-factor sign-in. */
110     @Mock
111     private StepUpGuard stepUpGuard;
112 
113     /** Reads the certificate of a bind request. */
114     @Mock
115     private PemCertificateReader pemCertificateReader;
116 
117     /** Real mapper. */
118     @Spy
119     private CertificateDtoMapper certificateDtoMapper = new CertificateDtoMapper();
120 
121     /** Controller under test. */
122     @InjectMocks
123     private CertificateController controller;
124 
125     /** Standalone MockMvc. */
126     private MockMvc mockMvc;
127 
128     /**
129      * Builds the MockMvc and resolves the account of every request.
130      */
131     @BeforeEach
132     void setUp() {
133         mockMvc = MockMvcBuilders.standaloneSetup(controller)
134                 .setControllerAdvice(new CertificateExceptionHandler(new MfaExceptionHandler())).build();
135         lenient().when(currentAccountResolver.resolve()).thenReturn(account);
136     }
137 
138     /**
139      * Verifies that the certificates of the account are listed without the account.
140      *
141      * @throws Exception if the request fails.
142      */
143     @Test
144     void listCertificates_whenAccountHasCertificates_answersThem() throws Exception {
145         when(listCertificatesUseCase.execute(account.id())).thenReturn(List.of(binding()));
146 
147         mockMvc.perform(get(PATH))
148                 .andExpect(status().isOk())
149                 .andExpect(jsonPath("$[0].id").value(bindingId.toString()))
150                 .andExpect(jsonPath("$[0].fingerprintSha256").value(FINGERPRINT))
151                 .andExpect(jsonPath("$[0].subjectDn").value("CN=alice"))
152                 .andExpect(jsonPath("$[0].userId").doesNotExist());
153     }
154 
155     /**
156      * Verifies that a certificate is bound after the step-up check and answered {@code 201}.
157      *
158      * @throws Exception if the request fails.
159      */
160     @Test
161     void bindCertificate_whenTheCertificateIsValid_answersCreatedAfterTheStepUpCheck() throws Exception {
162         when(pemCertificateReader.read(PEM)).thenReturn(new PresentedCertificate(FINGERPRINT, "CN=alice"));
163         when(bindCertificateUseCase.execute(account.id(), USERNAME, FINGERPRINT, "CN=alice")).thenReturn(binding());
164 
165         mockMvc.perform(post(PATH).contentType(MediaType.APPLICATION_JSON).content(BODY))
166                 .andExpect(status().isCreated())
167                 .andExpect(jsonPath("$.id").value(bindingId.toString()));
168 
169         final var order = inOrder(stepUpGuard, pemCertificateReader, bindCertificateUseCase);
170         order.verify(stepUpGuard).requireRecentMultiFactor(account.id());
171         order.verify(pemCertificateReader).read(PEM);
172         order.verify(bindCertificateUseCase).execute(account.id(), USERNAME, FINGERPRINT, "CN=alice");
173     }
174 
175     /**
176      * Verifies that nothing is read or bound without a recent multi-factor sign-in.
177      *
178      * @throws Exception if the request fails.
179      */
180     @Test
181     void bindCertificate_whenStepUpIsMissing_answersUnauthorizedWithTheChallenge() throws Exception {
182         doThrow(new InsufficientUserAuthenticationException(AuthenticationContextClass.MULTI_FACTOR,
183                 Duration.ofMinutes(STEP_UP_MINUTES))).when(stepUpGuard).requireRecentMultiFactor(account.id());
184 
185         mockMvc.perform(post(PATH).contentType(MediaType.APPLICATION_JSON).content(BODY))
186                 .andExpect(status().isUnauthorized())
187                 .andExpect(header().exists("WWW-Authenticate"));
188         verifyNoInteractions(pemCertificateReader, bindCertificateUseCase);
189     }
190 
191     /**
192      * Verifies that a value that is not a usable certificate is answered {@code 400} as a problem.
193      *
194      * @throws Exception if the request fails.
195      */
196     @Test
197     void bindCertificate_whenTheValueIsInvalid_answersBadRequest() throws Exception {
198         when(pemCertificateReader.read(PEM)).thenThrow(new InvalidCertificateException("bad"));
199 
200         mockMvc.perform(post(PATH).contentType(MediaType.APPLICATION_JSON).content(BODY))
201                 .andExpect(status().isBadRequest())
202                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:invalid-certificate"));
203         verifyNoInteractions(bindCertificateUseCase);
204     }
205 
206     /**
207      * Verifies that a certificate of another account is answered {@code 409} as a problem.
208      *
209      * @throws Exception if the request fails.
210      */
211     @Test
212     void bindCertificate_whenTheCertificateBelongsToAnotherAccount_answersConflict() throws Exception {
213         when(pemCertificateReader.read(PEM)).thenReturn(new PresentedCertificate(FINGERPRINT, "CN=alice"));
214         when(bindCertificateUseCase.execute(any(), any(), any(), any()))
215                 .thenThrow(new CertificateAlreadyBoundException("taken"));
216 
217         mockMvc.perform(post(PATH).contentType(MediaType.APPLICATION_JSON).content(BODY))
218                 .andExpect(status().isConflict())
219                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:certificate-already-bound"));
220     }
221 
222     /**
223      * Verifies that a request without the certificate is a bad request, never read.
224      *
225      * @throws Exception if the request fails.
226      */
227     @Test
228     void bindCertificate_whenTheCertificateIsMissing_answersBadRequest() throws Exception {
229         mockMvc.perform(post(PATH).contentType(MediaType.APPLICATION_JSON).content("{}"))
230                 .andExpect(status().isBadRequest());
231         verifyNoInteractions(pemCertificateReader, bindCertificateUseCase);
232     }
233 
234     /**
235      * Verifies that a certificate is unbound after the step-up check and answered {@code 204}.
236      *
237      * @throws Exception if the request fails.
238      */
239     @Test
240     void unbindCertificate_whenTheCertificateExists_answersNoContentAfterTheStepUpCheck() throws Exception {
241         mockMvc.perform(delete(PATH + "/" + bindingId)).andExpect(status().isNoContent())
242                 .andExpect(content().string(""));
243 
244         final var order = inOrder(stepUpGuard, unbindCertificateUseCase);
245         order.verify(stepUpGuard).requireRecentMultiFactor(account.id());
246         order.verify(unbindCertificateUseCase).execute(account.id(), USERNAME, bindingId);
247     }
248 
249     /**
250      * Verifies that a certificate is not unbound without a recent multi-factor sign-in.
251      *
252      * @throws Exception if the request fails.
253      */
254     @Test
255     void unbindCertificate_whenStepUpIsMissing_answersUnauthorized() throws Exception {
256         doThrow(new InsufficientUserAuthenticationException(AuthenticationContextClass.MULTI_FACTOR,
257                 Duration.ofMinutes(STEP_UP_MINUTES))).when(stepUpGuard).requireRecentMultiFactor(account.id());
258 
259         mockMvc.perform(delete(PATH + "/" + bindingId)).andExpect(status().isUnauthorized());
260         verifyNoInteractions(unbindCertificateUseCase);
261     }
262 
263     /**
264      * Verifies that a certificate the account does not have is answered {@code 404}.
265      *
266      * @throws Exception if the request fails.
267      */
268     @Test
269     void unbindCertificate_whenTheCertificateIsNotTheAccounts_answersNotFound() throws Exception {
270         doThrow(new CertificateNotFoundException("none")).when(unbindCertificateUseCase)
271                 .execute(account.id(), USERNAME, bindingId);
272 
273         mockMvc.perform(delete(PATH + "/" + bindingId))
274                 .andExpect(status().isNotFound())
275                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:certificate-not-found"));
276     }
277 
278     /**
279      * Verifies that the last way to sign in is answered {@code 409}.
280      *
281      * @throws Exception if the request fails.
282      */
283     @Test
284     void unbindCertificate_whenItIsTheLastSignInMethod_answersConflict() throws Exception {
285         doThrow(new LastSignInMethodException("last")).when(unbindCertificateUseCase)
286                 .execute(account.id(), USERNAME, bindingId);
287 
288         mockMvc.perform(delete(PATH + "/" + bindingId))
289                 .andExpect(status().isConflict())
290                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:last-sign-in-method"));
291     }
292 
293     /**
294      * Builds the binding used by the tests.
295      *
296      * @return the binding.
297      */
298     private CertificateBinding binding() {
299         return new CertificateBinding(bindingId, account.id(), FINGERPRINT, "CN=alice", CREATED_AT);
300     }
