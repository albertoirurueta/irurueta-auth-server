64  /** Unit tests of {@link RegisteredJwksClientAssertionDecoderFactory}, with keys generated at run time. */

71      /** Size of the generated RSA keys. */
72      private static final int KEY_SIZE = 2048;

74      /** Seconds the assertions live. */
75      private static final long LIFETIME_SECONDS = 60L;

96      static void generateKeys() throws JOSEException {

98          otherKey = new RSAKeyGenerator(KEY_SIZE).keyID("other-key").generate();

101     /** Creates the factory and the server context the default validators read. */
102     @BeforeEach
103     void setUp() {
104         factory = new RegisteredJwksClientAssertionDecoderFactory(replayValidator);

106         lenient().when(context.getIssuer()).thenReturn(ISSUER);
107         lenient().when(context.getAuthorizationServerSettings())

118     /** An assertion signed by a registered key decodes, and the replay validator is consulted. */
119     @Test
120     void createDecoder_whenClientHasRegisteredJwks_decodesAssertionSignedByARegisteredKey() throws Exception {
121         when(replayValidator.validate(any())).thenReturn(OAuth2TokenValidatorResult.success());
122         final var client = clientWithJwks(clientKey, SignatureAlgorithm.RS256);
123 
124         final var jwt = factory.createDecoder(client).decode(assertion(client, clientKey));
125 
126         assertEquals(client.getClientId(), jwt.getSubject());
127         verify(replayValidator).validate(any());
128     }
129 
130     /** A replayed assertion is refused. */
131     @Test
132     void createDecoder_whenReplayValidatorFails_refusesTheAssertion() throws Exception {
133         when(replayValidator.validate(any())).thenReturn(OAuth2TokenValidatorResult.failure(
134                 new OAuth2Error(OAuth2ErrorCodes.INVALID_CLIENT)));
135         final var client = clientWithJwks(clientKey, SignatureAlgorithm.RS256);
136         final var decoder = factory.createDecoder(client);
137         final var assertion = assertion(client, clientKey);
138 
139         assertThrows(JwtException.class, () -> decoder.decode(assertion));
140     }

172     /** A client whose registered keys changed gets a new decoder, which does not accept the old key. */
173     @Test
174     void createDecoder_whenJwksChange_returnsANewDecoder() throws Exception {
175         final var original = clientWithJwks(clientKey, SignatureAlgorithm.RS256);
176         final var rotated = RegisteredClient.from(original)
177                 .clientSettings(settings(otherKey, SignatureAlgorithm.RS256)).build();
178 
179         final var oldDecoder = factory.createDecoder(original);
180         final var newDecoder = factory.createDecoder(rotated);
181         final var assertion = assertion(original, clientKey);
182 
183         assertNotSame(oldDecoder, newDecoder);
184         assertThrows(JwtException.class, () -> newDecoder.decode(assertion));
185     }

245     /**
246      * Builds a client with registered keys.
247      *
248      * @param key       the registered public key.
249      * @param algorithm the signing algorithm.
250      * @return the client.
251      */
252     private static RegisteredClient clientWithJwks(final RSAKey key, final SignatureAlgorithm algorithm) {
253         return clientBuilder().clientSettings(settings(key, algorithm)).build();
254     }
255 
256     /**
257      * Builds the settings of a client with registered keys.
258      *
259      * @param key       the registered public key.
260      * @param algorithm the signing algorithm.
261      * @return the settings.
262      */
263     private static ClientSettings settings(final RSAKey key, final SignatureAlgorithm algorithm) {
264         return ClientSettings.builder()
265                 .setting(RegisteredJwksClientAssertionDecoderFactory.JWKS_SETTING,
266                         new JWKSet(key.toPublicJWK()).toString())
267                 .tokenEndpointAuthenticationSigningAlgorithm(algorithm).build();
268     }
269 
270     /**
271      * Starts a client builder with a fixed identifier.
272      *
273      * @return the builder.
274      */
275     private static RegisteredClient.Builder clientBuilder() {
276         return RegisteredClient.withId("id-1").clientId("client-1")
277                 .clientAuthenticationMethod(ClientAuthenticationMethod.PRIVATE_KEY_JWT)
278                 .authorizationGrantType(AuthorizationGrantType.CLIENT_CREDENTIALS);
279     }

281     /**
282      * Signs an assertion for a client.
283      *
284      * @param client the client the assertion names.
285      * @param signer the key that signs.
286      * @return the serialized assertion.
287      * @throws JOSEException if it cannot be signed.
288      */
289     private static String assertion(final RegisteredClient client, final RSAKey signer) throws JOSEException {
290         return assertion(client, signer, ISSUER);
291     }
292 
293     /**
294      * Builds an assertion of the client for the given audience, signed by the given key.
295      *
296      * @param client   the client.
297      * @param signer   the signing key.
298      * @param audience the {@code aud} of the assertion.
299      * @return the serialized assertion.
300      * @throws JOSEException if the assertion cannot be signed.
301      */
302     private static String assertion(final RegisteredClient client, final RSAKey signer, final String audience)
303             throws JOSEException {
304         final var now = Instant.now();
305         final var claims = new JWTClaimsSet.Builder().issuer(client.getClientId()).subject(client.getClientId())
306                 .audience(audience).jwtID(UUID.randomUUID().toString()).issueTime(Date.from(now))
307                 .expirationTime(Date.from(now.plusSeconds(LIFETIME_SECONDS))).build();
308         final var jwt = new SignedJWT(new JWSHeader.Builder(JWSAlgorithm.RS256).keyID(signer.getKeyID()).build(),
309                 claims);
310         jwt.sign(new RSASSASigner(signer));
311         return jwt.serialize();
312     }
