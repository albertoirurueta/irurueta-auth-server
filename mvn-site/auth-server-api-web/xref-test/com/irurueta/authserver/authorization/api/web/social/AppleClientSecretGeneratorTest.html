44  /**
45   * Unit tests of {@link AppleClientSecretGenerator}. The key is generated here at run time; none is committed.
46   */
47  class AppleClientSecretGeneratorTest {
48  
49      /** Client id of the tests. */
50      private static final String CLIENT_ID = "com.example.service";
51  
52      /** Team id of the tests. */
53      private static final String TEAM_ID = "TEAM123456";
54  
55      /** Key id of the tests. */
56      private static final String KEY_ID = "KEY1234567";
57  
58      /** Lifetime of a secret, in minutes. */
59      private static final long TTL_MINUTES = 10L;
60  
61      /** The instant of the clock. */
62      private static final Instant NOW = Instant.parse("2026-01-01T10:00:00Z");
63  
64      /**
65       * Verifies the content and the signature of a generated secret: the claims Apple requires and an ES256 signature
66       * that the public key verifies.
67       *
68       * @throws GeneralSecurityException if the key cannot be generated or the signature cannot be verified.
69       * @throws ParseException           if the secret is not a JWT.
70       * @throws JOSEException            if the signature cannot be verified.
71       */
72      @Test
73      void generate_whenCalled_returnsAnEs256JwtWithTheClaimsAppleRequires()
74              throws GeneralSecurityException, ParseException, JOSEException {
75          final var generator = KeyPairGenerator.getInstance("EC");
76          generator.initialize(new ECGenParameterSpec("secp256r1"));
77          final var pair = generator.generateKeyPair();
78          final var secrets = new AppleClientSecretGenerator(settings(), (ECPrivateKey) pair.getPrivate(),
79                  Clock.fixed(NOW, ZoneOffset.UTC));
80  
81          final var jwt = SignedJWT.parse(secrets.generate());
82  
83          assertEquals(JWSAlgorithm.ES256, jwt.getHeader().getAlgorithm());
84          assertEquals(KEY_ID, jwt.getHeader().getKeyID());
85          assertEquals(TEAM_ID, jwt.getJWTClaimsSet().getIssuer());
86          assertEquals(CLIENT_ID, jwt.getJWTClaimsSet().getSubject());
87          assertEquals(List.of("https://appleid.apple.com"), jwt.getJWTClaimsSet().getAudience());
88          assertEquals(NOW.getEpochSecond(), jwt.getJWTClaimsSet().getIssueTime().toInstant().getEpochSecond());
89          assertEquals(NOW.plus(Duration.ofMinutes(TTL_MINUTES)).getEpochSecond(),
90                  jwt.getJWTClaimsSet().getExpirationTime().toInstant().getEpochSecond());
91          assertTrue(jwt.verify(new ECDSAVerifier((ECPublicKey) pair.getPublic())));
92          assertEquals(CLIENT_ID, secrets.clientId());
93      }
94  
95      /**
96       * Verifies that the secret is cached while it is young and renewed after half of its lifetime.
97       *
98       * @throws GeneralSecurityException if the key cannot be generated.
99       */
100     @Test
101     void generate_whenCalledAgain_reusesTheSecretUntilHalfItsLifetimeHasPassed() throws GeneralSecurityException {
102         final var generator = KeyPairGenerator.getInstance("EC");
103         generator.initialize(new ECGenParameterSpec("secp256r1"));
104         final var key = (ECPrivateKey) generator.generateKeyPair().getPrivate();
105         final var clock = new MutableClock(NOW);
106         final var secrets = new AppleClientSecretGenerator(settings(), key, clock);
107 
108         final var first = secrets.generate();
109         clock.advance(Duration.ofMinutes(TTL_MINUTES / 2 - 1));
110         final var second = secrets.generate();
111         clock.advance(Duration.ofMinutes(2));
112         final var third = secrets.generate();
113 
114         assertSame(first, second);
115         assertNotEquals(first, third);
116     }
117 
118     /**
119      * Verifies that a key on a curve the signer does not support is refused when the generator is built.
120      */
121     @Test
122     void constructor_whenKeyCannotSignWithEs256_throwsIllegalArgumentException() {
123         final var key = mock(ECPrivateKey.class);
124         final var clock = Clock.fixed(NOW, ZoneOffset.UTC);
125         final var settings = settings();
126 
127         assertThrows(IllegalArgumentException.class, () -> new AppleClientSecretGenerator(settings, key, clock));
128     }
129 
130     /**
131      * Builds the settings of Apple.
132      *
133      * @return the settings.
134      */
135     private static SocialProperties.Provider settings() {
136         return new SocialProperties.Provider(CLIENT_ID, null, false, null, null, TEAM_ID, KEY_ID, "pem",
137                 Duration.ofMinutes(TTL_MINUTES));
138     }
139 
140     /**
141      * A clock a test moves forward. It is a Java clock, not a double of this repository's ports.
142      */
143     private static final class MutableClock extends Clock {
144 
145         /** The current instant. */
146         private Instant now;
147 
148         /**
149          * Creates the clock.
150          *
151          * @param now initial instant.
152          */
153         MutableClock(final Instant now) {
154             this.now = now;
155         }
156 
157         /**
158          * Moves the clock forward.
159          *
160          * @param duration how far.
161          */
162         void advance(final Duration duration) {
163             now = now.plus(duration);
164         }
165 
166         /**
167          * Gets the zone.
168          *
169          * @return UTC.
170          */
171         @Override
172         public ZoneId getZone() {
173             return ZoneOffset.UTC;
174         }
175 
176         /**
177          * Gets a clock in another zone.
178          *
179          * @param zone the zone.
180          * @return this clock.
181          */
182         @Override
183         public Clock withZone(final ZoneId zone) {
184             return this;
185         }
186 
187         /**
188          * Gets the current instant.
189          *
190          * @return the instant.
191          */
192         @Override
193         public Instant instant() {
194             return now;
195         }
196     }
