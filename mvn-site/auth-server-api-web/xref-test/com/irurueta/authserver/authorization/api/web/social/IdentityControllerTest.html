58  /**
59   * Unit tests of {@link IdentityController}, with a standalone MockMvc.
60   */
61  @ExtendWith(MockitoExtension.class)
62  class IdentityControllerTest {
63  
64      /** Username used by the tests. */
65      private static final String USERNAME = "alice";
66  
67      /** Path of the collection. */
68      private static final String PATH = "/api/account/identities";
69  
70      /** Maximum age of the step-up, in minutes. */
71      private static final int STEP_UP_MINUTES = 5;
72  
73      /** Account of the requests. */
74      private final CurrentAccount account = new CurrentAccount(UUID.randomUUID(), USERNAME);
75  
76      /** Identifier of the identity the tests unlink. */
77      private final UUID identityId = UUID.randomUUID();
78  
79      /** Resolves the account of the request. */
80      @Mock
81      private CurrentAccountResolver currentAccountResolver;
82  
83      /** Lists the identities. */
84      @Mock
85      private ListIdentitiesUseCase listIdentitiesUseCase;
86  
87      /** Unlinks an identity. */
88      @Mock
89      private UnlinkIdentityUseCase unlinkIdentityUseCase;
90  
91      /** Requires a recent multi-factor sign-in. */
92      @Mock
93      private StepUpGuard stepUpGuard;
94  
95      /** Real mapper. */
96      @Spy
97      private IdentityDtoMapper identityDtoMapper = new IdentityDtoMapper();
98  
99      /** Controller under test. */
100     @InjectMocks
101     private IdentityController controller;
102 
103     /** Standalone MockMvc. */
104     private MockMvc mockMvc;
105 
106     /**
107      * Builds the MockMvc and resolves the account of every request.
108      */
109     @BeforeEach
110     void setUp() {
111         mockMvc = MockMvcBuilders.standaloneSetup(controller)
112                 .setControllerAdvice(new IdentityExceptionHandler(new MfaExceptionHandler())).build();
113         lenient().when(currentAccountResolver.resolve()).thenReturn(account);
114     }
115 
116     /**
117      * Verifies that the identities of the account are listed without the subject.
118      *
119      * @throws Exception if the request fails.
120      */
121     @Test
122     void listIdentities_whenAccountHasIdentities_answersThem() throws Exception {
123         when(listIdentitiesUseCase.execute(account.id())).thenReturn(List.of(new FederatedIdentity(identityId,
124                 account.id(), IdentityProvider.GOOGLE, "subject", "alice@example.org",
125                 Instant.parse("2026-01-01T00:00:00Z"), null)));
126 
127         mockMvc.perform(get(PATH))
128                 .andExpect(status().isOk())
129                 .andExpect(jsonPath("$[0].id").value(identityId.toString()))
130                 .andExpect(jsonPath("$[0].provider").value("google"))
131                 .andExpect(jsonPath("$[0].email").value("alice@example.org"))
132                 .andExpect(jsonPath("$[0].subject").doesNotExist());
133     }
134 
135     /**
136      * Verifies that an account without identities gets an empty list.
137      *
138      * @throws Exception if the request fails.
139      */
140     @Test
141     void listIdentities_whenAccountHasNone_answersAnEmptyList() throws Exception {
142         when(listIdentitiesUseCase.execute(account.id())).thenReturn(List.of());
143 
144         mockMvc.perform(get(PATH)).andExpect(status().isOk()).andExpect(content().json("[]"));
145     }
146 
147     /**
148      * Verifies that an unlinked identity is answered {@code 204}, after the step-up check.
149      *
150      * @throws Exception if the request fails.
151      */
152     @Test
153     void unlinkIdentity_whenIdentityExists_answersNoContentAfterTheStepUpCheck() throws Exception {
154         mockMvc.perform(delete(PATH + "/" + identityId)).andExpect(status().isNoContent())
155                 .andExpect(content().string(""));
156 
157         final var order = inOrder(stepUpGuard, unlinkIdentityUseCase);
158         order.verify(stepUpGuard).requireRecentMultiFactor(account.id());
159         order.verify(unlinkIdentityUseCase).execute(account.id(), USERNAME, identityId);
160     }
161 
162     /**
163      * Verifies that an identity is not unlinked without a recent multi-factor sign-in.
164      *
165      * @throws Exception if the request fails.
166      */
167     @Test
168     void unlinkIdentity_whenStepUpIsMissing_answersUnauthorizedWithTheChallenge() throws Exception {
169         doThrow(new InsufficientUserAuthenticationException(AuthenticationContextClass.MULTI_FACTOR,
170                 Duration.ofMinutes(STEP_UP_MINUTES))).when(stepUpGuard).requireRecentMultiFactor(account.id());
171 
172         mockMvc.perform(delete(PATH + "/" + identityId))
173                 .andExpect(status().isUnauthorized())
174                 .andExpect(header().exists("WWW-Authenticate"))
175                 .andExpect(jsonPath("$.type").value("urn:problem:mfa:insufficient-user-authentication"));
176         verifyNoInteractions(unlinkIdentityUseCase);
177     }
178 
179     /**
180      * Verifies that unlinking an identity the account does not have is answered {@code 404} as a problem.
181      *
182      * @throws Exception if the request fails.
183      */
184     @Test
185     void unlinkIdentity_whenIdentityIsNotTheAccounts_answersNotFound() throws Exception {
186         doThrow(new FederatedIdentityNotFoundException("none")).when(unlinkIdentityUseCase)
187                 .execute(account.id(), USERNAME, identityId);
188 
189         mockMvc.perform(delete(PATH + "/" + identityId))
190                 .andExpect(status().isNotFound())
191                 .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_PROBLEM_JSON))
192                 .andExpect(jsonPath("$.type").value("urn:problem:identity:identity-not-found"));
193     }
194 
195     /**
196      * Verifies that unlinking the last way to sign in is answered {@code 409} as a problem.
197      *
198      * @throws Exception if the request fails.
199      */
200     @Test
201     void unlinkIdentity_whenIdentityIsTheLastSignInMethod_answersConflict() throws Exception {
202         doThrow(new LastSignInMethodException("last")).when(unlinkIdentityUseCase)
203                 .execute(account.id(), USERNAME, identityId);
204 
205         mockMvc.perform(delete(PATH + "/" + identityId))
206                 .andExpect(status().isConflict())
207                 .andExpect(jsonPath("$.type").value("urn:problem:identity:last-sign-in-method"));
208     }
209 
210     /**
211      * Verifies that an identifier that is not a UUID is a bad request, not a server error.
212      *
213      * @throws Exception if the request fails.
214      */
215     @Test
216     void unlinkIdentity_whenIdIsNotAUuid_answersBadRequest() throws Exception {
217         mockMvc.perform(delete(PATH + "/not-a-uuid")).andExpect(status().isBadRequest());
218         verifyNoInteractions(unlinkIdentityUseCase);
219     }
