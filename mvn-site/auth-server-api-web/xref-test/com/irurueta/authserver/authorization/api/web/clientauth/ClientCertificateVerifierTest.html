277     /**
278      * Checks that a context is refused with invalid_client.
279      *
280      * @param context the context.
281      */
282     private void assertInvalidClient(final OAuth2ClientAuthenticationContext context) {
283         final var error = assertThrows(OAuth2AuthenticationException.class, () -> verifier.accept(context));

288     /**
289      * Builds the settings of a client authenticated by a CA-issued certificate.
290      *
291      * @param subjectDn the registered subject DN.
292      * @return the settings.
293      */
294     private static ClientSettings tlsSettings(final String subjectDn) {
295         return ClientSettings.builder().x509CertificateSubjectDN(subjectDn).build();
296     }
297 
298     /**
299      * Builds the settings of a client with the certificate's key registered as its JWK set.
300      *
301      * @param certificate the certificate whose key is registered.
302      * @return the settings.
303      * @throws JOSEException if the key cannot be converted.
304      */
305     private static ClientSettings jwksSettings(final X509Certificate certificate) throws JOSEException {
306         return ClientSettings.builder().setting(RegisteredJwksClientAssertionDecoderFactory.JWKS_SETTING,
307                 new JWKSet(jwkOf(certificate)).toString()).build();
308     }
309 
310     /**
311      * Converts the public key of a certificate to a JWK.
312      *
313      * @param certificate the certificate.
314      * @return the public JWK.
315      * @throws JOSEException if the key is neither an RSA nor an EC key.
316      */
317     private static JWK jwkOf(final X509Certificate certificate) throws JOSEException {
318         final var key = certificate.getPublicKey();
319         if (key instanceof RSAPublicKey rsa) {
320             return new RSAKey.Builder(rsa).build();
321         }
322         if (key instanceof ECPublicKey ec) {
323             return new ECKey.Builder(Curve.forECParameterSpec(ec.getParams()), ec).build();
324         }
325         throw new JOSEException("Unsupported key type " + key.getAlgorithm());
326     }
327 
328     /**
329      * Builds the authentication context of a client presenting a chain.
330      *
331      * @param method the authentication method.
332      * @param settings the client's settings.
333      * @param chain  the presented certificates, the leaf first.
334      * @return the context.
335      */
336     private static OAuth2ClientAuthenticationContext context(final ClientAuthenticationMethod method,
337             final ClientSettings settings, final X509Certificate... chain) {
338         final var registered = client(settings);
339         return OAuth2ClientAuthenticationContext
340                 .with(new OAuth2ClientAuthenticationToken(registered, method, chain))
341                 .registeredClient(registered).build();
342     }
343 
344     /**
345      * Builds a client.
346      *
347      * @param settings the client's settings.
348      * @return the client.
349      */
350     private static RegisteredClient client(final ClientSettings settings) {
351         return RegisteredClient.withId("id-1").clientId(CLIENT_NAME)
352                 .clientAuthenticationMethod(ClientAuthenticationMethod.TLS_CLIENT_AUTH)
353                 .authorizationGrantType(AuthorizationGrantType.CLIENT_CREDENTIALS).clientSettings(settings).build();
354     }

356     /**
357      * Finds the PEM file of the authority next to the identity's key store.
358      *
359      * @param identity an identity issued by the authority.
360      * @return the file.
361      */
362     private static Path authorityPem(final TestPki.Identity identity) {
363         return identity.keyStore().getParent().resolve("ca.pem");
