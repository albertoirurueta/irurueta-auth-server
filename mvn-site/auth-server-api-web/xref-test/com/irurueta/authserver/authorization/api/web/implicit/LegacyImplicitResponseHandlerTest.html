16  package com.irurueta.authserver.authorization.api.web.implicit;

18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertFalse;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  import static org.mockito.Mockito.verify;
23  import static org.mockito.Mockito.verifyNoInteractions;
24  
25  import java.time.Instant;
26  import java.util.Set;
27  import org.junit.jupiter.api.Test;
28  import org.junit.jupiter.api.extension.ExtendWith;
29  import org.mockito.InjectMocks;
30  import org.mockito.Mock;
31  import org.mockito.junit.jupiter.MockitoExtension;
32  import org.springframework.mock.web.MockHttpServletRequest;
33  import org.springframework.mock.web.MockHttpServletResponse;
34  import org.springframework.security.authentication.TestingAuthenticationToken;
35  import org.springframework.security.oauth2.core.OAuth2AccessToken;
36  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
37  
38  /**
39   * Unit tests of {@link LegacyImplicitResponseHandler}.
40   */
41  @ExtendWith(MockitoExtension.class)
42  class LegacyImplicitResponseHandlerTest {
43  
44      /** Status of a redirect. */
45      private static final int FOUND = 302;
46  
47      /** Registered redirect URI. */
48      private static final String REDIRECT_URI = "http://localhost:9000/legacy-implicit/";
49  
50      /** Lifetime of the token in seconds. */
51      private static final long LIFETIME = 300;
52  
53      /** Answers the results that are not the flow's. */
54      @Mock
55      private AuthenticationSuccessHandler delegate;
56  
57      /** Handler under test. */
58      @InjectMocks
59      private LegacyImplicitResponseHandler handler;
60  
61      /**
62       * Verifies that a granted request redirects with the token, its type, lifetime, scope and state in the fragment,
63       * encoded, and with the headers that keep it private.
64       *
65       * @throws Exception never.
66       */
67      @Test
68      void onAuthenticationSuccess_whenTheTokenIsGranted_redirectsWithTheFragment() throws Exception {
69          final var now = Instant.now();
70          final var accessToken = new OAuth2AccessToken(OAuth2AccessToken.TokenType.BEARER, "a.b-c", now,
71                  now.plusSeconds(LIFETIME), Set.of("documents.read"));
72          final var result = new LegacyImplicitAccessTokenToken(accessToken, Set.of("documents.read"), REDIRECT_URI,
73                  "s t&=");
74          final var response = new MockHttpServletResponse();
75  
76          handler.onAuthenticationSuccess(new MockHttpServletRequest(), response, result);
77  
78          assertEquals(FOUND, response.getStatus());
79          assertEquals(REDIRECT_URI + "#access_token=a.b-c&token_type=Bearer&expires_in=300&scope=documents.read"
80                  + "&state=s%20t%26%3D", response.getRedirectedUrl());
81          assertEquals("no-store", response.getHeader("Cache-Control"));
82          assertEquals("no-referrer", response.getHeader("Referrer-Policy"));
83          verifyNoInteractions(delegate);
84      }
85  
86      /**
87       * Verifies that a granted request without state has no state in the fragment.
88       *
89       * @throws Exception never.
90       */
91      @Test
92      void onAuthenticationSuccess_whenThereIsNoState_omitsIt() throws Exception {
93          final var now = Instant.now();
94          final var accessToken = new OAuth2AccessToken(OAuth2AccessToken.TokenType.BEARER, "t", now,
95                  now.plusSeconds(LIFETIME), Set.of("documents.read"));
96          final var response = new MockHttpServletResponse();
97  
98          handler.onAuthenticationSuccess(new MockHttpServletRequest(), response,
99                  new LegacyImplicitAccessTokenToken(accessToken, Set.of("documents.read"), REDIRECT_URI, null));
100 
101         assertFalse(response.getRedirectedUrl().contains("state"));
102     }
103 
104     /**
105      * Verifies that a refusal redirects with the error and the state in the fragment.
106      *
107      * @throws Exception never.
108      */
109     @Test
110     void onAuthenticationSuccess_whenTheRequestIsRefused_redirectsWithTheError() throws Exception {
111         final var response = new MockHttpServletResponse();
112 
113         handler.onAuthenticationSuccess(new MockHttpServletRequest(), response,
114                 new LegacyImplicitErrorToken("invalid_scope", "bad scope", REDIRECT_URI, "abc"));
115 
116         assertEquals(FOUND, response.getStatus());
117         assertEquals(REDIRECT_URI + "#error=invalid_scope&error_description=bad%20scope&state=abc",
118                 response.getRedirectedUrl());
119         assertEquals("no-store", response.getHeader("Cache-Control"));
120         assertEquals("no-referrer", response.getHeader("Referrer-Policy"));
121     }
122 
123     /**
124      * Verifies that a refusal without description or state has only the error.
125      *
126      * @throws Exception never.
127      */
128     @Test
129     void onAuthenticationSuccess_whenTheRefusalHasNoDescriptionOrState_sendsOnlyTheError() throws Exception {
130         final var response = new MockHttpServletResponse();
131 
132         handler.onAuthenticationSuccess(new MockHttpServletRequest(), response,
133                 new LegacyImplicitErrorToken("invalid_request", null, REDIRECT_URI, null));
134 
135         assertEquals(REDIRECT_URI + "#error=invalid_request", response.getRedirectedUrl());
136     }
137 
138     /**
139      * Verifies that any other result goes to the delegate and gets none of the headers.
140      *
141      * @throws Exception never.
142      */
143     @Test
144     void onAuthenticationSuccess_whenTheResultIsAnotherOne_delegates() throws Exception {
145         final var request = new MockHttpServletRequest();
146         final var response = new MockHttpServletResponse();
147         final var other = new TestingAuthenticationToken("alice", "secret");
148 
149         handler.onAuthenticationSuccess(request, response, other);
150 
151         verify(delegate).onAuthenticationSuccess(request, response, other);
152         assertNull(response.getRedirectedUrl());
153         assertTrue(response.getHeaderNames().isEmpty());
154     }
155 }
