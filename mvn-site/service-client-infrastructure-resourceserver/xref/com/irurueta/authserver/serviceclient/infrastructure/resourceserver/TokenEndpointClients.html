18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import lombok.RequiredArgsConstructor;
21  import org.springframework.beans.factory.ObjectProvider;
22  import org.springframework.http.HttpHeaders;
23  import org.springframework.http.client.ClientHttpRequestFactory;
24  import org.springframework.http.converter.FormHttpMessageConverter;
25  import org.springframework.security.oauth2.client.endpoint.AbstractOAuth2AuthorizationGrantRequest;
26  import org.springframework.security.oauth2.client.endpoint.AbstractRestClientOAuth2AccessTokenResponseClient;
27  import org.springframework.security.oauth2.client.endpoint.NimbusJwtClientAuthenticationParametersConverter;
28  import org.springframework.security.oauth2.client.endpoint.RestClientClientCredentialsTokenResponseClient;
29  import org.springframework.security.oauth2.client.endpoint.RestClientTokenExchangeTokenResponseClient;
30  import org.springframework.security.oauth2.client.http.OAuth2ErrorResponseErrorHandler;
31  import org.springframework.security.oauth2.core.http.converter.OAuth2AccessTokenResponseHttpMessageConverter;
32  import org.springframework.stereotype.Component;
33  import org.springframework.util.LinkedMultiValueMap;
34  import org.springframework.web.client.RestClient;
35  
36  /**
37   * Builds the clients of the token endpoint for the two grants, set up for the configured authentication method: the
38   * default for {@code client_secret_basic}; the signed assertion for {@code private_key_jwt}; a REST client over the
39   * SSL bundle (so the client certificate is presented) for the mutual TLS methods; plus a DPoP proof header when DPoP
40   * is enabled.
41   */
42  @Component
43  @RequiredArgsConstructor
44  public class TokenEndpointClients {
45  
46      /** Settings of the service client at the authorization server. */
47      private final ServiceClientAuthProperties properties;
48  
49      /** Signing key of the assertion. */
50      private final PrivateKeyJwkResolver jwkResolver;
51  
52      /** Request factory that presents the client certificate. */
53      private final SslRequestFactories requestFactories;
54  
55      /** Creates the DPoP proofs; absent unless DPoP is enabled. */
56      private final ObjectProvider<DPoPProofs> dpopProofs;
57  
58      /**
59       * Builds the client of the client credentials grant.
60       *
61       * @return the client.
62       */
63      public RestClientClientCredentialsTokenResponseClient clientCredentials() {
64          final var client = new RestClientClientCredentialsTokenResponseClient();
65          customize(client);
66          return client;
67      }
68  
69      /**
70       * Builds the client of the token exchange grant.
71       *
72       * @return the client.
73       */
74      public RestClientTokenExchangeTokenResponseClient tokenExchange() {
75          final var client = new RestClientTokenExchangeTokenResponseClient();
76          customize(client);
77          return client;
78      }
79  
80      /**
81       * Sets a token response client up for the configured authentication method.
82       *
83       * @param client the client.
84       * @param <T>    grant request of the client.
85       */
86      private <T extends AbstractOAuth2AuthorizationGrantRequest> void customize(
87              final AbstractRestClientOAuth2AccessTokenResponseClient<T> client) {
88          requestFactories.create().ifPresent(factory -> client.setRestClient(tokenRestClient(factory)));
89          if (properties.method() == AuthMethod.PRIVATE_KEY_JWT) {
90              client.addParametersConverter(new NimbusJwtClientAuthenticationParametersConverter<T>(jwkResolver));
91          }
92          if (properties.method().isMutualTls()) {
93              // The client is identified by its certificate: nothing to add to the request but, as it is, the client
94              // refuses any method other than basic, post and none. Adding any parameters converter (this one adds
95              // nothing) lifts that check; the default converter still sends client_id.
96              client.addParametersConverter(request -> new LinkedMultiValueMap<>());
97          }
98          final var proofs = dpopProofs.getIfAvailable();
99          if (proofs != null) {
100             client.addHeadersConverter(request -> {
101                 final var headers = new HttpHeaders();
102                 headers.set(DPoPClientHttpRequestInterceptor.DPOP_HEADER,
103                         proofs.proof("POST", properties.tokenUri()));
104                 return headers;
105             });
106         }
107     }
108 
109     /**
110      * Builds the REST client of the token endpoint over a request factory, with the message converters and the error
111      * handler of Spring Security's default one.
112      *
113      * @param factory the request factory.
114      * @return the REST client.
115      */
116     private static RestClient tokenRestClient(final ClientHttpRequestFactory factory) {
117         return RestClient.builder()
118                 .requestFactory(factory)
119                 .configureMessageConverters(converters -> {
120                     converters.addCustomConverter(new FormHttpMessageConverter());
121                     converters.addCustomConverter(new OAuth2AccessTokenResponseHttpMessageConverter());
122                 })
123                 .defaultStatusHandler(new OAuth2ErrorResponseErrorHandler())
124                 .build();
125     }
126 }
