41  /**
42   * The OAuth2 client of the service client: its registrations, the authorized client manager that caches the access
43   * tokens and renews them shortly before they expire, and the documents client that sends them. Nothing here depends
44   * on a profile: every setting has a default, and a mutual TLS or {@code private_key_jwt} setup only fails when its
45   * key or bundle is missing at the moment it is needed.
46   */
47  @Configuration(proxyBeanMethods = false)
48  @EnableConfigurationProperties({ResourceServerProperties.class, ServiceClientAuthProperties.class,
49          ServiceClientKeyProperties.class})
50  public class OAuth2ClientConfiguration {
51  
52      /**
53       * Registrations of the client credentials and token exchange grants.
54       *
55       * @param mapper     builds the registrations.
56       * @param properties settings of the service client.
57       * @return the repository.
58       */
59      @Bean
60      ClientRegistrationRepository serviceClientRegistrations(final ServiceClientRegistrationMapper mapper,
61              final ServiceClientAuthProperties properties) {
62          return new InMemoryClientRegistrationRepository(
63                  List.of(mapper.clientCredentials(properties), mapper.tokenExchange(properties)));
64      }
65  
66      /**
67       * The cache of the authorized clients, so a token is reused while it is valid.
68       *
69       * @param registrations the registrations.
70       * @return the service.
71       */
72      @Bean
73      OAuth2AuthorizedClientService serviceClientAuthorizedClients(final ClientRegistrationRepository registrations) {
74          return new InMemoryOAuth2AuthorizedClientService(registrations);
75      }
76  
77      /**
78       * The authorized client manager. It works without a servlet request, as there is none in the CLI. A cached token
79       * is renewed once less than {@code service-client.auth.refresh-before} of its life is left.
80       *
81       * @param registrations   the registrations.
82       * @param clients         the cache of the authorized clients.
83       * @param tokenClients    the clients of the token endpoint.
84       * @param principalMapper builds the principal of the service client.
85       * @param properties      settings of the service client.
86       * @return the manager.
87       */
88      @Bean
89      OAuth2AuthorizedClientManager serviceClientAuthorizedClientManager(
90              final ClientRegistrationRepository registrations, final OAuth2AuthorizedClientService clients,
91              final TokenEndpointClients tokenClients, final PrincipalMapper principalMapper,
92              final ServiceClientAuthProperties properties) {
93          // tag::client-credentials-manager[]
94          final var clientCredentials = OAuth2AuthorizedClientProviderBuilder.builder()
95                  .clientCredentials(grant -> grant
96                          .accessTokenResponseClient(tokenClients.clientCredentials())
97                          .clockSkew(properties.refreshBefore()))
98                  .build();
99          final var clientCredentialsManager =
100                 new AuthorizedClientServiceOAuth2AuthorizedClientManager(registrations, clients);
101         clientCredentialsManager.setAuthorizedClientProvider(clientCredentials);
102         // end::client-credentials-manager[]
103 
104         // tag::token-exchange-client[]
105         // The builder has no token exchange method, so its provider is added with provider(...). The subject token is
106         // the principal object of the call (see PrincipalMapper.user) and the actor token is the client's own client
107         // credentials token, which is how the exchanged token names the client as actor.
108         final var exchange = new TokenExchangeOAuth2AuthorizedClientProvider();
109         exchange.setAccessTokenResponseClient(tokenClients.tokenExchange());
110         exchange.setClockSkew(properties.refreshBefore());
111         exchange.setActorTokenResolver(context -> {
112             final var actor = clientCredentialsManager.authorize(OAuth2AuthorizeRequest
113                     .withClientRegistrationId(ServiceClientRegistrationMapper.CLIENT_CREDENTIALS_REGISTRATION_ID)
114                     .principal(principalMapper.serviceClient(properties.clientId()))
115                     .build());
116             return actor == null ? null : actor.getAccessToken();
117         });
118         // end::token-exchange-client[]
119 
120         final var manager = new AuthorizedClientServiceOAuth2AuthorizedClientManager(registrations, clients);
121         manager.setAuthorizedClientProvider(OAuth2AuthorizedClientProviderBuilder.builder()
122                 .provider(clientCredentials)
123                 .provider(exchange)
124                 .build());
125         return manager;
126     }
127 
128     /**
129      * Generates the key pair that signs the DPoP proofs, for the life of the process. Present only when {@code
130      * service-client.auth.dpop} is {@code true}.
131      *
132      * @return the proof creator.
133      * @throws JOSEException if the key cannot be generated.
134      */
135     @Bean
136     @ConditionalOnProperty(prefix = "service-client.auth", name = "dpop", havingValue = "true")
137     DPoPProofs serviceClientDPoPProofs() throws JOSEException {
138         final var key = new ECKeyGenerator(Curve.P_256).generate();
139         return new DPoPProofs(new DefaultDPoPProofFactory(key, JWSAlgorithm.ES256));
140     }
141 
142     /**
143      * The documents client of the service client itself, whose requests carry the client credentials token.
144      *
145      * @param documentsApis   builds documents clients.
146      * @param principalMapper builds the principal of the service client.
147      * @param properties      settings of the service client.
148      * @return the documents client.
149      */
150     @Bean
151     DocumentsApi serviceClientDocumentsApi(final DocumentsApis documentsApis, final PrincipalMapper principalMapper,
152             final ServiceClientAuthProperties properties) {
153         return documentsApis.forPrincipal(ServiceClientRegistrationMapper.CLIENT_CREDENTIALS_REGISTRATION_ID,
154                 principalMapper.serviceClient(properties.clientId()));
155     }
156 }
