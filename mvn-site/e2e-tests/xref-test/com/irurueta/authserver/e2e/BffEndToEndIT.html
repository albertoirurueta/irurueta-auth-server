42  /**
43   * End-to-end test of the single page application through the real servers: the packaged {@code auth-server},
44   * {@code resource-server} and {@code bff} run on random ports against a PostgreSQL container, and the demo user
45   * drives the SPA in headless Chromium. The {@code bff} client requires consent, so the flow goes through the login
46   * and the consent pages. The browser reaches every application through {@code localhost}, where Chromium accepts the
47   * {@code Secure} session cookie over plain HTTP.
48   *
49   * <p>Every response the page receives during the whole flow, sign-out included, is recorded and scanned for tokens:
50   * bodies, the web storages and the cookies must never contain one. Response headers and URLs are scanned too, with
51   * exactly two exclusions, both parts of the RP-initiated logout: the {@code Location} header of the {@code 302} that
52   * answers {@code POST /bff/logout}, and the URL of the {@code /connect/logout} request the browser then makes. They
53   * carry the ID token as {@code id_token_hint}, as OpenID Connect RP-Initiated Logout prescribes, in a top-level
54   * navigation started by a form submission, which no script of the page can read. The test asserts that this
55   * navigation happened, so the exclusions never hide a different flow.
56   */
57  class BffEndToEndIT {

59      /** Name of the session cookie of the BFF. */
60      private static final String SESSION_COOKIE = "__Host-SESSION";

62      /** Matches the log line printed once the BFF is up. */
63      private static final Pattern BFF_STARTED = Pattern.compile("Started BffApplication in");

65      /** Matches the log line printed once the resource server is up. */

68      /** Matches a JSON Web Token: three dot-separated base64url segments, the first one a JSON header. */
69      private static final Pattern JWT_SHAPE = Pattern.compile("eyJ[\\w-]+\\.[\\w-]+\\.[\\w-]*");

71      /** Query that returns every token value the authorization server has issued. */
72      private static final String ISSUED_TOKENS = "SELECT access_token_value FROM oauth2_authorization"
73              + " WHERE access_token_value IS NOT NULL UNION ALL SELECT refresh_token_value FROM oauth2_authorization"
74              + " WHERE refresh_token_value IS NOT NULL UNION ALL SELECT oidc_id_token_value FROM oauth2_authorization"
75              + " WHERE oidc_id_token_value IS NOT NULL";

77      /** Query that counts the BFF sessions of an authenticated user. */
78      private static final String AUTHENTICATED_SESSIONS =
79              "SELECT count(*) FROM spring_session WHERE principal_name IS NOT NULL";

117     /**
118      * Starts the database and the three applications, registering the BFF's real address in the {@code bff} client.
119      *
120      * @throws Exception if an application or the container cannot be started.
121      */
122     @BeforeAll
123     static void startApplications() throws Exception {
124         final var bffPort = RunningAuthServer.freePort();
125         final var resourcePort = RunningAuthServer.freePort();
126         bffUrl = "http://localhost:" + bffPort;
127         authServer = RunningAuthServer.start(List.of(
128                 "--authserver.clients.bff.redirect-uris=" + bffUrl + "/login/oauth2/code/bff",
129                 "--authserver.clients.bff.post-logout-redirect-uris=" + bffUrl + "/"));
130 
131         resourceServer = ManagedApp.start(RunningAuthServer.jar("resourceserver.jar"),
132                 List.of("--server.port=" + resourcePort), List.of("AUTH_ISSUER=" + authServer.issuer()));
133         resourceServer.await(RESOURCE_SERVER_STARTED, RunningAuthServer.STARTUP_TIMEOUT);
134 
135         final var bffArguments = new ArrayList<>(authServer.bffDatabaseArguments());
136         bffArguments.add("--server.port=" + bffPort);
137         bffArguments.add("--bff.resource-server.base-uri=http://localhost:" + resourcePort);
138         bff = ManagedApp.start(RunningAuthServer.jar("bff.jar"), bffArguments,
139                 List.of("AUTH_ISSUER=" + authServer.issuer(), "BFF_CLIENT_SECRET=" + authServer.clientSecret()));
140         bff.await(BFF_STARTED, RunningAuthServer.STARTUP_TIMEOUT);
141     }

143     /** Stops the applications and the database. */
144     @AfterAll
145     static void stopApplications() {
146         if (bff != null) {
147             bff.stop();
148         }
149         if (resourceServer != null) {
150             resourceServer.stop();
151         }
152         if (authServer != null) {
153             authServer.close();
154         }

157     /**
158      * Signs in through the login and consent pages, reads the profile and the documents through the BFF, signs out,
159      * checks that the browser lands back on the signed-out SPA and that the protected calls are then refused, and
160      * scans everything the browser received, before and after the sign-out, for tokens.
161      *
162      * @throws Exception if a step of the flow fails.
163      */
164     @Test
165     void spa_whenUserSignsInAndOut_showsProfileAndDocumentsThenRefusesCalls() throws Exception {
166         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch();
167              var context = browser.newContext()) {
168             final var page = context.newPage();
169             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
170             final var observed = recordResponses(page);
171 
172             signIn(page);
173             consent(page, "#allow");
174 
175             page.waitForSelector("#profile:not([hidden])");
176             assertEquals("demo", page.textContent("#profile-subject"));
177             page.click("#load-documents");
178             page.waitForSelector("#documents:not([hidden])");
179             assertTrue(page.isHidden("#error"), "no error is shown when the documents are loaded");
180             assertEquals(1, authenticatedSessions());
181 
182             final var issued = issuedTokens();
183             assertSessionCookieIsHardened(page, context, observed);
184             assertNoTokenReachedTheBrowser(page, context, observed, issued);
185 
186             page.click("#sign-out");
187             page.waitForSelector("#signed-out:not([hidden])");
188             assertEquals(bffUrl + "/", page.url(), "the browser is back on the SPA, not on the login page");
189             assertTrue(page.isHidden("#profile"));
190             assertRpInitiatedLogoutHappened(observed);
191             assertEquals(HTTP_UNAUTHORIZED, context.request().get(bffUrl + "/api/documents").status());
192             assertEquals(HTTP_UNAUTHORIZED, context.request().get(bffUrl + "/bff/user").status());
193             assertEquals(0, authenticatedSessions());
194 
195             issued.addAll(issuedTokens());
196             assertNoTokenReachedTheBrowser(page, context, observed, issued);
197         }
198     }

200     /**
201      * Denies the consent and checks that the SPA reports the failed sign-in and that no session is created.
202      *
203      * @throws Exception if a step of the flow fails.
204      */
205     @Test
206     void spa_whenUserDeniesConsent_showsSignInFailureAndCreatesNoSession() throws Exception {
207         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch();
208              var context = browser.newContext()) {
209             final var page = context.newPage();
210             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
211 
212             signIn(page);
213             consent(page, "#deny");
214 
215             page.waitForURL(url -> url.startsWith(bffUrl + "/?error=signin"));
216             page.waitForSelector("#error:not([hidden])");
217             assertTrue(page.textContent("#error").contains("Sign-in failed"), page.textContent("#error"));
218             assertTrue(page.isHidden("#profile"));
219             assertEquals(HTTP_UNAUTHORIZED, context.request().get(bffUrl + "/bff/user").status());
220             assertEquals(0, authenticatedSessions());
221         }

224     /**
225      * Opens the SPA, which shows its signed-out state without leaving the page, clicks <em>Sign in</em> and signs the
226      * demo user in on the login page of the authorization server.
227      *
228      * @param page the page of the browser.
229      */
230     private static void signIn(final Page page) {
231         page.navigate(bffUrl + "/");
232         page.waitForSelector("#signed-out:not([hidden])");
233         assertEquals(bffUrl + "/", page.url(), "the first visit does not start the sign-in by itself");
234         page.click("#sign-in");
235         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
236         page.fill("#username", "demo");
237         page.fill("#password", authServer.demoPassword());
238         page.click("button[type=submit]");
239         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/oauth2/consent"));
240     }

273     /**
274      * Records every response the page receives: method, URL, headers (including {@code Set-Cookie}) and body.
275      *
276      * @param page the page of the browser.
277      * @return the recorded responses, in arrival order; filled as the page works.
278      */
279     private static List<ObservedResponse> recordResponses(final Page page) {
280         final var observed = new ArrayList<ObservedResponse>();
281         page.onResponse(response -> observed.add(ObservedResponse.of(response)));
282         return observed;
283     }

285     /**
286      * Asserts that the sign-out went through the RP-initiated logout: {@code POST /bff/logout} answered a redirect to
287      * the end session endpoint with an ID token hint, and the browser followed it. These two are what the token scan
288      * excludes from its header and URL checks.
289      *
290      * @param observed the responses recorded during the flow.
291      */
292     private static void assertRpInitiatedLogoutHappened(final List<ObservedResponse> observed) {
293         final var logout = observed.stream().filter(BffEndToEndIT::isBffLogout).findFirst().orElseThrow();
294         assertEquals(HTTP_FOUND, logout.status());
295         final var location = logout.headers().get(LOCATION);
296         assertTrue(location.startsWith(authServer.issuer() + END_SESSION_PATH), location);
297         assertTrue(location.contains("id_token_hint="), "the end session request carries the ID token hint");
298         assertTrue(observed.stream().anyMatch(BffEndToEndIT::isEndSessionRequest),
299                 "the browser followed the redirect to the end session endpoint");
300     }
301 
302     /**
303      * Asserts that the session cookie is {@code HttpOnly}, {@code Secure} and {@code SameSite=Lax}, and not readable by
304      * scripts, while the user is signed in.
305      *
306      * @param page     the page of the browser, on the SPA.
307      * @param context  the browser context.
308      * @param observed the responses recorded so far.
309      */
310     private static void assertSessionCookieIsHardened(
311             final Page page, final BrowserContext context, final List<ObservedResponse> observed) {
312         assertFalse(String.valueOf(page.evaluate("document.cookie")).contains(SESSION_COOKIE),
313                 "the session cookie is not readable by scripts");
314         final Cookie session = context.cookies().stream().filter(cookie -> SESSION_COOKIE.equals(cookie.name))
315                 .findFirst().orElseThrow();
316         assertTrue(session.httpOnly, "HttpOnly");
317         assertTrue(session.secure, "Secure");
318         assertEquals(SameSiteAttribute.LAX, session.sameSite);
319         assertTrue(observed.stream().anyMatch(response -> response.url().contains("/bff/user")));
320         assertTrue(observed.stream().anyMatch(response -> response.url().contains("/api/documents")
321                 && response.body().contains("[]")), "the documents response was scanned");
322         assertEquals(HTTP_OK, context.request().get(bffUrl + "/bff/user").status());
323     }

325     /**
326      * Asserts that no token is visible to the browser: not in the web storages, the cookies (the script-readable ones
327      * and the cookie jar as a whole) or any recorded response (body, headers and URL, with the two exclusions of the
328      * RP-initiated logout described on this class).
329      *
330      * @param page     the page of the browser, on the SPA.
331      * @param context  the browser context.
332      * @param observed the responses recorded so far.
333      * @param issued   the token values the authorization server has issued.
334      */
335     private static void assertNoTokenReachedTheBrowser(final Page page, final BrowserContext context,
336             final List<ObservedResponse> observed, final Set<String> issued) {
337         assertTrue(issued.stream().anyMatch(token -> JWT_SHAPE.matcher(token).matches()),
338                 "the authorization server issued a JWT, so the scan pattern is meaningful");
339         assertTrue(observed.size() > 1, "responses were recorded");
340 
341         final var sources = new LinkedHashMap<String, String>();
342         for (var index = 0; index < observed.size(); index++) {
343             final var response = observed.get(index);
344             final var label = "response #" + index + " (" + response.method() + " " + response.status() + ")";
345             if (!isEndSessionRequest(response)) {
346                 sources.put(label + " URL", response.url());
347             }
348             final var headers = new HashMap<>(response.headers());
349             if (isBffLogout(response)) {
350                 headers.remove(LOCATION);
351             }
352             sources.put(label + " headers", headers.toString());
353             sources.put(label + " body", response.body());
354         }
355         sources.put("localStorage", String.valueOf(page.evaluate("JSON.stringify(Object.entries(localStorage))")));
356         sources.put("sessionStorage",
357                 String.valueOf(page.evaluate("JSON.stringify(Object.entries(sessionStorage))")));
358         sources.put("document.cookie", String.valueOf(page.evaluate("document.cookie")));
359         context.cookies().forEach(cookie -> sources.put("cookie " + cookie.name, cookie.value));
360         sources.forEach((label, content) -> {
361             assertFalse(JWT_SHAPE.matcher(content).find(), "JWT-shaped value in " + label);
362             issued.forEach(token -> assertFalse(content.contains(token), "an issued token is visible in " + label));
363         });
364     }

366     /**
367      * Tells whether a response answers the sign-out form of the SPA.
368      *
369      * @param response a recorded response.
370      * @return {@code true} for the response to {@code POST /bff/logout}.
371      */
372     private static boolean isBffLogout(final ObservedResponse response) {
373         return "POST".equals(response.method()) && (bffUrl + BFF_LOGOUT_PATH).equals(response.url());
374     }
375 
376     /**
377      * Tells whether a response answers the request to the end session endpoint of the authorization server.
378      *
379      * @param response a recorded response.
380      * @return {@code true} for the response to {@code /connect/logout}.
381      */
382     private static boolean isEndSessionRequest(final ObservedResponse response) {
383         return response.url().startsWith(authServer.issuer() + END_SESSION_PATH);
384     }

386     /**
387      * A response the page received, captured when it arrived.
388      *
389      * @param method  method of the request.
390      * @param url     URL of the request.
391      * @param status  HTTP status.
392      * @param headers response headers, names in lower case (including {@code Set-Cookie}).
393      * @param body    the body, or a marker when the browser has none (redirects, navigated-away documents).
394      */
395     private record ObservedResponse(String method, String url, int status, Map<String, String> headers,
396             String body) {
397 
398         /**
399          * Captures a response.
400          *
401          * @param response the response.
402          * @return its recorded form.
403          */
404         static ObservedResponse of(final Response response) {
405             return new ObservedResponse(response.request().method(), response.url(), response.status(),
406                     Map.copyOf(response.allHeaders()), bodyOf(response));
407         }
408 
409         /**
410          * Reads the body of a response.
411          *
412          * @param response the response.
413          * @return the body, or a marker when the browser has none.
414          */
415         private static String bodyOf(final Response response) {
416             try {
417                 return response.text();
418             } catch (final PlaywrightException unavailable) {
419                 // redirects and navigated-away documents have no body to read: their headers are still scanned
420                 return "(no body)";
421             }
422         }
