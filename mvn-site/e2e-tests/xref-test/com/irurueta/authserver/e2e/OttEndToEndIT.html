35  /**
36   * End-to-end test of the one-time tokens with a real browser: the packaged {@code auth-server} (dev profile) runs on
37   * a random port against PostgreSQL and a Mailpit container that receives its e-mail. Headless Chromium visits a page
38   * that needs a session, is sent to the login page and signs in with the code the e-mail carries, which the test reads
39   * from the Mailpit HTTP API; in another scenario it signs in with the magic link, which a mail scanner can fetch
40   * (twice) without spending it; in the last one a signed-in user enrols a phone with the code of the SMS the logging
41   * sender recorded in the development inbox. Each scenario has a user of its own, copied from the demo user.
42   */
43  class OttEndToEndIT {

45      /** Longest wait for a page transition, in milliseconds. */
46      private static final double PAGE_TIMEOUT_MILLIS = 30_000;
47  
48      /** HTTP status of a successful response. */
49      private static final int HTTP_OK = 200;
50  
51      /** Path of the phone page, which needs a session. */
52      private static final String PHONE_PATH = "/account/phone";
53  
54      /** Path of the second-factor page. */
55      private static final String SECOND_FACTOR_PATH = "/login/second-factor";
56  
57      /** Path of the page where a code is typed. */
58      private static final String CODE_PATH = "/login/ott";
59  
60      /** Path of the development SMS inbox. */
61      private static final String INBOX_PATH = "/dev/sms-inbox";
62  
63      /** Selector of the button of the one-time token form of the login page. */
64      private static final String SEND_BUTTON = "form.ott-login button[type=submit]";
65  
66      /** Selector of the submit button of the page the browser lands on to type a code or confirm a link. */
67      private static final String SUBMIT_BUTTON = "main form button[type=submit]";
68  
69      /** Matches the code of the verification SMS in the HTML of the inbox. */
70      private static final Pattern SMS_CODE = Pattern.compile("Your verification code is ([0-9]+)");
71  
72      /** Longest wait for the verification SMS, which is sent after the request that asked for it, in milliseconds. */
73      private static final long SMS_WAIT_MILLIS = 15_000L;
74  
75      /** Pause between two looks at the development inbox, in milliseconds. */
76      private static final long SMS_POLL_MILLIS = 200L;

78      /** Number of digits of the national number of a generated Spanish mobile phone. */
79      private static final int PHONE_DIGITS = 8;

81      /** Number of values of a decimal digit. */
82      private static final int DECIMAL = 10;

94       * Starts Mailpit and the authorization server, whose mail sender points at it.

100         mailpit = Mailpit.newContainer();

103         authServer = RunningAuthServer.start(List.of(
104                 "--spring.mail.host=" + mailpit.getHost(),
105                 "--spring.mail.port=" + mailpit.getMappedPort(Mailpit.SMTP_PORT)));

119     /**
120      * Signs a user in with the code of an e-mail: the browser is sent to the login page by the page it asked for,
121      * asks for a code, types the one Mailpit received and lands on the page it originally asked for.
122      *
123      * @throws Exception if the user cannot be created.
124      */
125     @Test
126     void signIn_whenCodeFromEmailIsTyped_landsOnTheRequestedPage() throws Exception {
127         final var user = newUser();
128         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
129             final var page = newPage(browser);
130 
131             signInWithCode(page, user);
132 
133             assertTrue(page.textContent("h1").contains("Your phone"), page.url());
134         }
135     }

137     /**
138      * Signs a user in with the magic link of an e-mail: a scanner that fetches the link (twice) only gets the page that
139      * asks for a confirmation, and the browser that confirms signs in and lands on the page it asked for.
140      *
141      * @throws Exception if the user cannot be created.
142      */
143     @Test
144     void signIn_whenMagicLinkIsConfirmed_aScannerFetchDoesNotSpendIt() throws Exception {
145         final var user = newUser();
146         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
147             final var page = newPage(browser);
148             requestToken(page, user, "link");
149             final var link = mail.awaitMessageTo(user + "@example.com").link();
150 
151             final var scanner = playwright.request().newContext();
152             for (var scan = 0; scan < 2; scan++) {
153                 final var response = scanner.get(link);
154                 assertEquals(HTTP_OK, response.status());
155                 assertTrue(response.text().contains("Continue"));
156             }
157             page.navigate(link);
158             page.click(SUBMIT_BUTTON);
159             page.waitForURL(url -> url.startsWith(authServer.issuer() + PHONE_PATH));
160 
161             assertTrue(page.textContent("h1").contains("Your phone"), page.url());
162         }
163     }

165     /**
166      * Enrols a phone: a signed-in user types a number, reads the code of the verification SMS in the development inbox,
167      * types it, and is then asked for a second factor because the verified phone makes it required: the page offers the
168      * code by SMS, and the database holds the phone.
169      *
170      * @throws Exception if the user cannot be created or the database cannot be read.
171      */
172     @Test
173     void phoneEnrolment_whenCodeFromDevInboxIsTyped_verifiesThePhone() throws Exception {
174         final var user = newUser();
175         final var national = nationalNumber();
176         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
177             final var page = newPage(browser);
178             signInWithCode(page, user);
179 
180             page.fill("#phone-number", "+34 6" + national);
181             page.click("section.phone-enrol form[action$='/account/phone'] button[type=submit]");
182             page.waitForURL(url -> url.contains("status=pending"));
183             final var code = awaitCodeOf(playwright, "+346" + national);
184             page.fill("#phone-code", code);
185             page.click("section.phone-enrol form[action$='/verify'] button[type=submit]");
186             // the phone is now a confirmed second factor, so a session with only the e-mail code is asked for one
187             page.waitForURL(url -> url.equals(authServer.issuer() + SECOND_FACTOR_PATH));
188 
189             assertTrue(page.textContent("main").contains("Send me a code by SMS"));
190             assertEquals(List.of("1"), authServer.query("SELECT count(*) FROM user_phones WHERE phone_e164 = '+346"
191                     + national + "'"));
192         }
193     }

207     /**

219     /**
220      * Visits the phone page, which needs a session, and asks for a token on the login page it is sent to.
221      *
222      * @param page the page.
223      * @param user the name of the user.
224      * @param kind {@code code} or {@code link}.
225      */
226     private static void requestToken(final Page page, final String user, final String kind) {
227         page.navigate(authServer.issuer() + PHONE_PATH);
228         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
229         page.fill("#ott-username", user);
230         page.check("form.ott-login input[name=kind][value=" + kind + "]");
231         page.click(SEND_BUTTON);
232         page.waitForURL(url -> url.equals(authServer.issuer() + CODE_PATH));
233     }

235     /**
236      * Signs in with the code the e-mail carries and waits for the page the browser originally asked for.
237      *
238      * @param page the page.
239      * @param user the name of the user.
240      */
241     private static void signInWithCode(final Page page, final String user) {
242         requestToken(page, user, "code");
243         page.fill("#ott-code", mail.awaitMessageTo(user + "@example.com").code());
244         page.click(SUBMIT_BUTTON);
245         page.waitForURL(url -> url.startsWith(authServer.issuer() + PHONE_PATH));
246     }

248     /**
249      * Waits for a verification SMS sent to a number to show in the development inbox, and reads its code.
250      *
251      * @param playwright the Playwright instance, whose request context reads the inbox.
252      * @param phone      number the message was sent to, in E.164 form.
253      * @return the digits of the code of the last such message.
254      * @throws InterruptedException if the wait is interrupted.
255      */
256     private static String awaitCodeOf(final Playwright playwright, final String phone) throws InterruptedException {
257         final var deadline = System.currentTimeMillis() + SMS_WAIT_MILLIS;
258         final var requests = playwright.request().newContext();
259         var inbox = requests.get(authServer.issuer() + INBOX_PATH).text();
260         var code = lastCodeOf(inbox, phone);
261         while (code == null && System.currentTimeMillis() < deadline) {
262             Thread.sleep(SMS_POLL_MILLIS);
263             inbox = requests.get(authServer.issuer() + INBOX_PATH).text();
264             code = lastCodeOf(inbox, phone);
265         }
266         assertTrue(code != null, "The inbox holds no verification code for the number: " + inbox);
267         return code;
268     }

270     /**
271      * Reads the code of the last verification SMS sent to a number from the HTML of the development inbox.
272      *
273      * @param inbox HTML of the inbox.
274      * @param phone number the message was sent to, in E.164 form.
275      * @return the digits of the code, or {@code null} when the inbox holds none for the number.
276      */
277     private static String lastCodeOf(final String inbox, final String phone) {
278         String code = null;
279         for (final var row : inbox.split("<tr")) {
280             if (row.contains(phone)) {
281                 final var matcher = SMS_CODE.matcher(row);
282                 if (matcher.find()) {
283                     code = matcher.group(1);
284                 }
285             }
286         }
287         return code;
288     }

290     /**
291      * Generates the digits that follow the {@code 6} of a Spanish mobile number.
292      *
293      * @return the digits.
294      */
295     private static String nationalNumber() {
296         final var random = new SecureRandom();
297         final var builder = new StringBuilder();
298         for (var i = 0; i < PHONE_DIGITS; i++) {
299             builder.append(random.nextInt(DECIMAL));
300         }
301         return builder.toString();
302     }
