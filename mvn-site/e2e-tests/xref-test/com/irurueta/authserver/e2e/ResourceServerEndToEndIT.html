43  /**
44   * End-to-end test with a real token: the packaged {@code auth-server} and {@code resource-server} run on random
45   * ports against a PostgreSQL container, the demo user signs in with the authorization code flow and PKCE in
46   * headless Chromium (the redirect to the client is intercepted, no client application is needed), the code is
47   * exchanged for an access token, and that token is presented to the resource server. A token issued for another
48   * audience must be rejected. A second resource server runs the {@code opaque} profile and validates reference tokens
49   * through the authorization server's introspection endpoint.
50   */
51  class ResourceServerEndToEndIT {

128     /**
129      * Starts the authorization server, registers the test clients and starts the resource server.
130      *
131      * @throws Exception if an application or the container cannot be started.
132      */
133     @BeforeAll
134     static void startApplications() throws Exception {
135         resourcePort = RunningAuthServer.freePort();
136         opaquePort = RunningAuthServer.freePort();
137         // the seeded bff client may read documents as it is; only its redirect URI is the intercepted one
138         authServer = RunningAuthServer.start(List.of("--authserver.clients.bff.redirect-uris=" + REDIRECT_URI));
139 
140         // a copy of the bff client for another audience shows that the resource server tells the audiences apart;
141         // another copy gets reference tokens, for the resource server in its opaque profile
142         copyClient(OTHER_AUDIENCE_CLIENT, "replace(client_settings, 'resource-server', 'another-api')",
143                 "token_settings");
144         copyClient(REFERENCE_CLIENT, "client_settings", "replace(token_settings, 'self-contained', 'reference')");
145 
146         resourceServer = ManagedApp.start(RunningAuthServer.jar("resourceserver.jar"),
147                 List.of("--server.port=" + resourcePort), List.of("AUTH_ISSUER=" + authServer.issuer()));
148         // the opaque variant introspects with the bff client's own credentials: any confidential client will do
149         opaqueResourceServer = ManagedApp.start(RunningAuthServer.jar("resourceserver.jar"),
150                 List.of("--server.port=" + opaquePort, "--spring.profiles.active=opaque"),
151                 List.of("AUTH_ISSUER=" + authServer.issuer(),
152                         "AUTH_INTROSPECTION_URI=" + authServer.issuer() + "/oauth2/introspect",
153                         "INTROSPECTION_CLIENT_ID=" + ACCEPTED_CLIENT,
154                         "INTROSPECTION_CLIENT_SECRET=" + authServer.clientSecret()));
155         resourceServer.await(RESOURCE_SERVER_STARTED, RunningAuthServer.STARTUP_TIMEOUT);
156         opaqueResourceServer.await(RESOURCE_SERVER_STARTED, RunningAuthServer.STARTUP_TIMEOUT);
157     }

199     /**
200      * Verifies that the resource server in its {@code opaque} profile accepts a real reference token for its
201      * audience: the authorization server introspects it with its audience and roles, and rejects a JWT for another
202      * audience as well.
203      *
204      * @throws Exception if a flow step fails.
205      */
206     @Test
207     void documents_whenOpaqueProfileGetsAReferenceTokenForTheAudience_returnsOk() throws Exception {
208         final var token = obtainAccessToken(REFERENCE_CLIENT);
209 
210         assertFalse(token.contains("."), "a reference token is not a JWT");
211         assertEquals(HTTP_OK, listDocuments(opaqueUri(), token));
212         assertEquals(HTTP_UNAUTHORIZED, listDocuments(opaqueUri(), obtainAccessToken(OTHER_AUDIENCE_CLIENT)));
213     }

228     /**
229      * Runs the authorization code flow with PKCE for the demo user and returns the access token.
230      *
231      * @param clientId client that requests the token.
232      * @return the access token.
233      * @throws Exception if a flow step fails.
234      */
235     private static String obtainAccessToken(final String clientId) throws Exception {
236         final var verifier = randomVerifier();
237         final var authorizeUrl = authServer.issuer() + "/oauth2/authorize?response_type=code&client_id=" + clientId
238                 + "&redirect_uri=" + encode(REDIRECT_URI) + "&scope=" + encode("openid documents.read")
239                 + "&state=xyz&code_challenge=" + s256(verifier) + "&code_challenge_method=S256";
240         final String code;
241         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
242             code = signInAndCaptureCode(browser, authorizeUrl);
243         }
244 
245         final var credentials = Base64.getEncoder().encodeToString(
246                 (clientId + ":" + authServer.clientSecret()).getBytes(StandardCharsets.UTF_8));
247         final var request = HttpRequest.newBuilder(URI.create(authServer.issuer() + "/oauth2/token"))
248                 .header("Authorization", "Basic " + credentials)
249                 .header("Content-Type", "application/x-www-form-urlencoded")
250                 .POST(HttpRequest.BodyPublishers.ofString("grant_type=authorization_code&code=" + encode(code)
251                         + "&redirect_uri=" + encode(REDIRECT_URI) + "&code_verifier=" + verifier))
252                 .build();
253         final var response = HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString());
254         assertEquals(HTTP_OK, response.statusCode(), response.body());
255         final var matcher = ACCESS_TOKEN.matcher(response.body());
256         assertNotNull(matcher.find() ? matcher.group(1) : null, "the token endpoint answers with an access token");
257         return matcher.group(1);
258     }

260     /**
261      * Signs the demo user in and returns the authorization code that the server sends to the redirect URI. The
262      * request to that URI is intercepted, so no client has to listen there.
263      *
264      * @param browser      the browser.
265      * @param authorizeUrl authorization request URL.
266      * @return the authorization code.
267      */
268     private static String signInAndCaptureCode(final Browser browser, final String authorizeUrl) {
269         final var page = browser.newPage();
270         page.route(REDIRECT_URI + "**", Route::abort);
271         page.navigate(authorizeUrl);
272 
273         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
274         page.fill("#username", "demo");
275         page.fill("#password", authServer.demoPassword());
276         // the bff client requires consent (and so do the copies of it): allow every requested scope
277         final var redirect = page.waitForRequest(REDIRECT_URI + "**", () -> {
278             page.click("button[type=submit]");
279             allowWhenAsked(page);
280         });
281 
282         final var query = URI.create(redirect.url()).getRawQuery();
283         for (final var pair : query.split("&")) {
284             if (pair.startsWith("code=")) {
285                 return pair.substring("code=".length());
286             }
287         }
288         throw new IllegalStateException("The redirect carries no authorization code: " + redirect.url());
289     }

291     /**
292      * Allows the requested scopes when the consent page is shown. The page is skipped when the user already granted
293      * the scopes to the client in an earlier flow of this run.
294      *
295      * @param page the page of the browser, after the credentials were submitted.
296      */
297     private static void allowWhenAsked(final Page page) {
298         try {
299             page.waitForURL(url -> url.startsWith(authServer.issuer() + "/oauth2/consent"),
300                     new Page.WaitForURLOptions().setTimeout(CONSENT_WAIT_MILLIS));
301         } catch (final TimeoutError notAsked) {
302             return;
303         }
304         page.click("#allow");
305     }

318     /**
319      * Lists the documents of a resource server with a bearer token.
320      *
321      * @param uri   URI of the documents endpoint.
322      * @param token the access token.
323      * @return the HTTP status of the answer.
324      * @throws Exception if the request fails.
325      */
326     private static int listDocuments(final URI uri, final String token) throws Exception {
327         final var request = HttpRequest.newBuilder(uri).header("Authorization", "Bearer " + token)
328                 .GET().build();
329         return HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.discarding()).statusCode();
330     }

350     /**
351      * Registers a copy of the seeded {@code bff} client under another id, with the same secret.
352      *
353      * @param clientId       id of the copy.
354      * @param clientSettings SQL expression of the copy's client settings.
355      * @param tokenSettings  SQL expression of the copy's token settings.
356      * @throws Exception if the statement cannot be run.
357      */
358     private static void copyClient(final String clientId, final String clientSettings, final String tokenSettings)
359             throws Exception {
360         authServer.sql("INSERT INTO oauth2_registered_client (id, client_id, client_secret, client_name, "
361                 + "client_authentication_methods, authorization_grant_types, redirect_uris, scopes, "
362                 + "client_settings, token_settings) SELECT gen_random_uuid()::text, '" + clientId
363                 + "', client_secret, client_name, client_authentication_methods, authorization_grant_types, "
364                 + "redirect_uris, scopes, " + clientSettings + ", " + tokenSettings
365                 + " FROM oauth2_registered_client WHERE client_id = '" + ACCEPTED_CLIENT + "'");
366     }

368     /**
369      * Generates a PKCE code verifier.
370      *
371      * @return the verifier.
372      */
373     private static String randomVerifier() {
374         final var bytes = new byte[VERIFIER_BYTES];
375         new SecureRandom().nextBytes(bytes);
376         return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
377     }
378 
379     /**
380      * Computes the S256 PKCE code challenge of a verifier.
381      *
382      * @param verifier code verifier.
383      * @return the code challenge.
384      * @throws NoSuchAlgorithmException if SHA-256 is not available.
385      */
386     private static String s256(final String verifier) throws NoSuchAlgorithmException {
387         final var digest = MessageDigest.getInstance("SHA-256").digest(verifier.getBytes(StandardCharsets.US_ASCII));
388         return Base64.getUrlEncoder().withoutPadding().encodeToString(digest);
389     }
