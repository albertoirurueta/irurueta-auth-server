17  package com.irurueta.authserver.resourceserver.api.web;
18  
19  import io.swagger.v3.oas.models.Components;
20  import io.swagger.v3.oas.models.OpenAPI;
21  import io.swagger.v3.oas.models.info.Info;
22  import io.swagger.v3.oas.models.info.License;
23  import io.swagger.v3.oas.models.security.SecurityScheme;
24  import org.springdoc.core.models.GroupedOpenApi;
25  import org.springframework.context.annotation.Bean;
26  import org.springframework.context.annotation.Configuration;
27  
28  /**
29   * Runtime OpenAPI document of the HTTP API of the resource server, served by springdoc at {@value #API_DOCS_PATH}.
30   * The contract {@code apis/resource-server/rest-server/resource-server-api-v1.yaml} is the source of truth: the
31   * operations are described by the interface generated from it, and this class only adds what springdoc cannot read
32   * from it (the info block and the security scheme). {@code OpenApiDriftIT} fails when the document served differs
33   * from the contract.
34   */
35  @Configuration(proxyBeanMethods = false)
36  public class OpenApiConfig {
37  
38      /** Path of the OpenAPI document. */
39      public static final String API_DOCS_PATH = "/v3/api-docs";
40  
41      /** Name of the bearer-token security scheme. */
42      public static final String BEARER_SCHEME = "bearer";
43  
44      /**
45       * Describes the API.
46       *
47       * @return the OpenAPI info block.
48       */
49      @Bean
50      public OpenAPI resourceServerOpenApi() {
51          return new OpenAPI()
52                  .info(new Info()
53                          .title("Irurueta Resource Server API")
54                          .description("Protected REST API of the example resource server.")
55                          .version("1.0.0")
56                          .license(new License().name("Apache License 2.0")
57                                  .url("https://www.apache.org/licenses/LICENSE-2.0.txt")))
58                  .components(new Components().addSecuritySchemes(BEARER_SCHEME, new SecurityScheme()
59                          .type(SecurityScheme.Type.HTTP)
60                          .scheme("bearer")
61                          .bearerFormat("JWT")));
62      }
63  
64      /**
65       * Groups the application's own API paths.
66       *
67       * @return the group of API paths.
68       */
69      @Bean
70      public GroupedOpenApi apiGroup() {
71          return GroupedOpenApi.builder()
72                  .group("api")
73                  .pathsToMatch("/api/**")
74                  .build();
75      }
76  }
