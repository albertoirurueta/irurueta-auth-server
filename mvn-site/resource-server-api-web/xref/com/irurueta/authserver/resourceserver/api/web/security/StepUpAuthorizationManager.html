17  package com.irurueta.authserver.resourceserver.api.web.security;
18  
19  import java.time.Clock;
20  import java.time.Instant;
21  import java.util.Arrays;
22  import java.util.function.Supplier;
23  import lombok.RequiredArgsConstructor;
24  import org.springframework.security.authorization.AuthorizationDecision;
25  import org.springframework.security.authorization.AuthorizationManager;
26  import org.springframework.security.authorization.AuthorizationResult;
27  import org.springframework.security.core.Authentication;
28  import org.springframework.security.oauth2.server.resource.authentication.AbstractOAuth2TokenAuthenticationToken;
29  import org.springframework.security.web.access.intercept.RequestAuthorizationContext;
30  
31  /**
32   * Requires that the bearer token proves a recent strong authentication (RFC 9470): its {@code acr} claim must be
33   * one of the configured (space-separated) values and its {@code auth_time} claim no older than the configured maximum
34   * age. The claims are read from the token attributes, so it serves JWTs (their claims) and opaque tokens (the
35   * introspection answer) alike.
36   *
37   * <p>A caller without the {@code requiredAuthority} is denied with an ordinary decision (answered
38   * {@code 403 insufficient_scope}); a caller that has it but fails the step-up check is denied with a
39   * {@link StepUpRequiredDecision}, which {@link StepUpAccessDeniedHandler} answers with the RFC 9470 challenge.
40   */
41  // tag::resource-server-step-up[]
42  @RequiredArgsConstructor
43  public class StepUpAuthorizationManager implements AuthorizationManager<RequestAuthorizationContext> {
44  
45      /** Claim carrying the authentication context class reference. */
46      static final String ACR_CLAIM = "acr";
47  
48      /** Claim carrying the time of the user's authentication, in epoch seconds. */
49      static final String AUTH_TIME_CLAIM = "auth_time";
50  
51      /** Separator of the configured {@code acr-values}: any run of whitespace. */
52      private static final String ACR_VALUES_SEPARATOR = "\\s+";
53  
54      /** Authority the caller needs before the step-up requirement is evaluated. */
55      private final String requiredAuthority;
56  
57      /** Step-up policy. */
58      private final StepUpProperties properties;
59  
60      /** Source of the current time. */
61      private final Clock clock;
62  
63      @Override
64      public AuthorizationResult authorize(final Supplier<? extends Authentication> authentication,
65              final RequestAuthorizationContext context) {
66          final var current = authentication.get();
67          if (current == null || !current.isAuthenticated() || current.getAuthorities().stream()
68                  .noneMatch(authority -> requiredAuthority.equals(authority.getAuthority()))) {
69              return new AuthorizationDecision(false);
70          }
71          if (!properties.enabled() || satisfied(current)) {
72              return new AuthorizationDecision(true);
73          }
74          return new StepUpRequiredDecision();
75      }
76  
77      /**
78       * Tells whether the token proves the required authentication.
79       *
80       * @param authentication the authenticated bearer token.
81       * @return {@code true} when the {@code acr} matches and {@code auth_time} is recent enough.
82       */
83      private boolean satisfied(final Authentication authentication) {
84          if (!(authentication instanceof AbstractOAuth2TokenAuthenticationToken<?> token)) {
85              return false;
86          }
87          final var attributes = token.getTokenAttributes();
88          if (!isAccepted(attributes.get(ACR_CLAIM))) {
89              return false;
90          }
91          final var authenticatedAt = toInstant(attributes.get(AUTH_TIME_CLAIM));
92          return authenticatedAt != null
93                  && !authenticatedAt.plus(properties.maxAge()).isBefore(clock.instant());
94      }
95  
96      /**
97       * Tells whether an {@code acr} value is one of the configured ones. The configured {@code acr-values} is a
98       * space-separated list, like the {@code acr_values} parameter of OpenID Connect, so the token's single value must
99       * be a member of it, never equal to the whole list.
100      *
101      * @param acr the {@code acr} claim of the token.
102      * @return {@code true} when the claim is a text listed in the configured values.
103      */
104     private boolean isAccepted(final Object acr) {
105         return acr instanceof String value
106                 && Arrays.asList(properties.acrValues().trim().split(ACR_VALUES_SEPARATOR)).contains(value);
107     }
108 
109     /**
110      * Converts an {@code auth_time} value, which is epoch seconds in a JWT or an introspection answer.
111      *
112      * @param value the claim value.
113      * @return the instant, or {@code null} when the value is missing or of an unexpected type.
114      */
115     private static Instant toInstant(final Object value) {
116         return switch (value) {
117             case Instant instant -> instant;
118             case Number number -> Instant.ofEpochSecond(number.longValue());
119             case null, default -> null;
120         };
121     }
122 }
