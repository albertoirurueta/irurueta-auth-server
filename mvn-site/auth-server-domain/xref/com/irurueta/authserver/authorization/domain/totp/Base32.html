16  package com.irurueta.authserver.authorization.domain.totp;
17  
18  import java.nio.charset.StandardCharsets;
19  
20  /**
21   * Base32 encoding (RFC 4648) without padding, the form authenticator applications expect for a shared secret.
22   */
23  public final class Base32 {
24  
25      /** The Base32 alphabet. */
26      private static final String ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
27  
28      /** Bits carried by each byte. */
29      private static final int BITS_PER_BYTE = 8;
30  
31      /** Bits carried by each character. */
32      private static final int BITS_PER_CHAR = 5;
33  
34      /** Mask selecting the bits of one byte, to read it as unsigned. */
35      private static final int BYTE_MASK = 0xff;
36  
37      /** Mask selecting the bits of one character. */
38      private static final int CHAR_MASK = 0x1f;
39  
40      /**
41       * Not instantiable: a holder of static helpers.
42       */
43      private Base32() {
44      }
45  
46      /**
47       * Encodes bytes.
48       *
49       * @param data the bytes to encode; must not be {@code null}.
50       * @return the encoded text, upper case and without padding.
51       */
52      public static String encode(final byte[] data) {
53          final var result = new StringBuilder((data.length * BITS_PER_BYTE + BITS_PER_CHAR - 1) / BITS_PER_CHAR);
54          var buffer = 0;
55          var bits = 0;
56          for (final var value : data) {
57              buffer = buffer << BITS_PER_BYTE | value & BYTE_MASK;
58              bits += BITS_PER_BYTE;
59              while (bits >= BITS_PER_CHAR) {
60                  bits -= BITS_PER_CHAR;
61                  result.append(ALPHABET.charAt(buffer >> bits & CHAR_MASK));
62              }
63          }
64          if (bits > 0) {
65              result.append(ALPHABET.charAt(buffer << BITS_PER_CHAR - bits & CHAR_MASK));
66          }
67          return new String(result.toString().getBytes(StandardCharsets.US_ASCII), StandardCharsets.US_ASCII);
68      }
69  }
