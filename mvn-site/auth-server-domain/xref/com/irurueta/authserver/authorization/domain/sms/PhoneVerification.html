16  package com.irurueta.authserver.authorization.domain.sms;
17  
18  import static com.irurueta.authserver.authorization.domain.Preconditions.requireText;
19  
20  import java.time.Instant;
21  import java.util.UUID;
22  
23  /**
24   * A pending verification of a phone number: the code that was sent to it, hashed, until the user types it.
25   *
26   * @param userId    identifier of the user verifying the number.
27   * @param phone     the number being verified.
28   * @param codeHash  hash of the code sent, never the code itself.
29   * @param attempts  number of wrong submissions so far.
30   * @param expiresAt instant after which the code is no longer valid.
31   */
32  public record PhoneVerification(UUID userId, PhoneNumber phone, String codeHash, int attempts, Instant expiresAt) {
33  
34      /**
35       * Validates the verification.
36       *
37       * @throws IllegalArgumentException if the user, number or expiry is missing, the hash is blank or the attempts
38       *                                  are negative.
39       */
40      public PhoneVerification {
41          if (userId == null) {
42              throw new IllegalArgumentException("userId must not be null");
43          }
44          if (phone == null) {
45              throw new IllegalArgumentException("phone must not be null");
46          }
47          requireText(codeHash, "codeHash");
48          if (attempts < 0) {
49              throw new IllegalArgumentException("attempts must not be negative");
50          }
51          if (expiresAt == null) {
52              throw new IllegalArgumentException("expiresAt must not be null");
53          }
54      }
55  
56      /**
57       * Tells whether the verification has expired.
58       *
59       * @param now current instant.
60       * @return {@code true} when it expires at or before {@code now}.
61       */
62      public boolean isExpiredAt(final Instant now) {
63          return !expiresAt.isAfter(now);
64      }
65  }
