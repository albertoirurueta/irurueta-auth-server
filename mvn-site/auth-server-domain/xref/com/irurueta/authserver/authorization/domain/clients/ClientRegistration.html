17  package com.irurueta.authserver.authorization.domain.clients;
18  
19  import static com.irurueta.authserver.authorization.domain.Preconditions.requireText;
20  import static com.irurueta.authserver.authorization.domain.Preconditions.requireTextIfPresent;
21  
22  import java.time.Duration;
23  import java.util.List;
24  import java.util.Objects;
25  
26  /**
27   * An OAuth 2.1 client registered with the authorization server. The client secret is not part of the model: it is
28   * only ever handled encoded, through the repository.
29   *
30   * @param clientId           public identifier of the client.
31   * @param name               human readable name.
32   * @param confidential       {@code true} for a client able to keep a secret, {@code false} for a public client.
33   * @param authMethods        client authentication methods, for example {@code client_secret_basic} or {@code none}.
34   * @param grantTypes         grant types the client may use, for example {@code authorization_code}.
35   * @param redirectUris       redirect URIs the client may use.
36   * @param scopes             scopes the client may request.
37   * @param requireProofKey    whether the authorization code flow must use PKCE.
38   * @param audiences          logical identifiers of the APIs the access tokens are meant for (the {@code aud} claim);
39   *                           the client id when none is given.
40   * @param accessTokenTtl     lifetime of access tokens.
41   * @param refreshTokenTtl    lifetime of refresh tokens.
42   * @param accessTokenFormat  {@code self-contained} or {@code reference}.
43   * @param reuseRefreshTokens {@code true} to keep a refresh token after use, {@code false} to rotate it.
44   * @param postLogoutRedirectUris URIs an end-user may be redirected to after an OpenID Connect RP-initiated logout;
45   *                           a {@code post_logout_redirect_uri} that is not listed is never honoured.
46   * @param requireConsent     whether the end-user must explicitly grant the requested scopes on a consent page.
47   * @param mfaRequired        whether every user signing in through the client must complete a second factor.
48   * @param jwkSetUrl          URL of the client's public JWK set, used to verify {@code private_key_jwt} assertions and
49   *                           self-signed certificates; {@code null} when none.
50   * @param jwks               the client's registered public JWK set as JSON (public keys only), alternative to
51   *                           {@code jwkSetUrl}; {@code null} when none.
52   * @param tokenEndpointSigningAlgorithm JWS algorithm the client signs its {@code private_key_jwt} assertions with;
53   *                           {@code null} when unrestricted.
54   * @param tlsClientAuthSubjectDn expected subject distinguished name of the certificate of a
55   *                           {@code tls_client_auth} client; {@code null} when none.
56   * @param certificateBoundAccessTokens whether access tokens are bound to the client certificate ({@code cnf.x5t#S256}).
57   */
58  public record ClientRegistration(
59          String clientId, String name, boolean confidential, List<String> authMethods, List<String> grantTypes,
60          List<String> redirectUris, List<String> scopes, boolean requireProofKey, List<String> audiences,
61          Duration accessTokenTtl, Duration refreshTokenTtl, String accessTokenFormat, boolean reuseRefreshTokens,
62          List<String> postLogoutRedirectUris, boolean requireConsent, boolean mfaRequired, String jwkSetUrl, String jwks,
63          String tokenEndpointSigningAlgorithm, String tlsClientAuthSubjectDn, boolean certificateBoundAccessTokens) {
64  
65      /** Access token format of self-contained (JWT) tokens. */
66      public static final String FORMAT_SELF_CONTAINED = "self-contained";
67  
68      /** Grant type of the legacy implicit flow. */
69      public static final String GRANT_IMPLICIT = "implicit";
70  
71      /** Grant type of refresh tokens. */
72      public static final String GRANT_REFRESH_TOKEN = "refresh_token";
73  
74      /** Authentication method of a client that proves possession of a registered private key. */
75      public static final String AUTH_PRIVATE_KEY_JWT = "private_key_jwt";
76  
77      /** Authentication method of mutual TLS with a certificate issued by a trusted authority. */
78      public static final String AUTH_TLS_CLIENT_AUTH = "tls_client_auth";
79  
80      /** Authentication method of mutual TLS with a self-signed certificate. */
81      public static final String AUTH_SELF_SIGNED_TLS_CLIENT_AUTH = "self_signed_tls_client_auth";
82  
83      /** Prefix of the authentication methods that use a client secret. */
84      public static final String AUTH_CLIENT_SECRET_PREFIX = "client_secret_";
85  
86      /** Access token format of opaque, reference tokens. */
87      public static final String FORMAT_REFERENCE = "reference";
88  
89      /**
90       * Validates the registration and defensively copies its collections.
91       *
92       * @throws IllegalArgumentException if the id, name or format is blank, a collection is null or has a null
93       *                                  element, a lifetime is missing or not positive, or a public client does not
94       *                                  require PKCE (unless its only grant type is {@code implicit}), an
95       *                                  {@code implicit} client is confidential, requires consent or has refresh
96       *                                  tokens, a {@code tls_client_auth} client has no subject DN, a
97       *                                  {@code private_key_jwt} or {@code self_signed_tls_client_auth} client has
98       *                                  neither {@code jwks} nor {@code jwkSetUrl}, or a given optional text is blank.
99       */
100     public ClientRegistration {
101         requireText(clientId, "clientId");
102         requireText(name, "name");
103         requireText(accessTokenFormat, "accessTokenFormat");
104         requireNoNulls(authMethods, "authMethods");
105         authMethods = List.copyOf(authMethods);
106         requireNoNulls(grantTypes, "grantTypes");
107         grantTypes = List.copyOf(grantTypes);
108         requireNoNulls(redirectUris, "redirectUris");
109         redirectUris = List.copyOf(redirectUris);
110         requireNoNulls(scopes, "scopes");
111         scopes = List.copyOf(scopes);
112         requireNoNulls(audiences, "audiences");
113         audiences = List.copyOf(audiences);
114         if (audiences.isEmpty()) {
115             audiences = List.of(clientId);
116         }
117         requireNoNulls(postLogoutRedirectUris, "postLogoutRedirectUris");
118         postLogoutRedirectUris = List.copyOf(postLogoutRedirectUris);
119         if (isNotPositive(accessTokenTtl) || isNotPositive(refreshTokenTtl)) {
120             throw new IllegalArgumentException("token lifetimes must be positive");
121         }
122         requireTextIfPresent(jwkSetUrl, "jwkSetUrl");
123         requireTextIfPresent(jwks, "jwks");
124         requireTextIfPresent(tokenEndpointSigningAlgorithm, "tokenEndpointSigningAlgorithm");
125         requireTextIfPresent(tlsClientAuthSubjectDn, "tlsClientAuthSubjectDn");
126         if (!confidential && !requireProofKey && !grantTypes.equals(List.of(GRANT_IMPLICIT))) {
127             throw new IllegalArgumentException("a public client must require PKCE");
128         }
129         if (grantTypes.contains(GRANT_IMPLICIT)) {
130             requireValidImplicit(confidential, requireConsent, grantTypes);
131         }
132         if (authMethods.contains(AUTH_TLS_CLIENT_AUTH) && tlsClientAuthSubjectDn == null) {
133             throw new IllegalArgumentException("tls_client_auth requires a subject DN");
134         }
135         if ((authMethods.contains(AUTH_PRIVATE_KEY_JWT) || authMethods.contains(AUTH_SELF_SIGNED_TLS_CLIENT_AUTH))
136                 && jwks == null && jwkSetUrl == null) {
137             throw new IllegalArgumentException(
138                     "private_key_jwt and self_signed_tls_client_auth require jwks or jwkSetUrl");
139         }
140         if (authMethods.contains(AUTH_PRIVATE_KEY_JWT) && jwks != null && tokenEndpointSigningAlgorithm == null) {
141             throw new IllegalArgumentException(
142                     "private_key_jwt with registered jwks requires tokenEndpointSigningAlgorithm");
143         }
144     }
145 
146     /**
147      * Tells whether the client may use the legacy implicit grant, which only the legacy demo client seeded under the
148      * {@code legacy-implicit} profile may have.
149      *
150      * @return {@code true} when the grant types contain {@code implicit}.
151      */
152     public boolean usesImplicit() {
153         return grantTypes.contains(GRANT_IMPLICIT);
154     }
155 
156     /**
157      * Tells whether the client authenticates with a client secret.
158      *
159      * @return {@code true} when any authentication method is a {@code client_secret_*} one.
160      */
161     public boolean usesClientSecret() {
162         return authMethods.stream().anyMatch(method -> method.startsWith(AUTH_CLIENT_SECRET_PREFIX));
163     }
164 
165     /**
166      * Refuses an {@code implicit} client that is confidential, requires consent or may use refresh tokens.
167      *
168      * @param confidential whether the client is confidential.
169      * @param requireConsent whether consent is required.
170      * @param grantTypes the grant types of the client.
171      */
172     private static void requireValidImplicit(
173             final boolean confidential, final boolean requireConsent, final List<String> grantTypes) {
174         if (confidential) {
175             throw new IllegalArgumentException("an implicit client must be public");
176         }
177         if (requireConsent) {
178             throw new IllegalArgumentException("an implicit client cannot require consent");
179         }
180         if (grantTypes.contains(GRANT_REFRESH_TOKEN)) {
181             throw new IllegalArgumentException("an implicit client cannot have refresh tokens");
182         }
183     }
184 
185     /**
186      * Tells whether a duration is missing, zero or negative.
187      *
188      * @param duration the duration.
189      * @return {@code true} when it is not a positive duration.
190      */
191     private static boolean isNotPositive(final Duration duration) {
192         return duration == null || duration.isZero() || duration.isNegative();
193     }
194 
195     /**
196      * Rejects a null list and a list with null elements.
197      *
198      * @param values the list.
199      * @param name   name of the list, for the message.
200      */
201     private static void requireNoNulls(final List<String> values, final String name) {
202         if (values == null || values.stream().anyMatch(Objects::isNull)) {
203             throw new IllegalArgumentException(name + " must not be null or contain null elements");
204         }
205     }
