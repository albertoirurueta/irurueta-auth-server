17  package com.irurueta.authserver.authorization.domain.clients;
18  
19  import static com.irurueta.authserver.authorization.domain.Preconditions.requireText;
20  
21  import java.time.Duration;
22  import java.util.List;
23  import java.util.Objects;
24  
25  /**
26   * An OAuth 2.1 client registered with the authorization server. The client secret is not part of the model: it is
27   * only ever handled encoded, through the repository.
28   *
29   * @param clientId           public identifier of the client.
30   * @param name               human readable name.
31   * @param confidential       {@code true} for a client able to keep a secret, {@code false} for a public client.
32   * @param authMethods        client authentication methods, for example {@code client_secret_basic} or {@code none}.
33   * @param grantTypes         grant types the client may use, for example {@code authorization_code}.
34   * @param redirectUris       redirect URIs the client may use.
35   * @param scopes             scopes the client may request.
36   * @param requireProofKey    whether the authorization code flow must use PKCE.
37   * @param audiences          logical identifiers of the APIs the access tokens are meant for (the {@code aud} claim);
38   *                           the client id when none is given.
39   * @param accessTokenTtl     lifetime of access tokens.
40   * @param refreshTokenTtl    lifetime of refresh tokens.
41   * @param accessTokenFormat  {@code self-contained} or {@code reference}.
42   * @param reuseRefreshTokens {@code true} to keep a refresh token after use, {@code false} to rotate it.
43   * @param postLogoutRedirectUris URIs an end-user may be redirected to after an OpenID Connect RP-initiated logout;
44   *                           a {@code post_logout_redirect_uri} that is not listed is never honoured.
45   * @param requireConsent     whether the end-user must explicitly grant the requested scopes on a consent page.
46   * @param mfaRequired        whether every user signing in through the client must complete a second factor.
47   */
48  public record ClientRegistration(
49          String clientId, String name, boolean confidential, List<String> authMethods, List<String> grantTypes,
50          List<String> redirectUris, List<String> scopes, boolean requireProofKey, List<String> audiences,
51          Duration accessTokenTtl, Duration refreshTokenTtl, String accessTokenFormat, boolean reuseRefreshTokens,
52          List<String> postLogoutRedirectUris, boolean requireConsent, boolean mfaRequired) {
53  
54      /** Access token format of self-contained (JWT) tokens. */
55      public static final String FORMAT_SELF_CONTAINED = "self-contained";
56  
57      /** Access token format of opaque, reference tokens. */
58      public static final String FORMAT_REFERENCE = "reference";
59  
60      /**
61       * Validates the registration and defensively copies its collections.
62       *
63       * @throws IllegalArgumentException if the id, name or format is blank, a collection is null or has a null
64       *                                  element, a lifetime is missing or not positive, or a public client does not
65       *                                  require PKCE.
66       */
67      public ClientRegistration {
68          requireText(clientId, "clientId");
69          requireText(name, "name");
70          requireText(accessTokenFormat, "accessTokenFormat");
71          requireNoNulls(authMethods, "authMethods");
72          authMethods = List.copyOf(authMethods);
73          requireNoNulls(grantTypes, "grantTypes");
74          grantTypes = List.copyOf(grantTypes);
75          requireNoNulls(redirectUris, "redirectUris");
76          redirectUris = List.copyOf(redirectUris);
77          requireNoNulls(scopes, "scopes");
78          scopes = List.copyOf(scopes);
79          requireNoNulls(audiences, "audiences");
80          audiences = List.copyOf(audiences);
81          if (audiences.isEmpty()) {
82              audiences = List.of(clientId);
83          }
84          requireNoNulls(postLogoutRedirectUris, "postLogoutRedirectUris");
85          postLogoutRedirectUris = List.copyOf(postLogoutRedirectUris);
86          if (isNotPositive(accessTokenTtl) || isNotPositive(refreshTokenTtl)) {
87              throw new IllegalArgumentException("token lifetimes must be positive");
88          }
89          if (!confidential && !requireProofKey) {
90              throw new IllegalArgumentException("a public client must require PKCE");
91          }
92      }
93  
94      /**
95       * Tells whether a duration is missing, zero or negative.
96       *
97       * @param duration the duration.
98       * @return {@code true} when it is not a positive duration.
99       */
100     private static boolean isNotPositive(final Duration duration) {
101         return duration == null || duration.isZero() || duration.isNegative();
102     }
103 
104     /**
105      * Rejects a null list and a list with null elements.
106      *
107      * @param values the list.
108      * @param name   name of the list, for the message.
109      */
110     private static void requireNoNulls(final List<String> values, final String name) {
111         if (values == null || values.stream().anyMatch(Objects::isNull)) {
112             throw new IllegalArgumentException(name + " must not be null or contain null elements");
113         }
114     }
115 }
