17  package com.irurueta.authserver.authorization.application.sms;
18  
19  import com.irurueta.authserver.authorization.domain.sms.PhoneCountryResolver;
20  import com.irurueta.authserver.authorization.domain.sms.PhoneNumber;
21  import com.irurueta.authserver.authorization.domain.sms.SmsCostLedgerRepository;
22  import com.irurueta.authserver.authorization.domain.sms.SmsCostPolicy;
23  import java.time.Clock;
24  import lombok.RequiredArgsConstructor;
25  import org.springframework.context.ApplicationEventPublisher;
26  import org.springframework.stereotype.Service;
27  import org.springframework.transaction.annotation.Transactional;
28  
29  /**
30   * Requests an SMS inside the cost policy: the country must be served and the spending of the window and of the day,
31   * including this message, must stay within their caps.
32   *
33   * <p>The cost is reserved by {@link SmsCostLedgerRepository#tryRecord}, which checks both caps and records the message
34   * in one atomic step, so concurrent senders cannot overshoot a budget and a refused message costs nothing. The message
35   * itself is not sent here: an {@link SmsRequested} event is published and sent after the transaction commits, off the
36   * request thread, so the gateway is never called inside a transaction and a rolled-back request sends nothing. A
37   * refusal does not roll the surrounding transaction back ({@code noRollbackFor}): this use case is joined by its
38   * callers, and a rollback-only mark here would undo what they saved.
39   */
40  @Service
41  @Transactional(noRollbackFor = SmsRefusedException.class)
42  @RequiredArgsConstructor
43  public class SendSmsOneTimeCodeUseCase {
44  
45      /** Port that finds the country of a number. */
46      private final PhoneCountryResolver phoneCountryResolver;
47  
48      /** Policy of countries, costs and caps. */
49      private final SmsCostPolicy smsCostPolicy;
50  
51      /** Port that records the spend. */
52      private final SmsCostLedgerRepository smsCostLedgerRepository;
53  
54      /** Publishes the request to send the message. */
55      private final ApplicationEventPublisher eventPublisher;
56  
57      /** Source of the current time. */
58      private final Clock clock;
59  
60      /**
61       * Reserves the cost of a message and requests it.
62       *
63       * @param to   destination number.
64       * @param text text of the message.
65       * @throws SmsCountryNotAllowedException if the country of the number is unknown or not served.
66       * @throws SmsBudgetExceededException    if the message would exceed the cap of the window or of the day.
67       */
68      public void execute(final PhoneNumber to, final String text) {
69          final var country = phoneCountryResolver.countryOf(to)
70                  .filter(smsCostPolicy::isCountryAllowed)
71                  .orElseThrow(SmsCountryNotAllowedException::new);
72          if (!smsCostLedgerRepository.tryRecord(clock.instant(), country, smsCostPolicy.costMicros(country),
73                  smsCostPolicy.windowCap(), smsCostPolicy.dailyCap())) {
74              throw new SmsBudgetExceededException();
75          }
76          eventPublisher.publishEvent(new SmsRequested(to, text));
77      }
78  }
