16  package com.irurueta.authserver.authorization.application.totp;
17  
18  import com.irurueta.authserver.authorization.application.mfa.CheckSecondFactorLockoutUseCase;
19  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
20  import com.irurueta.authserver.authorization.application.mfa.RegisterSecondFactorFailureUseCase;
21  import com.irurueta.authserver.authorization.application.mfa.ResetSecondFactorFailuresUseCase;
22  import com.irurueta.authserver.authorization.domain.mfa.SecondFactorLockedException;
23  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
24  import java.util.List;
25  import java.util.Optional;
26  import java.util.UUID;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.stereotype.Service;
29  import org.springframework.transaction.annotation.Transactional;
30  
31  /**
32   * Use case that confirms the enrolment of an authenticator app inside the sign-in, where the code also stands as the
33   * second factor of that sign-in: like {@link VerifyTotpUseCase}, a locked second factor is refused, a wrong code
34   * counts towards the lockout and a right one clears the counter and is audited.
35   */
36  @Service
37  @Transactional
38  @RequiredArgsConstructor
39  public class ConfirmTotpAtSignInUseCase {
40  
41      /** Use case that confirms the enrolment. */
42      private final ConfirmTotpUseCase confirmTotpUseCase;
43  
44      /** Use case that tells whether the second factor is locked (and serializes the attempts of a user). */
45      private final CheckSecondFactorLockoutUseCase checkSecondFactorLockoutUseCase;
46  
47      /** Use case that counts a failure. */
48      private final RegisterSecondFactorFailureUseCase registerSecondFactorFailureUseCase;
49  
50      /** Use case that clears the failures. */
51      private final ResetSecondFactorFailuresUseCase resetSecondFactorFailuresUseCase;
52  
53      /** Use case that records the audit events. */
54      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
55  
56      // tag::totp-sign-in-enrolment[]
57      /**
58       * Confirms the enrolment with the first code of the authenticator app.
59       *
60       * @param userId   identifier of the user.
61       * @param username name of the user, for the audit log.
62       * @param code     the code as typed.
63       * @return the clear-text recovery codes, to be shown once; empty when the code is wrong (the failure is counted).
64       * @throws SecondFactorLockedException if the second factor of the user is locked.
65       * @throws com.irurueta.authserver.authorization.domain.totp.TotpNotEnrolledException    if no enrolment was
66       *                                                                                      started.
67       * @throws com.irurueta.authserver.authorization.domain.totp.TotpAlreadyEnabledException if the enrolment was
68       *                                                                                      already confirmed.
69       */
70      public Optional<List<String>> execute(final UUID userId, final String username, final String code) {
71          if (checkSecondFactorLockoutUseCase.execute(userId)) {
72              throw new SecondFactorLockedException("The second factor is locked after too many failed attempts");
73          }
74          final var recoveryCodes = confirmTotpUseCase.tryConfirm(userId, code);
75          if (recoveryCodes.isPresent()) {
76              resetSecondFactorFailuresUseCase.execute(userId);
77              recordSecurityEventUseCase.execute(SecurityEventType.SECOND_FACTOR_SUCCESS, userId, username,
78                      "totp enrolment");
79          } else {
80              registerSecondFactorFailureUseCase.execute(userId, username);
81          }
82          return recoveryCodes;
83      }
84      // end::totp-sign-in-enrolment[]
85  }
