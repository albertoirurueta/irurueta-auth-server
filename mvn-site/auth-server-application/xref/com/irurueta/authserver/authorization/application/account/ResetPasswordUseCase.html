22  import com.irurueta.authserver.authorization.domain.account.InvalidAccountTokenException;
23  import com.irurueta.authserver.authorization.domain.account.TokenGenerator;
24  import com.irurueta.authserver.authorization.domain.password.CompromisedPasswordCheck;
25  import com.irurueta.authserver.authorization.domain.password.CompromisedPasswordException;

27  import com.irurueta.authserver.authorization.domain.password.PasswordHasher;
28  import com.irurueta.authserver.authorization.domain.password.PasswordPolicy;
29  import com.irurueta.authserver.authorization.domain.password.UserAccountRepository;
30  import java.time.Clock;
31  import lombok.RequiredArgsConstructor;
32  import org.springframework.stereotype.Service;
33  import org.springframework.transaction.annotation.Transactional;
34  
35  /**
36   * Use case that chooses a new password with the single-use token received by e-mail. Every check (token, account,
37   * password policy including the username rule, compromised-password check) runs before the token is consumed, so a
38   * rejected password never burns the token; consuming it, storing the new hash and revoking every token of the user
39   * then happen in one transaction.
40   */
41  @Service
42  @RequiredArgsConstructor
43  public class ResetPasswordUseCase {
44  
45      /** Port that stores the accounts. */
46      private final UserAccountRepository userAccountRepository;
47  
48      /** Port that stores the reset tokens. */
49      private final AccountTokenRepository accountTokenRepository;
50  
51      /** Port that creates and hashes tokens. */
52      private final TokenGenerator tokenGenerator;
53  
54      /** Port that hashes passwords. */
55      private final PasswordHasher passwordHasher;
56  
57      /** Port that checks passwords against leaked ones. */
58      private final CompromisedPasswordCheck compromisedPasswordCheck;
59  
60      /** Use case that revokes the tokens of the user. */
61      private final RevokeUserAuthorizationsUseCase revokeUserAuthorizationsUseCase;
62  
63      /** Source of the current time. */
64      private final Clock clock;
65  
66      /** Rules a new password must follow. */
67      private final PasswordPolicy passwordPolicy;
68  
69      // tag::account-reset-password[]
70      /**
71       * Resets the password of the user the token was issued to. The token is looked up without being consumed, every
72       * check runs, and only then is the token consumed; the consumption is atomic, so of two concurrent resets with
73       * the same token only one succeeds.
74       *
75       * @param token       the reset token received by e-mail.
76       * @param newPassword the clear-text new password.
77       * @throws InvalidPasswordException     if the password breaks the policy.
78       * @throws CompromisedPasswordException if the password is known to be compromised.
79       * @throws InvalidAccountTokenException if the token is unknown, expired, used or its account is gone.
80       */
81      @Transactional
82      public void execute(final String token, final String newPassword) {
83          final var hashedToken = tokenGenerator.hash(token);
84          final var now = clock.instant();
85          final var stored = accountTokenRepository
86                  .findUsable(hashedToken, AccountToken.PURPOSE_PASSWORD_RESET, now)
87                  .orElseThrow(InvalidAccountTokenException::new);
88          final var account = userAccountRepository.findById(stored.userId())
89                  .orElseThrow(InvalidAccountTokenException::new);
90          requirePolicy(account.username(), newPassword);
91          requireNotCompromised(newPassword);
92          accountTokenRepository.consume(hashedToken, AccountToken.PURPOSE_PASSWORD_RESET, now)
93                  .orElseThrow(InvalidAccountTokenException::new);
94          userAccountRepository.updatePassword(account.id(), passwordHasher.encode(newPassword));
95          revokeUserAuthorizationsUseCase.execute(account.username());
96      }
97      // end::account-reset-password[]
98  
99      /**
100      * Rejects a password that breaks the policy.
101      *
102      * @param username name of the account the password is for.
103      * @param password the clear-text password.
104      */
105     private void requirePolicy(final String username, final String password) {
106         final var violations = passwordPolicy.violations(username, password);
107         if (!violations.isEmpty()) {
108             throw new InvalidPasswordException(violations);
109         }
110     }
111 
112     /**
113      * Rejects a compromised password.
114      *
115      * @param password the clear-text password.
116      */
117     private void requireNotCompromised(final String password) {
118         if (compromisedPasswordCheck.isCompromised(password)) {
119             throw new CompromisedPasswordException("The chosen password is known to be compromised");
120         }
121     }
122 }
