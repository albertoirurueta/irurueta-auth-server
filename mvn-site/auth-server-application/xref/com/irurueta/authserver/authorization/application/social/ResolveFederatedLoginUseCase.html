39  /**
40   * Use case that resolves a sign-in with an external provider to a local account. The provider's stable subject, never
41   * its e-mail address, finds the account. The identity is locked first
42   * ({@link FederatedIdentityRepository#lockSubject}), so two concurrent first sign-ins of the same person are
43   * serialized and the second one finds the link of the first instead of failing on the unique constraint. In order:
44   * <ol>
45   *     <li>an identity that is already linked signs in its account, unless the account is disabled;</li>
46   *     <li>otherwise the provider must vouch for the e-mail address: an unverified address never links and never
47   *     provisions;</li>
48   *     <li>an account that has that address and has verified it too is linked to the identity, unless it is disabled:
49   *     a disabled account is refused before anything is linked or audited;</li>
50   *     <li>an address that belongs to an account that never verified it is refused, so a pre-registered account cannot
51   *     capture the victim's sign-in;</li>
52   *     <li>otherwise a new, enabled account is provisioned, whose password is a random one nobody knows.</li>
53   * </ol>
54   */
55  @Service
56  @Transactional
57  @RequiredArgsConstructor
58  public class ResolveFederatedLoginUseCase {
59  
60      /** Longest username the {@code users} table accepts. */
61      static final int MAX_USERNAME_LENGTH = 100;
62  
63      /** Port that stores the links. */
64      private final FederatedIdentityRepository federatedIdentityRepository;
65  
66      /** Port that stores the accounts. */
67      private final UserAccountRepository userAccountRepository;
68  
69      /** Port that remembers the accounts that only hold a generated password. */
70      private final GeneratedPasswordRepository generatedPasswordRepository;
71  
72      /** Port that hashes the generated password. */
73      private final PasswordHasher passwordHasher;
74  
75      /** Port that generates the unguessable secret the generated password is made of. */
76      private final TokenGenerator tokenGenerator;
77  
78      /** Rules of linking and provisioning. */
79      private final FederatedLinkingPolicy federatedLinkingPolicy;
80  
81      /** Use case that records the audit events. */
82      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
83  
84      /** Source of the current time. */
85      private final Clock clock;
86  
87      /**
88       * Resolves a sign-in.
89       *
90       * @param provider      provider that authenticated the user.
91       * @param subject       stable identifier the provider assigned to the user.
92       * @param email         e-mail address the provider reported, or {@code null}.
93       * @param emailVerified whether the provider says it verified the address.
94       * @return the account the sign-in resolved to and how it was found.
95       * @throws FederatedLoginRefusedException if the identity is not linked and the provider does not vouch for a usable
96       *                                        e-mail address, the address belongs to an unverified account, or the
97       *                                        account is disabled.
98       */
99      public FederatedLogin execute(
100             final IdentityProvider provider, final String subject, final String email, final boolean emailVerified) {
101         final var now = clock.instant();
102         federatedIdentityRepository.lockSubject(provider, subject);
103         final var existing = federatedIdentityRepository.findByProviderAndSubject(provider, subject);
104         if (existing.isPresent()) {
105             return signIn(existing.get(), email, now);
106         }
107         if (!federatedLinkingPolicy.mayProvision(email, emailVerified)) {
108             throw new FederatedLoginRefusedException("The provider did not verify an e-mail address");
109         }
110         final var sameAddress = userAccountRepository.findByEmail(email);
111         if (sameAddress.isPresent()) {
112             final var account = sameAddress.get();
113             if (!federatedLinkingPolicy.mayLinkByEmail(account, email, emailVerified)) {
114                 throw new FederatedLoginRefusedException("The e-mail address belongs to an unverified account");
115             }
116             requireMaySignIn(account);
117             return new FederatedLogin(account, link(account, provider, subject, email, now),
118                     FederatedLoginOutcome.LINKED);
119         }
120         final var account = provision(provider, email);
121         return new FederatedLogin(account, link(account, provider, subject, email, now),
122                 FederatedLoginOutcome.PROVISIONED);
123     }
124 
125     /**
126      * Signs in the account of an identity that is already linked, remembering the sign-in.
127      *
128      * @param identity the stored link.
129      * @param email    e-mail address the provider reported now, or {@code null}.
130      * @param now      current instant.
131      * @return the resolution.
132      */
133     private FederatedLogin signIn(final FederatedIdentity identity, final String email, final Instant now) {
134         final var account = userAccountRepository.findById(identity.userId())
135                 .orElseThrow(() -> new FederatedLoginRefusedException("The linked account no longer exists"));
136         requireMaySignIn(account);
137         final var updated = federatedIdentityRepository.save(new FederatedIdentity(identity.id(), identity.userId(),
138                 identity.provider(), identity.subject(), email == null ? identity.email() : email,
139                 identity.linkedAt(), now));
140         return new FederatedLogin(account, updated, FederatedLoginOutcome.EXISTING);
141     }
142 
143     /**
144      * Refuses an account that may not sign in, before anything is linked, updated or audited.
145      *
146      * @param account the account the sign-in resolved to.
147      * @throws FederatedLoginRefusedException if the account is disabled.
148      */
149     private void requireMaySignIn(final UserAccount account) {
150         if (!federatedLinkingPolicy.maySignIn(account)) {
151             throw new FederatedLoginRefusedException("The account may not sign in");
152         }
153     }
154 
155     /**
156      * Links an identity to an account and audits it.
157      *
158      * @param account  the account.
159      * @param provider provider of the identity.
160      * @param subject  subject of the identity.
161      * @param email    e-mail address the provider reported.
162      * @param now      current instant.
163      * @return the stored link.
164      */
165     private FederatedIdentity link(final UserAccount account, final IdentityProvider provider, final String subject,
166                                    final String email, final Instant now) {
167         final var stored = federatedIdentityRepository.save(
168                 new FederatedIdentity(UUID.randomUUID(), account.id(), provider, subject, email, now, now));
169         recordSecurityEventUseCase.execute(SecurityEventType.IDENTITY_LINKED, account.id(), account.username(),
170                 provider.id());
171         return stored;
172     }
173 
174     /**
175      * Creates an enabled account from the provider's verified e-mail address. Its password is the hash of a random
176      * secret that is discarded, and the account is marked as having no password of its own.
177      *
178      * @param provider provider that vouched for the address.
179      * @param email    verified e-mail address.
180      * @return the new account.
181      */
182     private UserAccount provision(final IdentityProvider provider, final String email) {
183         final var id = UUID.randomUUID();
184         final var username = userAccountRepository.existsByUsername(email) || email.length() > MAX_USERNAME_LENGTH
185                 ? provider.id() + "-" + id : email;
186         final var account = new UserAccount(id, username, passwordHasher.encode(tokenGenerator.generate()), true,
187                 List.of(RegisterAccountUseCase.DEFAULT_AUTHORITY), email, true, UserAccount.DEFAULT_TENANT);
188         userAccountRepository.save(account);
189         generatedPasswordRepository.markGenerated(id);
190         return account;
191     }
