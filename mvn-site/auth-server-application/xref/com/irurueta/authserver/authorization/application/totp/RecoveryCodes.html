17  package com.irurueta.authserver.authorization.application.totp;
18  
19  import java.security.SecureRandom;
20  import java.util.Locale;
21  import org.springframework.stereotype.Component;
22  
23  /**
24   * Creates recovery codes (Crockford base-32 alphabet, so no {@code I}, {@code L}, {@code O} or {@code U}) and
25   * normalizes what a user types back into that form.
26   */
27  @Component
28  public class RecoveryCodes {
29  
30      /** Number of characters of a code. */
31      public static final int CODE_LENGTH = 10;
32  
33      /** Crockford's base-32 alphabet. */
34      private static final String ALPHABET = "0123456789ABCDEFGHJKMNPQRSTVWXYZ";
35  
36      /** Source of randomness. */
37      private final SecureRandom random = new SecureRandom();
38  
39      /**
40       * Creates a new random code.
41       *
42       * @return the code, {@link #CODE_LENGTH} characters long.
43       */
44      public String newCode() {
45          final var code = new StringBuilder(CODE_LENGTH);
46          for (var i = 0; i < CODE_LENGTH; i++) {
47              code.append(ALPHABET.charAt(random.nextInt(ALPHABET.length())));
48          }
49          return code.toString();
50      }
51  
52      /**
53       * Normalizes a typed code: upper case, without separators, and with the look-alike letters Crockford maps
54       * ({@code O} to {@code 0}, {@code I} and {@code L} to {@code 1}).
55       *
56       * @param typed the code as typed.
57       * @return the canonical form.
58       */
59      public String normalize(final String typed) {
60          return typed.toUpperCase(Locale.ROOT)
61                  .replace("-", "")
62                  .replace(" ", "")
63                  .replace('O', '0')
64                  .replace('I', '1')
65                  .replace('L', '1');
66      }
67  }
