17  package com.irurueta.authserver.authorization.application.sms;
18  
19  import com.irurueta.authserver.authorization.application.mfa.HasConfirmedSecondFactorUseCase;
20  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
21  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
22  import com.irurueta.authserver.authorization.domain.mfa.TrustedDeviceRepository;
23  import com.irurueta.authserver.authorization.domain.sms.PhoneVerificationRepository;
24  import com.irurueta.authserver.authorization.domain.sms.UserPhoneRepository;
25  import java.util.UUID;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.stereotype.Service;
28  import org.springframework.transaction.annotation.Transactional;
29  
30  /**
31   * Removes the phone of a user, verified and pending. Removing a verified phone removes a second factor, so it is
32   * audited, and when that leaves the user without any confirmed second factor every trusted device of the user is
33   * revoked too, as deleting the last passkey does: a remembered browser stands in for a second factor the user no
34   * longer has.
35   */
36  @Service
37  @Transactional
38  @RequiredArgsConstructor
39  public class RemovePhoneUseCase {
40  
41      /** Detail of the audit event. */
42      static final String AUDIT_DETAIL = "phone";
43  
44      /** Port that stores the verified phones. */
45      private final UserPhoneRepository userPhoneRepository;
46  
47      /** Port that stores the pending verifications. */
48      private final PhoneVerificationRepository phoneVerificationRepository;
49  
50      /** Use case telling whether the user still has a confirmed second factor. */
51      private final HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
52  
53      /** Port that stores the trusted devices. */
54      private final TrustedDeviceRepository trustedDeviceRepository;
55  
56      /** Use case that records the audit events. */
57      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
58  
59      /**
60       * Removes the phone and any pending verification of a user.
61       *
62       * @param userId   identifier of the user.
63       * @param username name of the user, for the audit log.
64       */
65      public void execute(final UUID userId, final String username) {
66          final var hadVerifiedPhone = userPhoneRepository.findByUserId(userId).isPresent();
67          userPhoneRepository.deleteByUserId(userId);
68          phoneVerificationRepository.deleteByUserId(userId);
69          if (!hadVerifiedPhone) {
70              return;
71          }
72          var detail = AUDIT_DETAIL;
73          if (!hasConfirmedSecondFactorUseCase.execute(userId)) {
74              final var revoked = trustedDeviceRepository.deleteByUser(userId);
75              detail = detail + "; trusted devices revoked: " + revoked;
76          }
77          recordSecurityEventUseCase.execute(SecurityEventType.SECOND_FACTOR_REMOVED, userId, username, detail);
78      }
79  }
