17  package com.irurueta.authserver.authorization.application.clients;
18  
19  import com.irurueta.authserver.authorization.domain.clients.ClientRegistration;
20  import com.irurueta.authserver.authorization.domain.clients.ClientRegistrationRepository;
21  import com.irurueta.authserver.authorization.domain.password.PasswordHasher;
22  import java.util.ArrayList;
23  import java.util.List;
24  import java.util.Map;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.stereotype.Service;
27  import org.springframework.transaction.annotation.Transactional;
28  import org.springframework.util.StringUtils;
29  
30  // tag::clients-seeding[]
31  /**
32   * Use case that registers the configured clients (built from {@link DefaultClientDefinitions}) that are missing. It
33   * is idempotent: a client that already exists is left untouched, including its secret.
34   */
35  @Service
36  @Transactional
37  @RequiredArgsConstructor
38  public class SeedRegisteredClientsUseCase {
39  
40      /** Port that stores the clients. */
41      private final ClientRegistrationRepository clientRegistrationRepository;
42  
43      /** Port that encodes secrets. */
44      private final PasswordHasher passwordHasher;
45  
46      /**
47       * Registers the missing clients among the given definitions.
48       *
49       * @param definitions   the clients this deployment seeds; a client that is not configured is not in the list.
50       * @param clientSecrets clear-text secrets by client id, taken from the environment; only needed for the
51       *                      confidential clients that do not exist yet.
52       * @return the ids of the clients that were created and of those skipped; a confidential client without a
53       * secret is skipped and does not prevent the other clients from being created.
54       */
55      public SeedResult execute(final List<ClientRegistration> definitions, final Map<String, String> clientSecrets) {
56          final var created = new ArrayList<String>();
57          final var skipped = new ArrayList<String>();
58          for (final var definition : definitions) {
59              if (clientRegistrationRepository.findByClientId(definition.clientId()).isPresent()) {
60                  continue;
61              }
62              String encodedSecret = null;
63              if (definition.confidential()) {
64                  final var secret = clientSecrets.get(definition.clientId());
65                  if (!StringUtils.hasText(secret)) {
66                      skipped.add(definition.clientId());
67                      continue;
68                  }
69                  encodedSecret = passwordHasher.encode(secret);
70              }
71              clientRegistrationRepository.save(definition, encodedSecret);
72              created.add(definition.clientId());
73          }
74          return new SeedResult(created, skipped);
75      }
76  }
