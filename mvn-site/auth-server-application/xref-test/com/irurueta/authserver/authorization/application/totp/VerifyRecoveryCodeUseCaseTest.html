46  /**
47   * Unit tests of {@link VerifyRecoveryCodeUseCase}.
48   */
49  @ExtendWith(MockitoExtension.class)
50  class VerifyRecoveryCodeUseCaseTest {
51  
52      /** Current instant used by the tests. */
53      private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
54  
55      /** Port that stores the recovery codes. */
56      @Mock
57      private RecoveryCodeRepository recoveryCodeRepository;
58  
59      /** Port that hashes the codes. */
60      @Mock
61      private PasswordHasher passwordHasher;
62  
63      /** Normalizer of typed codes. */
64      @Mock
65      private RecoveryCodes recoveryCodes;
66  
67      /** Use case that tells whether the user has confirmed a second factor. */
68      @Mock
69      private HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
70  
71      /** Use case that tells whether the second factor is locked. */
72      @Mock
73      private CheckSecondFactorLockoutUseCase checkSecondFactorLockoutUseCase;
74  
75      /** Use case that counts a failure. */
76      @Mock
77      private RegisterSecondFactorFailureUseCase registerSecondFactorFailureUseCase;
78  
79      /** Use case that clears the failures. */
80      @Mock
81      private ResetSecondFactorFailuresUseCase resetSecondFactorFailuresUseCase;
82  
83      /** Use case that records the audit events. */
84      @Mock
85      private RecordSecurityEventUseCase recordSecurityEventUseCase;
86  
87      /** Source of the current time. */
88      @Mock
89      private Clock clock;
90  
91      /** Use case under test. */
92      @InjectMocks
93      private VerifyRecoveryCodeUseCase useCase;
94  
95      /**
96       * Verifies that a locked second factor rejects the attempt without looking at the codes.
97       */
98      @Test
99      void execute_whenLocked_throwsSecondFactorLockedException() {
100         final var userId = UUID.randomUUID();
101         when(checkSecondFactorLockoutUseCase.execute(userId)).thenReturn(true);
102 
103         assertThrows(SecondFactorLockedException.class, () -> useCase.execute(userId, "alice", "CODE"));
104 
105         verify(recoveryCodeRepository, never()).findUnused(userId);
106     }
107 
108     /**
109      * Verifies that a matching unused code is consumed, the failures cleared and the use audited.
110      */
111     @Test
112     void execute_whenCodeMatches_consumesItAndResetsFailures() {
113         final var userId = UUID.randomUUID();
114         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(true);
115         final var other = new RecoveryCode(UUID.randomUUID(), userId, "other", null);
116         final var match = new RecoveryCode(UUID.randomUUID(), userId, "match", null);
117         when(clock.instant()).thenReturn(NOW);
118         when(recoveryCodes.normalize("abc")).thenReturn("ABC");
119         when(recoveryCodeRepository.findUnused(userId)).thenReturn(List.of(other, match));
120         when(passwordHasher.matches("ABC", "other")).thenReturn(false);
121         when(passwordHasher.matches("ABC", "match")).thenReturn(true);
122         when(recoveryCodeRepository.markUsed(match.id(), NOW)).thenReturn(true);
123 
124         assertTrue(useCase.execute(userId, "alice", "abc"));
125 
126         verify(resetSecondFactorFailuresUseCase).execute(userId);
127         verify(recordSecurityEventUseCase).execute(SecurityEventType.RECOVERY_CODE_USED, userId, "alice", null);
128     }
129 
130     /**
131      * Verifies that every code is checked even after a match, so the time does not reveal the position.
132      */
133     @Test
134     void execute_whenSeveralCodesMatch_checksAllAndConsumesTheFirst() {
135         final var userId = UUID.randomUUID();
136         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(true);
137         final var first = new RecoveryCode(UUID.randomUUID(), userId, "first", null);
138         final var second = new RecoveryCode(UUID.randomUUID(), userId, "second", null);
139         when(clock.instant()).thenReturn(NOW);
140         when(recoveryCodes.normalize("abc")).thenReturn("ABC");
141         when(recoveryCodeRepository.findUnused(userId)).thenReturn(List.of(first, second));
142         when(passwordHasher.matches("ABC", "first")).thenReturn(true);
143         when(passwordHasher.matches("ABC", "second")).thenReturn(true);
144         when(recoveryCodeRepository.markUsed(first.id(), NOW)).thenReturn(true);
145 
146         assertTrue(useCase.execute(userId, "alice", "abc"));
147 
148         verify(passwordHasher).matches("ABC", "second");
149     }
150 
151     /**
152      * Verifies that a wrong code counts as a failure.
153      */
154     @Test
155     void execute_whenNoCodeMatches_registersAFailure() {
156         final var userId = UUID.randomUUID();
157         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(true);
158         when(recoveryCodes.normalize("abc")).thenReturn("ABC");
159         when(recoveryCodeRepository.findUnused(userId))
160                 .thenReturn(List.of(new RecoveryCode(UUID.randomUUID(), userId, "other", null)));
161         when(passwordHasher.matches("ABC", "other")).thenReturn(false);
162 
163         assertFalse(useCase.execute(userId, "alice", "abc"));
164 
165         verify(registerSecondFactorFailureUseCase).execute(userId, "alice");
166         verify(resetSecondFactorFailuresUseCase, never()).execute(userId);
167     }
168 
169     /**
170      * Verifies that a code consumed concurrently by another request counts as a failure.
171      */
172     @Test
173     void execute_whenTheCodeWasJustConsumed_registersAFailure() {
174         final var userId = UUID.randomUUID();
175         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(true);
176         final var match = new RecoveryCode(UUID.randomUUID(), userId, "match", null);
177         when(clock.instant()).thenReturn(NOW);
178         when(recoveryCodes.normalize("abc")).thenReturn("ABC");
179         when(recoveryCodeRepository.findUnused(userId)).thenReturn(List.of(match));
180         when(passwordHasher.matches("ABC", "match")).thenReturn(true);
181         when(recoveryCodeRepository.markUsed(match.id(), NOW)).thenReturn(false);
182 
183         assertFalse(useCase.execute(userId, "alice", "abc"));
184 
185         verify(registerSecondFactorFailureUseCase).execute(userId, "alice");
186     }
187 
188     /**
189      * Verifies that a missing code is treated as an empty one.
190      */
191     @Test
192     void execute_whenCodeIsNull_registersAFailure() {
193         final var userId = UUID.randomUUID();
194         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(true);
195         when(recoveryCodes.normalize("")).thenReturn("");
196         when(recoveryCodeRepository.findUnused(userId)).thenReturn(List.of());
197 
198         assertFalse(useCase.execute(userId, "alice", null));
199 
200         verify(registerSecondFactorFailureUseCase).execute(userId, "alice");
201     }
202 
203     /**
204      * Verifies that a user without a confirmed second factor has no usable recovery code: the attempt fails, is
205      * counted, and the codes are not even looked at.
206      */
207     @Test
208     void execute_whenNoSecondFactorIsConfirmed_registersAFailure() {
209         final var userId = UUID.randomUUID();
210         when(hasConfirmedSecondFactorUseCase.execute(userId)).thenReturn(false);
211 
212         assertFalse(useCase.execute(userId, "alice", "abc"));
213 
214         verify(registerSecondFactorFailureUseCase).execute(userId, "alice");
215         verify(recoveryCodeRepository, never()).findUnused(userId);
216     }
