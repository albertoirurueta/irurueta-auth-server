136     /**
137      * Verifies that a registered account is disabled until its e-mail token is used, and that the token verifies
138      * the address, enables the account and can be used to sign in.
139      *
140      * @throws Exception if a request cannot be performed.
141      */
142     @Test
143     void register_whenNewAccount_isEnabledOnlyAfterTheEmailIsVerified() throws Exception {
144         final var username = "user-" + UUID.randomUUID();
145         final var email = username + "@example.com";
146         final var credential = AuthServerFixtures.randomValue();
147 
148         register(username, email, credential).andExpect(status().isAccepted());
149 
150         assertFalse(isEnabled(username));
151         final var token = ArgumentCaptor.<String>captor();
152         verify(accountMailer).sendEmailVerification(eq(email), token.capture());
153         mockMvc.perform(post("/login").with(csrf()).param("username", username).param("password", credential))
154                 .andExpect(redirectedUrl("/login?error"));
155 
156         mockMvc.perform(verifyRequest(token.getValue()))
157                 .andExpect(status().isNoContent());
158 
159         assertTrue(isEnabled(username));
160         assertTrue(jdbcClient.sql("SELECT email_verified FROM users WHERE username = ?").param(username)
161                 .query(Boolean.class).single());
162         mockMvc.perform(post("/login").with(csrf()).param("username", username).param("password", credential))
163                 .andExpect(redirectedUrl("/"));

488     /**
489      * Makes a request come from a new, random client address, so that the tests do not share a rate-limit counter.
490      *
491      * @return the request post-processor.
492      */
493     private static RequestPostProcessor fromNewClient() {
494         final var random = new SecureRandom();
495         final var address = "10." + random.nextInt(OCTET) + "." + random.nextInt(OCTET) + "."
496                 + random.nextInt(OCTET);
497         return request -> {
498             request.setRemoteAddr(address);
499             return request;
500         };
501     }

554     /**
555      * Reads the stored hash of a user.
556      *
557      * @param username name of the user.
558      * @return the stored hash.
559      */
560     private String storedHash(final String username) {
561         return jdbcClient.sql("""
562                 SELECT p.encoded_password FROM user_passwords p JOIN users u ON u.id = p.user_id
563                 WHERE u.username = ?
564                 """).param(username).query(String.class).single();
