88      /**
89       * Verifies that the login page offers the one-time token form: the identifier, the channel and the kind, posted
90       * to the token endpoint with the CSRF field and the sign-in purpose, below the password form.
91       *
92       * @throws Exception if a request cannot be performed.
93       */
94      @Test
95      void loginPage_whenRendered_offersTheOneTimeTokenForm() throws Exception {
96          getMockMvc().perform(get(LOGIN))
97                  .andExpect(status().isOk())
98                  .andExpect(content().string(containsString("Sign in with a code or a link")))
99                  .andExpect(content().string(containsString("action=\"/ott/generate\"")))
100                 .andExpect(content().string(containsString("name=\"purpose\" value=\"login\"")))
101                 .andExpect(content().string(containsString("name=\"channel\" value=\"sms\"")))
102                 .andExpect(content().string(containsString("name=\"kind\" value=\"link\"")))
103                 .andExpect(content().string(containsString("name=\"_csrf\"")))
104                 .andExpect(content().string(containsString("for=\"ott-username\"")));
105     }

107     /**
108      * Verifies that the second-factor page offers a code and a link by e-mail, and no SMS, to a user whose

183     /**
184      * Verifies the phone page and its posts: anonymous visitors are sent to sign in, a signed-in user sees the
185      * enrolment forms, a number is accepted (the answer never says whether it is registered), a wrong code and an
186      * invalid number are reported on the page and a status the page does not know is ignored.
187      *
188      * @throws Exception if a request cannot be performed.
189      */
190     @Test
191     void phonePage_whenUserIsSignedIn_enrolsAndReportsOutcomes() throws Exception {
192         getMockMvc().perform(get(PHONE_PAGE))
193                 .andExpect(status().is3xxRedirection())
194                 .andExpect(redirectedUrl("/login"));
195 
196         final var user = getFixtures().createUser();
197         final var session = new MockHttpSession();
198         getFixtures().signIn(user, session);
199 
200         getMockMvc().perform(get(PHONE_PAGE).session(session))
201                 .andExpect(status().isOk())
202                 .andExpect(header().string("Cache-Control", containsString("no-store")))
203                 .andExpect(content().string(containsString("Add a phone")))
204                 .andExpect(content().string(containsString("action=\"/account/phone\"")))
205                 .andExpect(content().string(containsString("action=\"/account/phone/verify\"")))
206                 .andExpect(content().string(containsString("name=\"_csrf\"")));
207 
208         getMockMvc().perform(form(PHONE_PAGE, session, "phone", "+34 600 111 222"))
209                 .andExpect(status().isSeeOther())
210                 .andExpect(redirectedUrl("/account/phone?status=pending"));
211         getMockMvc().perform(get(PHONE_PAGE).param("status", "pending").session(session))
212                 .andExpect(content().string(containsString("a code is on its way")));
213 
214         getMockMvc().perform(form(PHONE_PAGE + "/verify", session, "code", "000000"))
215                 .andExpect(status().isSeeOther())
216                 .andExpect(redirectedUrl("/account/phone?status=invalid-code"));
217         getMockMvc().perform(form(PHONE_PAGE, session, "phone", "not a number"))
218                 .andExpect(status().isSeeOther())
219                 .andExpect(redirectedUrl("/account/phone?status=invalid-phone"));
220         getMockMvc().perform(get(PHONE_PAGE).param("status", "<script>").session(session))
221                 .andExpect(content().string(not(containsString("<script>"))));
222         getMockMvc().perform(post(PHONE_PAGE + "/remove").session(session).with(csrf()))
223                 .andExpect(status().isSeeOther())
224                 .andExpect(redirectedUrl("/account/phone?status=removed"));
225     }

241     /**
242      * Verifies the template of the development SMS inbox (its controller only exists in the dev profile, which the
243      * integration tests do not run): the recorded messages are listed in a table with headers, and an empty inbox
244      * says so.
245      */
246     @Test
247     void devSmsInboxTemplate_whenRendered_listsTheMessagesOrSaysThereAreNone() {
248         RequestContextHolder.setRequestAttributes(
249                 new ServletRequestAttributes(new MockHttpServletRequest(), new MockHttpServletResponse()));
250         try {
251             final var entry = new SmsInboxEntry(new PhoneNumber("+34600111222"), "Your code is 123456",
252                     Instant.parse("2026-01-01T00:00:00Z"));
253             final var listed = pageRenderer.render("dev-sms-inbox", Map.of("messages", List.of(entry))).getBody();
254             final var empty = pageRenderer.render("dev-sms-inbox", Map.of("messages", List.of())).getBody();
255 
256             assertTrue(listed.contains("<th scope=\"col\">To</th>"));
257             assertTrue(listed.contains("+34600111222"));
258             assertTrue(listed.contains("Your code is 123456"));
259             assertTrue(empty.contains("No messages yet."));
260             assertFalse(empty.contains("<table"));
261         } finally {
262             RequestContextHolder.resetRequestAttributes();
263         }
264     }

266     /**
267      * Builds the post a browser makes when it submits a form of the phone page.
268      *
269      * @param path    path the form posts to.
270      * @param session the browser session.
271      * @param name    name of the only field.
272      * @param value   value of the field.
273      * @return the request.
274      */
275     private static MockHttpServletRequestBuilder form(final String path, final MockHttpSession session,
276             final String name, final String value) {
277         return post(path).session(session).with(csrf()).contentType(MediaType.APPLICATION_FORM_URLENCODED)
278                 .param(name, value);
279     }
