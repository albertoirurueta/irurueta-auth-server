36  /**
37   * Checks that the attempts to verify a user code are limited per signed-in user: a user code is short, so after a
38   * handful of attempts the endpoint answers {@code 429} with a {@code Retry-After} header, whatever the codes were.
39   */
40  @PostgresIT
41  @AutoConfigureMockMvc
42  class DeviceVerificationRateLimitIT {
43  
44      /** Attempts a user may make in a window, set for the test. */
45      private static final int PER_USER = 2;
46  
47      /** Path of the verification endpoint. */
48      private static final String VERIFICATION = "/oauth2/device_verification";
49  
50      /** MockMvc bound to the application context. */
51      @Autowired
52      private MockMvc mockMvc;
53  
54      /** Encoder of the application. */
55      @Autowired
56      private PasswordEncoder passwordEncoder;
57  
58      /** Port that stores the users. */
59      @Autowired
60      private UserAccountRepository userAccounts;
61  
62      /** Port that stores the client registrations. */
63      @Autowired
64      private ClientRegistrationRepository clientRegistrations;
65  
66      /**
67       * Sets a small limit for the run.
68       *
69       * @param registry registry of the properties of the context.
70       */
71      @DynamicPropertySource
72      static void limits(final DynamicPropertyRegistry registry) {
73          registry.add("authserver.device.rate-limit.per-user", () -> PER_USER);
74      }
75  
76      /**
77       * Verifies that the attempt over the limit is refused with {@code 429} and the time to wait.
78       *
79       * @throws Exception if a request cannot be performed.
80       */
81      @Test
82      void deviceVerification_whenTheUserExceedsTheLimit_answersTooManyRequests() throws Exception {
83          final var fixtures = new AuthServerFixtures(mockMvc, passwordEncoder, userAccounts, clientRegistrations);
84          final var user = fixtures.createUser();
85          final var session = new MockHttpSession();
86          fixtures.signIn(user, session);
87          for (var attempt = 0; attempt < PER_USER; attempt++) {
88              mockMvc.perform(post(VERIFICATION).session(session).with(csrf()).param("user_code", "ZZZZ-ZZZZ"))
89                      .andExpect(status().is3xxRedirection());
90          }
91  
92          final var refused = mockMvc.perform(post(VERIFICATION).session(session).with(csrf())
93                          .param("user_code", "ZZZZ-ZZZZ"))
94                  .andExpect(status().isTooManyRequests())
95                  .andExpect(header().exists("Retry-After")).andReturn();
96  
97          assertNotNull(refused.getResponse().getHeader("Retry-After"));
98      }
99  }
