16  package com.irurueta.authserver.authorization;
17  
18  import static org.junit.jupiter.api.Assertions.assertInstanceOf;
19  import static org.junit.jupiter.api.Assertions.assertNotNull;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  import com.irurueta.authserver.authorization.domain.sms.SmsSender;
24  import com.irurueta.authserver.authorization.infrastructure.sms.LoggingSmsSender;
25  import com.irurueta.authserver.authorization.infrastructure.sms.MissingSmsGatewayConfiguration;
26  import com.irurueta.authserver.authorization.infrastructure.sms.SmsAdapterConfiguration;
27  import java.time.Clock;
28  import org.junit.jupiter.api.Test;
29  import org.springframework.boot.test.context.runner.ApplicationContextRunner;
30  
31  /**
32   * Checks which {@link SmsSender} the profiles select: the logging one only under {@code dev} or {@code test}, none
33   * under a gateway profile (whose own configuration then provides it), and a failed startup with a message that names
34   * the profiles when none of them is active. It starts a minimal Spring context (the two configurations of the SMS
35   * adapter), hence an integration test.
36   */
37  class SmsSenderProfilesIT {
38  
39      /** Runner of a context holding only the configurations of the SMS adapter that the profiles choose between. */
40      private final ApplicationContextRunner runner = new ApplicationContextRunner()
41              .withUserConfiguration(SmsAdapterConfiguration.class, MissingSmsGatewayConfiguration.class)
42              .withBean(Clock.class, Clock::systemUTC);
43  
44      /**
45       * Verifies that the {@code test} profile gets the logging sender.
46       */
47      @Test
48      void smsSender_whenTestProfileIsActive_isTheLoggingSender() {
49          runner.withPropertyValues("spring.profiles.active=test")
50                  .run(context -> assertInstanceOf(LoggingSmsSender.class, context.getBean(SmsSender.class)));
51      }
52  
53      /**
54       * Verifies that the {@code dev} profile gets the logging sender.
55       */
56      @Test
57      void smsSender_whenDevProfileIsActive_isTheLoggingSender() {
58          runner.withPropertyValues("spring.profiles.active=dev")
59                  .run(context -> assertInstanceOf(LoggingSmsSender.class, context.getBean(SmsSender.class)));
60      }
61  
62      /**
63       * Verifies that a gateway profile, even next to {@code dev}, leaves the sender to the gateway's configuration and
64       * does not stop the startup.
65       */
66      @Test
67      void smsSender_whenGatewayProfileIsActive_isNotTheLoggingSender() {
68          runner.withPropertyValues("spring.profiles.active=dev,twilio")
69                  .run(context -> {
70                      assertNull(context.getStartupFailure(), "The startup must not fail");
71                      assertTrue(context.getBeansOfType(SmsSender.class).isEmpty());
72                  });
73      }
74  
75      /**
76       * Verifies that without any profile that provides a sender the startup fails, saying which profiles to activate.
77       */
78      @Test
79      void startup_whenNoSenderProfileIsActive_failsNamingTheProfiles() {
80          runner.run(context -> {
81              final var failure = context.getStartupFailure();
82              assertNotNull(failure, "The startup must fail");
83              var cause = (Throwable) failure;
84              while (cause.getCause() != null) {
85                  cause = cause.getCause();
86              }
87              assertInstanceOf(IllegalStateException.class, cause);
88              assertTrue(cause.getMessage().contains("'twilio' or 'sns'"), cause.getMessage());
89          });
90      }
91  }
