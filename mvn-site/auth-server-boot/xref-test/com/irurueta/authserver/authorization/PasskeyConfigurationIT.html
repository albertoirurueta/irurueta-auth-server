61  /**
62   * Checks the passkey sign-in as the login chain applies it: the endpoints the chain permits (with CSRF on), the
63   * passkey as the second factor after a password and as the only factor, the merge of the factors of the session, the
64   * resumption of the saved authorization request with a new session id, the claims of the tokens, the audit trail,
65   * the refusal of a disabled account, the answer to a failed or unknown sign-in and the account API. The relying party
66   * operations are mocked, so the tests need no authenticator: they exercise the filter, the provider, the
67   * multi-factor framework and the token customizers, not the cryptography of the credential, which the WebAuthn4J
68   * library and the end-to-end tests with a virtual authenticator cover.
69   */

128     /**
129      * Verifies the whole second-factor flow: a user with a passkey who signed in with the password is sent to the
130      * second-factor page, which offers the passkey; presenting it keeps the password factor (so the tokens report
131      * both and {@code mfa}), resumes the saved authorization request in the JSON the framework answers, changes the
132      * session id, records the sign-in, and issues the same roles, tenant and audiences as a password sign-in. The
133      * fresh multi-factor token then renames and deletes the passkey through the account API.
134      *
135      * @throws Exception if a request cannot be performed.
136      */
137     @Test
138     void signIn_whenPasskeyCompletesTheSecondFactor_mergesTheFactorsResumesTheRequestAndIssuesTheSameClaims()
139             throws Exception {
140         final var user = getFixtures().createUser();
141         final var passkey = seedPasskey(user.username(), "Laptop");
142         when(relyingParty.authenticate(any())).thenReturn(entityOf(user));
143         final var client = getFixtures().createCodeClient();
144         final var verifier = newVerifier();
145         final var session = new MockHttpSession();
146         final var authorizeUri = signInForAuthorization(user, client, verifier, Map.of(), session);
147         expectSecondFactorPage(authorizeUri, session);
148         assertTrue(secondFactorPage(session).contains("data-passkey-button"), "The page must offer the passkey");
149         final var sessionId = session.getId();
150         final var successesBefore = countEvents("LOGIN_SUCCESS");
151 
152         authenticateWithPasskey(session)
153                 .andExpect(status().isOk())
154                 .andExpect(jsonPath("$.authenticated").value(true))
155                 .andExpect(jsonPath("$.redirectUrl", containsString(client.clientId())));
156 
157         assertNotEquals(sessionId, session.getId(), "The session id must change when the passkey is presented");
158         assertEquals(successesBefore + 1, countEvents("LOGIN_SUCCESS"), "The sign-in must reach the audit trail");
159         final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
160         final var access = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
161         final var id = AuthServerFixtures.claimsOf(tokens.get("id_token").asString());
162         assertEquals(PASSWORD_AND_PASSKEY_AMR, strings(access.get(AMR)));
163         assertEquals(PASSWORD_AND_PASSKEY_AMR, strings(id.get(AMR)));
164         assertEquals(AuthenticationContextClass.MULTI_FACTOR, access.get(ACR).asString());
165         assertEquals(List.of("ROLE_USER"), strings(access.get(ROLES)));
166         assertEquals(AuthServerFixtures.TENANT, access.get(TENANT).asString());
167         assertSameClaimsAsAPasswordSignIn(access);
168 
169         final var bearer = BEARER + tokens.get("access_token").asString();
170         getMockMvc().perform(get(PASSKEYS_API).header(AUTHORIZATION, bearer))
171                 .andExpect(status().isOk())
172                 .andExpect(jsonPath("$[0].id").value(passkey.credentialId()))
173                 .andExpect(jsonPath("$[0].label").value("Laptop"));
174         getMockMvc().perform(patch(PASSKEYS_API + "/" + passkey.credentialId()).header(AUTHORIZATION, bearer)
175                         .contentType(MediaType.APPLICATION_JSON).content("{\"label\":\"Renamed\"}"))
176                 .andExpect(status().isOk())
177                 .andExpect(jsonPath("$.label").value("Renamed"));
178         getMockMvc().perform(delete(PASSKEYS_API + "/" + passkey.credentialId()).header(AUTHORIZATION, bearer))
179                 .andExpect(status().isNoContent());
180         assertEquals(0, countCredentials(passkey.credentialId()));
181     }

183     /**
184      * Verifies that a passkey cannot step around the second-factor lockout: once the second factor of a user who
185      * signed in with the password is locked (after too many failed codes), the passkey post is answered {@code 401}
186      * before the assertion is even verified, the session stays at the second factor, and a passwordless sign-in of
187      * the same account (no user in the session yet) is not affected.
188      *
189      * @throws Exception if a request cannot be performed.
190      */
191     @Test
192     void signIn_whenSecondFactorIsLocked_refusesThePasskeyAsASecondFactor() throws Exception {
193         final var user = getFixtures().createUser();
194         seedPasskey(user.username(), "Laptop");
195         when(relyingParty.authenticate(any())).thenReturn(entityOf(user));
196         final var client = getFixtures().createCodeClient();
197         final var session = new MockHttpSession();
198         final var authorizeUri = signInForAuthorization(user, client, newVerifier(), Map.of(), session);
199         expectSecondFactorPage(authorizeUri, session);
200         final var userId = userAccounts.findByUsername(user.username()).orElseThrow().id();
201         for (var attempt = 0; attempt < mfaProperties.maxFailedAttempts(); attempt++) {
202             registerSecondFactorFailureUseCase.execute(userId, user.username());
203         }
204 
205         authenticateWithPasskey(session).andExpect(status().isUnauthorized());
206 
207         verify(relyingParty, never()).authenticate(any());
208         expectSecondFactorPage(authorizeUri, session);
209         authenticateWithPasskey(new MockHttpSession())
210                 .andExpect(status().isOk())
211                 .andExpect(jsonPath("$.authenticated").value(true));
212     }

214     /**
215      * Verifies the passkey as the only factor: it satisfies the policy on its own, the tokens report {@code hwk} and,
216      * because only one factor was presented, the password {@code acr}, so the step-up of the account API (which
217      * demands the multi-factor class) refuses it with the RFC 9470 challenge, and presenting the same passkey again
218      * on the second-factor page does not change that: a user whose only second factor is a passkey cannot reach the
219      * multi-factor class with it alone.
220      *
221      * @throws Exception if a request cannot be performed.
222      */
223     @Test
224     void signIn_whenPasskeyIsTheOnlyFactor_isStrongEnoughButNeverReachesTheMultiFactorClass() throws Exception {
225         final var user = getFixtures().createUser();
226         final var passkey = seedPasskey(user.username(), "Laptop");
227         when(relyingParty.authenticate(any())).thenReturn(entityOf(user));
228         final var client = getFixtures().createCodeClient();
229         final var verifier = newVerifier();
230         final var session = new MockHttpSession();
231         final var authorizeUri = AuthServerFixtures.authorizeUri(client, verifier, Map.of());
232         getMockMvc().perform(get(authorizeUri).session(session)).andExpect(status().is3xxRedirection());
233 
234         authenticateWithPasskey(session)
235                 .andExpect(status().isOk())
236                 .andExpect(jsonPath("$.redirectUrl", containsString(client.clientId())));
237 
238         final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
239         final var access = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
240         assertEquals(List.of("hwk"), strings(access.get(AMR)));
241         assertEquals(AuthenticationContextClass.PASSWORD, access.get(ACR).asString());
242         assertEquals(List.of("ROLE_USER"), strings(access.get(ROLES)));
243         assertEquals(AuthServerFixtures.TENANT, access.get(TENANT).asString());
244         assertSameClaimsAsAPasswordSignIn(access);
245 
246         final var bearer = BEARER + tokens.get("access_token").asString();
247         getMockMvc().perform(get(PASSKEYS_API).header(AUTHORIZATION, bearer))
248                 .andExpect(status().isOk())
249                 .andExpect(jsonPath("$[0].id").value(passkey.credentialId()));
250         getMockMvc().perform(delete(PASSKEYS_API + "/" + passkey.credentialId()).header(AUTHORIZATION, bearer))
251                 .andExpect(status().isUnauthorized())
252                 .andExpect(header().string(HttpHeaders.WWW_AUTHENTICATE,
253                         containsString("insufficient_user_authentication")));
254         assertEquals(1, countCredentials(passkey.credentialId()));
255 
256         final var stepUpUri = AuthServerFixtures.authorizeUri(client, newVerifier(),
257                 Map.of("acr_values", AuthenticationContextClass.MULTI_FACTOR));
258         expectSecondFactorPage(stepUpUri, session);
259         authenticateWithPasskey(session).andExpect(status().isOk());
260         expectSecondFactorPage(stepUpUri, session);
261     }

284     /**
285      * Verifies that an assertion the relying party rejects and a passkey of an account that does not exist are
286      * answered identically: {@code 401} without a body, so the sign-in does not reveal whether an account exists.
287      *
288      * @throws Exception if a request cannot be performed.
289      */
290     @Test
291     void signIn_whenAssertionIsRejectedOrAccountIsUnknown_answersTheSameFailure() throws Exception {
292         when(relyingParty.authenticate(any())).thenThrow(new IllegalArgumentException("bad signature"));
293         final var rejected = authenticateWithPasskey(new MockHttpSession()).andReturn().getResponse();
294 
295         doReturn(entityNamed("ghost-" + AuthServerFixtures.randomValue())).when(relyingParty).authenticate(any());
296         final var unknown = authenticateWithPasskey(new MockHttpSession()).andReturn().getResponse();
297 
298         assertEquals(HttpStatus.UNAUTHORIZED.value(), rejected.getStatus());
299         assertEquals(rejected.getStatus(), unknown.getStatus());
300         assertEquals(rejected.getContentAsString(), unknown.getContentAsString());
301         assertEquals("", unknown.getContentAsString());
302         assertNotNull(rejected.getHeaderNames());
303         assertEquals(rejected.getHeaderNames().stream().filter(name -> !"Set-Cookie".equals(name)).sorted().toList(),
304                 unknown.getHeaderNames().stream().filter(name -> !"Set-Cookie".equals(name)).sorted().toList());
305     }

318     /**
319      * Fails unless the claims of a passkey sign-in carry the same roles, tenant and audiences as those of a password
320      * sign-in of an equivalent user.
321      *
322      * @param passkeyClaims the claims of the access token of the passkey sign-in.
323      * @throws Exception if a request cannot be performed.
324      */
325     private void assertSameClaimsAsAPasswordSignIn(final JsonNode passkeyClaims)
326             throws Exception {
327         final var other = getFixtures().createUser();
328         final var tokens = getFixtures().authorizationCodeTokens(other, getFixtures().createCodeClient());
329         final var password = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
330         assertEquals(strings(password.get(ROLES)), strings(passkeyClaims.get(ROLES)));
331         assertEquals(password.get(TENANT).asString(), passkeyClaims.get(TENANT).asString());
332         assertEquals(strings(password.get(AUD)), strings(passkeyClaims.get(AUD)));
333     }
