37  /**
38   * Checks that a password alone is never enough when multi-factor authentication is required: by the client
39   * ({@code mfa.required} setting), by the user ({@code mfa_required} column) or because the user enrolled a second
40   * factor. The browser is sent to the second-factor page and no authorization code is issued; a user who has no
41   * second factor yet enrols one there, sees the recovery codes once and resumes the sign-in.
42   */
43  class MfaPasswordOnlyForcedIT extends AbstractMfaFlow {
44  
45      /** Marker of the enrolment form in the second-factor page. */
46      private static final String ENROL_FIELD = "name=\"enroll\"";
47  
48      /** Port that stores the recovery codes. */
49      @Autowired
50      private RecoveryCodeRepository recoveryCodes;
51  
52      /** Port that stores user accounts. */
53      @Autowired
54      private UserAccountRepository userAccounts;
55  
56      /**
57       * Verifies that a client that requires MFA sends a password-only user to the second-factor page, which offers
58       * the TOTP form, and keeps doing so.
59       *
60       * @throws Exception if a request cannot be performed.
61       */
62      @Test
63      void authorize_whenClientRequiresMfaAndOnlyPasswordWasPresented_redirectsToSecondFactorPage() throws Exception {
64          final var user = getFixtures().createUser();
65          final var client = getFixtures().createMfaCodeClient();
66          final var session = new MockHttpSession();
67          final var authorizeUri = signInForAuthorization(user, client, newVerifier(), Map.of(), session);
68  
69          expectSecondFactorPage(authorizeUri, session);
70          expectSecondFactorPage(authorizeUri, session);
71          getMockMvc().perform(get(SECOND_FACTOR_PAGE).session(session))
72                  .andExpect(status().isOk())
73                  .andExpect(content().string(containsString("name=\"code\"")));
74      }
75  
76      /**
77       * Verifies that an account that requires MFA is sent to the second-factor page even when the client does not.
78       *
79       * @throws Exception if a request cannot be performed.
80       */
81      @Test
82      void authorize_whenUserRequiresMfaAndOnlyPasswordWasPresented_redirectsToSecondFactorPage() throws Exception {
83          final var user = getFixtures().createUser();
84          getFixtures().requireMfa(user, getTools());
85          final var client = getFixtures().createCodeClient();
86          final var session = new MockHttpSession();
87          final var authorizeUri = signInForAuthorization(user, client, newVerifier(), Map.of(), session);
88  
89          expectSecondFactorPage(authorizeUri, session);
90      }
91  
92      /**
93       * Verifies that a user who enrolled an authenticator app is always asked for it, even when neither the client nor
94       * an administrator requires it.
95       *
96       * @throws Exception if a request cannot be performed.
97       */
98      @Test
99      void authorize_whenUserHasAConfirmedSecondFactor_redirectsToSecondFactorPage() throws Exception {
100         final var enrolled = getFixtures().createEnrolledUser(getTools());
101         final var client = getFixtures().createCodeClient();
102         final var session = new MockHttpSession();
103         final var authorizeUri = signInForAuthorization(enrolled.user(), client, newVerifier(), Map.of(), session);
104 
105         expectSecondFactorPage(authorizeUri, session);
106         final var page = secondFactorPage(session);
107         assertTrue(page.contains("name=\"code\""));
108         assertFalse(page.contains(ENROL_FIELD), "An enrolled user must not be offered a new enrolment");
109     }
110 
111     // tag::mfa-enrolment-in-login-it[]
112     /**
113      * Verifies that a user who must use a second factor but has none enrols an authenticator app on the second-factor
114      * page, sees the recovery codes exactly once and then resumes the authorization request, whose tokens report the
115      * second factor.
116      *
117      * @throws Exception if a request cannot be performed.
118      */
119     @Test
120     void secondFactor_whenUserMustEnrol_enrolsShowsTheRecoveryCodesOnceAndResumes() throws Exception {
121         final var user = getFixtures().createUser();
122         getFixtures().requireMfa(user, getTools());
123         final var userId = userAccounts.findByUsername(user.username()).orElseThrow().id();
124         final var client = getFixtures().createCodeClient();
125         final var session = new MockHttpSession();
126         final var verifier = newVerifier();
127         final var authorizeUri = signInForAuthorization(user, client, verifier, Map.of(), session);
128         expectSecondFactorPage(authorizeUri, session);
129 
130         final var page = secondFactorPage(session);
131         assertTrue(page.contains(ENROL_FIELD) && page.contains("otpauth://totp/")
132                 && page.contains("src=\"data:image/png;base64,"), "The page must offer the enrolment");
133 
134         final var enrolled = postEnrolment(session, getTools().codeNow(userId, 0));
135         assertEquals(RecoveryCodesPageController.RECOVERY_CODES_PATH, enrolled.getResponse().getRedirectedUrl());
136 
137         final var codesPage = getMockMvc().perform(get(RecoveryCodesPageController.RECOVERY_CODES_PATH)
138                 .session(session)).andExpect(status().isOk()).andReturn().getResponse().getContentAsString();
139         assertEquals(GenerateRecoveryCodesUseCase.CODE_COUNT, recoveryCodes.findUnused(userId).size());
140         assertEquals(GenerateRecoveryCodesUseCase.CODE_COUNT, codesPage.split("<li><code>", -1).length - 1);
141         assertTrue(codesPage.contains("/oauth2/authorize?"), "The Continue link must resume the authorization");
142         getMockMvc().perform(get(RecoveryCodesPageController.RECOVERY_CODES_PATH).session(session))
143                 .andExpect(status().isOk())
144                 .andExpect(content().string(not(containsString("<li><code>"))));
145 
146         final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
147         assertEquals(AuthenticationContextClass.MULTI_FACTOR,
148                 AuthServerFixtures.claimsOf(tokens.get("access_token").asString()).get(ACR).asString());
149     }
150     // end::mfa-enrolment-in-login-it[]

152     /**
153      * Verifies that, when nobody requires MFA, the password alone still obtains the code (the control of the tests
154      * above).
155      *
156      * @throws Exception if a request cannot be performed.
157      */
158     @Test
159     void authorize_whenNobodyRequiresMfa_issuesTheCodeAfterThePassword() throws Exception {
160         final var user = getFixtures().createUser();
161         final var client = getFixtures().createCodeClient();
162         final var session = new MockHttpSession();
163         final var authorizeUri = signInForAuthorization(user, client, newVerifier(), Map.of(), session);
164 
165         final var redirect = getMockMvc().perform(get(authorizeUri).session(session))
166                 .andExpect(status().is3xxRedirection()).andReturn().getResponse().getRedirectedUrl();
167 
168         assertTrue(redirect != null && redirect.startsWith(AuthServerFixtures.REDIRECT_URI + "?code="));
169     }
