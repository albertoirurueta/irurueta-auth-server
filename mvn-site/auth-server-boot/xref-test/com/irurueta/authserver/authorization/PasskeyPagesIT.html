36  /**
37   * Checks the pages the passkeys add or extend, rendered through the real {@code PageRenderer} in English: the
38   * passkey section of the login page and its script (which the content security policy must allow without a nonce),
39   * the passkey action of the second-factor page and the account page that registers passkeys.
40   */
41  class PasskeyPagesIT extends AbstractPasskeyFlow {
42  
43      /** Path of the login page. */
44      private static final String LOGIN = "/login";
45  
46      /** Path of the script. */
47      private static final String SCRIPT = "/js/passkeys.js";
48  
49      /** Matches a script element that has no {@code src}, that is, an inline script. */
50      private static final Pattern INLINE_SCRIPT = Pattern.compile("<script(?![^>]*\\bsrc=)[^>]*>");
51  
52      /**
53       * Pins the locale of the texts the tests assert on.
54       */
55      @BeforeEach
56      void pinTheLocale() {
57          LocaleContextHolder.setLocale(Locale.ENGLISH);
58      }
59  
60      /**
61       * Restores the locale.
62       */
63      @AfterEach
64      void restoreTheLocale() {
65          LocaleContextHolder.resetLocaleContext();
66      }
67  
68      /**
69       * Verifies that the login page offers the passkey: a field the browser fills with passkeys (conditional
70       * mediation), a button, the CSRF header and token and the URLs of the WebAuthn endpoints for the script, and no
71       * inline script, since the policy only allows same-origin resources.
72       *
73       * @throws Exception if a request cannot be performed.
74       */
75      @Test
76      void loginPage_whenRendered_offersPasskeysWithAnExternalScript() throws Exception {
77          final var html = getMockMvc().perform(get(LOGIN))
78                  .andExpect(status().isOk())
79                  .andExpect(header().string("Content-Security-Policy", containsString("default-src 'self'")))
80                  .andExpect(content().string(containsString("autocomplete=\"username webauthn\"")))
81                  .andExpect(content().string(containsString("data-passkey-button")))
82                  .andExpect(content().string(containsString("Sign in with a passkey")))
83                  .andExpect(content().string(containsString("src=\"/js/passkeys.js\"")))
84                  .andExpect(content().string(containsString("data-csrf-header=\"X-CSRF-TOKEN\"")))
85                  .andExpect(content().string(containsString("data-csrf-token=\"")))
86                  .andExpect(content().string(containsString("data-options-url=\"/webauthn/authenticate/options\"")))
87                  .andExpect(content().string(containsString("data-login-url=\"/login/webauthn\"")))
88                  .andExpect(content().string(containsString("data-error-url=\"/login?error\"")))
89                  .andExpect(content().string(containsString("autocomplete=\"current-password\"")))
90                  .andReturn().getResponse().getContentAsString();
91  
92          assertEquals(false, INLINE_SCRIPT.matcher(html).find(), "The page must not carry an inline script");
93      }
94  
95      /**
96       * Verifies that the script is served to anybody (the login page needs it before sign-in) and reads the URLs of
97       * the endpoints from the markup instead of hard-coding them.
98       *
99       * @throws Exception if a request cannot be performed.
100      */
101     @Test
102     void script_whenRequestedAnonymously_isServed() throws Exception {
103         getMockMvc().perform(get(SCRIPT))
104                 .andExpect(status().isOk())
105                 .andExpect(content().string(containsString("container.dataset.optionsUrl")))
106                 .andExpect(content().string(containsString("container.dataset.loginUrl")))
107                 .andExpect(content().string(containsString("container.dataset.registerUrl")))
108                 .andExpect(content().string(not(containsString("\"/login/webauthn\""))))
109                 .andExpect(content().string(not(containsString("\"/webauthn/"))));
110     }

112     /**
113      * Verifies that the second-factor page only offers a passkey to a user who has one.
114      *
115      * @throws Exception if a request cannot be performed.
116      */
117     @Test
118     void secondFactorPage_whenUserHasAPasskey_offersItAndOtherwiseDoesNot() throws Exception {
119         final var enrolled = getFixtures().createEnrolledUser(getTools());
120         final var session = signInWithThePasswordOnly(enrolled.user());
121         getMockMvc().perform(get(SECOND_FACTOR_PAGE).session(session))
122                 .andExpect(status().isOk())
123                 .andExpect(content().string(not(containsString("data-passkey-button"))));
124 
125         seedPasskey(enrolled.user().username(), "Laptop");
126 
127         getMockMvc().perform(get(SECOND_FACTOR_PAGE).session(session))
128                 .andExpect(status().isOk())
129                 .andExpect(content().string(containsString("data-passkey-button")))
130                 .andExpect(content().string(containsString("Use a passkey")))
131                 .andExpect(content().string(containsString("data-error-url=\"/login/second-factor?error\"")))
132                 .andExpect(content().string(containsString("data-options-url=\"/webauthn/authenticate/options\"")))
133                 .andExpect(content().string(containsString("data-login-url=\"/login/webauthn\"")))
134                 .andExpect(content().string(containsString("name=\"code\"")));
135     }

137     /**
138      * Verifies that a user whose only second factor is a passkey gets the passkey action on the second-factor page
139      * and neither the enrolment of an authenticator app nor its code forms, and that no authenticator app secret is
140      * generated for them by rendering the page.
141      *
142      * @throws Exception if a request cannot be performed.
143      */
144     @Test
145     void secondFactorPage_whenThePasskeyIsTheOnlySecondFactor_offersItWithoutEnrollingAnAuthenticatorApp()
146             throws Exception {
147         final var user = getFixtures().createUser();
148         seedPasskey(user.username(), "Laptop");
149         final var session = signInWithThePasswordOnly(user);
150 
151         getMockMvc().perform(get(SECOND_FACTOR_PAGE).session(session))
152                 .andExpect(status().isOk())
153                 .andExpect(content().string(containsString("data-passkey-button")))
154                 .andExpect(content().string(not(containsString("totp-enrolment"))))
155                 .andExpect(content().string(not(containsString("otpauth://"))))
156                 .andExpect(content().string(not(containsString("/login/second-factor/totp"))));
157 
158         assertEquals(0L, getJdbc().sql("""
159                         SELECT count(*) FROM totp_credentials t JOIN users u ON u.id = t.user_id
160                         WHERE u.username = ?""").param(user.username()).query(Long.class).single(),
161                 "Rendering the page must not create an authenticator app secret");
162     }

164     /**
165      * Verifies the account page: anonymous visitors are sent to sign in, a session that has not completed the second
166      * factor is sent to the second-factor page, and a session that satisfies the policy sees its passkeys and the
167      * registration helper (a name field of at most 100 characters and a button).
168      *
169      * @throws Exception if a request cannot be performed.
170      */
171     @Test
172     void accountPage_whenRequested_isGuardedAndListsThePasskeysWithTheRegistrationHelper() throws Exception {
173         getMockMvc().perform(get(ACCOUNT_PAGE))
174                 .andExpect(status().is3xxRedirection())
175                 .andExpect(redirectedUrl("/login"));
176 
177         final var enrolled = getFixtures().createEnrolledUser(getTools());
178         seedPasskey(enrolled.user().username(), "Laptop");
179         final MockHttpSession pending = signInWithThePasswordOnly(enrolled.user());
180         getMockMvc().perform(get(ACCOUNT_PAGE).session(pending))
181                 .andExpect(status().is3xxRedirection())
182                 .andExpect(redirectedUrl("/login/second-factor"));
183 
184         final var session = signInCompletingTheSecondFactor(enrolled);
185         getMockMvc().perform(get(ACCOUNT_PAGE).session(session))
186                 .andExpect(status().isOk())
187                 .andExpect(content().string(containsString("<title>Passkeys</title>")))
188                 .andExpect(content().string(containsString("Your passkeys")))
189                 .andExpect(content().string(containsString("Laptop")))
190                 .andExpect(content().string(containsString("data-passkey-register")))
191                 .andExpect(content().string(containsString("data-passkey-label")))
192                 .andExpect(content().string(containsString("maxlength=\"100\"")))
193                 .andExpect(content().string(containsString("src=\"/js/passkeys.js\"")))
194                 .andExpect(content().string(containsString("data-csrf-header=\"X-CSRF-TOKEN\"")))
195                 .andExpect(content().string(containsString("data-options-url=\"/webauthn/register/options\"")))
196                 .andExpect(content().string(containsString("data-register-url=\"/webauthn/register\"")));
197     }
