33  /**
34   * Checks what trusting a provider for multi-factor authentication changes, against a fake Google configured with
35   * {@code trust-mfa}: the provider's own multi-factor sign-in counts only when the ID token really reports {@code mfa}
36   * in its {@code amr}, and then the user with an authenticator app is not asked for a code again and the tokens list
37   * the upstream factors; a sign-in the provider did not vouch for as multi-factor still goes through the second-factor
38   * step. Every client id and secret is generated at run time.
39   */
40  class SocialTrustedProviderIT extends AbstractSocialFlow {
41  
42      /** Client id of Google, generated for the run. */
43      private static final String GOOGLE_CLIENT_ID = "google-" + UUID.randomUUID();
44  
45      /** Issuer of the ID tokens of Google. */
46      private static final String GOOGLE_ISSUER = "https://accounts.google.com";
47  
48      /** Name of the claim with the authentication methods the provider used. */
49      private static final String UPSTREAM_AMR = "amr";
50  
51      /**
52       * Configures Google as a provider trusted for multi-factor authentication.
53       *
54       * @param registry registry of the properties of the context.
55       */
56      @DynamicPropertySource
57      static void providers(final DynamicPropertyRegistry registry) {
58          registry.add("authserver.social.providers.google.client-id", () -> GOOGLE_CLIENT_ID);
59          registry.add("authserver.social.providers.google.client-secret", () -> UUID.randomUUID().toString());
60          registry.add("authserver.social.providers.google.trust-mfa", () -> "true");
61      }
62  
63      /**
64       * Verifies that a sign-in the trusted provider reports as multi-factor completes the authorization without the
65       * second-factor step, and that the tokens show the upstream factors and the multi-factor class.
66       *
67       * @throws Exception if a request cannot be performed.
68       */
69      @Test
70      void authorize_whenTheTrustedProviderReportsMfa_skipsTheSecondFactorAndReportsTheUpstreamFactors()
71              throws Exception {
72          final var enrolled = getFixtures().createEnrolledUser(getTools());
73          final var session = new MockHttpSession();
74          final var client = getFixtures().createCodeClient();
75          final var verifier = newVerifier();
76          final var authorizeUri = beginAuthorization(client, verifier, session);
77          final var claims = FakeIdentityProviders.userClaims(
78                  "subject-" + UUID.randomUUID(), enrolled.user().email(), true);
79          claims.put(UPSTREAM_AMR, List.of("pwd", "mfa"));
80  
81          signInWithOidc(session, "google", GOOGLE_ISSUER, GOOGLE_CLIENT_ID, claims);
82          final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
83  
84          final var access = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
85          assertEquals(List.of("fed", "mfa", "pwd"), strings(access.get(AMR)));
86          assertEquals(AuthenticationContextClass.MULTI_FACTOR, access.get(ACR).asString());
87          assertEquals("google", access.get("idp").asString());
88          assertEquals(enrolled.user().username(), access.get(SUB).asString());
89      }
90  
91      /**
92       * Verifies that the same strong sign-in also satisfies a step-up request for the multi-factor class: the
93       * authorization endpoint applies the rule its tokens' {@code acr} comes from, so the user is not sent to the
94       * second-factor page.
95       *
96       * @throws Exception if a request cannot be performed.
97       */
98      @Test
99      void authorize_whenTheTrustedProviderReportsMfaAndStepUpIsRequested_issuesTheCodeWithoutSecondFactor()
100             throws Exception {
101         final var enrolled = getFixtures().createEnrolledUser(getTools());
102         final var session = new MockHttpSession();
103         final var client = getFixtures().createCodeClient();
104         final var verifier = newVerifier();
105         final var authorizeUri = beginAuthorization(client, verifier, session);
106         final var claims = FakeIdentityProviders.userClaims(
107                 "subject-" + UUID.randomUUID(), enrolled.user().email(), true);
108         claims.put(UPSTREAM_AMR, List.of("pwd", "mfa"));
109         signInWithOidc(session, "google", GOOGLE_ISSUER, GOOGLE_CLIENT_ID, claims);
110         completeAuthorization(authorizeUri, client, verifier, session);
111 
112         final var stepUpVerifier = newVerifier();
113         final var stepUpUri = AuthServerFixtures.authorizeUri(client, stepUpVerifier,
114                 Map.of("acr_values", AuthenticationContextClass.MULTI_FACTOR));
115         final var tokens = completeAuthorization(stepUpUri, client, stepUpVerifier, session);
116 
117         final var access = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
118         assertEquals(AuthenticationContextClass.MULTI_FACTOR, access.get(ACR).asString());
119     }
120 
121     /**
122      * Verifies that a trusted provider that does not report {@code mfa} is an ordinary primary factor: the user with
123      * an authenticator app is still sent to the second-factor page.
124      *
125      * @throws Exception if a request cannot be performed.
126      */
127     @Test
128     void authorize_whenTheTrustedProviderDoesNotReportMfa_asksForTheSecondFactor() throws Exception {
129         final var enrolled = getFixtures().createEnrolledUser(getTools());
130         final var session = new MockHttpSession();
131         final var client = getFixtures().createCodeClient();
132         final var verifier = newVerifier();
133         final var authorizeUri = beginAuthorization(client, verifier, session);
134         final var claims = FakeIdentityProviders.userClaims(
135                 "subject-" + UUID.randomUUID(), enrolled.user().email(), true);
136         claims.put(UPSTREAM_AMR, List.of("pwd"));
137 
138         signInWithOidc(session, "google", GOOGLE_ISSUER, GOOGLE_CLIENT_ID, claims);
139 
140         expectSecondFactorPage(authorizeUri, session);
141     }
142 
143     /**
144      * Verifies that a trusted provider that sends no {@code amr} at all is an ordinary primary factor too.
145      *
146      * @throws Exception if a request cannot be performed.
147      */
148     @Test
149     void authorize_whenTheTrustedProviderSendsNoAmr_asksForTheSecondFactor() throws Exception {
150         final var enrolled = getFixtures().createEnrolledUser(getTools());
151         final var session = new MockHttpSession();
152         final var client = getFixtures().createCodeClient();
153         final var verifier = newVerifier();
154         final var authorizeUri = beginAuthorization(client, verifier, session);
155 
156         signInWithOidc(session, "google", GOOGLE_ISSUER, GOOGLE_CLIENT_ID,
157                 FakeIdentityProviders.userClaims("subject-" + UUID.randomUUID(), enrolled.user().email(), true));
158 
159         expectSecondFactorPage(authorizeUri, session);
160         final var redirect = getMockMvc().perform(get(authorizeUri).session(session))
161                 .andExpect(status().is3xxRedirection()).andReturn().getResponse().getRedirectedUrl();
162         assertNotNull(redirect);
163         assertTrue(redirect.endsWith(SECOND_FACTOR_PAGE));
164     }
