16  package com.irurueta.authserver.authorization;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertTrue;
20  import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
21  
22  import com.irurueta.authserver.authorization.domain.mfa.AuthenticationContextClass;
23  import java.time.Instant;
24  import java.util.Map;
25  import java.util.concurrent.TimeUnit;
26  import org.junit.jupiter.api.Test;
27  import org.springframework.mock.web.MockHttpSession;
28  
29  /**
30   * Checks the OpenID Connect step-up parameters of the authorization endpoint: {@code acr_values} asks for a second
31   * factor the session has not presented and {@code max_age} forces a new sign-in when the last one is too old.
32   */
33  class StepUpIT extends AbstractMfaFlow {
34  
35      /** Time to let pass so a {@code max_age} of zero is exceeded, in milliseconds. */
36      private static final long AGE_MILLIS = 1_100L;
37  
38      /** A {@code max_age} that a fresh sign-in satisfies, in seconds. */
39      private static final String LONG_MAX_AGE = "300";
40  
41      /**
42       * Verifies that asking for the multi-factor class sends a password-only session to the second-factor page and
43       * that, once the code is posted, the tokens carry that class.
44       *
45       * @throws Exception if a request cannot be performed.
46       */
47      @Test
48      void authorize_whenAcrValuesAskForMfa_requiresTheSecondFactor() throws Exception {
49          final var enrolled = getFixtures().createEnrolledUser(getTools());
50          final var client = getFixtures().createCodeClient();
51          final var session = new MockHttpSession();
52          final var verifier = newVerifier();
53          final var authorizeUri = signInForAuthorization(enrolled.user(), client, verifier,
54                  Map.of("acr_values", AuthenticationContextClass.MULTI_FACTOR), session);
55  
56          expectSecondFactorPage(authorizeUri, session);
57          final var posted = postSecondFactor(session, getTools().codeNow(enrolled.id(), 0), false, false);
58          assertAccepted(posted);
59          final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
60  
61          assertEquals(AuthenticationContextClass.MULTI_FACTOR,
62                  AuthServerFixtures.claimsOf(tokens.get("access_token").asString()).get("acr").asString());
63      }
64  
65      /**
66       * Verifies that a request for the password class is satisfied by a password-only session of a user without a
67       * second factor (one with a second factor is always asked for it).
68       *
69       * @throws Exception if a request cannot be performed.
70       */
71      @Test
72      void authorize_whenAcrValuesAskForPassword_isSatisfiedByThePassword() throws Exception {
73          final var user = getFixtures().createUser();
74          final var client = getFixtures().createCodeClient();
75          final var session = new MockHttpSession();
76          final var verifier = newVerifier();
77          final var authorizeUri = signInForAuthorization(user, client, verifier,
78                  Map.of("acr_values", AuthenticationContextClass.PASSWORD), session);
79  
80          final var tokens = completeAuthorization(authorizeUri, client, verifier, session);
81  
82          assertEquals(AuthenticationContextClass.PASSWORD,
83                  AuthServerFixtures.claimsOf(tokens.get("access_token").asString()).get("acr").asString());
84      }
85  
86      /**
87       * Verifies that a session older than {@code max_age} is sent to sign in again, and that the new sign-in carries
88       * a newer {@code auth_time}.
89       *
90       * @throws Exception if a request cannot be performed.
91       */
92      @Test
93      void authorize_whenMaxAgeIsExceeded_requiresANewSignIn() throws Exception {
94          final var user = getFixtures().createUser();
95          final var client = getFixtures().createCodeClient();
96          final var session = new MockHttpSession();
97          final var verifier = newVerifier();
98          final var tooOld = signInForAuthorization(user, client, verifier, Map.of("max_age", "0"), session);
99          TimeUnit.MILLISECONDS.sleep(AGE_MILLIS);
100         final var reauthenticatedAfter = Instant.now().getEpochSecond();
101 
102         final var redirect = mockMvcRedirect(tooOld, session);
103         assertTrue(redirect.endsWith("/login"), "Expected a redirect to the login page but got " + redirect);
104 
105         final var fresh = AuthServerFixtures.authorizeUri(client, verifier, Map.of("max_age", LONG_MAX_AGE));
106         getFixtures().signIn(user, session);
107         final var tokens = completeAuthorization(fresh, client, verifier, session);
108         final var authTime = AuthServerFixtures.claimsOf(tokens.get("access_token").asString())
109                 .get("auth_time").asLong();
110         assertTrue(authTime >= reauthenticatedAfter, "auth_time must be that of the new sign-in");
111     }
112 
113     /**
114      * Requests the authorization endpoint and answers where it redirects.
115      *
116      * @param authorizeUri the authorization request.
117      * @param session      the browser session.
118      * @return the redirect location.
119      * @throws Exception if the request cannot be performed.
120      */
121     private String mockMvcRedirect(final String authorizeUri, final MockHttpSession session) throws Exception {
122         return getMockMvc().perform(get(authorizeUri).session(session))
123                 .andReturn().getResponse().getRedirectedUrl();
124     }
125 }
