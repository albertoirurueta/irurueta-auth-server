43  /**
44   * Checks the passkeys endpoints of the account API against the real application and the real tables: the list, the
45   * rename and the deletion of the caller's passkeys, the step-up challenge that guards the deletion of an account with
46   * a confirmed second factor, and that an account never sees or touches the passkeys of another one. The requests
47   * carry a {@code jwt()} for the owner of the account, so the tests do not depend on the token endpoint. Every test
48   * creates its own users.
49   */
50  class PasskeyAccountIT extends AbstractPasskeyFlow {

53      private static final String PASSKEYS_API = "/api/account/passkeys";

56      private static final String BLANK_LABEL = "{\"label\":\"   \"}";
57  
58      /** Body of a rename request. */

62      private static final String NOT_FOUND_TYPE = "urn:problem:passkeys:passkey-not-found";
63  
64      /** Type of the problem answered for a label that is refused. */

68      private static final int LABEL_TOO_LONG = 101;

103      * without passkeys.

111         final var first = seedPasskey(user.username(), "Laptop");

118                 .andExpect(jsonPath("$[?(@.id == '" + second.credentialId() + "')].label").value("Phone"))
119                 .andExpect(jsonPath("$[0].publicKey").doesNotExist());
120         getMockMvc().perform(get(PASSKEYS_API).with(bearerOf(empty.username())))
121                 .andExpect(status().isOk())
122                 .andExpect(jsonPath("$.length()").value(0));

170      * Verifies that a recent multi-factor sign-in deletes a passkey, and that the passkey is gone from the table and

191     /**
192      * Verifies that an account with a confirmed second factor answers the RFC 9470 challenge to a token that shows
193      * no recent multi-factor sign-in (password only, or too old), deletes nothing, and deletes the passkey once the
194      * token shows one.
195      *
196      * @throws Exception if a request cannot be performed.
197      */
198     @Test
199     void deletePasskey_whenAccountHasASecondFactor_needsARecentMultiFactorSignIn() throws Exception {
200         final var enrolled = getFixtures().createEnrolledUser(getTools());
201         final var username = enrolled.user().username();
202         final var passkey = seedPasskey(username, "Laptop");
203         final var path = PASSKEYS_API + "/" + passkey.credentialId();
204         final var old = jwt().jwt(token -> token.subject(username)
205                 .claim(ACR, AuthenticationContextClass.MULTI_FACTOR)
206                 .claim(AUTH_TIME, Instant.now().minusSeconds(OLD_SIGN_IN_SECONDS).getEpochSecond()));
207 
208         for (final var token : List.<RequestPostProcessor>of(bearerOf(username), old)) {
209             getMockMvc().perform(delete(path).with(token))
210                     .andExpect(status().isUnauthorized())
211                     .andExpect(header().string(HttpHeaders.WWW_AUTHENTICATE,
212                             containsString("insufficient_user_authentication")))
213                     .andExpect(header().string(HttpHeaders.WWW_AUTHENTICATE,
214                             containsString(AuthenticationContextClass.MULTI_FACTOR)))
215                     .andExpect(jsonPath("$.type").value("urn:problem:mfa:insufficient-user-authentication"));
216         }
217         assertEquals(1, countCredentials(passkey.credentialId()), "Nothing may be removed without the step-up");
218 
219         getMockMvc().perform(delete(path).with(stepUpBearerOf(username))).andExpect(status().isNoContent());
220         assertEquals(0, countCredentials(passkey.credentialId()));
221     }

242     /**
243      * Verifies that deleting a passkey keeps the trusted devices while another passkey still stands as a second
244      * factor, and revokes every one of them once the last passkey (the only second factor) is gone, as disabling the
245      * authenticator app does; another account's trusted device is never touched.
246      *
247      * @throws Exception if a request cannot be performed.
248      */
249     @Test
250     void deletePasskey_whenTheLastSecondFactorIsDeleted_revokesTheTrustedDevices() throws Exception {
251         final var user = getFixtures().createUser();
252         final var other = getFixtures().createUser();
253         final var userId = idOf(user.username());
254         final var otherId = idOf(other.username());
255         final var first = seedPasskey(user.username(), "Laptop");
256         final var second = seedPasskey(user.username(), "Phone");
257         createTrustedDeviceUseCase.execute(userId, user.username(), BROWSER);
258         createTrustedDeviceUseCase.execute(otherId, other.username(), BROWSER);
259 
260         getMockMvc().perform(delete(PASSKEYS_API + "/" + first.credentialId()).with(stepUpBearerOf(user.username())))
261                 .andExpect(status().isNoContent());
262         assertEquals(1, trustedDevices.findByUser(userId, Instant.now()).size(),
263                 "A remaining passkey still stands as a second factor");
264 
265         getMockMvc().perform(delete(PASSKEYS_API + "/" + second.credentialId()).with(stepUpBearerOf(user.username())))
266                 .andExpect(status().isNoContent());
267         assertEquals(0, trustedDevices.findByUser(userId, Instant.now()).size());
268         assertEquals(1, trustedDevices.findByUser(otherId, Instant.now()).size());
269         assertEquals(1, getJdbc().sql("SELECT count(*) FROM security_events WHERE user_id = ? AND detail = ?")
270                 .params(userId, "passkey; trusted devices revoked: 1").query(Long.class).single());
271     }

295     /**
296      * Verifies the ownership: a passkey of another account is never listed, renamed or deleted, is answered exactly
297      * as one that does not exist, and survives untouched.
298      *
299      * @throws Exception if a request cannot be performed.
300      */
301     @Test
302     void passkeys_whenPasskeyBelongsToAnotherAccount_areNeverListedRenamedOrDeleted() throws Exception {
303         final var user = getFixtures().createUser();
304         final var other = getFixtures().createUser();
305         final var own = seedPasskey(user.username(), "Own");
306         final var foreign = seedPasskey(other.username(), "Someone else's");
307         final var path = PASSKEYS_API + "/" + foreign.credentialId();
308 
309         getMockMvc().perform(get(PASSKEYS_API).with(bearerOf(user.username())))
310                 .andExpect(status().isOk())
311                 .andExpect(jsonPath("$.length()").value(1))
312                 .andExpect(jsonPath("$[0].id").value(own.credentialId()));
313         getMockMvc().perform(patch(path).with(bearerOf(user.username())).contentType(MediaType.APPLICATION_JSON)
314                         .content(RENAMED_LABEL))
315                 .andExpect(status().isNotFound())
316                 .andExpect(jsonPath("$.type").value(NOT_FOUND_TYPE));
317         getMockMvc().perform(delete(path).with(stepUpBearerOf(user.username())))
318                 .andExpect(status().isNotFound())
319                 .andExpect(jsonPath("$.type").value(NOT_FOUND_TYPE));
320 
321         assertEquals(1, countCredentials(foreign.credentialId()));
322         getMockMvc().perform(get(PASSKEYS_API).with(bearerOf(other.username())))
323                 .andExpect(jsonPath("$[0].label").value("Someone else's"));
324     }

326     /**
327      * Resolves the identifier of an account.
328      *
329      * @param username name of the account.
330      * @return its identifier.
331      */
332     private UUID idOf(final String username) {
333         return userAccounts.findByUsername(username).orElseThrow().id();
334     }
335 
336     /**
337      * Builds the bearer token of a user.
338      *
339      * @param username the subject of the token.
340      * @return the request post-processor.
341      */
342     private static RequestPostProcessor bearerOf(final String username) {
343         return jwt().jwt(token -> token.subject(username));
344     }
345 
346     /**
347      * Builds the bearer token of a user who signed in with a second factor just now.
348      *
349      * @param username the subject of the token.
350      * @return the request post-processor.
351      */
352     private static RequestPostProcessor stepUpBearerOf(final String username) {
353         return jwt().jwt(token -> token.subject(username)
354                 .claim(ACR, AuthenticationContextClass.MULTI_FACTOR)
355                 .claim(AUTH_TIME, Instant.now().getEpochSecond()));
356     }
