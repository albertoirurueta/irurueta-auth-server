16  package com.irurueta.authserver.testsupport;
17  
18  import org.testcontainers.containers.GenericContainer;
19  import org.testcontainers.containers.wait.strategy.Wait;
20  import org.testcontainers.utility.DockerImageName;
21  
22  /**
23   * Describes the Mailpit container the one-time token tests read the e-mail of: the pinned image, its two ports and a
24   * factory that builds the container and the client of its HTTP API. Mailpit accepts any message over SMTP and exposes
25   * what it received over HTTP, so a test reads the code or the link the application mailed.
26   */
27  public final class Mailpit {
28  
29      /** Docker image of Mailpit, pinned to a version so a run never depends on a moving tag. */
30      public static final String IMAGE = "axllent/mailpit:v1.29.0";
31  
32      /** Port the SMTP server listens on inside the container. */
33      public static final int SMTP_PORT = 1025;
34  
35      /** Port the HTTP API listens on inside the container. */
36      public static final int API_PORT = 8025;
37  
38      /** Path that answers once the HTTP API is ready. */
39      private static final String READY_PATH = "/api/v1/info";
40  
41      /** Prevents instantiation of this utility class. */
42      private Mailpit() {
43      }
44  
45      /**
46       * Creates a Mailpit container, not yet started.
47       *
48       * @return the container, which waits for the HTTP API to answer before it counts as started.
49       */
50      public static GenericContainer<?> newContainer() {
51          return new GenericContainer<>(DockerImageName.parse(IMAGE))
52                  .withExposedPorts(SMTP_PORT, API_PORT)
53                  .waitingFor(Wait.forHttp(READY_PATH).forPort(API_PORT));
54      }
55  
56      /**
57       * Builds the client of the HTTP API of a started container.
58       *
59       * @param container a started Mailpit container.
60       * @return the client.
61       */
62      public static MailpitClient clientOf(final GenericContainer<?> container) {
63          return new MailpitClient("http://" + container.getHost() + ":" + container.getMappedPort(API_PORT));
64      }
65  }
