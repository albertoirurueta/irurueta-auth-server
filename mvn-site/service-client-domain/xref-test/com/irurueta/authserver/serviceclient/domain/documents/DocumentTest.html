17  package com.irurueta.authserver.serviceclient.domain.documents;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertThrows;
21  
22  import org.junit.jupiter.api.Test;
23  
24  /**
25   * Unit tests of {@link Document}.
26   */
27  class DocumentTest {
28  
29      /** Identifier used by the tests. */
30      private static final String ID = "doc-1";
31  
32      /** Owner used by the tests. */
33      private static final String OWNER = "alice";
34  
35      /** Title used by the tests. */
36      private static final String TITLE = "Title";
37  
38      /** A valid document keeps its components. */
39      @Test
40      void constructor_whenValid_keepsComponents() {
41          final var document = new Document(ID, OWNER, TITLE);
42  
43          assertEquals(ID, document.id());
44          assertEquals(OWNER, document.owner());
45          assertEquals(TITLE, document.title());
46      }
47  
48      /** A null id is rejected. */
49      @Test
50      void constructor_whenIdIsNull_throwsIllegalArgumentException() {
51          final var error = assertThrows(IllegalArgumentException.class, () -> new Document(null, OWNER, TITLE));
52  
53          assertEquals("id is required", error.getMessage());
54      }
55  
56      /** A blank owner is rejected. */
57      @Test
58      void constructor_whenOwnerIsBlank_throwsIllegalArgumentException() {
59          assertThrows(IllegalArgumentException.class, () -> new Document(ID, " ", TITLE));
60      }
61  
62      /** A null title is rejected. */
63      @Test
64      void constructor_whenTitleIsNull_throwsIllegalArgumentException() {
65          assertThrows(IllegalArgumentException.class, () -> new Document(ID, OWNER, null));
66      }
67  
68      /** Two documents with the same components are equal. */
69      @Test
70      void equalsAndHashCode_whenFieldsMatch_areEqual() {
71          final var first = new Document(ID, OWNER, TITLE);
72          final var second = new Document(ID, OWNER, TITLE);
73  
74          assertEquals(first, second);
75          assertEquals(first.hashCode(), second.hashCode());
76      }
77  }
