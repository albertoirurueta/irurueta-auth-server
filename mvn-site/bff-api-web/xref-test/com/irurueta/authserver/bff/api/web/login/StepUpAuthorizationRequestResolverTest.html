16  package com.irurueta.authserver.bff.api.web.login;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertFalse;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  import static org.junit.jupiter.api.Assertions.assertSame;
22  import static org.junit.jupiter.api.Assertions.assertTrue;
23  import static org.mockito.Mockito.when;
24  
25  import java.util.Map;
26  import org.junit.jupiter.api.BeforeEach;
27  import org.junit.jupiter.api.Test;
28  import org.junit.jupiter.api.extension.ExtendWith;
29  import org.mockito.InjectMocks;
30  import org.mockito.Mock;
31  import org.mockito.junit.jupiter.MockitoExtension;
32  import org.springframework.mock.web.MockHttpServletRequest;
33  import org.springframework.mock.web.MockHttpSession;
34  import org.springframework.security.oauth2.client.web.OAuth2AuthorizationRequestResolver;
35  import org.springframework.security.oauth2.core.endpoint.OAuth2AuthorizationRequest;
36  
37  /**
38   * Unit tests of {@link StepUpAuthorizationRequestResolver}.
39   */
40  @ExtendWith(MockitoExtension.class)
41  class StepUpAuthorizationRequestResolverTest {
42  
43      /** Identifier of the client registration. */
44      private static final String REGISTRATION_ID = "bff";
45  
46      /** Authentication context class asked for. */
47      private static final String ACR_VALUES = "urn:irurueta:acr:mfa";
48  
49      /** Maximum age, in seconds, asked for. */
50      private static final long MAX_AGE = 300L;
51  
52      /** A parameter the delegate added, which must be kept. */
53      private static final String NONCE = "nonce-value";
54  
55      /** Builds the authorization request. */
56      @Mock
57      private OAuth2AuthorizationRequestResolver delegate;
58  
59      /** Resolver under test. */
60      @InjectMocks
61      private StepUpAuthorizationRequestResolver resolver;
62  
63      /** The current request. */
64      private MockHttpServletRequest request;
65  
66      /** Authorization request of the delegate. */
67      private OAuth2AuthorizationRequest authorizationRequest;
68  
69      /**
70       * Builds the request and the authorization request of the delegate.
71       */
72      @BeforeEach
73      void setUp() {
74          request = new MockHttpServletRequest();
75          authorizationRequest = OAuth2AuthorizationRequest.authorizationCode()
76                  .authorizationUri("http://localhost:9000/oauth2/authorize")
77                  .clientId(REGISTRATION_ID)
78                  .redirectUri("http://localhost:8080/login/oauth2/code/bff")
79                  .state("state")
80                  .additionalParameters(Map.of("nonce", NONCE))
81                  .build();
82      }
83  
84      /**
85       * Verifies that the step-up values of the session are added to the authorization request, which keeps what the
86       * delegate set, and are removed from the session.
87       */
88      @Test
89      void resolve_whenSessionHoldsStepUpValues_addsThemOnce() {
90          final var session = sessionWith(new StepUpLoginParameters(ACR_VALUES, MAX_AGE));
91          when(delegate.resolve(request)).thenReturn(authorizationRequest);
92  
93          final var resolved = resolver.resolve(request);
94  
95          assertEquals(ACR_VALUES, resolved.getAdditionalParameters()
96                  .get(StepUpAuthorizationRequestResolver.ACR_VALUES_PARAMETER));
97          assertEquals(String.valueOf(MAX_AGE), resolved.getAdditionalParameters()
98                  .get(StepUpAuthorizationRequestResolver.MAX_AGE_PARAMETER));
99          assertEquals(NONCE, resolved.getAdditionalParameters().get("nonce"));
100         assertTrue(resolved.getAuthorizationRequestUri().contains("max_age=" + MAX_AGE));
101         assertNull(session.getAttribute(StepUpAuthorizationRequestResolver.STEP_UP_ATTRIBUTE));
102     }
103 
104     /**
105      * Verifies that only the values given are added.
106      */
107     @Test
108     void resolve_whenOnlyMaxAgeIsKept_addsOnlyMaxAge() {
109         sessionWith(new StepUpLoginParameters(null, MAX_AGE));
110         when(delegate.resolve(request, REGISTRATION_ID)).thenReturn(authorizationRequest);
111 
112         final var parameters = resolver.resolve(request, REGISTRATION_ID).getAdditionalParameters();
113 
114         assertFalse(parameters.containsKey(StepUpAuthorizationRequestResolver.ACR_VALUES_PARAMETER));
115         assertEquals(String.valueOf(MAX_AGE), parameters.get(StepUpAuthorizationRequestResolver.MAX_AGE_PARAMETER));
116     }
117 
118     /**
119      * Verifies that {@code acrValues} alone is added without a maximum age.
120      */
121     @Test
122     void resolve_whenOnlyAcrValuesAreKept_addsOnlyAcrValues() {
123         sessionWith(new StepUpLoginParameters(ACR_VALUES, null));
124         when(delegate.resolve(request)).thenReturn(authorizationRequest);
125 
126         final var parameters = resolver.resolve(request).getAdditionalParameters();
127 
128         assertEquals(ACR_VALUES, parameters.get(StepUpAuthorizationRequestResolver.ACR_VALUES_PARAMETER));
129         assertFalse(parameters.containsKey(StepUpAuthorizationRequestResolver.MAX_AGE_PARAMETER));
130     }
131 
132     /**
133      * Verifies that the authorization request of the delegate is returned unchanged without a session.
134      */
135     @Test
136     void resolve_whenThereIsNoSession_returnsTheDelegatesRequest() {
137         when(delegate.resolve(request)).thenReturn(authorizationRequest);
138 
139         assertSame(authorizationRequest, resolver.resolve(request));
140     }
141 
142     /**
143      * Verifies that the authorization request of the delegate is returned unchanged when the session holds no
144      * step-up values.
145      */
146     @Test
147     void resolve_whenSessionHoldsNoStepUpValues_returnsTheDelegatesRequest() {
148         request.setSession(new MockHttpSession());
149         when(delegate.resolve(request, REGISTRATION_ID)).thenReturn(authorizationRequest);
150 
151         assertSame(authorizationRequest, resolver.resolve(request, REGISTRATION_ID));
152     }
153 
154     /**
155      * Verifies that a request that is not an authorization request stays one, and the step-up values are kept for
156      * the next authorization request.
157      */
158     @Test
159     void resolve_whenDelegateResolvesNothing_returnsNullAndKeepsTheValues() {
160         final var stepUp = new StepUpLoginParameters(ACR_VALUES, MAX_AGE);
161         final var session = sessionWith(stepUp);
162 
163         assertNull(resolver.resolve(request));
164         assertSame(stepUp, session.getAttribute(StepUpAuthorizationRequestResolver.STEP_UP_ATTRIBUTE));
165     }
166 
167     /**
168      * Gives the request a session holding step-up values.
169      *
170      * @param stepUp the step-up values.
171      * @return the session.
172      */
173     private MockHttpSession sessionWith(final StepUpLoginParameters stepUp) {
174         final var session = new MockHttpSession();
175         session.setAttribute(StepUpAuthorizationRequestResolver.STEP_UP_ATTRIBUTE, stepUp);
176         request.setSession(session);
177         return session;
178     }
179 }
