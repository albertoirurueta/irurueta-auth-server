18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  import static org.mockito.Mockito.verify;
22  import static org.mockito.Mockito.when;
23  
24  import com.irurueta.authserver.bff.api.web.v1.generated.model.DocumentResponse;
25  import com.irurueta.authserver.bff.application.documents.DeleteDocumentUseCase;
26  import com.irurueta.authserver.bff.application.documents.ListDocumentsUseCase;
27  import com.irurueta.authserver.bff.domain.documents.Document;
28  import java.util.List;
29  import org.junit.jupiter.api.Test;
30  import org.junit.jupiter.api.extension.ExtendWith;
31  import org.mockito.InjectMocks;
32  import org.mockito.Mock;
33  import org.mockito.Spy;
34  import org.mockito.junit.jupiter.MockitoExtension;
35  import org.springframework.http.HttpStatus;
36  
37  /**
38   * Unit tests of {@link DocumentsController}.
39   */
40  @ExtendWith(MockitoExtension.class)
41  class DocumentsControllerTest {
42  
43      /** Identifier of the document. */
44      private static final String ID = "7";
45  
46      /** Lists the documents. */
47      @Mock
48      private ListDocumentsUseCase listDocumentsUseCase;
49  
50      /** Deletes a document. */
51      @Mock
52      private DeleteDocumentUseCase deleteDocumentUseCase;
53  
54      /** Maps the documents; the real one. */
55      @Spy
56      private DocumentDtoMapper documentDtoMapper;
57  
58      /** Controller under test. */
59      @InjectMocks
60      private DocumentsController controller;
61  
62      /**
63       * Verifies that the documents of the use case are answered as DTOs.
64       */
65      @Test
66      void listDocuments_whenUserHasDocuments_answersThemAsDtos() {
67          when(listDocumentsUseCase.execute()).thenReturn(List.of(new Document(ID, "alice", "Title", "Body")));
68  
69          final var response = controller.listDocuments();
70  
71          assertEquals(HttpStatus.OK, response.getStatusCode());
72          assertEquals(List.of(new DocumentResponse(ID, "alice", "Title", "Body")), response.getBody());
73      }
74  
75      /**
76       * Verifies that a deletion runs the use case and answers 204 without a body.
77       */
78      @Test
79      void deleteDocument_whenDeleted_answersNoContent() {
80          final var response = controller.deleteDocument(ID);
81  
82          verify(deleteDocumentUseCase).execute(ID);
83          assertEquals(HttpStatus.NO_CONTENT, response.getStatusCode());
84          assertNull(response.getBody());
85      }
86  }
