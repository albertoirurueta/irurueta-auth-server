16  package com.irurueta.authserver.bff.api.web.documents;
17  
18  import com.irurueta.authserver.bff.api.web.login.StepUpLoginParameters;
19  import com.irurueta.authserver.bff.domain.documents.DocumentsStepUpRequiredException;
20  import java.util.Arrays;
21  import java.util.LinkedHashMap;
22  import java.util.Map;
23  import java.util.stream.Collectors;
24  import org.springframework.stereotype.Component;
25  
26  /**
27   * Maps a step-up request of the resource server onto the extra members of the {@code step-up-required} problem, the
28   * values the SPA passes back to {@code GET /bff/login}. They are values the resource server chose, so they are kept
29   * within what that endpoint accepts, otherwise the login would answer {@code 400} and the user could never step up:
30   * <ul>
31   *   <li>{@code acrValues}: only the references of the accepted format are kept, and none at all when the result would
32   *       be empty or longer than the login allows;</li>
33   *   <li>{@code maxAge}: capped at the login's maximum, which is stricter than what the resource server asked for, so
34   *       the new sign-in still satisfies it.</li>
35   * </ul>
36   */
37  @Component
38  public class StepUpProblemMapper {
39  
40      /** Problem member carrying the authentication context classes. */
41      static final String ACR_VALUES_PROPERTY = "acrValues";
42  
43      /** Problem member carrying the maximum age, in seconds. */
44      static final String MAX_AGE_PROPERTY = "maxAge";
45  
46      /** Separator of the authentication context class references: any run of whitespace. */
47      private static final String ACR_VALUES_SEPARATOR = "\\s+";
48  
49      /**
50       * Builds the extra problem members of a step-up request.
51       *
52       * @param exception the step-up request, with the values the resource server asked for.
53       * @return the members, in a stable order; empty when the resource server named no usable value.
54       */
55      public Map<String, Object> toProblemProperties(final DocumentsStepUpRequiredException exception) {
56          final var properties = new LinkedHashMap<String, Object>();
57          exception.getAcrValues()
58                  .map(StepUpProblemMapper::acceptedAcrValues)
59                  .filter(acrValues -> !acrValues.isEmpty()
60                          && acrValues.length() <= StepUpLoginParameters.MAX_ACR_VALUES_LENGTH)
61                  .ifPresent(acrValues -> properties.put(ACR_VALUES_PROPERTY, acrValues));
62          exception.getMaxAge().ifPresent(maxAge -> properties.put(MAX_AGE_PROPERTY,
63                  Math.min(maxAge.toSeconds(), StepUpLoginParameters.MAX_AGE_LIMIT_SECONDS)));
64          return properties;
65      }
66  
67      /**
68       * Keeps the authentication context class references of the format the login accepts.
69       *
70       * @param acrValues space-separated references.
71       * @return the accepted ones, separated by one space; empty when none is.
72       */
73      private static String acceptedAcrValues(final String acrValues) {
74          return Arrays.stream(acrValues.trim().split(ACR_VALUES_SEPARATOR))
75                  .filter(acr -> StepUpLoginParameters.ACR_VALUE.matcher(acr).matches())
76                  .collect(Collectors.joining(" "));
77      }
78  }
