17  package com.irurueta.authserver.bff.api.web;
18  
19  import io.swagger.v3.oas.models.Components;
20  import io.swagger.v3.oas.models.OpenAPI;
21  import io.swagger.v3.oas.models.info.Info;
22  import io.swagger.v3.oas.models.info.License;
23  import io.swagger.v3.oas.models.security.SecurityScheme;
24  import org.springdoc.core.models.GroupedOpenApi;
25  import org.springframework.context.annotation.Bean;
26  import org.springframework.context.annotation.Configuration;
27  
28  /**
29   * Runtime OpenAPI document of the JSON API of the BFF, served by springdoc at {@value #API_DOCS_PATH} to a signed-in
30   * user. The contract {@code apis/bff/rest-server/bff-api-v1.yaml} is the source of truth: the operations are
31   * described by the interface generated from it, and this class only adds what springdoc cannot read from it (the
32   * info block and the security scheme). {@code OpenApiDriftIT} fails when the document served differs from the
33   * contract.
34   */
35  @Configuration(proxyBeanMethods = false)
36  public class OpenApiConfig {
37  
38      /** Path of the OpenAPI document. */
39      public static final String API_DOCS_PATH = "/v3/api-docs";
40  
41      /** Name of the session-cookie security scheme the contract's operations refer to. */
42      public static final String SESSION_SCHEME = "session";
43  
44      /** Name of the CSRF security scheme, which protects every state-changing request. */
45      public static final String CSRF_SCHEME = "csrf";
46  
47      /** Name of the session cookie (set by {@code SessionConfiguration} of {@code bff/infrastructure/session}). */
48      static final String SESSION_COOKIE = "__Host-SESSION";
49  
50      /** Name of the header carrying the CSRF token. */
51      static final String CSRF_HEADER = "X-XSRF-TOKEN";
52  
53      /**
54       * Describes the API.
55       *
56       * @return the OpenAPI info block and security scheme.
57       */
58      @Bean
59      public OpenAPI bffOpenApi() {
60          return new OpenAPI()
61                  .info(new Info()
62                          .title("Irurueta BFF API")
63                          .description("JSON endpoints of the example backend for frontend.")
64                          .version("1.0.0")
65                          .license(new License().name("Apache License 2.0")
66                                  .url("https://www.apache.org/licenses/LICENSE-2.0.txt")))
67                  .components(new Components()
68                          .addSecuritySchemes(SESSION_SCHEME, new SecurityScheme()
69                                  .type(SecurityScheme.Type.APIKEY)
70                                  .in(SecurityScheme.In.COOKIE)
71                                  .name(SESSION_COOKIE))
72                          .addSecuritySchemes(CSRF_SCHEME, new SecurityScheme()
73                                  .type(SecurityScheme.Type.APIKEY)
74                                  .in(SecurityScheme.In.HEADER)
75                                  .name(CSRF_HEADER)));
76      }
77  
78      /**
79       * Groups the BFF's own JSON paths ({@code /bff/**}) and the documents API ({@code /api/**}), leaving out the
80       * OAuth 2.0 login endpoints.
81       *
82       * @return the group of API paths.
83       */
84      @Bean
85      public GroupedOpenApi apiGroup() {
86          return GroupedOpenApi.builder()
87                  .group("api")
88                  .pathsToMatch("/bff/**", "/api/**")
89                  .build();
90      }
91  }
