17  package com.irurueta.authserver.bff.api.web.login;
18  
19  import jakarta.servlet.http.HttpServletRequest;
20  import jakarta.servlet.http.HttpServletResponse;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.security.core.Authentication;
23  import org.springframework.security.web.authentication.SimpleUrlAuthenticationSuccessHandler;
24  import org.springframework.stereotype.Component;
25  
26  // tag::bff-login-success[]
27  /**
28   * Sends the browser, once the OAuth2 login has succeeded, back to the {@code returnTo} that {@link BffLoginController}
29   * validated and kept in the session. The attribute is read once and removed, and validated again by
30   * {@link ReturnToValidator} before it is used, so a value that reached the session any other way can never become an
31   * open redirect; when it is absent or not allowed, the browser goes to the root of the application.
32   */
33  @Component
34  @RequiredArgsConstructor
35  public class ReturnToAuthenticationSuccessHandler extends SimpleUrlAuthenticationSuccessHandler {
36  
37      /** Session attribute, owned by the BFF, holding the validated {@code returnTo} of the login in progress. */
38      public static final String RETURN_TO_ATTRIBUTE = ReturnToAuthenticationSuccessHandler.class.getName()
39              + ".RETURN_TO";
40  
41      /** Where the browser goes when there is no allowed {@code returnTo}: the root of the application. */
42      static final String DEFAULT_TARGET = "/";
43  
44      /** Decides which {@code returnTo} values are safe. */
45      private final ReturnToValidator returnToValidator;
46  
47      /**
48       * Takes the {@code returnTo} out of the session and decides where the browser goes.
49       *
50       * @param request        the request of the login callback.
51       * @param response       the response.
52       * @param authentication the new authentication.
53       * @return the validated {@code returnTo}, or {@link #DEFAULT_TARGET}.
54       */
55      @Override
56      protected String determineTargetUrl(final HttpServletRequest request, final HttpServletResponse response,
57              final Authentication authentication) {
58          final var session = request.getSession(false);
59          if (session == null) {
60              return DEFAULT_TARGET;
61          }
62          final var returnTo = session.getAttribute(RETURN_TO_ATTRIBUTE);
63          session.removeAttribute(RETURN_TO_ATTRIBUTE);
64          return returnTo instanceof String target && returnToValidator.isAllowed(target) ? target : DEFAULT_TARGET;
65      }
66  }
67  // end::bff-login-success[]
