37  /**
38   * Integration tests of the sender-constrained tokens the resource server accepts, over a real TLS handshake: the
39   * server presents a certificate issued by an in-test authority and asks the client for one
40   * ({@code server.ssl.client-auth=want}), so a token bound to a certificate ({@code cnf.x5t#S256}, RFC 8705) is only
41   * accepted from the connection that presents that certificate, and a token bound to a key ({@code cnf.jkt}, RFC 9449)
42   * only with a DPoP proof signed by it, never as a plain bearer token. Every key and certificate is generated at run
43   * time; the key store of the server doubles as its trust store because it also holds the authority's certificate.
44   * TLS is pinned to 1.2 because the JSSE implementation of TLS 1.3 has no post-handshake authentication, which
45   * optional client authentication needs, so a client certificate would never reach the server.
46   */
47  @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
48  class BoundTokenIT {
49  
50      /** Issuer claim of the test tokens. */
51      private static final String ISSUER = "http://localhost:9000";
52  
53      /** Subject of the test tokens. */
54      private static final String SUBJECT = "demo";
55  
56      /** Audience the resource server accepts. */
57      private static final List<String> AUDIENCES = List.of("resource-server");
58  
59      /** Scope of the test tokens. */
60      private static final String SCOPE = "documents.read";
61  
62      /** Status of a successful answer. */
63      private static final int OK = 200;
64  
65      /** Status of a refused token. */
66      private static final int UNAUTHORIZED = 401;
67  
68      /** The certificate authority of the test, generated for the run. */
69      private static final TestPki PKI = TestPki.create("test-authority-" + UUID.randomUUID());
70  
71      /** The server's identity, generated for the run. */
72      private static final TestPki.Identity SERVER = PKI.issueServer("localhost");
73  
74      /** The identity of the client the tokens are bound to. */
75      private static final TestPki.Identity CLIENT = PKI.issueClient("bound-client");
76  
77      /** Another identity, trusted by the same authority. */
78      private static final TestPki.Identity OTHER_CLIENT = PKI.issueClient("other-client");
79  
80      /** Port of the running server. */
81      @LocalServerPort
82      private int port;
83  
84      /**
85       * Points the decoder at the generated signing key and turns TLS with optional client authentication on.
86       *
87       * @param registry registry of the test properties.
88       * @throws IOException if the PEM file cannot be written.
89       */
90      @DynamicPropertySource
91      static void tls(final DynamicPropertyRegistry registry) throws IOException {
92          TestTokens.publicKeyLocation(registry);
93          registry.add("server.ssl.key-store", () -> SERVER.keyStore().toString());
94          registry.add("server.ssl.key-store-password", SERVER::password);
95          registry.add("server.ssl.key-store-type", () -> "PKCS12");
96          registry.add("server.ssl.key-alias", () -> "entity");
97          registry.add("server.ssl.trust-store", () -> SERVER.keyStore().toString());
98          registry.add("server.ssl.trust-store-password", SERVER::password);
99          registry.add("server.ssl.trust-store-type", () -> "PKCS12");
100         registry.add("server.ssl.client-auth", () -> "want");
101         registry.add("server.ssl.enabled-protocols", () -> "TLSv1.2");
102     }
103 
104     /**
105      * Removes the generated key stores.
106      */
107     @AfterAll
108     static void removeKeys() {
109         PKI.close();
110     }

112     /**
113      * A certificate-bound token sent over the connection that presents its certificate is accepted.
114      *
115      * @throws Exception if the request cannot be performed.
116      */
117     @Test
118     void protectedEndpoint_whenCertificateBoundTokenComesWithItsCertificate_returnsOk() throws Exception {
119         final var response = get(CLIENT, certificateBoundToken(CLIENT), null);
120 
121         assertEquals(OK, response.statusCode());
122         assertEquals(SUBJECT, response.body());
123     }
124 
125     /**
126      * A certificate-bound token sent over a connection without a client certificate is refused.
127      *
128      * @throws Exception if the request cannot be performed.
129      */
130     @Test
131     void protectedEndpoint_whenCertificateBoundTokenComesWithoutCertificate_returnsUnauthorized() throws Exception {
132         final var response = get(null, certificateBoundToken(CLIENT), null);
133 
134         assertEquals(UNAUTHORIZED, response.statusCode());
135     }
136 
137     /**
138      * A certificate-bound token sent with another certificate is refused.
139      *
140      * @throws Exception if the request cannot be performed.
141      */
142     @Test
143     void protectedEndpoint_whenCertificateBoundTokenComesWithAnotherCertificate_returnsUnauthorized()
144             throws Exception {
145         final var response = get(OTHER_CLIENT, certificateBoundToken(CLIENT), null);
146 
147         assertEquals(UNAUTHORIZED, response.statusCode());
148     }
149 
150     /**
151      * A DPoP-bound token with a valid proof is accepted.
152      *
153      * @throws Exception if the request cannot be performed.
154      */
155     @Test
156     void protectedEndpoint_whenDpopBoundTokenComesWithValidProof_returnsOk() throws Exception {
157         final var key = TestTokens.dpopKey();
158         final var token = dpopBoundToken(key);
159         final var proof = TestTokens.dpopProof(key, "GET", url(), token);
160 
161         final var response = get(null, "DPoP " + token, proof);
162 
163         assertEquals(OK, response.statusCode());
164         assertEquals(SUBJECT, response.body());
165     }
166 
167     /**
168      * A DPoP-bound token sent as a plain bearer token is refused, even with a proof.
169      *
170      * @throws Exception if the request cannot be performed.
171      */
172     @Test
173     void protectedEndpoint_whenDpopBoundTokenIsSentAsBearer_returnsUnauthorized() throws Exception {
174         final var key = TestTokens.dpopKey();
175         final var token = dpopBoundToken(key);
176         final var proof = TestTokens.dpopProof(key, "GET", url(), token);
177 
178         final var response = get(null, "Bearer " + token, proof);
179 
180         assertEquals(UNAUTHORIZED, response.statusCode());
181     }
182 
183     /**
184      * A DPoP-bound token sent with the DPoP scheme but no proof is refused.
185      *
186      * @throws Exception if the request cannot be performed.
187      */
188     @Test
189     void protectedEndpoint_whenDpopBoundTokenComesWithoutProof_returnsUnauthorized() throws Exception {
190         final var response = get(null, "DPoP " + dpopBoundToken(TestTokens.dpopKey()), null);
191 
192         assertEquals(UNAUTHORIZED, response.statusCode());
193     }
194 
195     /**
196      * A proof signed by another key than the one the token is bound to is refused.
197      *
198      * @throws Exception if the request cannot be performed.
199      */
200     @Test
201     void protectedEndpoint_whenProofIsSignedByAnotherKey_returnsUnauthorized() throws Exception {
202         final var token = dpopBoundToken(TestTokens.dpopKey());
203         final var proof = TestTokens.dpopProof(TestTokens.dpopKey(), "GET", url(), token);
204 
205         final var response = get(null, "DPoP " + token, proof);
206 
207         assertEquals(UNAUTHORIZED, response.statusCode());
208     }

210     /**
211      * Builds a token bound to the certificate of an identity.
212      *
213      * @param identity the identity.
214      * @return the value of the {@code Authorization} header carrying the compact token as a bearer token.
215      */
216     private static String certificateBoundToken(final TestPki.Identity identity) {
217         return "Bearer " + TestTokens.tokenValue(ISSUER, SUBJECT, AUDIENCES, SCOPE, List.of(),
218                 TestTokens.certificateBound(identity.certificate().certificate()));
219     }
220 
221     /**
222      * Builds a token bound to a DPoP key.
223      *
224      * @param key the key.
225      * @return the compact token.
226      */
227     private static String dpopBoundToken(final ECKey key) {
228         return TestTokens.tokenValue(ISSUER, SUBJECT, AUDIENCES, SCOPE, List.of(), TestTokens.dpopBound(key));
229     }

231     /**
232      * Gives the URL of the protected endpoint.
233      *
234      * @return the URL.
235      */
236     private String url() {
237         return "https://localhost:" + port + ProtectedTestController.PATH;
238     }
239 
240     /**
241      * Calls the protected endpoint.
242      *
243      * @param identity      identity the client presents, {@code null} for none.
244      * @param authorization value of the {@code Authorization} header.
245      * @param proof         value of the {@code DPoP} header, {@code null} for none.
246      * @return the response.
247      * @throws Exception if the request cannot be performed.
248      */
249     private HttpResponse<String> get(final TestPki.Identity identity, final String authorization,
250                                      final String proof) throws Exception {
251         final var request = HttpRequest.newBuilder(URI.create(url())).header("Authorization", authorization);
252         if (proof != null) {
253             request.header("DPoP", proof);
254         }
255         try (var client = HttpClient.newBuilder().sslContext(PKI.clientContext(identity)).build()) {
256             return client.send(request.GET().build(), HttpResponse.BodyHandlers.ofString());
257         }
258     }
