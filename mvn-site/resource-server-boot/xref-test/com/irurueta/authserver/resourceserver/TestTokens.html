54  /**
55   * Signs the JWTs the resource server integration tests present, with an RSA key generated when the class loads. The
56   * public half is handed to the application through
57   * {@code spring.security.oauth2.resourceserver.jwt.public-key-location}; Spring Boot only builds a decoder from that
58   * key when {@code issuer-uri} is blank (otherwise it resolves the issuer's metadata over HTTP), so the tests blank it.
59   * No key is ever committed.
60   */
61  final class TestTokens {
62  
63      /** Lifetime of the test tokens. */
64      static final Duration TOKEN_TTL = Duration.ofMinutes(5);
65  
66      /** Size in bits of the generated RSA key. */
67      private static final int KEY_SIZE = 2048;
68  
69      /** Number of base64 characters per line of a PEM document. */
70      private static final int PEM_LINE_LENGTH = 64;
71  
72      /** Key pair that signs the test tokens. */
73      private static final KeyPair KEY_PAIR = generateKeyPair();
74  
75      /** Utility class, not meant to be instantiated. */
76      private TestTokens() {
77      }
78  
79      /**
80       * Points the resource server's decoder at the public half of the generated key.
81       *
82       * @param registry registry of the test properties.
83       * @throws IOException if the PEM file cannot be written.
84       */
85      static void publicKeyLocation(final DynamicPropertyRegistry registry) throws IOException {
86          final var pem = Files.createTempFile("resource-server-test-key", ".pem");
87          pem.toFile().deleteOnExit();
88          Files.writeString(pem, toPem((RSAPublicKey) KEY_PAIR.getPublic()), StandardCharsets.US_ASCII);
89          registry.add("spring.security.oauth2.resourceserver.jwt.public-key-location", () -> pem.toUri().toString());
90          registry.add("spring.security.oauth2.resourceserver.jwt.issuer-uri", () -> "");
91      }
92  
93      /**
94       * Builds an {@code Authorization} header value carrying a token signed with the generated key.
95       *
96       * @param issuer    issuer claim of the token.
97       * @param subject   subject claim of the token.
98       * @param audiences audience claim of the token; empty to leave the claim out.
99       * @param scope     space-separated scopes of the token; empty to leave the claim out.
100      * @param roles     roles of the caller; empty to leave the claim out.
101      * @param extra     further claims of the token.
102      * @return the header value.
103      */
104     static String bearer(final String issuer, final String subject, final List<String> audiences, final String scope,
105                          final List<String> roles, final Map<String, Object> extra) {
106         return "Bearer " + tokenValue(issuer, subject, audiences, scope, roles, extra);
107     }
108 
109     /**
110      * Builds a token signed with the generated key, without any scheme prefix.
111      *
112      * @param issuer    issuer claim of the token.
113      * @param subject   subject claim of the token.
114      * @param audiences audience claim of the token; empty to leave the claim out.
115      * @param scope     space-separated scopes of the token; empty to leave the claim out.
116      * @param roles     roles of the caller; empty to leave the claim out.
117      * @param extra     further claims of the token.
118      * @return the compact token.
119      */
120     static String tokenValue(final String issuer, final String subject, final List<String> audiences,
121                              final String scope, final List<String> roles, final Map<String, Object> extra) {
122         final var rsaKey = new RSAKey.Builder((RSAPublicKey) KEY_PAIR.getPublic())
123                 .privateKey((RSAPrivateKey) KEY_PAIR.getPrivate())
124                 .build();
125         final var encoder = new NimbusJwtEncoder(new ImmutableJWKSet<>(new JWKSet(rsaKey)));
126         final var now = Instant.now();
127         final var claims = JwtClaimsSet.builder()
128                 .issuer(issuer)
129                 .subject(subject)
130                 .issuedAt(now)
131                 .expiresAt(now.plus(TOKEN_TTL));
132         if (!audiences.isEmpty()) {
133             claims.audience(audiences);
134         }
135         if (!scope.isEmpty()) {
136             claims.claim("scope", scope);
137         }
138         if (!roles.isEmpty()) {
139             claims.claim("roles", roles);
140         }
141         extra.forEach(claims::claim);
142         return encoder.encode(JwtEncoderParameters.from(claims.build())).getTokenValue();
143     }

145     /**
146      * Builds the {@code cnf} claim of a token bound to a TLS client certificate (RFC 8705).
147      *
148      * @param certificate the client certificate.
149      * @return the extra claims to pass to {@link #tokenValue}.
150      */
151     static Map<String, Object> certificateBound(final X509Certificate certificate) {
152         try {
153             final var digest = MessageDigest.getInstance("SHA-256").digest(certificate.getEncoded());
154             return Map.of("cnf", Map.of("x5t#S256", Base64.getUrlEncoder().withoutPadding().encodeToString(digest)));
155         } catch (final GeneralSecurityException e) {
156             throw new IllegalStateException("Cannot compute the certificate thumbprint", e);
157         }
158     }
159 
160     /**
161      * Generates an EC key pair for a DPoP client.
162      *
163      * @return the key, with its private half.
164      */
165     static ECKey dpopKey() {
166         try {
167             return new ECKeyGenerator(Curve.P_256).generate();
168         } catch (final JOSEException e) {
169             throw new IllegalStateException("Cannot generate the DPoP key", e);
170         }
171     }
172 
173     /**
174      * Builds the {@code cnf} claim of a token bound to a DPoP key (RFC 9449).
175      *
176      * @param key the DPoP key.
177      * @return the extra claims to pass to {@link #tokenValue}.
178      */
179     static Map<String, Object> dpopBound(final ECKey key) {
180         try {
181             return Map.of("cnf", Map.of("jkt", key.computeThumbprint().toString()));
182         } catch (final JOSEException e) {
183             throw new IllegalStateException("Cannot compute the key thumbprint", e);
184         }
185     }
186 
187     /**
188      * Builds a DPoP proof for a request.
189      *
190      * @param key         the DPoP key.
191      * @param method      HTTP method of the request.
192      * @param url         full URL of the request.
193      * @param accessToken the access token the proof is bound to ({@code ath}).
194      * @return the compact proof.
195      */
196     static String dpopProof(final ECKey key, final String method, final String url, final String accessToken) {
197         try {
198             final var hash = MessageDigest.getInstance("SHA-256")
199                     .digest(accessToken.getBytes(StandardCharsets.US_ASCII));
200             final var claims = new JWTClaimsSet.Builder()
201                     .jwtID(UUID.randomUUID().toString())
202                     .claim("htm", method)
203                     .claim("htu", url)
204                     .claim("ath", Base64.getUrlEncoder().withoutPadding().encodeToString(hash))
205                     .issueTime(new Date())
206                     .build();
207             final var header = new JWSHeader.Builder(JWSAlgorithm.ES256)
208                     .type(new JOSEObjectType("dpop+jwt"))
209                     .jwk(key.toPublicJWK())
210                     .build();
211             final var jwt = new SignedJWT(header, claims);
212             jwt.sign(new ECDSASigner(key));
213             return jwt.serialize();
214         } catch (final GeneralSecurityException | JOSEException e) {
215             throw new IllegalStateException("Cannot build the DPoP proof", e);
216         }
217     }
218 
219     /**
220      * Encodes a public key as a PEM document with {@code \n} line endings.
221      *
222      * @param key the public key.
223      * @return the PEM document.
224      */
225     private static String toPem(final RSAPublicKey key) {
226         final var encoder = Base64.getMimeEncoder(PEM_LINE_LENGTH, "\n".getBytes(StandardCharsets.US_ASCII));
227         return "-----BEGIN PUBLIC KEY-----\n" + encoder.encodeToString(key.getEncoded())
228                 + "\n-----END PUBLIC KEY-----\n";
229     }
230 
231     /**
232      * Generates the RSA key pair that signs the test tokens.
233      *
234      * @return the key pair.
235      */
236     private static KeyPair generateKeyPair() {
237         try {
238             final var generator = KeyPairGenerator.getInstance("RSA");
239             generator.initialize(KEY_SIZE);
240             return generator.generateKeyPair();
241         } catch (final NoSuchAlgorithmException e) {
242             throw new IllegalStateException("RSA is not available", e);
243         }
244     }
