46  /**
47   * Integration tests of {@link com.irurueta.authserver.resourceserver.api.web.security.SecurityConfig} through the
48   * whole filter chain, against the test-only {@link ProtectedTestController}.
49   *
50   * <p>The real-token tests sign JWTs with an RSA key generated here and point
51   * {@code spring.security.oauth2.resourceserver.jwt.public-key-location} at its public half. Spring Boot only builds
52   * a decoder from that key when {@code issuer-uri} is blank (otherwise it resolves the issuer's metadata over HTTP),
53   * so the tests blank it. The configured audiences still apply: the audience validator the application really uses
54   * is exercised without starting the authorization server. No key is ever committed.
55   */
56  @SpringBootTest
57  @AutoConfigureMockMvc
58  class SecurityConfigIT {
59  
60      /** Subject of the test bearer token. */
61      private static final String SUBJECT = "demo";
62  
63      /** Issuer claim of the test tokens. */
64      private static final String ISSUER = "http://localhost:9000";
65  
66      /** Audience the resource server accepts by default. */
67      private static final String ACCEPTED_AUDIENCE = "resource-server";
68  
69      /** Owner of the documents that are not the caller's. */
70      private static final String OTHER_OWNER = "someone-else";
71  
72      /** Audience of a token issued to another client. */
73      private static final String OTHER_AUDIENCE = "bff";
74  
75      /** Lifetime of the test tokens. */
76      private static final Duration TOKEN_TTL = TestTokens.TOKEN_TTL;
77  
78      /** Scope that allows reading documents. */
79      private static final String READ_SCOPE = "documents.read";
80  
81      /** Scope that allows deleting documents. */
82      private static final String DELETE_SCOPE = "documents.delete";
83  
84      /** The RFC 9470 challenge the default policy answers. */
85      private static final String STEP_UP_CHALLENGE = "Bearer error=\"insufficient_user_authentication\", "
86              + "acr_values=\"urn:irurueta:acr:mfa\", max_age=\"300\"";

192     /**
193      * Verifies that a caller sees only their own documents.
194      *
195      * @throws Exception if the request cannot be performed.
196      */
197     @Test
198     void documents_whenTokenHasTheReadScope_listsOnlyTheCallersDocuments() throws Exception {
199         documentRepository.save(new Document("it-own", SUBJECT, "Own", "text"));
200         documentRepository.save(new Document("it-foreign", "someone-else", "Foreign", "text"));
201 
202         mockMvc.perform(get(DocumentController.PATH)
203                         .header(HttpHeaders.AUTHORIZATION, bearer(ISSUER, List.of(ACCEPTED_AUDIENCE), READ_SCOPE)))
204                 .andExpect(status().isOk())
205                 .andExpect(jsonPath("$[?(@.id == 'it-own')]").exists())
206                 .andExpect(jsonPath("$[?(@.id == 'it-foreign')]").doesNotExist());
207     }

210      * Verifies that deleting needs the delete scope.
211      *
212      * @throws Exception if the request cannot be performed.
213      */
214     @Test
215     void documents_whenTokenLacksTheDeleteScope_deleteIsForbidden() throws Exception {
216         documentRepository.save(new Document("it-keep", SUBJECT, "Keep", "text"));

218         mockMvc.perform(delete(DocumentController.PATH + "/it-keep")
219                         .header(HttpHeaders.AUTHORIZATION, bearer(ISSUER, List.of(ACCEPTED_AUDIENCE), READ_SCOPE)))

221                 .andExpect(header().string(HttpHeaders.WWW_AUTHENTICATE, containsString("insufficient_scope")));

226     /**
227      * Verifies that a caller with the delete scope deletes a document, and that deleting it again finds nothing.
228      *
229      * @throws Exception if the request cannot be performed.
230      */
231     @Test
232     void documents_whenTokenHasTheDeleteScope_deletesOnceThenNotFound() throws Exception {
233         documentRepository.save(new Document("it-delete", SUBJECT, "Delete", "text"));
234         final var header = strongBearer(List.of(), Instant.now());

236         mockMvc.perform(delete(DocumentController.PATH + "/it-delete").header(HttpHeaders.AUTHORIZATION, header))

238         mockMvc.perform(delete(DocumentController.PATH + "/it-delete").header(HttpHeaders.AUTHORIZATION, header))
239                 .andExpect(status().isNotFound());

275     /**
276      * Verifies that a token with the delete scope but the password-level {@code acr} is answered with the RFC 9470
277      * step-up challenge and leaves the document alone.
278      *
279      * @throws Exception if the request cannot be performed.
280      */
281     @Test
282     void documents_whenAcrIsPassword_deleteRequiresStepUp() throws Exception {
283         documentRepository.save(new Document("it-step-up-acr", SUBJECT, "Keep", "text"));
284 
285         mockMvc.perform(delete(DocumentController.PATH + "/it-step-up-acr").header(HttpHeaders.AUTHORIZATION,
286                         bearer(ISSUER, List.of(ACCEPTED_AUDIENCE), DELETE_SCOPE, List.of(),
287                                 Map.of("acr", "urn:irurueta:acr:pwd", "auth_time", Instant.now().getEpochSecond()))))
288                 .andExpect(status().isUnauthorized())
289                 .andExpect(header().string(HttpHeaders.WWW_AUTHENTICATE, STEP_UP_CHALLENGE));
290         assertEquals(1, documentRepository.findByOwner(SUBJECT).stream()
291                 .filter(document -> "it-step-up-acr".equals(document.id())).count());
292     }

320     /**
321      * Builds an {@code Authorization} header value for a token with the delete scope and the strong {@code acr}.
322      *
323      * @param roles    roles of the caller; empty to leave the claim out.
324      * @param authTime time the user authenticated.
325      * @return the header value.
326      */
327     private static String strongBearer(final List<String> roles, final Instant authTime) {
328         return bearer(ISSUER, List.of(ACCEPTED_AUDIENCE), DELETE_SCOPE, roles,
329                 Map.of("acr", "urn:irurueta:acr:mfa", "auth_time", authTime.getEpochSecond()));
330     }
