18  package com.irurueta.authserver.serviceclient;
19  
20  import com.irurueta.authserver.serviceclient.application.documents.ListDocumentsUseCase;
21  import com.irurueta.authserver.serviceclient.infrastructure.resourceserver.ServiceTokens;
22  import lombok.RequiredArgsConstructor;
23  import org.springframework.boot.CommandLineRunner;
24  import org.springframework.context.annotation.Profile;
25  import org.springframework.stereotype.Component;
26  import tools.jackson.databind.json.JsonMapper;
27  
28  /**
29   * Command-line client of the OAuth 2.0 client credentials grant: obtains an access token with its own credentials,
30   * lists the documents the resource server shows it and prints each one as a JSON line.
31   * <p>
32   * <b>Running it from the packaged jar</b>:
33   * <pre>
34   * java -jar service-client-boot.jar --spring.profiles.active=client-credentials \
35   *     --service-client.auth.token-uri=http://localhost:9000/oauth2/token \
36   *     --service-client.resource-server.base-uri=http://localhost:8081
37   * </pre>
38   * with {@code SERVICE_CLIENT_ID} and {@code SERVICE_CLIENT_SECRET} (or {@code --service-client.auth.method} and its
39   * key or SSL bundle for the other authentication methods, see {@code application-client-credentials.yml}). With
40   * {@code --service-client.cli.print-access-token=true} a last {@code Access token: ...} line prints the token. On any
41   * failure the process ends with a non-zero exit code.
42   * <p>
43   * <b>Why it lives in {@code src/main}</b>: the end-to-end test runs the packaged service-client jar, so the runner
44   * has to be in it; it is a genuine, reusable tool of the application and not a test double.
45   */
46  @Component
47  @Profile(ClientCredentialsCli.PROFILE)
48  @RequiredArgsConstructor
49  public class ClientCredentialsCli implements CommandLineRunner {
50  
51      /** Name of the Spring profile that turns the runner on. */
52      public static final String PROFILE = "client-credentials";
53  
54      /** Lists the documents. */
55      private final ListDocumentsUseCase listDocumentsUseCase;
56  
57      /** Gives the token the call used. */
58      private final ServiceTokens serviceTokens;
59  
60      /** Writes the lines. */
61      private final CliOutput output;
62  
63      /** Writes a document as JSON. */
64      private final JsonMapper jsonMapper;
65  
66      /** The settings. */
67      private final CliProperties properties;
68  
69      /**
70       * Lists the documents and, when asked, prints the token.
71       *
72       * @param args ignored: settings arrive as Spring properties.
73       * @throws com.irurueta.authserver.serviceclient.domain.documents.DocumentAccessException if the token cannot be
74       *         obtained or the resource server refuses the call.
75       */
76      @Override
77      public void run(final String... args) {
78          listDocumentsUseCase.execute().forEach(document -> output.line(jsonMapper.writeValueAsString(document)));
79          if (properties.printAccessToken()) {
80              output.line("Access token: " + serviceTokens.clientCredentialsToken());
81          }
82      }
83  }
