17  package com.irurueta.authserver.serviceclient.device;
18  
19  import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
20  import com.fasterxml.jackson.annotation.JsonProperty;
21  import com.fasterxml.jackson.annotation.JsonSetter;
22  import com.fasterxml.jackson.annotation.Nulls;
23  
24  /**
25   * Response of the device authorization endpoint (RFC 8628 section 3.2).
26   *
27   * @param deviceCode              the code the CLI presents when polling; a secret, never printed.
28   * @param userCode                the code the user types (or confirms) on the verification page.
29   * @param verificationUri         the page where the user approves the request.
30   * @param verificationUriComplete the same page with the user code included; may be {@code null} (optional in the RFC).
31   * @param expiresIn               seconds the device code lives; the default when the server sends none or sends
32   *                                {@code null}.
33   * @param interval                minimum seconds between two polls; the default when the server sends none or
34   *                                sends {@code null} (Spring Authorization Server does not send one).
35   */
36  @JsonIgnoreProperties(ignoreUnknown = true)
37  public record DeviceAuthorization(
38          @JsonProperty("device_code") String deviceCode,
39          @JsonProperty("user_code") String userCode,
40          @JsonProperty("verification_uri") String verificationUri,
41          @JsonProperty("verification_uri_complete") String verificationUriComplete,
42          @JsonProperty("expires_in") @JsonSetter(nulls = Nulls.AS_EMPTY) long expiresIn,
43          @JsonProperty("interval") @JsonSetter(nulls = Nulls.AS_EMPTY) long interval) {
44  
45      /**
46       * Lifetime assumed when the server does not say (the Spring Authorization Server default is five minutes).
47       */
48      public static final long DEFAULT_EXPIRES_IN_SECONDS = 300L;
49  
50      /**
51       * Polling interval assumed when the server does not say (RFC 8628 section 3.2).
52       */
53      public static final long DEFAULT_INTERVAL_SECONDS = 5L;
54  
55      /**
56       * Validates the mandatory members and applies the defaults of the optional ones.
57       *
58       * @param deviceCode              the device code.
59       * @param userCode                the user code.
60       * @param verificationUri         the verification page.
61       * @param verificationUriComplete the verification page with the user code; optional.
62       * @param expiresIn               seconds the device code lives; zero or negative means the default.
63       * @param interval                seconds between polls; zero or negative means the default.
64       */
65      public DeviceAuthorization {
66          if (isBlank(deviceCode) || isBlank(userCode) || isBlank(verificationUri)) {
67              throw new DeviceFlowException("The device authorization response lacks device_code, user_code or "
68                      + "verification_uri");
69          }
70          expiresIn = expiresIn > 0 ? expiresIn : DEFAULT_EXPIRES_IN_SECONDS;
71          interval = interval > 0 ? interval : DEFAULT_INTERVAL_SECONDS;
72      }
73  
74      /**
75       * Describes the authorization without the device code, which is a secret.
76       *
77       * @return a description without secrets.
78       */
79      @Override
80      public String toString() {
81          return "DeviceAuthorization[userCode=" + userCode + ", verificationUri=" + verificationUri
82                  + ", expiresIn=" + expiresIn + ", interval=" + interval + "]";
83      }
84  
85      /**
86       * Tells whether a text is missing or blank.
87       *
88       * @param value the text.
89       * @return {@code true} when it has no content.
90       */
91      private static boolean isBlank(final String value) {
92          return value == null || value.isBlank();
93      }
94  }
