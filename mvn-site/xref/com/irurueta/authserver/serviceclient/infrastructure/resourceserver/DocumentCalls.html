18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import com.irurueta.authserver.serviceclient.domain.documents.Document;
21  import com.irurueta.authserver.serviceclient.domain.documents.DocumentAccessException;
22  import com.irurueta.authserver.serviceclient.infrastructure.resourceserver.v1.generated.api.DocumentsApi;
23  import java.util.List;
24  import lombok.RequiredArgsConstructor;
25  import lombok.extern.slf4j.Slf4j;
26  import org.springframework.security.oauth2.core.OAuth2AuthorizationException;
27  import org.springframework.stereotype.Component;
28  import org.springframework.web.client.RestClientException;
29  import org.springframework.web.client.RestClientResponseException;
30  
31  /**
32   * Lists the documents through a generated client and translates every failure into a {@link
33   * DocumentAccessException}, so no HTTP, OAuth2 or JSON type escapes through the ports. Messages and logs never carry
34   * a token.
35   */
36  @Slf4j
37  @Component
38  @RequiredArgsConstructor
39  public class DocumentCalls {
40  
41      /** Maps the generated model onto the domain. */
42      private final DocumentMapper documentMapper;
43  
44      // tag::service-client-gateway[]
45      /**
46       * Lists the documents the token behind the client may see.
47       *
48       * @param documentsApi the generated client, which sends the access token.
49       * @return the documents, empty when there are none.
50       * @throws DocumentAccessException if the token cannot be obtained, the resource server refuses it, cannot be
51       *                                 reached or answers something that is not a list of documents.
52       */
53      public List<Document> list(final DocumentsApi documentsApi) {
54          try {
55              final var documents = documentsApi.listDocuments();
56              return documents == null ? List.of() : documents.stream().map(documentMapper::toDomain).toList();
57          } catch (final OAuth2AuthorizationException e) {
58              log.warn("No access token for the resource server: {}", e.getError().getErrorCode());
59              throw new DocumentAccessException("The access token could not be obtained", e);
60          } catch (final RestClientResponseException e) {
61              log.warn("The resource server answered {}", e.getStatusCode().value());
62              throw new DocumentAccessException("The resource server answered " + e.getStatusCode().value(), e);
63          } catch (final RestClientException e) {
64              log.warn("The resource server cannot be reached: {}", e.getClass().getSimpleName());
65              throw new DocumentAccessException("The resource server is unavailable", e);
66          } catch (final IllegalArgumentException e) {
67              log.warn("The resource server answered an invalid document");
68              throw new DocumentAccessException("The resource server answered an invalid document", e);
69          }
70      }
71      // end::service-client-gateway[]
72  }
