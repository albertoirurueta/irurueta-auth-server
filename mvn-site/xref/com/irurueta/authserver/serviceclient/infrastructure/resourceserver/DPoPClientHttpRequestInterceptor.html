18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import java.io.IOException;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.http.HttpHeaders;
23  import org.springframework.http.HttpRequest;
24  import org.springframework.http.client.ClientHttpRequestExecution;
25  import org.springframework.http.client.ClientHttpRequestInterceptor;
26  import org.springframework.http.client.ClientHttpResponse;
27  import org.springframework.security.core.Authentication;
28  import org.springframework.security.oauth2.client.ClientAuthorizationException;
29  import org.springframework.security.oauth2.client.OAuth2AuthorizeRequest;
30  import org.springframework.security.oauth2.client.OAuth2AuthorizedClientManager;
31  import org.springframework.security.oauth2.core.OAuth2AccessToken;
32  import org.springframework.security.oauth2.core.OAuth2Error;
33  import org.springframework.security.oauth2.core.OAuth2ErrorCodes;
34  
35  /**
36   * Sends the access token of an authorized client as {@code Authorization: DPoP <token>} together with a {@code DPoP}
37   * proof that binds the request and the token, when the token is DPoP-bound. Spring Security's own interceptor can
38   * only set a bearer header, so this one replaces it when DPoP is enabled. A call without a token is never sent.
39   */
40  @RequiredArgsConstructor
41  public class DPoPClientHttpRequestInterceptor implements ClientHttpRequestInterceptor {
42  
43      /** Name of the header that carries the proof. */
44      public static final String DPOP_HEADER = "DPoP";
45  
46      /** Scheme of a DPoP-bound token in the {@code Authorization} header. */
47      private static final String DPOP_SCHEME = "DPoP ";
48  
49      /** Obtains (and renews) the access token. */
50      private final OAuth2AuthorizedClientManager manager;
51  
52      /** Creates the proofs. */
53      private final DPoPProofs proofs;
54  
55      /** Registration of the client that obtains the token. */
56      private final String registrationId;
57  
58      /** Principal the token is cached under. */
59      private final Authentication principal;
60  
61      // tag::dpop-client[]
62      @Override
63      public ClientHttpResponse intercept(final HttpRequest request, final byte[] body,
64              final ClientHttpRequestExecution execution) throws IOException {
65          final var authorizeRequest = OAuth2AuthorizeRequest.withClientRegistrationId(registrationId)
66                  .principal(principal)
67                  .build();
68          final var authorizedClient = manager.authorize(authorizeRequest);
69          if (authorizedClient == null) {
70              throw new ClientAuthorizationException(new OAuth2Error(OAuth2ErrorCodes.ACCESS_DENIED,
71                      "The service client has no access token", null), registrationId);
72          }
73          final var token = authorizedClient.getAccessToken();
74          if (OAuth2AccessToken.TokenType.DPOP.equals(token.getTokenType())) {
75              request.getHeaders().set(HttpHeaders.AUTHORIZATION, DPOP_SCHEME + token.getTokenValue());
76              request.getHeaders().set(DPOP_HEADER,
77                      proofs.proof(request.getMethod().name(), request.getURI(), token.getTokenValue()));
78          } else {
79              request.getHeaders().setBearerAuth(token.getTokenValue());
80          }
81          return execution.execute(request, body);
82      }
83      // end::dpop-client[]
84  }
