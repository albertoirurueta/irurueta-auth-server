18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import java.net.URI;
21  import java.time.Duration;
22  import java.util.List;
23  import org.springframework.boot.context.properties.ConfigurationProperties;
24  
25  /**
26   * Settings of the service client at the authorization server ({@code service-client.auth.*}). Every one has a default
27   * so that the application starts under any profile; the CLI profiles supply the real values.
28   *
29   * @param method        how the client authenticates, {@code client_secret_basic} by default.
30   * @param clientId      identifier of the client, {@code service-client} by default.
31   * @param clientSecret  shared secret of {@code client_secret_basic}; empty for the other methods.
32   * @param tokenUri      token endpoint, {@code http://localhost:9000/oauth2/token} by default.
33   * @param scopes        scopes requested, {@code documents.read} by default.
34   * @param refreshBefore how long before its expiry a cached access token is renewed, {@code PT60S} by default.
35   * @param dpop          whether tokens are bound to a proof-of-possession key (RFC 9449).
36   * @param sslBundle     name of the Spring Boot SSL bundle that holds the client certificate and the trust material;
37   *                      needed by the mutual TLS methods (it is also used for the calls to the resource server), and
38   *                      used by the other methods only when set.
39   */
40  @ConfigurationProperties(prefix = "service-client.auth")
41  public record ServiceClientAuthProperties(AuthMethod method, String clientId, String clientSecret, URI tokenUri,
42          List<String> scopes, Duration refreshBefore, boolean dpop, String sslBundle) {
43  
44      /** Client identifier used when nothing is configured. */
45      public static final String DEFAULT_CLIENT_ID = "service-client";
46  
47      /** Token endpoint used when nothing is configured. */
48      public static final URI DEFAULT_TOKEN_URI = URI.create("http://localhost:9000/oauth2/token");
49  
50      /** Scope requested when nothing is configured. */
51      public static final String DEFAULT_SCOPE = "documents.read";
52  
53      /** Seconds before the expiry of a token at which it is renewed, when nothing is configured. */
54      public static final long DEFAULT_REFRESH_BEFORE_SECONDS = 60L;
55  
56      /**
57       * Applies the defaults and validates the values.
58       *
59       * @throws IllegalArgumentException if the refresh margin is negative.
60       */
61      public ServiceClientAuthProperties {
62          method = method == null ? AuthMethod.CLIENT_SECRET_BASIC : method;
63          clientId = clientId == null || clientId.isBlank() ? DEFAULT_CLIENT_ID : clientId;
64          clientSecret = clientSecret == null ? "" : clientSecret;
65          tokenUri = tokenUri == null ? DEFAULT_TOKEN_URI : tokenUri;
66          scopes = scopes == null || scopes.isEmpty() ? List.of(DEFAULT_SCOPE) : List.copyOf(scopes);
67          refreshBefore = refreshBefore == null ? Duration.ofSeconds(DEFAULT_REFRESH_BEFORE_SECONDS) : refreshBefore;
68          if (refreshBefore.isNegative()) {
69              throw new IllegalArgumentException("service-client.auth.refresh-before must not be negative");
70          }
71      }
72  
73      /**
74       * Describes the settings without the secret.
75       *
76       * @return the description.
77       */
78      @Override
79      public String toString() {
80          return "ServiceClientAuthProperties[method=" + method + ", clientId=" + clientId + ", tokenUri=" + tokenUri
81                  + ", scopes=" + scopes + ", refreshBefore=" + refreshBefore + ", dpop=" + dpop + ", sslBundle="
82                  + sslBundle + "]";
83      }
84  }
