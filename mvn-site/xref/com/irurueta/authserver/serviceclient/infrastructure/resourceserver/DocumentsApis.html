18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import com.irurueta.authserver.serviceclient.infrastructure.resourceserver.v1.generated.ApiClient;
21  import com.irurueta.authserver.serviceclient.infrastructure.resourceserver.v1.generated.api.DocumentsApi;
22  import lombok.RequiredArgsConstructor;
23  import org.springframework.security.core.Authentication;
24  import org.springframework.stereotype.Component;
25  import org.springframework.web.client.RestClient;
26  
27  /**
28   * Builds the documents client generated from the resource server's contract on top of a REST client that sends the
29   * access token of an authorized client (so the generated client's own authentication settings stay unused).
30   */
31  @Component
32  @RequiredArgsConstructor
33  public class DocumentsApis {
34  
35      /** Template of the REST clients; it is never used itself, only copied (the builder is a prototype bean). */
36      private final RestClient.Builder restClientBuilder;
37  
38      /** Where the resource server is. */
39      private final ResourceServerProperties properties;
40  
41      /** Request factory over the SSL bundle, when there is one. */
42      private final SslRequestFactories requestFactories;
43  
44      /** Interceptors that obtain and send the access token. */
45      private final ResourceServerInterceptors interceptors;
46  
47      // tag::service-client-documents-client[]
48      /**
49       * Builds a documents client whose requests carry the access token of one authorized client.
50       *
51       * @param registrationId registration of the OAuth2 client that obtains the token.
52       * @param principal      principal the token is cached under (the service client, or a user for a token exchange).
53       * @return the documents client.
54       */
55      public DocumentsApi forPrincipal(final String registrationId, final Authentication principal) {
56          final var builder = restClientBuilder.clone();
57          requestFactories.create().ifPresent(builder::requestFactory);
58          final var restClient = builder
59                  .requestInterceptors(list -> list.addAll(interceptors.create(registrationId, principal)))
60                  .build();
61          final var apiClient = new ApiClient(restClient)
62                  .setBasePath(properties.baseUri().toString().replaceFirst("/+$", ""));
63          return new DocumentsApi(apiClient);
64      }
65      // end::service-client-documents-client[]
66  }
