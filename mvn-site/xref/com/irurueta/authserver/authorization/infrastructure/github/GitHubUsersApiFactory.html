16  package com.irurueta.authserver.authorization.infrastructure.github;
17  
18  import com.irurueta.authserver.authorization.infrastructure.github.v1.generated.ApiClient;
19  import com.irurueta.authserver.authorization.infrastructure.github.v1.generated.api.UsersApi;
20  import org.springframework.stereotype.Component;
21  import org.springframework.web.client.RestClient;
22  
23  // tag::github-client-factory[]
24  /**
25   * Builds the users client generated from the GitHub contract for one call. The generated {@link ApiClient} keeps its
26   * base path and bearer token as mutable state, and both change with every sign-in (the base path comes from the
27   * registration, the token from the user), so a shared instance would mix the calls of concurrent sign-ins: each call
28   * gets its own client on top of one shared, thread-safe {@link RestClient}.
29   */
30  @Component
31  public class GitHubUsersApiFactory {
32  
33      /** HTTP client every generated client sends its requests through. */
34      private final RestClient restClient;
35  
36      /**
37       * Creates the factory.
38       *
39       * @param builder builder of REST clients configured by Spring Boot (message converters, observations).
40       */
41      public GitHubUsersApiFactory(final RestClient.Builder builder) {
42          restClient = builder.build();
43      }
44  
45      /**
46       * Builds the client of one call.
47       *
48       * @param basePath    base URI of GitHub's API, without a trailing slash.
49       * @param accessToken GitHub access token of the user, sent as the bearer token.
50       * @return the generated users client.
51       */
52      public UsersApi create(final String basePath, final String accessToken) {
53          final var apiClient = new ApiClient(restClient).setBasePath(basePath);
54          apiClient.setBearerToken(accessToken);
55          return new UsersApi(apiClient);
56      }
57  }
58  // end::github-client-factory[]
