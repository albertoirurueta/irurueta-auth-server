16  package com.irurueta.authserver.authorization.infrastructure.github;
17  
18  import com.irurueta.authserver.authorization.domain.social.VerifiedEmailLookup;
19  import java.util.Optional;
20  import lombok.RequiredArgsConstructor;
21  import lombok.extern.slf4j.Slf4j;
22  import org.springframework.stereotype.Component;
23  import org.springframework.util.StringUtils;
24  import org.springframework.web.client.RestClientException;
25  
26  // tag::github-verified-email[]
27  /**
28   * {@link VerifiedEmailLookup} over GitHub's {@code GET /user/emails}, called through the client generated from
29   * {@code apis/github/rest-server/github-api-v1.yaml}. The {@code email} of GitHub's profile is whatever the user chose
30   * to show and is not verified, so the address is read from that endpoint, which flags each address as primary and
31   * verified.
32   *
33   * <p>The base URI of the API is the user-info URI of the registration without its trailing {@code /user} (by default
34   * {@code https://api.github.com}), so a GitHub Enterprise registration, or the fake provider of the tests, is called
35   * where it is. Every failure (GitHub unreachable, an error status, an unreadable body, a user-info URI of another
36   * shape) means "no verified address", which refuses the sign-in; it is logged without the token or anything of the
37   * user.
38   */
39  @Slf4j
40  @Component
41  @RequiredArgsConstructor
42  public class GitHubVerifiedEmailLookup implements VerifiedEmailLookup {
43  
44      /** Path of the user endpoint below the base URI of GitHub's API. */
45      static final String USER_PATH = "/user";
46  
47      /** Builds the generated client of each call. */
48      private final GitHubUsersApiFactory usersApiFactory;
49  
50      /** Picks the verified primary address. */
51      private final GitHubEmailMapper emailMapper;
52  
53      /**
54       * Finds the verified primary address.
55       *
56       * @param userInfoUri URI of GitHub's user endpoint, as the registration configures it.
57       * @param accessToken GitHub access token of the user.
58       * @return the address, or an empty optional when there is none or GitHub cannot tell.
59       */
60      @Override
61      public Optional<String> verifiedPrimaryEmail(final String userInfoUri, final String accessToken) {
62          final var basePath = basePathOf(userInfoUri);
63          if (basePath.isEmpty() || !StringUtils.hasText(accessToken)) {
64              log.warn("The e-mail addresses of a GitHub user cannot be read: no usable user-info URI or token");
65              return Optional.empty();
66          }
67          try {
68              return emailMapper.verifiedPrimary(
69                      usersApiFactory.create(basePath.get(), accessToken).listEmailsForAuthenticatedUser());
70          } catch (final RestClientException e) {
71              log.warn("The e-mail addresses of a GitHub user could not be read: {}", e.getClass().getSimpleName());
72              return Optional.empty();
73          }
74      }
75  
76      /**
77       * Derives the base URI of the API from the user-info URI.
78       *
79       * @param userInfoUri URI of the user endpoint, possibly {@code null}.
80       * @return the URI without its trailing {@code /user} (and slashes), or an empty optional when it does not end so.
81       */
82      private static Optional<String> basePathOf(final String userInfoUri) {
83          if (!StringUtils.hasText(userInfoUri)) {
84              return Optional.empty();
85          }
86          final var trimmed = userInfoUri.strip().replaceFirst("/+$", "");
87          if (!trimmed.endsWith(USER_PATH)) {
88              return Optional.empty();
89          }
90          final var basePath = trimmed.substring(0, trimmed.length() - USER_PATH.length());
91          return StringUtils.hasText(basePath) ? Optional.of(basePath) : Optional.empty();
92      }
93  }
