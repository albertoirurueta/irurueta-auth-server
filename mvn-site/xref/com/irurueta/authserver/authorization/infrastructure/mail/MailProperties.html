17  package com.irurueta.authserver.authorization.infrastructure.mail;
18  
19  import java.util.Locale;
20  import org.springframework.boot.context.properties.ConfigurationProperties;
21  
22  /**
23   * Settings of the e-mails sent by {@link SmtpAccountMailer}. The SMTP connection itself is configured with the
24   * standard {@code spring.mail.*} properties and holds no credentials in this repository.
25   *
26   * @param from sender address of every message.
27   * @param baseUrl public base URL of the authorization server, used to build the links of the messages (without a
28   *                trailing slash).
29   * @param locale language of the messages; it is explicit configuration, never the platform's default.
30   */
31  // tag::mail-properties[]
32  @ConfigurationProperties(prefix = "authserver.mail")
33  public record MailProperties(String from, String baseUrl, Locale locale) {
34  
35      /** Sender used when none is configured. */
36      public static final String DEFAULT_FROM = "no-reply@localhost";
37  
38      /** Base URL used when none is configured. */
39      public static final String DEFAULT_BASE_URL = "http://localhost:9000";
40  
41      /**
42       * Applies the defaults of the missing values and removes a trailing slash of the base URL.
43       *
44       * @param from sender address, or {@code null} for {@link #DEFAULT_FROM}.
45       * @param baseUrl public base URL, or {@code null} for {@link #DEFAULT_BASE_URL}.
46       * @param locale language of the messages, or {@code null} for English.
47       */
48      public MailProperties {
49          from = from == null || from.isBlank() ? DEFAULT_FROM : from;
50          final var url = baseUrl == null || baseUrl.isBlank() ? DEFAULT_BASE_URL : baseUrl;
51          baseUrl = url.endsWith("/") ? url.substring(0, url.length() - 1) : url;
52          locale = locale == null ? Locale.ENGLISH : locale;
53      }
54  }
55  // end::mail-properties[]
