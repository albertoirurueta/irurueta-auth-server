17  package com.irurueta.authserver.authorization.infrastructure.keys;
18  
19  import com.irurueta.authserver.authorization.domain.totp.TotpSecretCipher;
20  import org.springframework.security.crypto.encrypt.BytesEncryptor;
21  import org.springframework.security.crypto.encrypt.Encryptors;
22  import org.springframework.stereotype.Component;
23  
24  // tag::totp-secret-cipher[]
25  /**
26   * {@link TotpSecretCipher} that encrypts TOTP secrets with AES-256 in GCM mode. Its key is derived from the
27   * key-encryption passphrase with a salt of its own ({@link PurposeKeyDerivation}, purpose {@value #PURPOSE}), so it
28   * never shares the key that protects the signing keys. Each encryption uses a random initialisation vector, which is
29   * stored in front of the cipher text.
30   */
31  @Component
32  public class AesGcmTotpSecretCipher implements TotpSecretCipher {
33  
34      /** Label of the purpose the key is derived for. */
35      public static final String PURPOSE = "totp-secret";
36  
37      /** The encryptor derived from the passphrase and the salt of this purpose. */
38      private final BytesEncryptor encryptor;
39  
40      /**
41       * Derives the encryption key.
42       *
43       * @param properties passphrase of the key-encryption secret.
44       * @param derivation derives the salt of this purpose from the configured salt.
45       */
46      public AesGcmTotpSecretCipher(final KeyEncryptionProperties properties, final PurposeKeyDerivation derivation) {
47          encryptor = Encryptors.stronger(properties.password(), derivation.saltFor(PURPOSE));
48      }
49  
50      @Override
51      public byte[] encrypt(final byte[] secret) {
52          return encryptor.encrypt(secret);
53      }
54  
55      @Override
56      public byte[] decrypt(final byte[] encrypted) {
57          try {
58              return encryptor.decrypt(encrypted);
59          } catch (final RuntimeException e) {
60              throw new IllegalStateException("The TOTP secret cannot be decrypted", e);
61          }
62      }
63  }
64  // end::totp-secret-cipher[]
