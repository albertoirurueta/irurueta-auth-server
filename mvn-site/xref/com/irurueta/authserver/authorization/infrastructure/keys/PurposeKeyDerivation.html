17  package com.irurueta.authserver.authorization.infrastructure.keys;
18  
19  import java.nio.charset.StandardCharsets;
20  import java.security.GeneralSecurityException;
21  import java.util.HexFormat;
22  import javax.crypto.Mac;
23  import javax.crypto.SecretKeyFactory;
24  import javax.crypto.spec.PBEKeySpec;
25  import javax.crypto.spec.SecretKeySpec;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.stereotype.Component;
28  
29  // tag::keys-purpose-derivation[]
30  /**
31   * Derives, from the key-encryption secret ({@link KeyEncryptionProperties}), a distinct salt and key for each purpose
32   * other than the encryption of signing keys, so no two uses share a key: a key exposed or misused for one purpose
33   * says nothing about the others. The salt of a purpose is the HMAC-SHA256 of the purpose label keyed with the
34   * configured salt; its key is derived from the passphrase and that salt with PBKDF2-HMAC-SHA256. The signing-key
35   * cipher ({@link AesGcmKeyCipher}) keeps using the configured salt itself, so the keys already stored still decrypt.
36   */
37  @Component
38  @RequiredArgsConstructor
39  public class PurposeKeyDerivation {
40  
41      /** Algorithm that derives the salt of a purpose. */
42      private static final String SALT_ALGORITHM = "HmacSHA256";
43  
44      /** Algorithm that derives the key of a purpose. */
45      private static final String KEY_ALGORITHM = "PBKDF2WithHmacSHA256";
46  
47      /** PBKDF2 iterations; a key is derived once per purpose, when its bean is created. */
48      private static final int ITERATIONS = 100_000;
49  
50      /** Size of a derived key in bits (AES-256 / HMAC-SHA256). */
51      private static final int KEY_BITS = 256;
52  
53      /** The passphrase and the salt of the key-encryption secret. */
54      private final KeyEncryptionProperties properties;
55  
56      /**
57       * Derives the salt of a purpose.
58       *
59       * @param purpose label of the purpose, for example {@code totp-secret}.
60       * @return the salt, hex encoded, as {@code Encryptors.stronger} expects it.
61       */
62      public String saltFor(final String purpose) {
63          try {
64              final var mac = Mac.getInstance(SALT_ALGORITHM);
65              mac.init(new SecretKeySpec(properties.salt().getBytes(StandardCharsets.UTF_8), SALT_ALGORITHM));
66              return HexFormat.of().formatHex(mac.doFinal(purpose.getBytes(StandardCharsets.UTF_8)));
67          } catch (final GeneralSecurityException e) {
68              throw new IllegalStateException("The salt of " + purpose + " cannot be derived", e);
69          }
70      }
71  
72      /**
73       * Derives the key of a purpose.
74       *
75       * @param purpose label of the purpose, for example {@code trusted-device-cookie}.
76       * @return the 256-bit key.
77       */
78      public byte[] keyFor(final String purpose) {
79          final var spec = new PBEKeySpec(properties.password().toCharArray(),
80                  HexFormat.of().parseHex(saltFor(purpose)), ITERATIONS, KEY_BITS);
81          try {
82              return SecretKeyFactory.getInstance(KEY_ALGORITHM).generateSecret(spec).getEncoded();
83          } catch (final GeneralSecurityException e) {
84              throw new IllegalStateException("The key of " + purpose + " cannot be derived", e);
85          } finally {
86              spec.clearPassword();
87          }
88      }
89  }
