17  package com.irurueta.authserver.authorization.infrastructure.persistence.clients;
18  
19  import com.irurueta.authserver.authorization.domain.clients.ClientRegistration;
20  import java.time.Instant;
21  import java.util.ArrayList;
22  import java.util.List;
23  import org.springframework.security.oauth2.core.AuthorizationGrantType;
24  import org.springframework.security.oauth2.core.ClientAuthenticationMethod;
25  import org.springframework.security.oauth2.jose.jws.JwsAlgorithm;
26  import org.springframework.security.oauth2.jose.jws.SignatureAlgorithm;
27  import org.springframework.security.oauth2.server.authorization.client.RegisteredClient;
28  import org.springframework.security.oauth2.server.authorization.settings.ClientSettings;
29  import org.springframework.security.oauth2.server.authorization.settings.OAuth2TokenFormat;
30  import org.springframework.security.oauth2.server.authorization.settings.TokenSettings;
31  import org.springframework.stereotype.Component;
32  import org.springframework.util.StringUtils;
33  
34  /**
35   * Maps the domain {@link ClientRegistration} onto Spring Authorization Server's {@link RegisteredClient} and back.
36   * The audiences, which the framework has no field for, travel as the {@value #AUDIENCES_SETTING} custom client
37   * setting.
38   */
39  @Component
40  public class RegisteredClientMapper {
41  
42      /** Name of the custom client setting that holds the audiences of the access tokens issued to the client. */
43      public static final String AUDIENCES_SETTING = "settings.client.audiences";
44  
45      /**
46       * Name of the custom client setting that makes every user of the client complete a second factor. It is only
47       * written and read here: everybody else asks {@code ClientRegistration.mfaRequired()} through the client port.
48       */
49      public static final String MFA_REQUIRED_SETTING = "settings.client.mfa.required";
50  
51      /**
52       * Name of the custom client setting that holds the client's registered public JWK set as JSON, used to verify
53       * {@code private_key_jwt} assertions and self-signed certificates.
54       */
55      public static final String JWKS_SETTING = "settings.client.jwks";
56  
57      /** Authentication method that marks a public client. */
58      private static final String AUTH_METHOD_NONE = ClientAuthenticationMethod.NONE.getValue();
59  
60      /**
61       * Maps a registration to a registered client.
62       *
63       * @param registration the registration.
64       * @param id           internal identifier of the registered client.
65       * @param issuedAt     instant the client identifier was issued.
66       * @param secret       encoded secret, or {@code null}.
67       * @return the registered client.
68       * @throws IllegalArgumentException if the token endpoint signing algorithm is not a JWS signature algorithm.
69       */
70      public RegisteredClient toRegisteredClient(
71              final ClientRegistration registration, final String id, final Instant issuedAt, final String secret) {
72          final var clientSettings = ClientSettings.builder()
73                  .requireProofKey(registration.requireProofKey())
74                  .requireAuthorizationConsent(registration.requireConsent())
75                  .setting(AUDIENCES_SETTING, new ArrayList<>(registration.audiences()))
76                  .setting(MFA_REQUIRED_SETTING, registration.mfaRequired());
77          // tag::clients-auth-settings[]
78          if (registration.jwkSetUrl() != null) {
79              clientSettings.jwkSetUrl(registration.jwkSetUrl());
80          }
81          if (registration.jwks() != null) {
82              clientSettings.setting(JWKS_SETTING, registration.jwks());
83          }
84          if (registration.tokenEndpointSigningAlgorithm() != null) {
85              clientSettings.tokenEndpointAuthenticationSigningAlgorithm(
86                      signingAlgorithmOf(registration.tokenEndpointSigningAlgorithm()));
87          }
88          if (registration.tlsClientAuthSubjectDn() != null) {
89              clientSettings.x509CertificateSubjectDN(registration.tlsClientAuthSubjectDn());
90          }
91          // end::clients-auth-settings[]
92          final var builder = RegisteredClient.withId(id)
93                  .clientId(registration.clientId())
94                  .clientIdIssuedAt(issuedAt)
95                  .clientSecret(secret)
96                  .clientName(registration.name())
97                  .clientAuthenticationMethods(methods -> registration.authMethods()
98                          .forEach(method -> methods.add(new ClientAuthenticationMethod(method))))
99                  .authorizationGrantTypes(types -> registration.grantTypes()
100                         .forEach(type -> types.add(new AuthorizationGrantType(type))))
101                 .redirectUris(uris -> uris.addAll(registration.redirectUris()))
102                 .postLogoutRedirectUris(uris -> uris.addAll(registration.postLogoutRedirectUris()))
103                 .scopes(scopes -> scopes.addAll(registration.scopes()))
104                 .clientSettings(clientSettings.build())
105                 // tag::tokens-settings[]
106                 .tokenSettings(TokenSettings.builder()
107                         .accessTokenTimeToLive(registration.accessTokenTtl())
108                         .refreshTokenTimeToLive(registration.refreshTokenTtl())
109                         .accessTokenFormat(new OAuth2TokenFormat(registration.accessTokenFormat()))
110                         .reuseRefreshTokens(registration.reuseRefreshTokens())
111                         .x509CertificateBoundAccessTokens(registration.certificateBoundAccessTokens())
112                         .build());
113                 // end::tokens-settings[]
114         return builder.build();
115     }
116 
117     /**
118      * Maps a registered client to a registration. The framework keeps these collections as sets, so they are
119      * returned sorted to make the result deterministic. A client is confidential unless it authenticates with
120      * {@code none}.
121      *
122      * @param client the registered client.
123      * @return the registration.
124      */
125     public ClientRegistration toRegistration(final RegisteredClient client) {
126         final var authMethods = client.getClientAuthenticationMethods().stream()
127                 .map(ClientAuthenticationMethod::getValue).sorted().toList();
128         final var settings = client.getClientSettings();
129         final var tokenSettings = client.getTokenSettings();
130         return new ClientRegistration(client.getClientId(), client.getClientName(),
131                 !authMethods.contains(AUTH_METHOD_NONE), authMethods,
132                 client.getAuthorizationGrantTypes().stream().map(AuthorizationGrantType::getValue).sorted().toList(),
133                 client.getRedirectUris().stream().sorted().toList(), client.getScopes().stream().sorted().toList(),
134                 settings.isRequireProofKey(), audiencesOf(settings), tokenSettings.getAccessTokenTimeToLive(),
135                 tokenSettings.getRefreshTokenTimeToLive(), tokenSettings.getAccessTokenFormat().getValue(),
136                 tokenSettings.isReuseRefreshTokens(),
137                 client.getPostLogoutRedirectUris().stream().sorted().toList(),
138                 settings.isRequireAuthorizationConsent(), mfaRequiredOf(settings), settings.getJwkSetUrl(),
139                 jwksOf(settings), signingAlgorithmNameOf(settings), settings.getX509CertificateSubjectDN(),
140                 tokenSettings.isX509CertificateBoundAccessTokens());
141     }
142 
143     /**
144      * Reads the audiences custom setting.
145      *
146      * @param settings client settings.
147      * @return the audiences; empty when the setting is absent (the registration then defaults to the client id).
148      */
149     public List<String> audiencesOf(final ClientSettings settings) {
150         final var value = settings.getSetting(AUDIENCES_SETTING);
151         if (value instanceof List<?> audiences) {
152             return audiences.stream().map(String::valueOf).toList();
153         }
154         return List.of();
155     }
156 
157     /**
158      * Reads the second-factor custom setting.
159      *
160      * @param settings client settings.
161      * @return {@code true} only when the setting is present and true.
162      */
163     public boolean mfaRequiredOf(final ClientSettings settings) {
164         return Boolean.TRUE.equals(settings.getSetting(MFA_REQUIRED_SETTING));
165     }
166 
167     /**
168      * Reads the registered JWK set custom setting.
169      *
170      * @param settings client settings.
171      * @return the JWK set JSON, or {@code null} when the setting is absent.
172      */
173     public String jwksOf(final ClientSettings settings) {
174         final var value = settings.getSetting(JWKS_SETTING);
175         return value instanceof String jwks && StringUtils.hasText(jwks) ? jwks : null;
176     }
177 
178     /**
179      * Reads the token endpoint authentication signing algorithm.
180      *
181      * @param settings client settings.
182      * @return the algorithm name, or {@code null} when none is set.
183      */
184     private String signingAlgorithmNameOf(final ClientSettings settings) {
185         final JwsAlgorithm algorithm = settings.getTokenEndpointAuthenticationSigningAlgorithm();
186         return algorithm == null ? null : algorithm.getName();
187     }
188 
189     /**
190      * Resolves a JWS signature algorithm by name.
191      *
192      * @param name the algorithm name, for example {@code RS256}.
193      * @return the algorithm.
194      * @throws IllegalArgumentException if the name is not a JWS signature algorithm.
195      */
196     private SignatureAlgorithm signingAlgorithmOf(final String name) {
197         final var algorithm = SignatureAlgorithm.from(name);
198         if (algorithm == null) {
199             throw new IllegalArgumentException("unsupported token endpoint signing algorithm: " + name);
200         }
201         return algorithm;
202     }
