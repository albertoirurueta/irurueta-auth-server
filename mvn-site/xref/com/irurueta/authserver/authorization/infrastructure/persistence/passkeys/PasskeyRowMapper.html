17  package com.irurueta.authserver.authorization.infrastructure.persistence.passkeys;
18  
19  import com.irurueta.authserver.authorization.domain.passkeys.Passkey;
20  import java.sql.ResultSet;
21  import java.sql.SQLException;
22  import java.time.Instant;
23  import org.springframework.jdbc.core.RowMapper;
24  import org.springframework.stereotype.Component;
25  
26  /**
27   * Maps a row of the {@code user_credentials} table to a {@link Passkey}. The {@code created} and {@code last_used}
28   * columns are {@code timestamp with time zone}s that Spring Security's own repository writes as JDBC timestamps, so
29   * they are read back the same way, which keeps the instant whatever the time zone of the JVM.
30   */
31  @Component
32  public class PasskeyRowMapper implements RowMapper<Passkey> {
33  
34      /**
35       * Maps the current row.
36       *
37       * @param rs     result set positioned on the row.
38       * @param rowNum number of the row.
39       * @return the passkey; a row without a creation timestamp is given the epoch, since the domain requires one.
40       * @throws SQLException if a column cannot be read.
41       */
42      @Override
43      public Passkey mapRow(final ResultSet rs, final int rowNum) throws SQLException {
44          return new Passkey(rs.getString("credential_id"), rs.getString("user_entity_user_id"),
45                  rs.getString("label"), readInstant(rs, "created", Instant.EPOCH), readInstant(rs, "last_used", null),
46                  rs.getLong("signature_count"), rs.getBoolean("backup_eligible"), rs.getBoolean("backup_state"));
47      }
48  
49      /**
50       * Reads a {@code timestamp with time zone} column that may be null.
51       *
52       * @param rs       result set positioned on a row.
53       * @param column   name of the column.
54       * @param fallback value returned when the column is null.
55       * @return the instant, or the fallback.
56       * @throws SQLException if the column cannot be read.
57       */
58      private static Instant readInstant(final ResultSet rs, final String column, final Instant fallback)
59              throws SQLException {
60          final var value = rs.getTimestamp(column);
61          return value == null ? fallback : value.toInstant();
62      }
63  }
