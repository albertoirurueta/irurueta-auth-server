17  package com.irurueta.authserver.authorization.infrastructure.persistence.mfa;
18  
19  import static com.irurueta.authserver.authorization.infrastructure.persistence.JdbcTimestamps.toTimestamp;
20  
21  import com.irurueta.authserver.authorization.domain.mfa.SecondFactorAttemptRepository;
22  import com.irurueta.authserver.authorization.domain.mfa.SecondFactorLockout;
23  import java.time.Duration;
24  import java.time.Instant;
25  import java.util.Map;
26  import java.util.UUID;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.jdbc.core.simple.JdbcClient;
29  import org.springframework.stereotype.Repository;
30  
31  /**
32   * {@link SecondFactorAttemptRepository} over the {@code second_factor_attempts} table.
33   */
34  @Repository
35  @RequiredArgsConstructor
36  public class JdbcSecondFactorAttemptRepository implements SecondFactorAttemptRepository {
37  
38      /** Query returning the state of a user. */
39      private static final String FIND_ATTEMPTS = """
40              SELECT failed_count, locked_until FROM second_factor_attempts WHERE user_id = ?
41              """;
42  
43      // tag::mfa-lock-attempts[]
44      /** Statement that creates the clean state of a user unless one exists, so there is always a row to lock. */
45      private static final String ENSURE_ATTEMPTS = """
46              INSERT INTO second_factor_attempts (user_id, failed_count) VALUES (?, 0)
47              ON CONFLICT (user_id) DO NOTHING
48              """;
49  
50      /**
51       * Query returning the state of a user and locking its row until the end of the transaction: a concurrent
52       * verification of the same user waits here until the first one commits.
53       */
54      private static final String LOCK_ATTEMPTS = """
55              SELECT failed_count, locked_until FROM second_factor_attempts WHERE user_id = ? FOR UPDATE
56              """;
57      // end::mfa-lock-attempts[]
58  
59      /** Statement that clears the counter and the lock, keeping the row so later verifications can lock it. */
60      private static final String RESET_ATTEMPTS = """
61              UPDATE second_factor_attempts SET failed_count = 0, locked_until = NULL WHERE user_id = ?
62              """;
63  
64      // tag::mfa-record-failure[]
65      /**
66       * Single-statement upsert that counts one failure and sets the lock when the maximum is reached. A lock that
67       * has already ended restarts the counter at one. Concurrent failures serialize on the row, so none is lost.
68       */
69      private static final String RECORD_FAILURE = """
70              INSERT INTO second_factor_attempts AS t (user_id, failed_count, locked_until)
71              VALUES (:userId, 1, CASE WHEN 1 >= :maxAttempts THEN :lockedUntil ELSE NULL END)
72              ON CONFLICT (user_id) DO UPDATE SET
73                  failed_count = CASE WHEN t.locked_until IS NOT NULL AND t.locked_until <= :now
74                                      THEN 1 ELSE t.failed_count + 1 END,
75                  locked_until = CASE
76                      WHEN (CASE WHEN t.locked_until IS NOT NULL AND t.locked_until <= :now
77                                 THEN 1 ELSE t.failed_count + 1 END) >= :maxAttempts THEN :lockedUntil
78                      WHEN t.locked_until <= :now THEN NULL
79                      ELSE t.locked_until END
80              RETURNING failed_count, locked_until
81              """;
82      // end::mfa-record-failure[]
83  
84      /** JDBC client used to run the statements. */
85      private final JdbcClient jdbcClient;
86  
87      /** Maps a row to the state of a user. */
88      private final SecondFactorLockoutRowMapper secondFactorLockoutRowMapper;
89  
90      @Override
91      public SecondFactorLockout find(final UUID userId) {
92          return jdbcClient.sql(FIND_ATTEMPTS).param(userId).query(secondFactorLockoutRowMapper).optional()
93                  .orElseGet(SecondFactorLockout::none);
94      }
95  
96      /**
97       * Reads and locks the state of a user, in the caller's transaction (without one, the lock is released as soon
98       * as the query ends).
99       *
100      * @param userId identifier of the user.
101      * @return the state.
102      */
103     @Override
104     public SecondFactorLockout lockForUpdate(final UUID userId) {
105         jdbcClient.sql(ENSURE_ATTEMPTS).param(userId).update();
106         return jdbcClient.sql(LOCK_ATTEMPTS).param(userId).query(secondFactorLockoutRowMapper).optional()
107                 .orElseGet(SecondFactorLockout::none);
108     }
109 
110     @Override
111     public SecondFactorLockout recordFailure(final UUID userId, final int maxAttempts, final Duration lockout,
112                                              final Instant now) {
113         return jdbcClient.sql(RECORD_FAILURE)
114                 .params(Map.of("userId", userId, "maxAttempts", maxAttempts, "now", toTimestamp(now),
115                         "lockedUntil", toTimestamp(now.plus(lockout))))
116                 .query(secondFactorLockoutRowMapper)
117                 .single();
118     }
119 
120     @Override
121     public void reset(final UUID userId) {
122         jdbcClient.sql(RESET_ATTEMPTS).param(userId).update();
123     }
124 }
