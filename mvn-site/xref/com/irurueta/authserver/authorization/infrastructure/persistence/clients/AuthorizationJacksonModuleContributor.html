16  package com.irurueta.authserver.authorization.infrastructure.persistence.clients;
17  
18  import tools.jackson.databind.JacksonModule;
19  import tools.jackson.databind.jsontype.BasicPolymorphicTypeValidator;
20  
21  /**
22   * Extension point that adds a Jackson module to the mapper of the stored authorizations
23   * ({@link AuthorizationJsonMapperFactory}), for a feature whose authentication or attribute types the framework's
24   * security modules do not cover (for example the passkeys, whose module Spring Security leaves out of
25   * {@code SecurityJacksonModules}). Every bean of this type is applied; no shared file is edited.
26   *
27   * <p>It is deliberately not a plain {@link JacksonModule} bean: Spring Boot adds every such bean to the
28   * application-wide {@code JsonMapper}, which would widen the polymorphic typing of every JSON answer of the
29   * application, while the module only belongs to the mapper of the stored authorizations. It also needs the type
30   * validator of that mapper, which only the factory has.
31   */
32  @FunctionalInterface
33  public interface AuthorizationJacksonModuleContributor {
34  
35      /**
36       * Creates the module and allows, in the validator, the types it (de)serializes polymorphically. It is called
37       * before the framework's modules are created, while the validator can still be configured.
38       *
39       * @param validator builder of the polymorphic type validator of the authorization mapper.
40       * @return the module to add.
41       */
42      JacksonModule module(BasicPolymorphicTypeValidator.Builder validator);
43  }
