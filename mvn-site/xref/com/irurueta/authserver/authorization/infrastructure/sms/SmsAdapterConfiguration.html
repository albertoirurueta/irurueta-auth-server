17  package com.irurueta.authserver.authorization.infrastructure.sms;
18  
19  import com.google.i18n.phonenumbers.PhoneNumberUtil;
20  import com.irurueta.authserver.authorization.domain.sms.PhoneCountryResolver;
21  import com.irurueta.authserver.authorization.domain.sms.PhoneNormalizer;
22  import com.irurueta.authserver.authorization.domain.sms.SmsSender;
23  import java.time.Clock;
24  import org.springframework.boot.context.properties.EnableConfigurationProperties;
25  import org.springframework.context.annotation.Bean;
26  import org.springframework.context.annotation.Configuration;
27  import org.springframework.context.annotation.Profile;
28  
29  // tag::sms-adapters[]
30  /**
31   * Wires the phone normaliser and the logging {@link SmsSender} of development and tests. The profile expressions make
32   * the three senders (logging, Twilio, SNS) mutually exclusive in every combination of profiles: the logging one only
33   * under {@code dev} or {@code test} and when neither gateway profile is active, Twilio unless SNS is also active, and
34   * SNS whenever it is active. With none of those profiles there is no sender at all and
35   * {@link MissingSmsGatewayConfiguration} stops the startup: a deployment never silently keeps its messages in memory.
36   */
37  @Configuration(proxyBeanMethods = false)
38  @EnableConfigurationProperties(SmsOutboxProperties.class)
39  public class SmsAdapterConfiguration {
40  
41      /** Profiles under which the logging sender (and its outbox) is used: development or tests, and no gateway. */
42      public static final String LOGGING_SENDER_PROFILES = "(dev | test) & !twilio & !sns";
43  
44      /**
45       * Creates the phone normaliser.
46       *
47       * @return the libphonenumber based normaliser.
48       */
49      @Bean
50      public PhoneNormalizer phoneNormalizer() {
51          return new LibPhoneNumberNormalizer(PhoneNumberUtil.getInstance());
52      }
53  
54      /**
55       * Creates the resolver of the country of a phone number.
56       *
57       * @return the libphonenumber based resolver.
58       */
59      @Bean
60      public PhoneCountryResolver phoneCountryResolver() {
61          return new LibPhoneNumberCountryResolver(PhoneNumberUtil.getInstance());
62      }
63  
64      /**
65       * Creates the outbox of the logging sender.
66       *
67       * @param properties outbox settings.
68       * @return the outbox, readable by the development inbox.
69       */
70      @Bean
71      @Profile(LOGGING_SENDER_PROFILES)
72      public SmsOutbox smsOutbox(final SmsOutboxProperties properties) {
73          return new SmsOutbox(properties.capacity());
74      }
75  
76      /**
77       * Creates the logging sender of development and tests.
78       *
79       * @param outbox records the messages.
80       * @param clock  source of the recording instant.
81       * @return the logging sender.
82       */
83      @Bean
84      @Profile(LOGGING_SENDER_PROFILES)
85      public SmsSender loggingSmsSender(final SmsOutbox outbox, final Clock clock) {
86          return new LoggingSmsSender(outbox, clock);
87      }
88  }
