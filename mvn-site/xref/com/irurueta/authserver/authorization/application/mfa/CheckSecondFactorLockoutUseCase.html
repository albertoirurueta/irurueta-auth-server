17  package com.irurueta.authserver.authorization.application.mfa;
18  
19  import com.irurueta.authserver.authorization.domain.mfa.SecondFactorAttemptRepository;
20  import java.time.Clock;
21  import java.util.UUID;
22  import lombok.RequiredArgsConstructor;
23  import org.springframework.stereotype.Service;
24  import org.springframework.transaction.annotation.Transactional;
25  
26  // tag::mfa-lockout-check[]
27  /**
28   * Use case that tells whether the second factor of a user is locked after too many failed attempts. It locks the
29   * user's attempt row until the end of the caller's transaction, so the verifications of the same user (each a
30   * transactional use case that starts with this check) run one at a time: none of them can pass the check while
31   * another one is about to count a failure.
32   */
33  @Service
34  @Transactional
35  @RequiredArgsConstructor
36  public class CheckSecondFactorLockoutUseCase {
37  
38      /** Port that stores the failed-attempt counters. */
39      private final SecondFactorAttemptRepository secondFactorAttemptRepository;
40  
41      /** Source of the current time. */
42      private final Clock clock;
43  
44      /**
45       * Checks the lockout of a user.
46       *
47       * @param userId identifier of the user.
48       * @return {@code true} if the second factor is locked right now.
49       */
50      public boolean execute(final UUID userId) {
51          return secondFactorAttemptRepository.lockForUpdate(userId).isLocked(clock.instant());
52      }
53  }
54  // end::mfa-lockout-check[]
