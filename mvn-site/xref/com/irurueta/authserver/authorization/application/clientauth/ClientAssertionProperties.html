17  package com.irurueta.authserver.authorization.application.clientauth;
18  
19  import java.time.Duration;
20  import org.springframework.boot.context.properties.ConfigurationProperties;
21  import org.springframework.boot.context.properties.bind.DefaultValue;
22  import org.springframework.scheduling.support.CronExpression;
23  import org.springframework.util.StringUtils;
24  
25  /**
26   * Settings of the client assertions, bound from {@code authserver.client-assertion.*} once and validated at startup.
27   * A missing value takes its default, so the application starts without any setting.
28   *
29   * @param maxLifetime longest lifetime (exp minus iat, or exp minus now when iat is absent) an assertion may have
30   *                    (default 5 minutes).
31   * @param cleanup     the purge of the stored identifiers.
32   */
33  @ConfigurationProperties("authserver.client-assertion")
34  public record ClientAssertionProperties(
35          @DefaultValue("PT5M") Duration maxLifetime,
36          @DefaultValue Cleanup cleanup) {
37  
38      /** Default longest lifetime of an assertion, in minutes. */
39      private static final long DEFAULT_MAX_LIFETIME_MINUTES = 5L;
40  
41      /**
42       * Applies the defaults and validates the settings.
43       *
44       * @throws IllegalArgumentException if the lifetime is zero or negative or the cron expression is invalid.
45       */
46      public ClientAssertionProperties {
47          if (maxLifetime == null) {
48              maxLifetime = Duration.ofMinutes(DEFAULT_MAX_LIFETIME_MINUTES);
49          } else if (maxLifetime.isNegative() || maxLifetime.isZero()) {
50              throw new IllegalArgumentException("authserver.client-assertion.max-lifetime must be positive");
51          }
52          if (cleanup == null) {
53              cleanup = new Cleanup(null);
54          }
55      }
56  
57      /**
58       * Settings of the purge of expired identifiers.
59       *
60       * @param cron Spring cron expression saying when the purge runs (default every hour at minute 30).
61       */
62      public record Cleanup(@DefaultValue("0 30 * * * *") String cron) {
63  
64          /** Default cron expression. */
65          private static final String DEFAULT_CRON = "0 30 * * * *";
66  
67          /**
68           * Applies the default and validates the expression.
69           *
70           * @throws IllegalArgumentException if the expression is not a valid Spring cron expression.
71           */
72          public Cleanup {
73              if (!StringUtils.hasText(cron)) {
74                  cron = DEFAULT_CRON;
75              }
76              if (!CronExpression.isValidExpression(cron)) {
77                  throw new IllegalArgumentException("authserver.client-assertion.cleanup.cron is not a valid cron");
78              }
79          }
80      }
81  }
