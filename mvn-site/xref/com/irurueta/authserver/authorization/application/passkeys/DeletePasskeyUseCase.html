17  package com.irurueta.authserver.authorization.application.passkeys;
18  
19  import com.irurueta.authserver.authorization.application.mfa.HasConfirmedSecondFactorUseCase;
20  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
21  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
22  import com.irurueta.authserver.authorization.domain.mfa.TrustedDeviceRepository;
23  import com.irurueta.authserver.authorization.domain.passkeys.PasskeyNotFoundException;
24  import com.irurueta.authserver.authorization.domain.passkeys.PasskeyRepository;
25  import java.util.UUID;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.stereotype.Service;
28  import org.springframework.transaction.annotation.Transactional;
29  
30  /**
31   * Use case that deletes a passkey of a user and audits the removal. When that leaves the user without any confirmed
32   * second factor, every trusted device of the user is revoked too, as disabling TOTP does: a remembered browser stands
33   * in for a second factor the user no longer has.
34   */
35  @Service
36  @Transactional
37  @RequiredArgsConstructor
38  public class DeletePasskeyUseCase {
39  
40      /** Port that stores the passkeys. */
41      private final PasskeyRepository passkeyRepository;
42  
43      /** Use case telling whether the user still has a confirmed second factor. */
44      private final HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
45  
46      /** Port that stores the trusted devices. */
47      private final TrustedDeviceRepository trustedDeviceRepository;
48  
49      /** Use case that records the audit events. */
50      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
51  
52      /**
53       * Deletes a passkey; a passkey of another account is never touched.
54       *
55       * @param userId   identifier of the signed-in user, for the audit log.
56       * @param username name of the signed-in account.
57       * @param id       credential identifier.
58       * @throws PasskeyNotFoundException if the account has no passkey with that identifier.
59       */
60      public void execute(final UUID userId, final String username, final String id) {
61          final var ownerId = passkeyRepository.findUserIdByUsername(username)
62                  .orElseThrow(() -> new PasskeyNotFoundException("The account has no passkeys"));
63          if (!passkeyRepository.delete(id, ownerId)) {
64              throw new PasskeyNotFoundException("The account has no passkey " + id);
65          }
66          var detail = "passkey";
67          if (!hasConfirmedSecondFactorUseCase.execute(userId)) {
68              final var revoked = trustedDeviceRepository.deleteByUser(userId);
69              detail = detail + "; trusted devices revoked: " + revoked;
70          }
71          recordSecurityEventUseCase.execute(SecurityEventType.SECOND_FACTOR_REMOVED, userId, username, detail);
72      }
73  }
