17  package com.irurueta.authserver.authorization.application.keys;
18  
19  import com.irurueta.authserver.authorization.domain.keys.SigningKey;
20  import com.irurueta.authserver.authorization.domain.keys.SigningKeyGenerator;
21  import com.irurueta.authserver.authorization.domain.keys.SigningKeyRepository;
22  import com.irurueta.authserver.authorization.domain.keys.SigningKeyStatus;
23  import java.time.Clock;
24  import java.time.Duration;
25  import java.time.Instant;
26  import java.util.List;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.beans.factory.annotation.Value;
29  import org.springframework.stereotype.Service;
30  import org.springframework.transaction.annotation.Transactional;
31  
32  /**
33   * Use case that keeps the signing keys rotating. One call, made at start-up and then on a schedule, does whatever is
34   * due: it creates the {@code active} and {@code next} keys when they are missing, promotes {@code next} to
35   * {@code active} and retires the old {@code active} key once the rotation interval has elapsed, always leaves a
36   * fresh {@code next} key behind, and deletes keys retired for longer than the effective retention (see
37   * {@link SigningKeyRetention}). A second call right after the first changes nothing, and the whole call runs under
38   * the repository's rotation lock, so several instances can call it safely. {@link #executeForced()} is the
39   * administrator's path: it replaces the active key immediately, whatever its age.
40   */
41  @Service
42  @Transactional
43  @RequiredArgsConstructor
44  public class RotateSigningKeysUseCase {
45  
46      /** Port that stores the keys. */
47      private final SigningKeyRepository signingKeyRepository;
48  
49      /** Port that generates keys. */
50      private final SigningKeyGenerator signingKeyGenerator;
51  
52      /** Source of the current time. */
53      private final Clock clock;
54  
55      /** How long a key signs before it is replaced ({@code authserver.keys.rotation-interval}, default 30 days). */
56      @Value("${authserver.keys.rotation-interval:P30D}")
57      private final Duration rotationInterval;
58  
59      /** Decides how long a retired key stays published. */
60      private final SigningKeyRetention signingKeyRetention;
61  
62      // tag::keys-rotation[]
63      /**
64       * Does whatever the key lifecycle requires at the current instant: the scheduled path, which only rotates a key
65       * that has signed for the whole rotation interval.
66       */
67      public void execute() {
68          run(false);
69      }
70  
71      /**
72       * Rotates now, whatever the age of the active key: the active key is retired, the next key promoted and a new
73       * next key generated (for example after a suspected key compromise). Expired retired keys are deleted too.
74       */
75      public void executeForced() {
76          run(true);
77      }
78  
79      /**
80       * Applies the lifecycle transitions while holding the rotation lock.
81       *
82       * @param force whether the active key is replaced even when it is not due.
83       */
84      private void run(final boolean force) {
85          final var now = clock.instant();
86          final var retention = signingKeyRetention.effective();
87          signingKeyRepository.withRotationLock(() -> {
88              rotate(now, force, retention);
89              return null;
90          });
91      }
92  
93      /**
94       * Applies the lifecycle transitions that are due, or forced.
95       *
96       * @param now       current instant.
97       * @param force     whether the active key is replaced even when it is not due.
98       * @param retention how long a retired key stays published.
99       */
100     private void rotate(final Instant now, final boolean force, final Duration retention) {
101         final var keys = signingKeyRepository.findAll();
102         var active = first(keys, SigningKeyStatus.ACTIVE);
103         var next = first(keys, SigningKeyStatus.NEXT);
104         if (active != null && (force || isDue(active, now))) {
105             signingKeyRepository.save(active.retire(now));
106             active = null;
107         }
108         if (active == null) {
109             final var promoted = next != null ? next : signingKeyGenerator.generate(now);
110             signingKeyRepository.save(promoted.activate(now));
111             next = null;
112         }
113         if (next == null) {
114             signingKeyRepository.save(signingKeyGenerator.generate(now));
115         }
116         keys.stream()
117                 .filter(key -> key.status() == SigningKeyStatus.RETIRED && isExpired(key, now, retention))
118                 .forEach(key -> signingKeyRepository.deleteByKid(key.kid()));
119     }
120     // end::keys-rotation[]
121 
122     /**
123      * Finds the first key in a state.
124      *
125      * @param keys   the keys.
126      * @param status the state.
127      * @return the key, or {@code null} when none is in that state.
128      */
129     private static SigningKey first(final List<SigningKey> keys, final SigningKeyStatus status) {
130         return keys.stream().filter(key -> key.status() == status).findFirst().orElse(null);
131     }
132 
133     /**
134      * Tells whether the active key has signed for the whole rotation interval.
135      *
136      * @param active the active key.
137      * @param now    current instant.
138      * @return {@code true} when it must be replaced.
139      */
140     private boolean isDue(final SigningKey active, final Instant now) {
141         final var since = active.activatedAt() == null ? active.createdAt() : active.activatedAt();
142         return !since.plus(rotationInterval).isAfter(now);
143     }
144 
145     /**
146      * Tells whether a retired key has been retired for longer than the retention.
147      *
148      * @param retired   the retired key.
149      * @param now       current instant.
150      * @param retention how long a retired key stays published.
151      * @return {@code true} when it can be deleted.
152      */
153     private static boolean isExpired(final SigningKey retired, final Instant now, final Duration retention) {
154         final var since = retired.retiredAt() == null ? retired.createdAt() : retired.retiredAt();
155         return !since.plus(retention).isAfter(now);
156     }
157 }
