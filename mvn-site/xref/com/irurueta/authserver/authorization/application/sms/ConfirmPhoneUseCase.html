18  
19  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
20  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
21  import com.irurueta.authserver.authorization.domain.mfa.TrustedDeviceRepository;
22  import com.irurueta.authserver.authorization.domain.ott.OneTimeCodeHasher;
23  import com.irurueta.authserver.authorization.domain.sms.PhoneVerification;
24  import com.irurueta.authserver.authorization.domain.sms.PhoneVerificationRepository;
25  import com.irurueta.authserver.authorization.domain.sms.UserPhone;
26  import com.irurueta.authserver.authorization.domain.sms.UserPhoneRepository;
27  import java.time.Clock;
28  import java.time.Instant;
29  import java.util.UUID;
30  import lombok.RequiredArgsConstructor;
31  import org.springframework.stereotype.Service;
32  import org.springframework.transaction.annotation.Transactional;
33  
34  /**
35   * Confirms a phone with the code that was sent to it. Every way of failing returns the same {@code false}: no
36   * pending verification, an expired or burnt code, a wrong code and a number that another user verified meanwhile.
37   *
38   * <p>Confirming a number other than the verified one replaces a second factor: the replacement is audited as the
39   * removal of the old phone, and every trusted device of the user is revoked, as disabling TOTP does, because those
40   * browsers were remembered in place of a factor the user no longer has.
41   */
42  @Service
43  @Transactional
44  @RequiredArgsConstructor
45  public class ConfirmPhoneUseCase {
46  
47      /** Start of the detail of the audit event of a replaced phone, followed by the number of revoked devices. */
48      static final String REPLACED_DETAIL = "phone replaced; trusted devices revoked: ";
49  
50      /** Port that stores the verified phones. */
51      private final UserPhoneRepository userPhoneRepository;
52  
53      /** Port that stores the pending verification. */
54      private final PhoneVerificationRepository phoneVerificationRepository;
55  
56      /** Hasher of the verification code. */
57      private final OneTimeCodeHasher oneTimeCodeHasher;
58  
59      /** Port that stores the trusted devices. */
60      private final TrustedDeviceRepository trustedDeviceRepository;
61  
62      /** Use case that records the audit events. */
63      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
64  
65      /** SMS settings. */
66      private final SmsProperties smsProperties;
67  
68      /** Source of the current time. */
69      private final Clock clock;
70  
71      /**
72       * Confirms the phone of a user.
73       *
74       * @param userId   identifier of the user.
75       * @param username name of the user, for the audit log.
76       * @param code     the code as typed.
77       * @return {@code true} when the code was right and the phone is now verified.
78       */
79      public boolean execute(final UUID userId, final String username, final String code) {
80          final var now = clock.instant();
81          final var pending = phoneVerificationRepository.findByUserId(userId).orElse(null);
82          if (pending == null) {
83              return false;
84          }
85          if (pending.isExpiredAt(now)) {
86              phoneVerificationRepository.deleteByUserId(userId);
87              return false;
88          }
89          final var attempts = phoneVerificationRepository.incrementAttempts(userId);
90          if (attempts > smsProperties.verificationMaxAttempts()) {
91              phoneVerificationRepository.deleteByUserId(userId);
92              return false;
93          }
94          if (!oneTimeCodeHasher.matches(code, pending.codeHash())) {
95              if (attempts >= smsProperties.verificationMaxAttempts()) {
96                  phoneVerificationRepository.deleteByUserId(userId);
97              }
98              return false;
99          }
100         return verify(userId, username, pending, now);
101     }
102 
103     /**
104      * Stores the verified phone, clears the pending verification and handles the replacement of another number.
105      *
106      * @param userId   identifier of the user.
107      * @param username name of the user, for the audit log.
108      * @param pending  the verification that was confirmed.
109      * @param now      the current instant.
110      * @return {@code true} when the phone was stored, {@code false} when another user owns the number.
111      */
112     private boolean verify(final UUID userId, final String username, final PhoneVerification pending,
113             final Instant now) {
114         final var replaced = userPhoneRepository.findByUserId(userId)
115                 .filter(previous -> !previous.phone().equals(pending.phone()))
116                 .isPresent();
117         final var stored = userPhoneRepository.save(new UserPhone(userId, pending.phone(), now));
118         phoneVerificationRepository.deleteByUserId(userId);
119         if (stored && replaced) {
120             final var revoked = trustedDeviceRepository.deleteByUser(userId);
121             recordSecurityEventUseCase.execute(SecurityEventType.SECOND_FACTOR_REMOVED, userId, username,
122                     REPLACED_DETAIL + revoked);
123         }
124         return stored;
125     }
126 }
