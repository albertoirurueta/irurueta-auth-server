16  package com.irurueta.authserver.authorization.application.social;
17  
18  import com.irurueta.authserver.authorization.domain.passkeys.PasskeyRepository;
19  import com.irurueta.authserver.authorization.domain.social.FederatedIdentityRepository;
20  import com.irurueta.authserver.authorization.domain.social.GeneratedPasswordRepository;
21  import com.irurueta.authserver.authorization.domain.social.SignInMethods;
22  import com.irurueta.authserver.authorization.domain.social.SignInMethodsLock;
23  import com.irurueta.authserver.authorization.domain.x509.CertificateBindingRepository;
24  import java.util.UUID;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.stereotype.Service;
27  import org.springframework.transaction.annotation.Propagation;
28  import org.springframework.transaction.annotation.Transactional;
29  
30  // tag::social-lock-sign-in-methods[]
31  /**
32   * Use case that locks the sign-in methods of an account for the rest of the caller's transaction and counts them: a
33   * password its owner knows (not a generated one), the linked identities, a passkey and the bound certificates. Every
34   * use case that removes a sign-in method starts with it, so two concurrent removals of the same account are serialized
35   * and the second one counts what the first left. It only runs inside the caller's transaction, which holds the lock.
36   */
37  @Service
38  @Transactional(propagation = Propagation.MANDATORY)
39  @RequiredArgsConstructor
40  public class LockSignInMethodsUseCase {
41  
42      /** Port that serializes the removals of an account. */
43      private final SignInMethodsLock signInMethodsLock;
44  
45      /** Port that stores the links. */
46      private final FederatedIdentityRepository federatedIdentityRepository;
47  
48      /** Port that remembers the accounts that only hold a generated password. */
49      private final GeneratedPasswordRepository generatedPasswordRepository;
50  
51      /** Port that reads the passkeys. */
52      private final PasskeyRepository passkeyRepository;
53  
54      /** Port that reads the bound certificates. */
55      private final CertificateBindingRepository certificateBindingRepository;
56  
57      /**
58       * Locks the sign-in methods of an account and counts them.
59       *
60       * @param userId identifier of the account.
61       * @return the ways the account can sign in now, counted once the lock is held.
62       */
63      public SignInMethods execute(final UUID userId) {
64          signInMethodsLock.lock(userId);
65          return new SignInMethods(
66                  !generatedPasswordRepository.isGenerated(userId),
67                  federatedIdentityRepository.countByUser(userId),
68                  passkeyRepository.existsForAccount(userId),
69                  certificateBindingRepository.findByUser(userId).size());
70      }
71  }
72  // end::social-lock-sign-in-methods[]
