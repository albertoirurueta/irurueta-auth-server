16  package com.irurueta.authserver.authorization.application.social;
17  
18  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
19  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
20  import com.irurueta.authserver.authorization.domain.social.FederatedIdentity;
21  import com.irurueta.authserver.authorization.domain.social.FederatedIdentityAlreadyLinkedException;
22  import com.irurueta.authserver.authorization.domain.social.FederatedIdentityRepository;
23  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
24  import java.time.Clock;
25  import java.util.UUID;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.stereotype.Service;
28  import org.springframework.transaction.annotation.Transactional;
29  
30  /**
31   * Use case that links an identity of an external provider to the account of a signed-in user. Unlike the sign-in, it
32   * does not depend on the e-mail address: the user proved to own the account by being signed in, and the provider by
33   * having just authenticated. The identity is locked first, as the sign-in does, so a concurrent sign-in or link of the
34   * same identity is serialized with it.
35   */
36  @Service
37  @Transactional
38  @RequiredArgsConstructor
39  public class LinkIdentityUseCase {
40  
41      /** Port that stores the links. */
42      private final FederatedIdentityRepository federatedIdentityRepository;
43  
44      /** Use case that records the audit events. */
45      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
46  
47      /** Source of the current time. */
48      private final Clock clock;
49  
50      /**
51       * Links an identity to an account. Linking an identity the account already has changes nothing.
52       *
53       * @param userId   identifier of the signed-in user.
54       * @param username name of the signed-in account, for the audit log.
55       * @param provider provider that authenticated the user.
56       * @param subject  stable identifier the provider assigned to the user.
57       * @param email    e-mail address the provider reported, or {@code null}.
58       * @return the link.
59       * @throws FederatedIdentityAlreadyLinkedException if the identity belongs to another account.
60       */
61      public FederatedIdentity execute(final UUID userId, final String username, final IdentityProvider provider,
62                                       final String subject, final String email) {
63          federatedIdentityRepository.lockSubject(provider, subject);
64          final var existing = federatedIdentityRepository.findByProviderAndSubject(provider, subject);
65          if (existing.isPresent()) {
66              if (existing.get().userId().equals(userId)) {
67                  return existing.get();
68              }
69              throw new FederatedIdentityAlreadyLinkedException("The identity is linked to another account");
70          }
71          final var now = clock.instant();
72          final var stored = federatedIdentityRepository.save(
73                  new FederatedIdentity(UUID.randomUUID(), userId, provider, subject, email, now, null));
74          recordSecurityEventUseCase.execute(SecurityEventType.IDENTITY_LINKED, userId, username, provider.id());
75          return stored;
76      }
77  }
