17  package com.irurueta.authserver.authorization.application.account;
18  
19  import com.irurueta.authserver.authorization.domain.account.AccountToken;
20  import com.irurueta.authserver.authorization.domain.account.InvalidAccountTokenException;
21  import com.irurueta.authserver.authorization.domain.account.SignedTokenService;
22  import com.irurueta.authserver.authorization.domain.password.UserAccountRepository;
23  import java.util.UUID;
24  import lombok.RequiredArgsConstructor;
25  import org.springframework.stereotype.Service;
26  import org.springframework.transaction.annotation.Transactional;
27  
28  /**
29   * Use case that confirms an e-mail address from the signed token that was mailed to it, which also enables the
30   * account. A token is effective once: presenting it again, even before it expires, is rejected as an invalid token,
31   * so a replayed link can never re-enable an account that was disabled after its verification.
32   */
33  @Service
34  @RequiredArgsConstructor
35  public class VerifyEmailUseCase {
36  
37      /** Port that verifies the token. */
38      private final SignedTokenService signedTokenService;
39  
40      /** Port that stores the accounts. */
41      private final UserAccountRepository userAccountRepository;
42  
43      /**
44       * Verifies the e-mail address the token was issued for.
45       *
46       * @param token the signed verification token.
47       * @throws InvalidAccountTokenException if the token is not valid, or the address was already verified.
48       */
49      @Transactional
50      public void execute(final String token) {
51          final var subject = signedTokenService.verify(token, AccountToken.PURPOSE_EMAIL_VERIFICATION)
52                  .orElseThrow(InvalidAccountTokenException::new);
53          // single effective use: a replayed (still unexpired) token finds the address already verified
54          if (!userAccountRepository.markEmailVerified(parseUserId(subject))) {
55              throw new InvalidAccountTokenException();
56          }
57      }
58  
59      /**
60       * Parses the subject of a verified token.
61       *
62       * @param subject the subject.
63       * @return the user id.
64       * @throws InvalidAccountTokenException if the subject is not a user id.
65       */
66      private static UUID parseUserId(final String subject) {
67          try {
68              return UUID.fromString(subject);
69          } catch (IllegalArgumentException e) {
70              throw new InvalidAccountTokenException();
71          }
72      }
73  }
