16  package com.irurueta.authserver.authorization.application.x509;
17  
18  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
19  import com.irurueta.authserver.authorization.application.social.LockSignInMethodsUseCase;
20  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
21  import com.irurueta.authserver.authorization.domain.social.FederatedLinkingPolicy;
22  import com.irurueta.authserver.authorization.domain.social.LastSignInMethodException;
23  import com.irurueta.authserver.authorization.domain.x509.CertificateBindingRepository;
24  import com.irurueta.authserver.authorization.domain.x509.CertificateNotFoundException;
25  import java.util.UUID;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.stereotype.Service;
28  import org.springframework.transaction.annotation.Transactional;
29  
30  // tag::x509-unbind-use-case[]
31  /**
32   * Use case that unbinds a client certificate from an account and audits it. Like unlinking an identity, it refuses to
33   * remove the last way to sign in, counted under the lock of {@link LockSignInMethodsUseCase} so that concurrent
34   * removals of the same account are serialized.
35   */
36  @Service
37  @Transactional
38  @RequiredArgsConstructor
39  public class UnbindCertificateUseCase {
40  
41      /** Port that stores the bindings. */
42      private final CertificateBindingRepository certificateBindingRepository;
43  
44      /** Locks and counts the sign-in methods of the account. */
45      private final LockSignInMethodsUseCase lockSignInMethodsUseCase;
46  
47      /** Rules of removing a sign-in method. */
48      private final FederatedLinkingPolicy federatedLinkingPolicy;
49  
50      /** Use case that records the audit events. */
51      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
52  
53      /**
54       * Unbinds a certificate; a certificate of another account is never touched.
55       *
56       * @param userId   identifier of the account.
57       * @param username name of the account, for the audit log.
58       * @param id       identifier of the binding.
59       * @throws CertificateNotFoundException if the account has no such binding.
60       * @throws LastSignInMethodException    if the certificate is the last way the account can sign in.
61       */
62      public void execute(final UUID userId, final String username, final UUID id) {
63          final var methods = lockSignInMethodsUseCase.execute(userId);
64          if (certificateBindingRepository.findByUser(userId).stream().noneMatch(binding -> binding.id().equals(id))) {
65              throw new CertificateNotFoundException("The account has no certificate " + id);
66          }
67          if (!federatedLinkingPolicy.mayUnlink(methods.withoutCertificate())) {
68              throw new LastSignInMethodException("The certificate is the last way to sign in to the account");
69          }
70          if (!certificateBindingRepository.delete(id, userId)) {
71              throw new CertificateNotFoundException("The account has no certificate " + id);
72          }
73          recordSecurityEventUseCase.execute(SecurityEventType.CERTIFICATE_UNBOUND, userId, username, id.toString());
74      }
75  }
76  // end::x509-unbind-use-case[]
