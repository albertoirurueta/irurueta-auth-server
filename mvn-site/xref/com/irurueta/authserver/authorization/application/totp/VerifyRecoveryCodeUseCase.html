37  /**
38   * Use case that accepts a recovery code as a second factor. A code works once. Every stored code is checked, with no
39   * early exit, so the time taken does not reveal which one matched; failures count towards the lockout. A user without
40   * a confirmed second factor has no usable recovery code: the attempt fails and is counted.
41   */
42  @Service
43  @Transactional
44  @RequiredArgsConstructor
45  public class VerifyRecoveryCodeUseCase {
46  
47      /** Port that stores the recovery codes. */
48      private final RecoveryCodeRepository recoveryCodeRepository;
49  
50      /** Port that hashes the codes like passwords. */
51      private final PasswordHasher passwordHasher;
52  
53      /** Normalizer of typed codes. */
54      private final RecoveryCodes recoveryCodes;
55  
56      /** Use case that tells whether the user has confirmed a second factor. */
57      private final HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
58  
59      /** Use case that tells whether the second factor is locked. */
60      private final CheckSecondFactorLockoutUseCase checkSecondFactorLockoutUseCase;
61  
62      /** Use case that counts a failure. */
63      private final RegisterSecondFactorFailureUseCase registerSecondFactorFailureUseCase;
64  
65      /** Use case that clears the failures. */
66      private final ResetSecondFactorFailuresUseCase resetSecondFactorFailuresUseCase;
67  
68      /** Use case that records the audit events. */
69      private final RecordSecurityEventUseCase recordSecurityEventUseCase;
70  
71      /** Source of the current time. */
72      private final Clock clock;
73  
74      /**
75       * Verifies and consumes a recovery code.
76       *
77       * @param userId   identifier of the user.
78       * @param username name of the user, for the audit log.
79       * @param code     the code as typed.
80       * @return {@code true} if the code was valid (and is now used up); {@code false} when it was not, or the user has
81       *         no confirmed second factor.
82       * @throws SecondFactorLockedException if the second factor of the user is locked.
83       */
84      public boolean execute(final UUID userId, final String username, final String code) {
85          if (checkSecondFactorLockoutUseCase.execute(userId)) {
86              throw new SecondFactorLockedException("The second factor is locked after too many failed attempts");
87          }
88          if (!hasConfirmedSecondFactorUseCase.execute(userId)) {
89              registerSecondFactorFailureUseCase.execute(userId, username);
90              return false;
91          }
92          final var normalized = recoveryCodes.normalize(code == null ? "" : code);
93          Optional<RecoveryCode> match = Optional.empty();
94          for (final var candidate : recoveryCodeRepository.findUnused(userId)) {
95              final var matches = passwordHasher.matches(normalized, candidate.hashedCode());
96              if (matches && match.isEmpty()) {
97                  match = Optional.of(candidate);
98              }
99          }
100         if (match.isPresent() && recoveryCodeRepository.markUsed(match.get().id(), clock.instant())) {
101             resetSecondFactorFailuresUseCase.execute(userId);
102             recordSecurityEventUseCase.execute(SecurityEventType.RECOVERY_CODE_USED, userId, username, null);
103             return true;
104         }
105         registerSecondFactorFailureUseCase.execute(userId, username);
106         return false;
107     }
108 }
