17  package com.irurueta.authserver.authorization.application.totp;
18  
19  import com.irurueta.authserver.authorization.application.mfa.HasConfirmedSecondFactorUseCase;
20  import com.irurueta.authserver.authorization.domain.password.PasswordHasher;
21  import com.irurueta.authserver.authorization.domain.totp.RecoveryCode;
22  import com.irurueta.authserver.authorization.domain.totp.RecoveryCodeRepository;
23  import com.irurueta.authserver.authorization.domain.totp.TotpNotEnrolledException;
24  import java.util.List;
25  import java.util.UUID;
26  import java.util.stream.Stream;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.stereotype.Service;
29  import org.springframework.transaction.annotation.Transactional;
30  
31  /**
32   * Use case that creates a fresh set of recovery codes for a user, replacing any previous set. The clear-text codes
33   * are returned once; only their hashes are stored. Recovery codes stand in for a second factor, so they are only
34   * issued to a user who has confirmed one.
35   */
36  @Service
37  @Transactional
38  @RequiredArgsConstructor
39  public class GenerateRecoveryCodesUseCase {
40  
41      /** Number of codes in a set. */
42      public static final int CODE_COUNT = 10;
43  
44      /** Port that stores the recovery codes. */
45      private final RecoveryCodeRepository recoveryCodeRepository;
46  
47      /** Port that hashes the codes like passwords. */
48      private final PasswordHasher passwordHasher;
49  
50      /** Creator of the codes. */
51      private final RecoveryCodes recoveryCodes;
52  
53      /** Use case that tells whether the user has confirmed a second factor. */
54      private final HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
55  
56      /**
57       * Generates the codes.
58       *
59       * @param userId identifier of the user.
60       * @return the clear-text codes, to be shown to the user once.
61       * @throws TotpNotEnrolledException if the user has no confirmed second factor.
62       */
63      public List<String> execute(final UUID userId) {
64          if (!hasConfirmedSecondFactorUseCase.execute(userId)) {
65              throw new TotpNotEnrolledException("Recovery codes need a confirmed authenticator app");
66          }
67          final var codes = Stream.generate(recoveryCodes::newCode).limit(CODE_COUNT).toList();
68          recoveryCodeRepository.replaceAll(userId, codes.stream()
69                  .map(code -> new RecoveryCode(UUID.randomUUID(), userId, passwordHasher.encode(code), null))
70                  .toList());
71          return codes;
72      }
73  }
