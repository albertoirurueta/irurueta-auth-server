16  package com.irurueta.authserver.authorization.domain.mfa;
17  
18  import static com.irurueta.authserver.authorization.domain.Preconditions.requireText;
19  
20  /**
21   * A way of authenticating, as registered by a feature.
22   *
23   * @param id        stable identifier of the factor, for example {@code totp}.
24   * @param amrValue  value of the {@code amr} claim (RFC 8176) reported when the factor was used, for example
25   *                  {@code otp}.
26   * @param authority authority granted to an authentication that used the factor, for example {@code FACTOR_TOTP}.
27   * @param strength    how much the factor proves on its own.
28   * @param multiFactor whether the factor alone proves a multi-factor authentication, so a session holding only it
29   *                    reaches the multi-factor class ({@code acr}) and passes a step-up: for example a sign-in at a
30   *                    provider trusted for, and reporting, its own multi-factor authentication. Only a
31   *                    {@link FactorStrength#PRIMARY_STRONG} factor can be one; a strong factor that proves a single
32   *                    thing (a passkey whose user verification is not required, a certificate) is not.
33   */
34  public record AuthenticationFactor(String id, String amrValue, String authority, FactorStrength strength,
35          boolean multiFactor) {
36  
37      /**
38       * Validates the factor.
39       *
40       * @throws IllegalArgumentException if a text is blank, the strength is missing, or a factor that is not
41       *                                  {@link FactorStrength#PRIMARY_STRONG} claims to be multi-factor alone.
42       */
43      public AuthenticationFactor {
44          requireText(id, "id");
45          requireText(amrValue, "amrValue");
46          requireText(authority, "authority");
47          if (strength == null) {
48              throw new IllegalArgumentException("strength must not be null");
49          }
50          if (multiFactor && strength != FactorStrength.PRIMARY_STRONG) {
51              throw new IllegalArgumentException("only a PRIMARY_STRONG factor can be multi-factor alone");
52          }
53      }
54  
55      /**
56       * Creates a factor that does not prove a multi-factor authentication alone, the usual case.
57       *
58       * @param id        stable identifier of the factor.
59       * @param amrValue  value of the {@code amr} claim reported when the factor was used.
60       * @param authority authority granted to an authentication that used the factor.
61       * @param strength  how much the factor proves on its own.
62       * @throws IllegalArgumentException if a text is blank or the strength is missing.
63       */
64      public AuthenticationFactor(final String id, final String amrValue, final String authority,
65              final FactorStrength strength) {
66          this(id, amrValue, authority, strength, false);
67      }
68  }
