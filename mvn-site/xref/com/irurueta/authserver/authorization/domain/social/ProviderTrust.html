16  package com.irurueta.authserver.authorization.domain.social;
17  
18  import java.util.Collection;
19  import java.util.Objects;
20  
21  /**
22   * How far a provider's own multi-factor authentication is trusted. By default a federated sign-in is a primary factor
23   * and the normal second-factor flow applies; only a provider marked as trusted, whose upstream {@code amr} claim
24   * proves {@code mfa}, is accepted as a strong sign-in on its own.
25   *
26   * @param provider provider the setting belongs to.
27   * @param trustMfa whether the provider's multi-factor claim is trusted.
28   */
29  public record ProviderTrust(IdentityProvider provider, boolean trustMfa) {
30  
31      /** The {@code amr} value (RFC 8176) that says the provider authenticated the user with several factors. */
32      public static final String MFA_AMR = "mfa";
33  
34      /**
35       * Validates the trust setting.
36       *
37       * @throws IllegalArgumentException if the provider is missing.
38       */
39      public ProviderTrust {
40          if (provider == null) {
41              throw new IllegalArgumentException("provider must not be null");
42          }
43      }
44  
45      /**
46       * Tells whether a sign-in counts as strong, that is, whether it proves identity without a second factor.
47       *
48       * @param upstreamAmr {@code amr} values the provider reported, or {@code null} when it reported none.
49       * @return {@code true} when the provider is trusted and its {@code amr} contains {@value #MFA_AMR}.
50       */
51      public boolean isStrong(final Collection<String> upstreamAmr) {
52          return trustMfa && upstreamAmr != null
53                  && upstreamAmr.stream().filter(Objects::nonNull).anyMatch(MFA_AMR::equals);
54      }
55  }
