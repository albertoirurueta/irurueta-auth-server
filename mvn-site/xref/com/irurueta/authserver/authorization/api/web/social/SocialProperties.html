33  /**
34   * Settings of the sign-in with external providers, bound once from {@code authserver.social.*} and injected wherever
35   * one is needed; an invalid value stops the application at startup. A provider is enabled when its
36   * {@code client-id} has text: a blank one leaves the provider off, so the shipped {@code application.yml} can list all
37   * of them with the values of environment variables and nothing is enabled until a deployment supplies them. Secrets
38   * (the client secret, the Apple private key) are never kept in a committed file: they come from the environment only.
39   *
40   * <p>Whether a provider's own multi-factor authentication is trusted ({@code trust-mfa}) is per provider and off by
41   * default: the sign-in is then a primary factor and the normal second-factor flow applies.
42   *
43   * @param providers settings of each provider, by registration id ({@code google}, {@code microsoft}, {@code apple},
44   *                  {@code github}).
45   */
46  @ConfigurationProperties("authserver.social")
47  public record SocialProperties(Map<String, Provider> providers) {
48  
49      /** Tenant value of Entra ID that accepts work, school and personal accounts of any tenant. */
50      public static final String COMMON_TENANT = "common";
51  
52      /** Tenant value of Entra ID that accepts work and school accounts of any tenant. */
53      public static final String ORGANIZATIONS_TENANT = "organizations";
54  
55      /** Tenant value of Entra ID that accepts personal accounts only. */
56      public static final String CONSUMERS_TENANT = "consumers";
57  
58      /** Longest lifetime Apple accepts for a client secret: six months. */
59      public static final Duration MAX_APPLE_CLIENT_SECRET_TTL = Duration.ofDays(180);
60  
61      /** Default lifetime of the client secret JWT sent to Apple. */
62      public static final Duration DEFAULT_APPLE_CLIENT_SECRET_TTL = Duration.ofMinutes(5);
63  
64      /** Shape of a tenant id (a GUID): the issuer of a single-tenant token is built from it. */
65      private static final Pattern TENANT_ID_PATTERN = Pattern.compile(
66              "[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}");
67  
68      /** Characters a tenant (a GUID or a domain name) may contain: it becomes part of a URL. */
69      private static final Pattern TENANT_PATTERN = Pattern.compile("[A-Za-z0-9][A-Za-z0-9.-]*");
70  
71      /**
72       * Validates the settings and copies the map.
73       *
74       * @throws IllegalArgumentException if a key is not the registration id of a known provider, or an enabled provider
75       *                                  lacks a setting it needs or has an invalid one.
76       */
77      public SocialProperties {
78          final var copy = new LinkedHashMap<String, Provider>();
79          if (providers != null) {
80              providers.forEach((id, provider) -> {
81                  final var known = IdentityProvider.fromId(id).orElseThrow(() -> new IllegalArgumentException(
82                          "authserver.social.providers." + id + " is not a known provider"));
83                  if (provider != null) {
84                      validate(known, provider);
85                      copy.put(id, provider);
86                  }
87              });
88          }
89          providers = Map.copyOf(copy);
90      }
91  
92      /**
93       * Returns the enabled providers, in the order of {@link IdentityProvider}.
94       *
95       * @return the enabled providers and their settings; empty when none is configured.
96       */
97      public Map<IdentityProvider, Provider> enabledProviders() {
98          final var enabled = new EnumMap<IdentityProvider, Provider>(IdentityProvider.class);
99          providers.forEach((id, provider) -> {
100             if (provider.enabled()) {
101                 IdentityProvider.fromId(id).ifPresent(known -> enabled.put(known, provider));
102             }
103         });
104         return enabled;
105     }
106 
107     /**
108      * Returns the settings of an enabled provider.
109      *
110      * @param provider the provider.
111      * @return its settings, or an empty optional when it is not enabled.
112      */
113     public Optional<Provider> find(final IdentityProvider provider) {
114         return Optional.ofNullable(enabledProviders().get(provider));
115     }
116 
117     /**
118      * Returns how far the multi-factor authentication of a provider is trusted.
119      *
120      * @param provider the provider.
121      * @return the trust setting; the provider is not trusted when it is not enabled.
122      */
123     public ProviderTrust trust(final IdentityProvider provider) {
124         return new ProviderTrust(provider, find(provider).map(Provider::trustMfa).orElse(false));
125     }
126 
127     /**
128      * Checks the settings of a provider that is enabled.
129      *
130      * @param id       the provider.
131      * @param provider its settings.
132      */
133     private static void validate(final IdentityProvider id, final Provider provider) {
134         if (!provider.enabled()) {
135             return;
136         }
137         final var prefix = "authserver.social.providers." + id.id() + ".";
138         if (id == IdentityProvider.APPLE) {
139             require(provider.teamId(), prefix + "team-id");
140             require(provider.keyId(), prefix + "key-id");
141             require(provider.privateKey(), prefix + "private-key");
142             if (provider.clientSecretTtl().isNegative() || provider.clientSecretTtl().isZero()
143                     || provider.clientSecretTtl().compareTo(MAX_APPLE_CLIENT_SECRET_TTL) > 0) {
144                 throw new IllegalArgumentException(
145                         prefix + "client-secret-ttl must be positive and at most " + MAX_APPLE_CLIENT_SECRET_TTL);
146             }
147         } else {
148             require(provider.clientSecret(), prefix + "client-secret");
149         }
150         if (id == IdentityProvider.MICROSOFT) {
151             if (!TENANT_PATTERN.matcher(provider.tenant()).matches()) {
152                 throw new IllegalArgumentException(prefix + "tenant is not a valid tenant");
153             }
154             if (provider.allowedTenants().stream().anyMatch(tenant -> !TENANT_PATTERN.matcher(tenant).matches())) {
155                 throw new IllegalArgumentException(prefix + "allowed-tenants holds a value that is not a valid tenant");
156             }
157         }
158     }
159 
160     /**
161      * Requires a setting to have text.
162      *
163      * @param value value of the setting.
164      * @param name  full name of the setting, for the message.
165      */
166     private static void require(final String value, final String name) {
167         if (!StringUtils.hasText(value)) {
168             throw new IllegalArgumentException(name + " must not be blank when the provider is enabled");
169         }
170     }
171 
172     /**
173      * Settings of one provider. Only the ones that apply to the provider are read: the client secret by Google,
174      * Microsoft and GitHub, the tenant settings by Microsoft and the key settings by Apple (whose client secret is a
175      * JWT signed with the private key).
176      *
177      * @param clientId        client id registered with the provider; blank leaves the provider disabled.
178      * @param clientSecret    client secret (Google, Microsoft, GitHub), supplied from the environment.
179      * @param trustMfa        whether the provider's {@code amr} claim is trusted to prove several factors (default
180      *                        {@code false}).
181      * @param tenant          Entra ID tenant: {@code common} (the default), {@code organizations},
182      *                        {@code consumers}, or a tenant id or domain.
183      * @param allowedTenants  tenant ids an Entra ID multi-tenant sign-in is limited to; empty accepts any tenant.
184      * @param teamId          Apple developer team id.
185      * @param keyId           id of the Apple key the client secret is signed with.
186      * @param privateKey      PKCS#8 PEM private key (ES256) Apple issued, supplied from the environment; a literal
187      *                        {@code \n} stands for a line break.
188      * @param clientSecretTtl lifetime of the client secret JWT sent to Apple (default five minutes, at most six
189      *                        months).
190      */
191     public record Provider(
192             String clientId,
193             String clientSecret,
194             boolean trustMfa,
195             @DefaultValue(COMMON_TENANT) String tenant,
196             List<String> allowedTenants,
197             String teamId,
198             String keyId,
199             String privateKey,
200             @DefaultValue("PT5M") Duration clientSecretTtl) {
201 
202         /**
203          * Copies the tenants and fills the defaults of the unbound settings.
204          */
205         public Provider {
206             allowedTenants = allowedTenants == null ? List.of() : List.copyOf(allowedTenants);
207             tenant = StringUtils.hasText(tenant) ? tenant : COMMON_TENANT;
208             clientSecretTtl = clientSecretTtl == null ? DEFAULT_APPLE_CLIENT_SECRET_TTL : clientSecretTtl;
209         }
210 
211         /**
212          * Returns the tenants an Entra ID sign-in is limited to: the configured list, else the tenant itself when it is
213          * a tenant id, else none, which means any tenant (a multi-tenant application, or a tenant given by domain
214          * name, whose tokens carry its id).
215          *
216          * @return the permitted tenant ids; empty accepts any tenant.
217          */
218         public List<String> permittedTenants() {
219             if (!allowedTenants.isEmpty()) {
220                 return allowedTenants;
221             }
222             return TENANT_ID_PATTERN.matcher(tenant).matches() ? List.of(tenant) : List.of();
223         }
224 
225         /**
226          * Tells whether the provider is enabled.
227          *
228          * @return {@code true} when the client id has text.
229          */
230         public boolean enabled() {
231             return StringUtils.hasText(clientId);
232         }
233     }
234 }
