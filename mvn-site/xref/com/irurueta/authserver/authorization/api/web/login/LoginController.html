17  package com.irurueta.authserver.authorization.api.web.login;
18  
19  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
20  import com.irurueta.authserver.authorization.api.web.pages.v1.generated.api.LoginPageApi;
21  import jakarta.servlet.http.HttpServletRequest;
22  import java.util.Comparator;
23  import java.util.HashMap;
24  import java.util.List;
25  import org.springframework.http.ResponseEntity;
26  import org.springframework.security.core.AuthenticationException;
27  import org.springframework.security.web.WebAttributes;
28  import org.springframework.stereotype.Controller;
29  
30  /**
31   * Serves the login page, as the contract {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml} describes it
32   * through the generated {@link LoginPageApi}, supplying the ordered login method fragments and the outcome of the last
33   * attempt to the template.
34   */
35  @Controller
36  public class LoginController implements LoginPageApi {
37  
38      /**
39       * Path of the login page, which also receives the form post. The single source of the path for the security
40       * chains and the template, taken from the contract.
41       */
42      public static final String LOGIN_PATH = PATH_LOGIN_PAGE;
43  
44      /** Name of the login template. */
45      static final String VIEW_NAME = "login";
46  
47      /** Name of the model attribute that holds the path the login form posts to. */
48      static final String LOGIN_PATH_ATTRIBUTE = "loginPath";
49  
50      /** Name of the model attribute that holds the fragments of the login methods. */
51      static final String FRAGMENTS_ATTRIBUTE = "fragments";
52  
53      /** Name of the model attribute that holds the key of the message of a failed sign-in. */
54      static final String ERROR_MESSAGE_ATTRIBUTE = "errorMessageKey";
55  
56      /** Name of the model attribute that tells the user has just signed out. */
57      static final String LOGGED_OUT_ATTRIBUTE = "loggedOut";
58  
59      /** Fragments contributed by the authentication methods, sorted by order. */
60      private final List<LoginMethodFragment> fragments;
61  
62      /** Maps the last authentication exception to a message key. */
63      private final LoginErrorMessages errorMessages;
64  
65      /** Renders the page. */
66      private final PageRenderer pageRenderer;
67  
68      /** The current request, injected as a thread-bound proxy, whose session holds the last failure. */
69      private final HttpServletRequest request;
70  
71      /**
72       * Creates the controller.
73       *
74       * @param methods       fragments contributed by the authentication methods.
75       * @param errorMessages mapper of the authentication exceptions to message keys.
76       * @param pageRenderer  renderer of the page.
77       * @param request       the current request.
78       */
79      public LoginController(final List<LoginMethodFragment> methods, final LoginErrorMessages errorMessages,
80              final PageRenderer pageRenderer, final HttpServletRequest request) {
81          this.fragments = methods.stream().sorted(Comparator.comparingInt(LoginMethodFragment::order)).toList();
82          this.errorMessages = errorMessages;
83          this.pageRenderer = pageRenderer;
84          this.request = request;
85      }
86  
87      // tag::login-controller[]
88      /**
89       * Renders the login page.
90       *
91       * @param error  present (even if empty) when the previous attempt failed.
92       * @param logout present (even if empty) when the user has just signed out.
93       * @return the login page.
94       */
95      @Override
96      public ResponseEntity<String> loginPage(final String error, final String logout) {
97          final var model = new HashMap<String, Object>();
98          model.put(LOGIN_PATH_ATTRIBUTE, LOGIN_PATH);
99          model.put(FRAGMENTS_ATTRIBUTE, fragments);
100         if (error != null) {
101             model.put(ERROR_MESSAGE_ATTRIBUTE, errorMessages.keyOf(lastFailure()));
102         }
103         model.put(LOGGED_OUT_ATTRIBUTE, logout != null);
104         return pageRenderer.render(VIEW_NAME, model);
105     }
106     // end::login-controller[]
107 
108     /**
109      * Gets the exception of the previous failed attempt, which Spring Security keeps in the session.
110      *
111      * @return the exception, or {@code null} when there is no session or it holds none.
112      */
113     private AuthenticationException lastFailure() {
114         final var session = request.getSession(false);
115         final var failure = session == null ? null : session.getAttribute(WebAttributes.AUTHENTICATION_EXCEPTION);
116         return failure instanceof final AuthenticationException exception ? exception : null;
117     }
118 }
