39  // tag::phone-page-controller[]
40  /**
41   * Serves the phone account page and the session-protected form posts it uses, as the contract
42   * {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml} describes them through the generated
43   * {@link PhonePageApi}. A browser page has a session, not a bearer token, so it cannot call the REST API
44   * ({@code /api/account/phone}, which {@code ApiSecurityConfig} keeps bearer-only); the forms call the same use cases
45   * instead, with the same limits, and each answers by redirecting back to the page (post/redirect/get) with the outcome
46   * in the {@code status} parameter. The number is shown masked, as the REST API answers it.
47   */
48  @Controller
49  @RequiredArgsConstructor
50  public class PhonePageController implements PhonePageApi {
51  
52      /** Path of the page, taken from the contract. */
53      public static final String PHONE_PATH = PATH_PHONE_PAGE;
54  
55      /** Name of the template of the page. */
56      static final String VIEW_NAME = "account-phone";
57  
58      /** Name of the model attribute that holds the masked verified number, absent when there is none. */
59      static final String PHONE_ATTRIBUTE = "phone";
60  
61      /** Name of the model attribute that holds the outcome of the previous post. */
62      static final String STATUS_ATTRIBUTE = "status";
63  
64      /** Status of a code that was requested. */
65      static final String PENDING = "pending";
66  
67      /** Status of a number that was verified. */
68      static final String VERIFIED = "verified";
69  
70      /** Status of a number that was removed. */
71      static final String REMOVED = "removed";
72  
73      /** Status of a number that is not valid. */
74      static final String INVALID_PHONE = "invalid-phone";
75  
76      /** Status of a number of a country SMS messages are not sent to. */
77      static final String COUNTRY_NOT_ALLOWED = "country-not-allowed";
78  
79      /** Status of a wrong code. */
80      static final String INVALID_CODE = "invalid-code";
81  
82      /** Status of a request over a limit. */
83      static final String RATE_LIMITED = "rate-limited";
84  
85      /** Status of an enrolment or a removal that needs a recent multi-factor sign-in. */
86      static final String STEP_UP_REQUIRED = "step-up-required";
87  
88      /** The statuses the page tells; any other value of the parameter is ignored. */
89      private static final Set<String> STATUSES = Set.of(PENDING, VERIFIED, REMOVED, INVALID_PHONE,
90              COUNTRY_NOT_ALLOWED, INVALID_CODE, RATE_LIMITED, STEP_UP_REQUIRED);
91  
92      /** Resolves the account of the request. */
93      private final CurrentAccountResolver currentAccountResolver;
94  
95      /** Finds the verified phone of the account. */
96      private final FindUserPhoneUseCase findUserPhoneUseCase;
97  
98      /** Starts an enrolment. */
99      private final StartPhoneEnrolmentUseCase startPhoneEnrolmentUseCase;
100 
101     /** Confirms an enrolment. */
102     private final ConfirmPhoneUseCase confirmPhoneUseCase;
103 
104     /** Removes the phone. */
105     private final RemovePhoneUseCase removePhoneUseCase;
106 
107     /** Requires a recent multi-factor sign-in before the phone is replaced or removed. */
108     private final PhoneSessionStepUp stepUp;
109 
110     /** Limits the requests. */
111     private final OttRateLimiter rateLimiter;
112 
113     /** Maps the phone onto its masked form. */
114     private final PhoneDtoMapper phoneDtoMapper;
115 
116     /** Renders the page. */
117     private final PageRenderer pageRenderer;
118 
119     /** The current request, injected as a thread-bound proxy, which tells the address of the client. */
120     private final HttpServletRequest request;
121 
122     /**
123      * Renders the page.
124      *
125      * @param status outcome of the previous post, ignored when it is not one of the known ones.
126      * @return the page, never cached.
127      */
128     @Override
129     public ResponseEntity<String> phonePage(final String status) {
130         final var account = currentAccountResolver.resolve();
131         final var model = new HashMap<String, Object>();
132         findUserPhoneUseCase.execute(account.id())
133                 .ifPresent(phone -> model.put(PHONE_ATTRIBUTE, phoneDtoMapper.toResponse(phone).maskedNumber()));
134         if (status != null && STATUSES.contains(status)) {
135             model.put(STATUS_ATTRIBUTE, status);
136         }
137         final var page = pageRenderer.render(VIEW_NAME, model);
138         return ResponseEntity.status(page.getStatusCode())
139                 .headers(page.getHeaders())
140                 .cacheControl(CacheControl.noStore())
141                 .body(page.getBody());
142     }
143 
144     /**
145      * Sends the verification code to the number, when the session shows a recent enough multi-factor sign-in for an
146      * account that already has a confirmed second factor (the new number would replace the verified one).
147      *
148      * @param phone the number.
149      * @return a redirect to the page; the status is {@code pending} whether or not the number belongs to another
150      *         account.
151      */
152     @Override
153     public ResponseEntity<Void> startPhoneEnrolmentPage(final String phone) {
154         final var account = currentAccountResolver.resolve();
155         if (!stepUp.isSatisfied(account.id(), SecurityContextHolder.getContext().getAuthentication())) {
156             return redirect(STEP_UP_REQUIRED);
157         }
158         try {
159             rateLimiter.checkGeneration(request.getRemoteAddr(), PhoneController.ACCOUNT_KEY_PREFIX + account.id());
160             startPhoneEnrolmentUseCase.execute(account.id(), phone);
161             return redirect(PENDING);
162         } catch (final OttRateLimitExceededException e) {
163             return redirect(RATE_LIMITED);
164         } catch (final InvalidPhoneNumberException e) {
165             return redirect(INVALID_PHONE);
166         } catch (final SmsCountryNotAllowedException e) {
167             return redirect(COUNTRY_NOT_ALLOWED);
168         }
169     }
170 
171     /**
172      * Confirms the number with the code.
173      *
174      * @param code the code received by SMS.
175      * @return a redirect to the page, with the outcome.
176      */
177     @Override
178     public ResponseEntity<Void> verifyPhonePage(final String code) {
179         final var account = currentAccountResolver.resolve();
180         try {
181             rateLimiter.checkVerification(request.getRemoteAddr(), PhoneController.ACCOUNT_KEY_PREFIX + account.id());
182         } catch (final OttRateLimitExceededException e) {
183             return redirect(RATE_LIMITED);
184         }
185         return redirect(confirmPhoneUseCase.execute(account.id(), account.username(), code) ? VERIFIED : INVALID_CODE);
186     }
187 
188     /**
189      * Removes the phone, when the session shows a recent enough multi-factor sign-in.
190      *
191      * @return a redirect to the page, with the outcome.
192      */
193     @Override
194     public ResponseEntity<Void> removePhonePage() {
195         final var account = currentAccountResolver.resolve();
196         if (!stepUp.isSatisfied(account.id(), SecurityContextHolder.getContext().getAuthentication())) {
197             return redirect(STEP_UP_REQUIRED);
198         }
199         removePhoneUseCase.execute(account.id(), account.username());
200         return redirect(REMOVED);
201     }
202 
203     /**
204      * Builds the redirect back to the page.
205      *
206      * @param status the outcome.
207      * @return {@code 303 See Other} to the page.
208      */
209     private ResponseEntity<Void> redirect(final String status) {
210         return ResponseEntity.status(HttpStatus.SEE_OTHER)
211                 .header(HttpHeaders.LOCATION, URI.create(PHONE_PATH + "?" + STATUS_ATTRIBUTE + "=" + status).toString())
212                 .build();
213     }
214 }
