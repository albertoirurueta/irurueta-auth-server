17  package com.irurueta.authserver.authorization.api.web.passkeys;
18  
19  import com.irurueta.authserver.authorization.api.web.mfa.CurrentAccountResolver;
20  import com.irurueta.authserver.authorization.api.web.mfa.StepUpGuard;
21  import com.irurueta.authserver.authorization.api.web.v1.generated.api.PasskeysApi;
22  import com.irurueta.authserver.authorization.api.web.v1.generated.model.Passkey;
23  import com.irurueta.authserver.authorization.api.web.v1.generated.model.RenamePasskeyRequest;
24  import com.irurueta.authserver.authorization.application.passkeys.DeletePasskeyUseCase;
25  import com.irurueta.authserver.authorization.application.passkeys.ListPasskeysUseCase;
26  import com.irurueta.authserver.authorization.application.passkeys.RenamePasskeyUseCase;
27  import java.util.List;
28  import lombok.RequiredArgsConstructor;
29  import org.springframework.http.ResponseEntity;
30  import org.springframework.web.bind.annotation.RestController;
31  
32  /**
33   * Manages the passkeys of the authenticated account (the operations of the contract). Registering one is not here:
34   * it is the WebAuthn ceremony of the security chain, which the account page drives.
35   */
36  @RestController
37  @RequiredArgsConstructor
38  public class PasskeyController implements PasskeysApi {
39  
40      /** Resolves the account of the request. */
41      private final CurrentAccountResolver currentAccountResolver;
42  
43      /** Lists the passkeys of an account. */
44      private final ListPasskeysUseCase listPasskeysUseCase;
45  
46      /** Renames a passkey. */
47      private final RenamePasskeyUseCase renamePasskeyUseCase;
48  
49      /** Deletes a passkey. */
50      private final DeletePasskeyUseCase deletePasskeyUseCase;
51  
52      /** Requires a recent multi-factor sign-in before a passkey is deleted. */
53      private final StepUpGuard stepUpGuard;
54  
55      /** Maps the passkeys onto the contract's DTOs. */
56      private final PasskeyDtoMapper passkeyDtoMapper;
57  
58      /**
59       * Lists the passkeys of the authenticated account.
60       *
61       * @return {@code 200 OK} with the passkeys, possibly none.
62       */
63      @Override
64      public ResponseEntity<List<Passkey>> listPasskeys() {
65          final var account = currentAccountResolver.resolve();
66          return ResponseEntity.ok(passkeyDtoMapper.toResponses(listPasskeysUseCase.execute(account.username())));
67      }
68  
69      /**
70       * Renames a passkey of the authenticated account.
71       *
72       * @param id      credential identifier of the passkey.
73       * @param request the new label.
74       * @return {@code 200 OK} with the renamed passkey.
75       * @throws com.irurueta.authserver.authorization.domain.passkeys.PasskeyNotFoundException
76       *         if the account has no such passkey.
77       */
78      @Override
79      public ResponseEntity<Passkey> renamePasskey(final String id, final RenamePasskeyRequest request) {
80          final var account = currentAccountResolver.resolve();
81          return ResponseEntity.ok(passkeyDtoMapper.toResponse(
82                  renamePasskeyUseCase.execute(account.username(), id, request.label())));
83      }
84  
85      /**
86       * Deletes a passkey of the authenticated account.
87       *
88       * @param id credential identifier of the passkey.
89       * @return {@code 204 No Content}.
90       * @throws com.irurueta.authserver.authorization.domain.passkeys.PasskeyNotFoundException
91       *         if the account has no such passkey.
92       * @throws com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException
93       *         if the account has a confirmed second factor and the token does not show a recent multi-factor
94       *         sign-in.
95       */
96      @Override
97      public ResponseEntity<Void> deletePasskey(final String id) {
98          final var account = currentAccountResolver.resolve();
99          stepUpGuard.requireRecentMultiFactor(account.id());
100         deletePasskeyUseCase.execute(account.id(), account.username(), id);
101         return ResponseEntity.noContent().build();
102     }
103 }
