39  /**
40   * Serves the account page of the linked identities and the form post that starts linking another one, as the contract
41   * {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml} describes them through the generated
42   * {@link IdentitiesPageApi}. The post records the link marker ({@link PendingIdentityLinks}) and redirects to the
43   * provider's sign-in; the provider's answer then links the identity to the signed-in account (see
44   * {@link FederatedAccountResolver}). The page exists whether or not a provider is configured; without one it lists the
45   * linked identities only. The chain keeps both behind the multi-factor policy ({@link IdentitiesPageChainCustomizer}).
46   */
47  @Controller
48  @RequiredArgsConstructor
49  public class IdentitiesPageController implements IdentitiesPageApi {
50  
51      /** Name of the template of the page. */
52      static final String VIEW_NAME = "account-identities";
53  
54      /** Name of the model attribute that holds the linked identities. */
55      static final String IDENTITIES_ATTRIBUTE = "identities";
56  
57      /** Name of the model attribute that holds the providers that can be linked. */
58      static final String PROVIDERS_ATTRIBUTE = "providers";
59  
60      /** Name of the model attribute that holds the outcome of the previous link. */
61      static final String STATUS_ATTRIBUTE = "status";
62  
63      /** Status of an identity that was linked. */
64      static final String LINKED = "linked";
65  
66      /** Status of an identity that belongs to another account. */
67      static final String ALREADY_LINKED = "already-linked";
68  
69      /** Status of a provider that is not configured. */
70      static final String UNKNOWN_PROVIDER = "unknown-provider";
71  
72      /** Path that starts the sign-in with a provider, answered by Spring Security. */
73      static final String AUTHORIZATION_PATH = "/oauth2/authorization/";
74  
75      /** The statuses the page tells; any other value of the parameter is ignored. */
76      private static final Set<String> STATUSES = Set.of(LINKED, ALREADY_LINKED, UNKNOWN_PROVIDER);
77  
78      /** Resolves the account of the request. */
79      private final CurrentAccountResolver currentAccountResolver;
80  
81      /** Lists the identities of an account. */
82      private final ListIdentitiesUseCase listIdentitiesUseCase;
83  
84      /** Maps the identities onto the view model. */
85      private final IdentityViewMapper identityViewMapper;
86  
87      /** Keeps the link markers. */
88      private final PendingIdentityLinks pendingIdentityLinks;
89  
90      /** Registrations of the configured providers; absent while none is configured. */
91      private final ObjectProvider<ClientRegistrationRepository> clientRegistrations;
92  
93      /** Renders the page. */
94      private final PageRenderer pageRenderer;
95  
96      /** The current request, injected as a thread-bound proxy, which holds the session. */
97      private final HttpServletRequest request;
98  
99      /**
100      * Renders the page.
101      *
102      * @param status outcome of the previous link, ignored when it is not one of the known ones.
103      * @return the page, never cached.
104      */
105     @Override
106     public ResponseEntity<String> identitiesPage(final String status) {
107         final var account = currentAccountResolver.resolve();
108         final var model = new HashMap<String, Object>();
109         model.put(IDENTITIES_ATTRIBUTE, identityViewMapper.toViews(listIdentitiesUseCase.execute(account.id())));
110         model.put(PROVIDERS_ATTRIBUTE, configuredProviders().stream().map(IdentityProvider::id).toList());
111         if (status != null && STATUSES.contains(status)) {
112             model.put(STATUS_ATTRIBUTE, status);
113         }
114         final var page = pageRenderer.render(VIEW_NAME, model);
115         return ResponseEntity.status(page.getStatusCode())
116                 .headers(page.getHeaders())
117                 .cacheControl(CacheControl.noStore())
118                 .body(page.getBody());
119     }
120 
121     /**
122      * Records the link marker and sends the browser to the provider's sign-in.
123      *
124      * @param provider registration id of the provider.
125      * @return a redirect to the provider's sign-in, or to the page with {@code unknown-provider}.
126      */
127     @Override
128     public ResponseEntity<Void> linkIdentityPage(final String provider) {
129         final var account = currentAccountResolver.resolve();
130         final var configured = IdentityProvider.fromId(provider).filter(configuredProviders()::contains);
131         if (configured.isEmpty()) {
132             return redirect(PATH_IDENTITIES_PAGE + "?" + STATUS_ATTRIBUTE + "=" + UNKNOWN_PROVIDER);
133         }
134         pendingIdentityLinks.begin(request.getSession(), configured.get(), account.id(), account.username());
135         return redirect(AUTHORIZATION_PATH + configured.get().id());
136     }
137 
138     /**
139      * Lists the providers that are configured, in the order of {@link IdentityProvider}.
140      *
141      * @return the configured providers, possibly none.
142      */
143     private List<IdentityProvider> configuredProviders() {
144         return Optional.ofNullable(clientRegistrations.getIfAvailable())
145                 .map(repository -> Arrays.stream(IdentityProvider.values())
146                         .filter(candidate -> repository.findByRegistrationId(candidate.id()) != null)
147                         .toList())
148                 .orElseGet(List::of);
149     }
150 
151     /**
152      * Answers a redirect that the browser follows with a GET.
153      *
154      * @param location the target, relative to the application.
155      * @return the redirect.
156      */
157     private ResponseEntity<Void> redirect(final String location) {
158         return ResponseEntity.status(HttpStatus.SEE_OTHER)
159                 .header(HttpHeaders.LOCATION, request.getContextPath() + location)
160                 .build();
161     }
162 }
