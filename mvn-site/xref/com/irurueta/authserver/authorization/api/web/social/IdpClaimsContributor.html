21  import java.util.List;
22  import java.util.Map;

25  import org.springframework.core.annotation.Order;
26  import org.springframework.security.core.GrantedAuthority;

28  import org.springframework.security.oauth2.core.oidc.IdTokenClaimNames;
29  import org.springframework.security.oauth2.server.authorization.token.OAuth2TokenContext;
30  import org.springframework.stereotype.Component;

32  // tag::social-claims[]
33  /**
34   * Adds to the tokens of a user who signed in with an external provider the claim {@code idp} (the registration id of
35   * the provider). The {@code sub} is the local account's username, because the principal's name is (see
36   * {@link FederatedUserMapper}), so a token of a federated sign-in is indistinguishable from any other in what it says
37   * about the user.
38   *
39   * <p>When the provider is trusted for its multi-factor authentication and proved it (the strong factor,
40   * {@link FederatedFactor#STRONG_AUTHORITY}), the upstream {@code amr} values are added to the {@code amr} claim the
41   * {@link com.irurueta.authserver.authorization.api.web.mfa.MfaClaimsContributor} wrote (each value once, so the
42   * repeated ones of the factor and of the provider are listed once, and {@code mfa} is there exactly once). The
43   * {@code acr} is not set here: the strong factor is {@link
44   * com.irurueta.authserver.authorization.domain.mfa.AuthenticationFactor#multiFactor() multi-factor alone}, so the
45   * multi-factor contributor already wrote the multi-factor class, from the same rule the step-up checks use. The
46   * upstream {@code amr} of a provider that is not trusted is never copied: it would claim a strength the server did not
47   * accept.
48   */
49  @Component
50  @Order(IdpClaimsContributor.ORDER)
51  @RequiredArgsConstructor
52  public class IdpClaimsContributor implements JwtClaimsContributor {
53  
54      /** Order of the contributor: after the multi-factor one, whose {@code amr} it extends. */
55      public static final int ORDER = 110;
56  
57      /** Name of the claim that says which provider authenticated the user. */
58      public static final String IDP_CLAIM = "idp";
59  
60      /** Reads the upstream {@code amr}. */
61      private final FederatedUserMapper userMapper;
62  
63      /**
64       * Adds the claims.
65       *
66       * @param context the token context.
67       * @param claims  the claims, which this method extends.
68       */
69      @Override
70      public void contribute(final OAuth2TokenContext context, final Map<String, Object> claims) {
71          if (!(context.getPrincipal() instanceof final OAuth2AuthenticationToken token)) {
72              return;
73          }
74          claims.put(IDP_CLAIM, token.getAuthorizedClientRegistrationId());
75          final var strong = token.getAuthorities().stream().map(GrantedAuthority::getAuthority)
76                  .anyMatch(FederatedFactor.STRONG_AUTHORITY::equals);
77          if (!strong) {
78              return;
79          }
80          // a mutable list on purpose: the authorization store only deserializes allow-listed types
81          final var amr = new ArrayList<String>(existingAmr(claims));
82          final var upstream = userMapper.upstreamAmr(Objects.requireNonNull(token.getPrincipal()));
83          upstream.stream().filter(value -> !amr.contains(value)).forEach(amr::add);
84          claims.put(IdTokenClaimNames.AMR, amr);
85      }
86  
87      /**
88       * Reads the {@code amr} already in the claims.
89       *
90       * @param claims the claims.
91       * @return its values; empty when there is none.
92       */
93      private static List<String> existingAmr(final Map<String, Object> claims) {
94          return claims.get(IdTokenClaimNames.AMR) instanceof final List<?> values
95                  ? values.stream().map(String::valueOf).distinct().toList() : List.of();
96      }
97  }
