16  package com.irurueta.authserver.authorization.api.web.social;
17  
18  import com.irurueta.authserver.authorization.api.web.login.LoginChainCustomizer;
19  import com.irurueta.authserver.authorization.api.web.mfa.MfaAuthorizationManager;
20  import com.irurueta.authserver.authorization.api.web.pages.v1.generated.api.IdentitiesPageApi;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
23  
24  /**
25   * Keeps the identities page and its link post behind the multi-factor policy, as the passkeys and phone pages are: a
26   * session that only presented the password of an account that still has to complete its second factor is sent to the
27   * second-factor page instead of being able to add a way to sign in to the account.
28   */
29  @RequiredArgsConstructor
30  public class IdentitiesPageChainCustomizer implements LoginChainCustomizer {
31  
32      /** Order of this customizer in the login chain, after the X.509 one ({@code 6}). */
33      public static final int ORDER = 7;
34  
35      /** Applies the multi-factor policy. */
36      private final MfaAuthorizationManager authorizationManager;
37  
38      /**
39       * Customizes the login/UI chain.
40       *
41       * @param http security builder of the chain.
42       */
43      @Override
44      public void customize(final HttpSecurity http) {
45          http.authorizeHttpRequests(authorize -> authorize
46                  .requestMatchers(IdentitiesPageApi.PATH_IDENTITIES_PAGE, IdentitiesPageApi.PATH_IDENTITIES_PAGE + "/**")
47                  .access(authorizationManager));
48      }
49  }
