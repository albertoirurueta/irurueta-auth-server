39  /**
40   * Self-service account endpoints: registration, e-mail verification, password change and password reset. The paths,
41   * status codes and bodies are those of the contract {@code apis/auth-server/rest-server/auth-server-api-v1.yaml},
42   * through the generated {@link AccountApi}. Failures are answered as RFC 9457 problem details by
43   * {@link AccountExceptionHandler}. No answer reveals whether an e-mail address belongs to an account.
44   */
45  @Slf4j
46  @RestController
47  @RequiredArgsConstructor
48  public class AccountController implements AccountApi {
49  
50      /** Registers accounts. */
51      private final RegisterAccountUseCase registerAccountUseCase;
52  
53      /** Verifies e-mail addresses. */
54      private final VerifyEmailUseCase verifyEmailUseCase;
55  
56      /** Changes the password of an authenticated account. */
57      private final ChangePasswordUseCase changePasswordUseCase;
58  
59      /** Sends password reset links. */
60      private final RequestPasswordResetUseCase requestPasswordResetUseCase;
61  
62      /** Resets forgotten passwords. */
63      private final ResetPasswordUseCase resetPasswordUseCase;
64  
65      /** Limits the unauthenticated requests per client and per e-mail address. */
66      private final AccountRateLimiter accountRateLimiter;
67  
68      /**
69       * Registers an account. The answer is the same whether or not the address or the username was already taken.
70       *
71       * @param request registration data.
72       * @return {@code 202 Accepted}.
73       */
74      @Override
75      public ResponseEntity<Void> register(final RegisterRequest request) {
76          accountRateLimiter.check(clientAddress(), request.email());
77          registerAccountUseCase.execute(request.username(), request.email(), request.password());
78          return ResponseEntity.accepted().build();
79      }
80  
81      /**
82       * Verifies an e-mail address.
83       *
84       * @param token verification token received by e-mail.
85       * @return {@code 204 No Content}.
86       */
87      @Override
88      public ResponseEntity<Void> verify(final String token) {
89          accountRateLimiter.check(clientAddress(), null);
90          verifyEmailUseCase.execute(token);
91          return ResponseEntity.noContent().build();
92      }
93  
94      /**
95       * Changes the password of the authenticated account, the principal of the bearer token.
96       *
97       * @param request current and new passwords.
98       * @return {@code 204 No Content}.
99       */
100     @Override
101     public ResponseEntity<Void> changePassword(final ChangePasswordRequest request) {
102         final var principal = currentRequest().getUserPrincipal();
103         changePasswordUseCase.execute(principal.getName(), request.currentPassword(), request.newPassword());
104         return ResponseEntity.noContent().build();
105     }
106 
107     /**
108      * Asks for a password reset link. The answer is the same whether or not the address belongs to an account, and
109      * even when the link cannot be delivered: only a registered address is ever mailed, so a delivery failure must
110      * not tell the caller that the address is registered (the use case has rolled back its token by then).
111      *
112      * @param request e-mail address of the account.
113      * @return {@code 202 Accepted}.
114      */
115     @Override
116     public ResponseEntity<Void> requestPasswordReset(final PasswordResetRequest request) {
117         accountRateLimiter.check(clientAddress(), request.email());
118         try {
119             requestPasswordResetUseCase.execute(request.email());
120         } catch (final AccountMailException e) {
121             log.warn("A password reset link could not be delivered: {}", e.getMessage());
122         }
123         return ResponseEntity.accepted().build();
124     }
125 
126     /**
127      * Resets a forgotten password.
128      *
129      * @param request reset token and new password.
130      * @return {@code 204 No Content}.
131      */
132     @Override
133     public ResponseEntity<Void> resetPassword(final ResetPasswordRequest request) {
134         accountRateLimiter.check(clientAddress(), null);
135         resetPasswordUseCase.execute(request.token(), request.newPassword());
136         return ResponseEntity.noContent().build();
137     }
138 
139     /**
140      * Returns the address of the client making the current request, the key of the rate limit. The generated
141      * interface carries only the contract's parameters, so the request is taken from the request context.
142      *
143      * @return the client address.
144      */
145     private static String clientAddress() {
146         return currentRequest().getRemoteAddr();
147     }
148 
149     /**
150      * Returns the request being handled.
151      *
152      * @return the current HTTP request.
153      */
154     private static HttpServletRequest currentRequest() {
155         return ((ServletRequestAttributes) RequestContextHolder.currentRequestAttributes()).getRequest();
156     }
157 }
