17  package com.irurueta.authserver.authorization.api.web.passkeys;
18  
19  import lombok.RequiredArgsConstructor;
20  import org.springframework.security.config.ObjectPostProcessor;
21  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
22  import org.springframework.security.web.authentication.session.SessionAuthenticationStrategy;
23  import org.springframework.security.web.webauthn.authentication.WebAuthnAuthenticationFilter;
24  
25  /**
26   * Hands the session authentication strategy of the chain to the WebAuthn filter. {@code WebAuthnConfigurer} leaves
27   * its filter with the do-nothing strategy, so a passkey sign-in would keep the session id of the visitor (session
28   * fixation) and the CSRF token it was rendered with. The chain's strategy changes the id and renews the token, as the
29   * one of the form login does. The strategy is a shared object that only exists once every configurer has been
30   * initialised, and the configurer post-processes its filter while it is configured, so it is read then.
31   */
32  @RequiredArgsConstructor
33  public class SessionStrategyPostProcessor implements ObjectPostProcessor<WebAuthnAuthenticationFilter> {
34  
35      /** Security builder of the chain, whose shared strategy is read. */
36      private final HttpSecurity http;
37  
38      /**
39       * Sets the shared strategy, when the chain has one, on the filter.
40       *
41       * @param filter the filter the configurer built.
42       * @param <O>    type of the filter.
43       * @return the same filter.
44       */
45      @Override
46      public <O extends WebAuthnAuthenticationFilter> O postProcess(final O filter) {
47          final var strategy = http.getSharedObject(SessionAuthenticationStrategy.class);
48          if (strategy != null) {
49              filter.setSessionAuthenticationStrategy(strategy);
50          }
51          return filter;
52      }
53  }
