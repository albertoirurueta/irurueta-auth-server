17  package com.irurueta.authserver.authorization.api.web.social;
18  
19  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
20  import java.util.Arrays;
21  import org.springframework.context.annotation.Condition;
22  import org.springframework.context.annotation.ConditionContext;
23  import org.springframework.core.env.Environment;
24  import org.springframework.core.type.AnnotatedTypeMetadata;
25  import org.springframework.util.StringUtils;
26  
27  /**
28   * Condition that holds when at least one social provider is configured, that is, when some
29   * {@code authserver.social.providers.<id>.client-id} has text. Every bean of the social login hangs from it, so with
30   * no provider configured (the default) the application has no social login at all: no client registration
31   * repository (which cannot be empty), no filter and no login page entry.
32   */
33  public class SocialProvidersCondition implements Condition {
34  
35      /**
36       * Tells whether a provider is configured.
37       *
38       * @param environment environment to read the setting from.
39       * @param provider    the provider.
40       * @return {@code true} when the provider's client id has text.
41       */
42      static boolean isConfigured(final Environment environment, final IdentityProvider provider) {
43          return StringUtils.hasText(
44                  environment.getProperty("authserver.social.providers." + provider.id() + ".client-id"));
45      }
46  
47      /**
48       * Tells whether any provider is configured.
49       *
50       * @param context  the condition context.
51       * @param metadata metadata of the annotated element.
52       * @return {@code true} when at least one provider is configured.
53       */
54      @Override
55      public boolean matches(final ConditionContext context, final AnnotatedTypeMetadata metadata) {
56          return Arrays.stream(IdentityProvider.values())
57                  .anyMatch(provider -> isConfigured(context.getEnvironment(), provider));
58      }
59  }
