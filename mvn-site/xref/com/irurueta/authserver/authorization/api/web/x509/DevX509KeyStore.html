16  package com.irurueta.authserver.authorization.api.web.x509;
17  
18  import java.io.IOException;
19  import java.io.UncheckedIOException;
20  import java.nio.charset.StandardCharsets;
21  import java.nio.file.FileAlreadyExistsException;
22  import java.nio.file.Files;
23  import java.nio.file.Path;
24  import java.nio.file.attribute.PosixFilePermissions;
25  import java.security.SecureRandom;
26  import java.time.Duration;
27  import java.util.HexFormat;
28  import java.util.List;
29  import java.util.concurrent.TimeUnit;
30  import lombok.extern.slf4j.Slf4j;
31  
32  /**
33   * Provides the server certificate of the dedicated TLS port when the {@code dev} profile is active and none is
34   * configured, so a developer does not have to create one: a self-signed certificate for {@code localhost} is generated
35   * once per machine with the JDK's {@code keytool}, stored in a key store under {@code ~/.irurueta-auth-server/} that
36   * only the owner can read (its password in a file next to it, as {@code DevKeyEncryptionSecret} keeps its secret) and
37   * reused at every start. Nothing is ever written inside the repository.
38   *
39   * <p>Why it lives in {@code src/main} although it only serves development: the end-to-end tests run the packaged jar
40   * with the {@code dev} profile, and a class under {@code src/test} never reaches it. It is inert everywhere else: the
41   * bean exists only under the {@code dev} profile, and the port itself is off unless
42   * {@code authserver.x509.port} is set. The clients' certificates are never validated by the connector (the
43   * fingerprint binding is what identifies them), so the key store holds the server's certificate only.
44   */
45  @Slf4j
46  public class DevX509KeyStore {
47  
48      /** Directory, under the user's home, that holds the per-machine files. */
49      static final String DIRECTORY = ".irurueta-auth-server";
50  
51      /** Name of the key store file. */
52      static final String KEY_STORE_FILE = "x509-tls.p12";
53  
54      /** Name of the file that holds the password of the key store. */
55      static final String PASSWORD_FILE = "x509-tls.password";
56  
57      /** Type of the generated key store. */
58      static final String KEY_STORE_TYPE = "PKCS12";
59  
60      /** POSIX permissions of the generated files. */
61      private static final String OWNER_ONLY = "rw-------";
62  
63      /** Number of random bytes of the generated password. */
64      private static final int PASSWORD_BYTES = 32;
65  
66      /** Validity of the generated certificate, in days. */
67      private static final String VALIDITY_DAYS = "825";
68  
69      /** Longest time the generation may take. */
70      private static final Duration GENERATION_TIMEOUT = Duration.ofMinutes(1);
71  
72      /** Source of the generated password. */
73      private static final SecureRandom RANDOM = new SecureRandom();
74  
75      /**
76       * Reads the per-machine key store, generating it first when it does not exist yet, in the home directory of the
77       * user.
78       *
79       * @return the key store.
80       * @throws UncheckedIOException  if the files cannot be read or created.
81       * @throws IllegalStateException if {@code keytool} fails.
82       */
83      public X509TlsKeyStore load() {
84          return load(Path.of(System.getProperty("user.home"), DIRECTORY));
85      }
86  
87      /**
88       * Reads the key store of a directory, generating it first when it does not exist yet.
89       *
90       * @param directory directory that holds the key store and its password.
91       * @return the key store.
92       * @throws UncheckedIOException  if the files cannot be read or created.
93       * @throws IllegalStateException if {@code keytool} fails.
94       */
95      X509TlsKeyStore load(final Path directory) {
96          final var keyStore = directory.resolve(KEY_STORE_FILE);
97          final var passwordFile = directory.resolve(PASSWORD_FILE);
98          try {
99              Files.createDirectories(directory);
100             if (!Files.exists(keyStore) || !Files.exists(passwordFile)) {
101                 create(keyStore, passwordFile);
102             }
103             return new X509TlsKeyStore(keyStore, Files.readString(passwordFile).strip(), KEY_STORE_TYPE);
104         } catch (final IOException e) {
105             throw new UncheckedIOException("Cannot read or create the development X.509 key store", e);
106         }
107     }
108 
109     /**
110      * Creates the password file and the key store, both readable by their owner only where the file system supports
111      * it.
112      *
113      * @param keyStore     the key store to create.
114      * @param passwordFile the file that holds its password.
115      * @throws IOException if a file cannot be written or {@code keytool} cannot run.
116      */
117     private void create(final Path keyStore, final Path passwordFile) throws IOException {
118         final var bytes = new byte[PASSWORD_BYTES];
119         RANDOM.nextBytes(bytes);
120         Files.deleteIfExists(keyStore);
121         Files.deleteIfExists(passwordFile);
122         try {
123             createOwnerOnly(passwordFile);
124             Files.writeString(passwordFile, HexFormat.of().formatHex(bytes) + "\n");
125             runKeytool(keyStore, passwordFile);
126             ownerOnly(keyStore);
127             log.warn("DEV ONLY: generated a self-signed TLS key store for the X.509 port in {}", keyStore);
128         } catch (final FileAlreadyExistsException e) {
129             log.debug("The development X.509 key store was created concurrently", e);
130         }
131     }
132 
133     /**
134      * Generates the key pair and its self-signed certificate.
135      *
136      * @param keyStore     the key store to create.
137      * @param passwordFile the file that holds its password.
138      * @throws IOException if {@code keytool} cannot run.
139      */
140     private void runKeytool(final Path keyStore, final Path passwordFile) throws IOException {
141         final var keytool = Path.of(System.getProperty("java.home"), "bin", "keytool").toString();
142         final var command = List.of(keytool, "-genkeypair", "-alias", "x509-tls", "-keyalg", "RSA", "-keysize", "3072",
143                 "-validity", VALIDITY_DAYS, "-dname", "CN=localhost", "-ext", "SAN=dns:localhost",
144                 "-storetype", KEY_STORE_TYPE, "-keystore", keyStore.toString(),
145                 "-storepass:file", passwordFile.toString());
146         final var process = new ProcessBuilder(command).redirectErrorStream(true).start();
147         try {
148             final var output = new String(process.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
149             if (!process.waitFor(GENERATION_TIMEOUT.toSeconds(), TimeUnit.SECONDS) || process.exitValue() != 0) {
150                 process.destroyForcibly();
151                 throw new IllegalStateException("keytool could not generate the X.509 key store: " + output);
152             }
153         } catch (final InterruptedException e) {
154             Thread.currentThread().interrupt();
155             process.destroyForcibly();
156             throw new IllegalStateException("Interrupted while generating the X.509 key store", e);
157         }
158     }
159 
160     /**
161      * Creates an empty file readable by its owner only, where the file system supports it.
162      *
163      * @param file the file to create.
164      * @throws IOException if the file cannot be created.
165      */
166     private static void createOwnerOnly(final Path file) throws IOException {
167         if (file.getFileSystem().supportedFileAttributeViews().contains("posix")) {
168             Files.createFile(file, PosixFilePermissions.asFileAttribute(PosixFilePermissions.fromString(OWNER_ONLY)));
169         } else {
170             Files.createFile(file);
171         }
172     }
173 
174     /**
175      * Restricts a file to its owner, where the file system supports it.
176      *
177      * @param file the file.
178      * @throws IOException if the permissions cannot be changed.
179      */
180     private static void ownerOnly(final Path file) throws IOException {
181         if (file.getFileSystem().supportedFileAttributeViews().contains("posix")) {
182             Files.setPosixFilePermissions(file, PosixFilePermissions.fromString(OWNER_ONLY));
183         }
184     }
185 }
