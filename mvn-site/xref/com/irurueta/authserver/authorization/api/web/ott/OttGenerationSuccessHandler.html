23  import jakarta.servlet.http.HttpServletRequest;
24  import jakarta.servlet.http.HttpServletResponse;
25  import java.io.IOException;
26  import java.util.UUID;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.security.authentication.ott.OneTimeToken;
29  import org.springframework.security.authentication.ott.OneTimeTokenAuthenticationToken;
30  import org.springframework.security.authentication.ott.OneTimeTokenService;
31  import org.springframework.security.core.Authentication;
32  import org.springframework.security.core.context.SecurityContextHolder;
33  import org.springframework.security.web.DefaultRedirectStrategy;
34  import org.springframework.security.web.RedirectStrategy;
35  import org.springframework.security.web.authentication.ott.OneTimeTokenGenerationSuccessHandler;

37  /**
38   * Delivers the token Spring's filter just generated, and answers every request the same way. The token the stock
39   * service stored is the one of a magic link; for a numeric code it is discarded, because a code cannot use that store
40   * (it is keyed by the token value alone, and a six-digit value collides across users), and
41   * {@link RequestOneTimeCodeUseCase} stores the code in its own table, bound to a new binding id kept in the session.
42   * Nothing is sent, and nothing differs for the caller, when the identifier is unknown, disabled or has no verified
43   * e-mail address or phone: the browser is always redirected to the page that says to check the inbox, so the response
44   * does not tell whether the account exists. The use cases only request the delivery, which happens after their
45   * transaction commits and off the request thread, so neither a slow mail server or SMS gateway nor a delivery failure
46   * shows in the response.
47   *
48   * <p>A token asked for as a second factor is delivered only through a channel {@link OttSecondFactorChannels} allows
49   * for the signed-in user (never the channel that signed the user in, and e-mail only to a user without another
50   * confirmed second factor), and a second-factor link only for a user name that fits next to the marker of
51   * {@link LinkTokenUsername}. Any other request is answered the same, but its token is discarded and nothing is sent.
52   */
53  @RequiredArgsConstructor
54  public class OttGenerationSuccessHandler implements OneTimeTokenGenerationSuccessHandler {
55  
56      /** Starts the numeric code of a channel. */
57      private final RequestOneTimeCodeUseCase requestOneTimeCodeUseCase;
58  
59      /** Requests the delivery of a magic link. */
60      private final RequestMagicLinkUseCase requestMagicLinkUseCase;
61  
62      /** Store of the link tokens, which the code requests discard their unused token from. */
63      private final OneTimeTokenService oneTimeTokenService;
64  
65      /** Builds the magic link. */
66      private final OttLinkFactory linkFactory;
67  
68      /** Decides which channels may be the second factor of the signed-in user. */
69      private final OttSecondFactorChannels secondFactorChannels;
70  
71      /** Redirects to the page that says to check the inbox. */
72      private final RedirectStrategy redirectStrategy = new DefaultRedirectStrategy();
73  
74      /**
75       * Delivers the token and redirects to the page that says to check the inbox.
76       *
77       * @param request  the request.
78       * @param response the response.
79       * @param token    the token Spring's filter generated.
80       * @throws IOException if the redirect cannot be sent.
81       */
82      @Override
83      public void handle(final HttpServletRequest request, final HttpServletResponse response, final OneTimeToken token)
84              throws IOException {
85          final var current = SecurityContextHolder.getContext().getAuthentication();
86          final var parameters = OttGenerationParameters.from(request, current);
87          final var identifier = LinkTokenUsername.identifierOf(token.getUsername());
88          final var bindingId = UUID.randomUUID().toString();
89          final var deliverable = isDeliverable(parameters, current);
90          if (parameters.kind() == OttKind.LINK && deliverable) {
91              requestMagicLinkUseCase.execute(identifier, linkFactory.confirmationLink(token.getTokenValue()));
92          } else {
93              // a code, or a link that is not sent, never uses the token of the stock service
94              oneTimeTokenService.consume(new OneTimeTokenAuthenticationToken(token.getTokenValue()));
95              if (deliverable) {
96                  requestOneTimeCodeUseCase.execute(identifier, parameters.channel(), parameters.purpose(), bindingId);
97              }
98          }
99          new PendingOtt(identifier, bindingId, parameters.kind(), parameters.purpose(), parameters.channel())
100                 .saveTo(request.getSession());
101         redirectStrategy.sendRedirect(request, response, OneTimeTokenPageApi.PATH_OTT_CODE_PAGE);
102     }
103 
104     /**
105      * Tells whether a token may be delivered: always for a sign-in; for a second factor only through a channel the
106      * signed-in user may use, and a link only when the user name fits the token store with its marker.
107      *
108      * @param parameters what the request asks for.
109      * @param current    the authentication of the request.
110      * @return {@code true} when the token may be delivered.
111      */
112     private boolean isDeliverable(final OttGenerationParameters parameters, final Authentication current) {
113         if (parameters.purpose() != OttPurpose.SECOND_FACTOR) {
114             return true;
115         }
116         if (parameters.kind() == OttKind.LINK && !LinkTokenUsername.fitsSecondFactorLink(current.getName())) {
117             return false;
118         }
119         return secondFactorChannels.isAllowed(current, parameters.channel());
120     }
121 }
