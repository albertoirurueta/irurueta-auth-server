17  package com.irurueta.authserver.authorization.api.web.mfa;
18  
19  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
20  import com.irurueta.authserver.authorization.api.web.pages.v1.generated.api.SecondFactorPageApi;
21  import jakarta.servlet.http.HttpServletRequest;
22  import java.util.Comparator;
23  import java.util.HashMap;
24  import java.util.List;
25  import org.springframework.http.ResponseEntity;
26  import org.springframework.security.authentication.LockedException;
27  import org.springframework.security.core.context.SecurityContextHolder;
28  import org.springframework.security.web.WebAttributes;
29  import org.springframework.stereotype.Controller;
30  
31  /**
32   * Serves the second-factor page, as the contract {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml}
33   * describes it through the generated {@link SecondFactorPageApi}, supplying the ordered fragments of the second
34   * factors, the model attributes each of them contributes for the signed-in user and the outcome of the last attempt to
35   * the template.
36   */
37  @Controller
38  public class SecondFactorController implements SecondFactorPageApi {
39  
40      /** Path of the second-factor page, taken from the contract. */
41      public static final String SECOND_FACTOR_PATH = PATH_SECOND_FACTOR_PAGE;
42  
43      /** Key of the message shown when a code is wrong. */
44      public static final String GENERIC_ERROR = "secondFactor.error.generic";
45  
46      /** Key of the message shown when the second factor is locked. */
47      public static final String LOCKED_ERROR = "secondFactor.error.locked";
48  
49      /** Name of the second-factor template. */
50      static final String VIEW_NAME = "second-factor";
51  
52      /** Name of the model attribute that holds the fragments of the second factors. */
53      static final String FRAGMENTS_ATTRIBUTE = "fragments";
54  
55      /** Name of the model attribute that holds the key of the message of a failed attempt. */
56      static final String ERROR_MESSAGE_ATTRIBUTE = "errorMessageKey";
57  
58      /** Fragments contributed by the second factors, sorted by order. */
59      private final List<SecondFactorFragment> fragments;
60  
61      /** Renders the page. */
62      private final PageRenderer pageRenderer;
63  
64      /** The current request, injected as a thread-bound proxy, whose session holds the last failure. */
65      private final HttpServletRequest request;
66  
67      /**
68       * Creates the controller.
69       *
70       * @param factors      fragments contributed by the second factors.
71       * @param pageRenderer renderer of the page.
72       * @param request      the current request.
73       */
74      public SecondFactorController(final List<SecondFactorFragment> factors, final PageRenderer pageRenderer,
75              final HttpServletRequest request) {
76          this.fragments = factors.stream().sorted(Comparator.comparingInt(SecondFactorFragment::order)).toList();
77          this.pageRenderer = pageRenderer;
78          this.request = request;
79      }
80  
81      // tag::mfa-second-factor-page[]
82      /**
83       * Renders the second-factor page.
84       *
85       * @param error present (even if empty) when the previous attempt failed.
86       * @return the second-factor page.
87       */
88      @Override
89      public ResponseEntity<String> secondFactorPage(final String error) {
90          final var model = new HashMap<String, Object>();
91          final var authentication = SecurityContextHolder.getContext().getAuthentication();
92          fragments.forEach(fragment -> model.putAll(fragment.modelAttributes(authentication)));
93          model.put(FRAGMENTS_ATTRIBUTE, fragments);
94          if (error != null) {
95              model.put(ERROR_MESSAGE_ATTRIBUTE, lastFailureWasLockout() ? LOCKED_ERROR : GENERIC_ERROR);
96          }
97          return pageRenderer.render(VIEW_NAME, model);
98      }
99      // end::mfa-second-factor-page[]
100 
101     /**
102      * Tells whether the previous attempt failed because the second factor is locked, which Spring Security keeps in
103      * the session.
104      *
105      * @return {@code true} when the session holds a {@link LockedException} as the last failure.
106      */
107     private boolean lastFailureWasLockout() {
108         final var session = request.getSession(false);
109         final var failure = session == null ? null : session.getAttribute(WebAttributes.AUTHENTICATION_EXCEPTION);
110         return failure instanceof LockedException;
111     }
112 }
