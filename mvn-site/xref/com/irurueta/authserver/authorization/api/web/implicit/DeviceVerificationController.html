16  package com.irurueta.authserver.authorization.api.web.implicit;
17  
18  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
19  import com.irurueta.authserver.authorization.api.web.pages.v1.generated.api.DeviceVerificationPageApi;
20  import java.util.HashMap;
21  import java.util.regex.Pattern;
22  import lombok.RequiredArgsConstructor;
23  import org.springframework.http.ResponseEntity;
24  import org.springframework.stereotype.Controller;
25  
26  /**
27   * Serves the device verification page, as the contract {@code apis/auth-server/web-pages/auth-server-pages-v1.yaml}
28   * describes it through the generated {@link DeviceVerificationPageApi}: the form where the user types the code the
29   * device shows. The form posts to the framework's own endpoint ({@code POST /oauth2/device_verification}); a request
30   * that already carries a code never reaches this controller, because that endpoint answers it. The page is also where
31   * the endpoint sends the browser back to, with {@code ?error} when the attempt failed and {@code ?success} when the
32   * device was approved.
33   */
34  @Controller
35  @RequiredArgsConstructor
36  public class DeviceVerificationController implements DeviceVerificationPageApi {
37  
38      /** Path of the page, and of the verification endpoint, taken from the contract. */
39      public static final String VERIFICATION_PATH = PATH_DEVICE_VERIFICATION_PAGE;
40  
41      /** Name of the parameter that tells the page that the device was just approved. */
42      public static final String SUCCESS_PARAMETER = "success";
43  
44      /** Name of the template of the page. */
45      static final String VIEW_NAME = "device-verification";
46  
47      /** Name of the model attribute that holds the code to prefill. */
48      static final String USER_CODE_ATTRIBUTE = "userCode";
49  
50      /** Name of the model attribute that tells whether the previous attempt failed. */
51      static final String ERROR_ATTRIBUTE = "failed";
52  
53      /** Name of the model attribute that tells whether the device was just approved. */
54      static final String SUCCESS_ATTRIBUTE = "approved";
55  
56      /** Shape of a user code worth prefilling: letters, digits and dashes. */
57      private static final Pattern PLAUSIBLE_CODE = Pattern.compile("[A-Za-z0-9-]{1,32}");
58  
59      /** Renders the page. */
60      private final PageRenderer pageRenderer;
61  
62      // tag::device-verification-controller[]
63      /**
64       * Renders the device verification page.
65       *
66       * @param userCode a code to prefill, ignored unless it looks like a user code.
67       * @param error    present when the previous attempt failed.
68       * @param success  present when the device was just approved.
69       * @return the page.
70       */
71      @Override
72      public ResponseEntity<String> deviceVerificationPage(final String userCode, final String error,
73              final String success) {
74          final var model = new HashMap<String, Object>();
75          model.put(ERROR_ATTRIBUTE, error != null);
76          model.put(SUCCESS_ATTRIBUTE, success != null);
77          if (userCode != null && PLAUSIBLE_CODE.matcher(userCode).matches()) {
78              model.put(USER_CODE_ATTRIBUTE, userCode);
79          }
80          return pageRenderer.render(VIEW_NAME, model);
81      }
82      // end::device-verification-controller[]
83  }
