16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.api.web.mfa.SecondFactorUserResolver;
19  import com.irurueta.authserver.authorization.application.mfa.HasConfirmedSecondFactorUseCase;
20  import com.irurueta.authserver.authorization.application.sms.HasVerifiedPhoneUseCase;
21  import com.irurueta.authserver.authorization.domain.ott.OttChannel;
22  import java.util.EnumSet;
23  import java.util.Set;
24  import lombok.RequiredArgsConstructor;
25  import org.springframework.security.core.Authentication;
26  import org.springframework.stereotype.Component;
27  
28  // tag::ott-second-factor-channels[]
29  /**
30   * Decides through which channels a one-time code or link may be the second factor of a signed-in user. The same rule
31   * decides what the second-factor page offers, whether a requested token is delivered at all and whether a presented
32   * one is accepted, so a channel that is not offered can neither be asked for nor used by posting the forms by hand:
33   * <ul>
34   * <li>never the channel that signed the user in (the same mailbox or phone would be proved twice);</li>
35   * <li>SMS only to a user with a verified phone;</li>
36   * <li>e-mail only to a user without any other confirmed second factor (authenticator app, passkey, verified phone).
37   * The password is reset by e-mail too, so a mailbox is not independent from the password: whoever controls it could
38   * otherwise skip the stronger factor the user enrolled. For a user whose multi-factor authentication was forced by an
39   * administrator before any factor was enrolled, the e-mail is then the only second factor available.</li>
40   * </ul>
41   */
42  @Component
43  @RequiredArgsConstructor
44  public class OttSecondFactorChannels {
45  
46      /** Rules about the session a second factor is presented in. */
47      private final PrimaryFactorSession primaryFactorSession;
48  
49      /** Resolves the identifier of the account of the signed-in user. */
50      private final SecondFactorUserResolver userResolver;
51  
52      /** Tells whether the user has a confirmed second factor of any kind. */
53      private final HasConfirmedSecondFactorUseCase hasConfirmedSecondFactorUseCase;
54  
55      /** Tells whether the user has a verified phone. */
56      private final HasVerifiedPhoneUseCase hasVerifiedPhoneUseCase;
57  
58      /**
59       * Gets the channels a signed-in user may use as the second factor.
60       *
61       * @param authentication the authentication of the session, which holds a primary factor.
62       * @return the channels; empty when none may be used.
63       */
64      public Set<OttChannel> allowed(final Authentication authentication) {
65          final var userId = userResolver.userIdOf(authentication.getName());
66          final var channels = EnumSet.noneOf(OttChannel.class);
67          if (!primaryFactorSession.signedInThrough(authentication, OttChannel.EMAIL)
68                  && !hasConfirmedSecondFactorUseCase.execute(userId)) {
69              channels.add(OttChannel.EMAIL);
70          }
71          if (!primaryFactorSession.signedInThrough(authentication, OttChannel.SMS)
72                  && hasVerifiedPhoneUseCase.execute(userId)) {
73              channels.add(OttChannel.SMS);
74          }
75          return channels;
76      }
77  
78      /**
79       * Tells whether a signed-in user may use a channel as the second factor.
80       *
81       * @param authentication the authentication of the session, which holds a primary factor.
82       * @param channel        the channel.
83       * @return {@code true} when the channel is one of {@link #allowed(Authentication)}.
84       */
85      public boolean isAllowed(final Authentication authentication, final OttChannel channel) {
86          return allowed(authentication).contains(channel);
87      }
88  }
