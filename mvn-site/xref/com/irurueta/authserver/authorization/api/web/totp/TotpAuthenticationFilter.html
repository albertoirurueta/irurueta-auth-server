45  /**
46   * Processes the post of the second-factor forms ({@code POST /login/second-factor/totp}): the TOTP form, the
47   * recovery-code form and the enrolment form (field {@code enroll}) of a user who has no authenticator app yet. The
48   * code is verified by {@link TotpAuthenticationProvider}; on success the session id is changed (and the CSRF token
49   * renewed, with the session authentication strategy of the chain, as the form login does), the enriched
50   * authentication is saved to the session, a device is remembered when the user asked for it and the browser resumes
51   * the request that was interrupted, or, after an enrolment, first sees its recovery codes once. On failure it is sent
52   * back to the second-factor page with {@code ?error}. The endpoint is owned by this filter, like {@code POST /login},
53   * so it is not part of the pages contract.
54   */
55  public class TotpAuthenticationFilter extends AbstractAuthenticationProcessingFilter {
56  
57      /** Path the second-factor forms post to. */
58      public static final String TOTP_PATH = SECOND_FACTOR_PATH + "/totp";
59  
60      /** Name of the form field that holds the code. */
61      public static final String CODE_PARAMETER = "code";
62  
63      /** Name of the form field present when the code is a recovery code. */
64      public static final String RECOVERY_PARAMETER = "recovery";
65  
66      /** Name of the form field present when the code confirms the enrolment of an authenticator app. */
67      public static final String ENROLL_PARAMETER = "enroll";
68  
69      /** Name of the form field present when the user asks to remember the device. */
70      public static final String REMEMBER_PARAMETER = "rememberDevice";
71  
72      /** Longest user agent description kept, the size of the column that stores it. */
73      static final int MAX_USER_AGENT_LENGTH = 255;
74  
75      /** Use case that remembers the device. */
76      private final CreateTrustedDeviceUseCase createTrustedDeviceUseCase;
77  
78      /** Resolves the identifier of the account of the signed-in user. */
79      private final SecondFactorUserResolver userResolver;
80  
81      /** Writes the cookie of a remembered device. */
82      private final TrustedDeviceCookieWriter cookieWriter;
83  
84      /** Keeps the recovery codes of an enrolment until they are shown. */
85      private final PendingRecoveryCodes pendingRecoveryCodes;
86  
87      /**
88       * Creates the filter. Until {@link TotpChainCustomizer} hands it the session authentication strategy of the
89       * chain, it changes the session id on success with a {@link ChangeSessionIdAuthenticationStrategy}.
90       *
91       * @param authenticationManager      manager that holds the {@link TotpAuthenticationProvider}.
92       * @param createTrustedDeviceUseCase use case that remembers the device.
93       * @param userResolver               resolves the identifier of the account of the signed-in user.
94       * @param cookieWriter               writes the cookie of a remembered device.
95       * @param pendingRecoveryCodes       keeps the recovery codes of an enrolment until they are shown.
96       */
97      public TotpAuthenticationFilter(final AuthenticationManager authenticationManager,
98              final CreateTrustedDeviceUseCase createTrustedDeviceUseCase, final SecondFactorUserResolver userResolver,
99              final TrustedDeviceCookieWriter cookieWriter, final PendingRecoveryCodes pendingRecoveryCodes) {
100         super(PathPatternRequestMatcher.withDefaults().matcher(HttpMethod.POST, TOTP_PATH), authenticationManager);
101         this.createTrustedDeviceUseCase = createTrustedDeviceUseCase;
102         this.userResolver = userResolver;
103         this.cookieWriter = cookieWriter;
104         this.pendingRecoveryCodes = pendingRecoveryCodes;
105         setSecurityContextRepository(new HttpSessionSecurityContextRepository());
106         setSessionAuthenticationStrategy(new ChangeSessionIdAuthenticationStrategy());
107         setAuthenticationSuccessHandler(new RecoveryCodesSuccessHandler(pendingRecoveryCodes,
108                 new SavedRequestAwareAuthenticationSuccessHandler()));
109         setAuthenticationFailureHandler(new SimpleUrlAuthenticationFailureHandler(SECOND_FACTOR_PATH + "?error"));
110     }
111 
112     // tag::totp-filter[]
113     /**
114      * Builds the verification request from the form and the authentication the user already holds, and keeps in the
115      * session the recovery codes an enrolment generates, until they are shown.
116      *
117      * @param request  the request.
118      * @param response the response.
119      * @return the authentication with the TOTP factor added.
120      * @throws AuthenticationException if nobody is signed in or the code is not accepted.
121      */
122     @Override
123     public Authentication attemptAuthentication(final HttpServletRequest request, final HttpServletResponse response)
124             throws AuthenticationException {
125         final var current = SecurityContextHolder.getContext().getAuthentication();
126         if (current == null || !current.isAuthenticated() || current instanceof AnonymousAuthenticationToken) {
127             throw new InsufficientAuthenticationException("The user has not signed in yet");
128         }
129         final var code = StringUtils.trimWhitespace(request.getParameter(CODE_PARAMETER));
130         final var verification = new TotpAuthenticationToken(current, code == null ? "" : code, kindOf(request));
131         final var verified = getAuthenticationManager().authenticate(verification);
132         if (!verification.issuedRecoveryCodes().isEmpty()) {
133             pendingRecoveryCodes.save(request, verification.issuedRecoveryCodes());
134         }
135         return verified;
136     }
137 
138     /**
139      * Remembers the device when the user asked for it, then saves the authentication and resumes the request.
140      *
141      * @param request    the request.
142      * @param response   the response.
143      * @param chain      the rest of the chain.
144      * @param authResult the verified authentication.
145      * @throws IOException      if the redirect cannot be sent.
146      * @throws ServletException if the success handler fails.
147      */
148     @Override
149     protected void successfulAuthentication(final HttpServletRequest request, final HttpServletResponse response,
150             final FilterChain chain, final Authentication authResult) throws IOException, ServletException {
151         if (request.getParameter(REMEMBER_PARAMETER) != null) {
152             final var created = createTrustedDeviceUseCase.execute(userResolver.userIdOf(authResult.getName()),
153                     authResult.getName(), userAgentSummary(request));
154             cookieWriter.write(response, created);
155         }
156         super.successfulAuthentication(request, response, chain, authResult);
157     }
158     // end::totp-filter[]
159 
160     /**
161      * Tells what kind of code a post carries, from its form fields.
162      *
163      * @param request the request.
164      * @return {@link TotpCodeKind#ENROLMENT} with the {@value #ENROLL_PARAMETER} field,
165      *         {@link TotpCodeKind#RECOVERY} with the {@value #RECOVERY_PARAMETER} field, {@link TotpCodeKind#TOTP}
166      *         otherwise.
167      */
168     private static TotpCodeKind kindOf(final HttpServletRequest request) {
169         if (request.getParameter(ENROLL_PARAMETER) != null) {
170             return TotpCodeKind.ENROLMENT;
171         }
172         return request.getParameter(RECOVERY_PARAMETER) != null ? TotpCodeKind.RECOVERY : TotpCodeKind.TOTP;
173     }
174 
175     /**
176      * Describes the browser of a request.
177      *
178      * @param request the request.
179      * @return the {@code User-Agent} header, cut to the size that is stored; {@code null} when there is none.
180      */
181     private static String userAgentSummary(final HttpServletRequest request) {
182         final var userAgent = request.getHeader(HttpHeaders.USER_AGENT);
183         if (!StringUtils.hasText(userAgent)) {
184             return null;
185         }
186         return userAgent.length() > MAX_USER_AGENT_LENGTH ? userAgent.substring(0, MAX_USER_AGENT_LENGTH) : userAgent;
187     }
