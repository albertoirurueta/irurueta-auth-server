16  package com.irurueta.authserver.authorization.api.web.x509;
17  
18  import java.util.List;
19  import org.springframework.boot.context.properties.ConfigurationProperties;
20  import org.springframework.boot.context.properties.bind.DefaultValue;
21  import org.springframework.security.web.util.matcher.IpAddressMatcher;
22  import org.springframework.util.StringUtils;
23  
24  // tag::x509-properties[]
25  /**
26   * Configuration of the sign-in with a client certificate ({@code authserver.x509.*}). It is off by default.
27   *
28   * @param enabled           whether the sign-in with a client certificate exists at all (default {@code false}).
29   * @param forwardedHeader   name of the request header in which a trusted reverse proxy forwards the client certificate
30   *                          it verified (PEM, URL-encoded as {@code $ssl_client_escaped_cert} of nginx does, or the
31   *                          base64 of the DER form); empty (the default) turns the forwarded transport off.
32   * @param trustedProxies    addresses or CIDR ranges of the proxies that may forward a certificate; required when the
33   *                          header is set, and a request from any other address never has its header read.
34   * @param port              port of the dedicated TLS connector that requests a client certificate; {@code 0} (the
35   *                          default) turns the direct transport off.
36   * @param amr               RFC 8176 {@code amr} value the factor reports: {@code sc} (smart card, the default) or
37   *                          {@code hwk} (a hardware-bound key).
38   * @param keyStore          path of the key store with the server certificate of the dedicated port; with the
39   *                          {@code dev} profile and none configured, a self-signed one is generated at the first start
40   *                          under {@code ~/.irurueta-auth-server/}.
41   * @param keyStorePassword  password of the key store (and of its key), from the environment, never from a file.
42   * @param keyStoreType      type of the key store (default {@code PKCS12}).
43   */
44  @ConfigurationProperties(prefix = "authserver.x509")
45  public record X509Properties(
46          @DefaultValue("false") boolean enabled,
47          @DefaultValue("") String forwardedHeader,
48          @DefaultValue List<String> trustedProxies,
49          @DefaultValue("0") int port,
50          @DefaultValue("sc") String amr,
51          @DefaultValue("") String keyStore,
52          @DefaultValue("") String keyStorePassword,
53          @DefaultValue("PKCS12") String keyStoreType) {
54  
55      /** {@code amr} value of a smart card. */
56      public static final String AMR_SMART_CARD = "sc";
57  
58      /** {@code amr} value of a key held in hardware. */
59      public static final String AMR_HARDWARE_KEY = "hwk";
60  
61      /** Highest TCP port. */
62      private static final int MAX_PORT = 65_535;
63  
64      /**
65       * Validates the configuration and copies the list.
66       *
67       * @throws IllegalArgumentException if the port is out of range, the {@code amr} value is not {@code sc} or
68       *                                  {@code hwk}, a trusted proxy is not an address or a range, or a forwarded
69       *                                  header is set without any trusted proxy.
70       */
71      public X509Properties {
72          trustedProxies = List.copyOf(trustedProxies);
73          if (port < 0 || port > MAX_PORT) {
74              throw new IllegalArgumentException("port must be between 0 and " + MAX_PORT);
75          }
76          if (!AMR_SMART_CARD.equals(amr) && !AMR_HARDWARE_KEY.equals(amr)) {
77              throw new IllegalArgumentException("amr must be " + AMR_SMART_CARD + " or " + AMR_HARDWARE_KEY);
78          }
79          trustedProxies.forEach(IpAddressMatcher::new);
80          if (StringUtils.hasText(forwardedHeader) && trustedProxies.isEmpty()) {
81              throw new IllegalArgumentException("a forwarded header needs at least one trusted proxy");
82          }
83      }
84  
85      /**
86       * Tells whether the certificate may arrive in a header.
87       *
88       * @return {@code true} when a header name is configured.
89       */
90      public boolean forwardedEnabled() {
91          return StringUtils.hasText(forwardedHeader);
92      }
93  
94      /**
95       * Tells whether the dedicated TLS connector is requested.
96       *
97       * @return {@code true} when a port is configured.
98       */
99      public boolean tlsEnabled() {
100         return port > 0;
101     }
102 }
