16  package com.irurueta.authserver.authorization.api.web.mfa;
17  
18  import com.irurueta.authserver.authorization.domain.mfa.AuthenticationFactor;
19  import com.irurueta.authserver.authorization.domain.mfa.MfaPolicy;
20  import java.time.Instant;
21  import java.util.Comparator;
22  import java.util.List;
23  import java.util.Objects;
24  import java.util.Optional;
25  import java.util.function.Predicate;
26  import java.util.stream.Collectors;
27  import lombok.RequiredArgsConstructor;
28  import org.springframework.security.core.Authentication;
29  import org.springframework.security.core.authority.FactorGrantedAuthority;
30  import org.springframework.stereotype.Component;
31  
32  // tag::mfa-factor-timeline[]
33  /**
34   * Reads, from the {@link FactorGrantedAuthority} instances of an authentication, the factors that prove something
35   * about it and when they were presented. It is the single place the claims ({@link MfaClaimsContributor}) and the
36   * step-up checks ({@link StepUpAuthorizationRequestConverter}) take {@code amr}, {@code acr} and {@code auth_time}
37   * from, so both always agree:
38   *
39   * <ul>
40   *   <li>only the factors of the {@link AuthenticationFactorRegistry} that {@link MfaPolicy#isReported} accepts count:
41   *       a remembered (trusted) device does not;</li>
42   *   <li>the authentication is multi-factor when {@link MfaPolicy#isMultiFactor} says so: at least two distinct
43   *       factors count, or one counted factor proves a multi-factor authentication alone;</li>
44   *   <li>{@code auth_time} is when the primary factor was presented (the earliest one when there are several), so a
45   *       second factor added later never makes the sign-in look newer; without a primary factor it is the earliest
46   *       counted factor.</li>
47   * </ul>
48   */
49  @Component
50  @RequiredArgsConstructor
51  public class FactorTimeline {
52  
53      /** Registry of the known factors. */
54      private final AuthenticationFactorRegistry factorRegistry;
55  
56      /** Policy that tells which factors are reported and which are primary. */
57      private final MfaPolicy mfaPolicy;
58  
59      /**
60       * Lists the factor authorities of an authentication that count, oldest first.
61       *
62       * @param authentication the authentication.
63       * @return the counted factor authorities, sorted by issue instant; empty when none.
64       */
65      public List<FactorGrantedAuthority> countedFactors(final Authentication authentication) {
66          return factorsMatching(authentication, mfaPolicy::isReported);
67      }
68  
69      /**
70       * Lists the {@code amr} values of the counted factors, in the order they were presented, without duplicates.
71       *
72       * @param authentication the authentication.
73       * @return the values; empty when no factor counts.
74       */
75      public List<String> amrValues(final Authentication authentication) {
76          return countedFactors(authentication).stream()
77                  .map(authority -> factorRegistry.findByAuthority(authority.getAuthority()))
78                  .flatMap(Optional::stream)
79                  .map(AuthenticationFactor::amrValue)
80                  .distinct()
81                  .toList();
82      }
83  
84      /**
85       * Tells whether an authentication is multi-factor, by {@link MfaPolicy#isMultiFactor} over its counted factors.
86       *
87       * @param authentication the authentication.
88       * @return {@code true} for a multi-factor authentication.
89       */
90      public boolean isMultiFactor(final Authentication authentication) {
91          return mfaPolicy.isMultiFactor(countedFactors(authentication).stream()
92                  .map(authority -> factorRegistry.findByAuthority(authority.getAuthority()))
93                  .flatMap(Optional::stream)
94                  .collect(Collectors.toSet()));
95      }
96  
97      /**
98       * Finds the instant of the authentication, the one {@code auth_time} reports and {@code max_age} is measured
99       * from.
100      *
101      * @param authentication the authentication.
102      * @return the issue instant of the earliest primary factor, or else of the earliest counted factor; empty when no
103      *         counted factor carries an instant.
104      */
105     public Optional<Instant> authTime(final Authentication authentication) {
106         return earliest(factorsMatching(authentication,
107                 factor -> mfaPolicy.isReported(factor) && mfaPolicy.isPrimary(factor)))
108                 .or(() -> earliest(countedFactors(authentication)));
109     }
110 
111     /**
112      * Lists the factor authorities of an authentication whose registered factor passes a test, oldest first.
113      *
114      * @param authentication the authentication.
115      * @param test           the test of the registered factor.
116      * @return the matching authorities.
117      */
118     private List<FactorGrantedAuthority> factorsMatching(final Authentication authentication,
119             final Predicate<AuthenticationFactor> test) {
120         return authentication.getAuthorities().stream()
121                 .filter(FactorGrantedAuthority.class::isInstance)
122                 .map(FactorGrantedAuthority.class::cast)
123                 .filter(authority -> factorRegistry.findByAuthority(authority.getAuthority()).filter(test).isPresent())
124                 .sorted(Comparator.comparing(FactorGrantedAuthority::getIssuedAt,
125                         Comparator.nullsLast(Comparator.naturalOrder())))
126                 .toList();
127     }
128 
129     /**
130      * Finds the earliest issue instant of some factor authorities.
131      *
132      * @param factors the authorities.
133      * @return the earliest instant, or empty when none carries one.
134      */
135     private static Optional<Instant> earliest(final List<FactorGrantedAuthority> factors) {
136         return factors.stream().map(FactorGrantedAuthority::getIssuedAt).filter(Objects::nonNull)
137                 .min(Comparator.naturalOrder());
138     }
139 }
