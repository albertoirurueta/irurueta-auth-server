17  package com.irurueta.authserver.authorization.api.web.social;
18  
19  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
20  import java.util.Objects;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.security.oauth2.client.authentication.OAuth2AuthenticationToken;
23  
24  // tag::social-factor-selector[]
25  /**
26   * Decides which factor a federated sign-in proved: the strong one when the provider is trusted for its multi-factor
27   * authentication ({@code authserver.social.providers.<id>.trust-mfa}) and its upstream {@code amr} contains
28   * {@code mfa}, the ordinary one otherwise. An upstream claim the server was not told to trust is never believed.
29   */
30  @RequiredArgsConstructor
31  public class FederatedFactorSelector {
32  
33      /** Settings of the providers, for the trust. */
34      private final SocialProperties properties;
35  
36      /** Reads the upstream {@code amr}. */
37      private final FederatedUserMapper userMapper;
38  
39      /**
40       * Selects the factor of a sign-in.
41       *
42       * @param token the authentication of the sign-in.
43       * @return {@link FederatedFactor#STRONG_AUTHORITY} or {@link FederatedFactor#AUTHORITY}.
44       */
45      public String authorityOf(final OAuth2AuthenticationToken token) {
46          return IdentityProvider.fromId(token.getAuthorizedClientRegistrationId())
47                  .map(properties::trust)
48                  .filter(trust -> trust.isStrong(userMapper.upstreamAmr(Objects.requireNonNull(token.getPrincipal()))))
49                  .map(trust -> FederatedFactor.STRONG_AUTHORITY)
50                  .orElse(FederatedFactor.AUTHORITY);
51      }
52  }
53  // end::social-factor-selector[]
