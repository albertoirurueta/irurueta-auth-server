16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import com.irurueta.authserver.authorization.api.web.mfa.AuthenticationFactorRegistry;
19  import com.irurueta.authserver.authorization.domain.mfa.FactorStrength;
20  import com.irurueta.authserver.authorization.domain.ott.OttChannel;
21  import lombok.RequiredArgsConstructor;
22  import org.springframework.security.core.Authentication;
23  import org.springframework.security.core.GrantedAuthority;
24  
25  /**
26   * Rules about the session a one-time token is used as a second factor in: it must already hold a primary factor, and
27   * the second factor must not be the very channel that signed the user in (an e-mail code after an e-mail sign-in
28   * would prove the same mailbox twice).
29   */
30  @RequiredArgsConstructor
31  public class PrimaryFactorSession {
32  
33      /** Registry that tells the strength of each factor authority. */
34      private final AuthenticationFactorRegistry factorRegistry;
35  
36      /**
37       * Tells whether an authentication is a signed-in user who presented a primary factor.
38       *
39       * @param authentication the authentication, or {@code null}.
40       * @return {@code true} when it holds a factor of strength {@link FactorStrength#PRIMARY} or
41       *         {@link FactorStrength#PRIMARY_STRONG}.
42       */
43      public boolean holdsPrimaryFactor(final Authentication authentication) {
44          return OttGenerationParameters.isSignedIn(authentication)
45                  && factorRegistry.factorsOf(authentication).stream()
46                  .anyMatch(factor -> factor.strength() == FactorStrength.PRIMARY
47                          || factor.strength() == FactorStrength.PRIMARY_STRONG);
48      }
49  
50      /**
51       * Tells whether a channel is the one that signed the user in.
52       *
53       * @param authentication the authentication of the session.
54       * @param channel        the channel of the second factor.
55       * @return {@code true} when the authentication holds the sign-in authority of the channel.
56       */
57      public boolean signedInThrough(final Authentication authentication, final OttChannel channel) {
58          final var authority = OttFactors.signInAuthority(channel);
59          return authentication.getAuthorities().stream().map(GrantedAuthority::getAuthority).anyMatch(authority::equals);
60      }
61  }
