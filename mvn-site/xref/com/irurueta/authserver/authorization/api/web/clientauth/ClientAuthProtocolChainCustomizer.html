16  package com.irurueta.authserver.authorization.api.web.clientauth;
17  
18  import com.irurueta.authserver.authorization.api.web.ProtocolChainCustomizer;
19  import java.util.List;
20  import lombok.RequiredArgsConstructor;
21  import org.springframework.security.authentication.AuthenticationProvider;
22  import org.springframework.security.config.annotation.web.configurers.oauth2.server.authorization.OAuth2AuthorizationServerConfigurer;
23  import org.springframework.security.oauth2.server.authorization.authentication.JwtClientAssertionAuthenticationProvider;
24  import org.springframework.security.oauth2.server.authorization.authentication.X509ClientCertificateAuthenticationProvider;
25  
26  // tag::clientauth-chain[]
27  /**
28   * Installs {@code private_key_jwt} with replay protection and the mutual-TLS client authentication in the protocol
29   * chain: the framework's assertion provider decodes with {@link RegisteredJwksClientAssertionDecoderFactory}, and its
30   * certificate provider verifies with {@link ClientCertificateVerifier}. The server metadata (the OpenID provider
31   * configuration and the OAuth 2.0 authorization server metadata) also advertises {@code mtls_endpoint_aliases}.
32   */
33  @RequiredArgsConstructor
34  public class ClientAuthProtocolChainCustomizer implements ProtocolChainCustomizer {
35  
36      /** Order of the customizer: after the multi-factor, step-up and device ones. */
37      public static final int ORDER = 3;
38  
39      /** Creates the decoders of the client assertions. */
40      private final RegisteredJwksClientAssertionDecoderFactory decoderFactory;
41  
42      /** Verifies the client certificates. */
43      private final ClientCertificateVerifier certificateVerifier;
44  
45      /** Builds the {@code mtls_endpoint_aliases} claim. */
46      private final MtlsEndpointAliases endpointAliases;
47  
48      /**
49       * Customizes the client authentication and the metadata.
50       *
51       * @param server configurer of the authorization server.
52       */
53      @Override
54      public void customize(final OAuth2AuthorizationServerConfigurer server) {
55          server.clientAuthentication(client -> client.authenticationProviders(this::configure));
56          server.authorizationServerMetadataEndpoint(metadata -> metadata.authorizationServerMetadataCustomizer(
57                  builder -> endpointAliases.contribute(builder::claim)));
58          server.oidc(oidc -> oidc.providerConfigurationEndpoint(configuration ->
59                  configuration.providerConfigurationCustomizer(builder -> endpointAliases.contribute(builder::claim))));
60      }
61  
62      /**
63       * Sets the decoder factory and the certificate verifier on the framework's providers.
64       *
65       * @param providers the providers of the client authentication.
66       */
67      private void configure(final List<AuthenticationProvider> providers) {
68          for (final var provider : providers) {
69              if (provider instanceof JwtClientAssertionAuthenticationProvider assertions) {
70                  assertions.setJwtDecoderFactory(decoderFactory);
71              } else if (provider instanceof X509ClientCertificateAuthenticationProvider certificates) {
72                  certificates.setCertificateVerifier(certificateVerifier);
73              }
74          }
75      }
76  }
77  // end::clientauth-chain[]
