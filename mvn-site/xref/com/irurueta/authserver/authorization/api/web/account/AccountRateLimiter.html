16  package com.irurueta.authserver.authorization.api.web.account;
17  
18  import com.irurueta.authserver.authorization.api.web.ratelimit.FixedWindowBuckets;
19  import java.time.Clock;
20  import java.util.Locale;
21  import org.springframework.boot.context.properties.EnableConfigurationProperties;
22  import org.springframework.stereotype.Component;
23  
24  // tag::account-rate-limit[]
25  /**
26   * Rate limiter of the unauthenticated account endpoints (register, verify, reset request, reset), built on
27   * <a href="https://bucket4j.com">Bucket4j</a> token buckets ({@link FixedWindowBuckets}). Requests are limited twice:
28   * per client address, and per e-mail address when the request names one, so neither a single client nor many clients
29   * aiming at one address can flood the server (and the mailboxes) with account mail or guess reset tokens. Each bucket
30   * holds the limit of its key and is refilled all at once every {@link AccountRateLimitProperties#window()}.
31   *
32   * <p>The buckets of each kind of key live in a Caffeine cache bounded to
33   * {@link AccountRateLimitProperties#maxTrackedKeys()} entries, which also drops a bucket idle for a whole window
34   * (it would be full again anyway). They live in this instance only: several instances each apply the limits on
35   * their own, until the buckets move to a shared store through a Bucket4j {@code ProxyManager} (for example
36   * {@code bucket4j-postgresql}). The client address is {@code request.getRemoteAddr()}, so behind a proxy
37   * {@code server.forward-headers-strategy} must be set for it to be the real client.
38   */
39  @Component
40  @EnableConfigurationProperties(AccountRateLimitProperties.class)
41  public final class AccountRateLimiter {
42  
43      /** Limits, window and bound of the tracked keys. */
44      private final AccountRateLimitProperties properties;
45  
46      /** Buckets per client address. */
47      private final FixedWindowBuckets clients;
48  
49      /** Buckets per (lower-cased) e-mail address. */
50      private final FixedWindowBuckets emails;
51  
52      /**
53       * Creates the limiter.
54       *
55       * @param properties limits, window and bound of the tracked keys.
56       * @param clock      source of the current time.
57       */
58      public AccountRateLimiter(final AccountRateLimitProperties properties, final Clock clock) {
59          this.properties = properties;
60          clients = new FixedWindowBuckets(clock, properties.window(), properties.maxTrackedKeys());
61          emails = new FixedWindowBuckets(clock, properties.window(), properties.maxTrackedKeys());
62      }
63  
64      /**
65       * Counts one request and rejects it once a limit is exceeded.
66       *
67       * @param clientAddress address of the calling client.
68       * @param email         e-mail address the request names, or {@code null}.
69       * @throws AccountRateLimitExceededException if the client or the e-mail address exceeded its limit.
70       */
71      public void check(final String clientAddress, final String email) {
72          if (!properties.enabled()) {
73              return;
74          }
75          consume(clients, clientAddress, properties.perClient());
76          if (email != null) {
77              consume(emails, email.toLowerCase(Locale.ROOT), properties.perEmail());
78          }
79      }
80  
81      /**
82       * Tells how many keys of each kind are tracked, after any pending eviction.
83       *
84       * @return the number of client keys plus the number of e-mail keys.
85       */
86      long trackedKeys() {
87          return clients.trackedKeys() + emails.trackedKeys();
88      }
89  
90      /**
91       * Takes one token from the bucket of a key, creating the bucket on first use.
92       *
93       * @param buckets buckets of one kind of key.
94       * @param key     the key.
95       * @param limit   requests the key may make per window.
96       * @throws AccountRateLimitExceededException if the bucket is empty.
97       */
98      private static void consume(final FixedWindowBuckets buckets, final String key, final int limit) {
99          final var wait = buckets.tryConsume(key, limit);
100         if (wait.isPresent()) {
101             throw new AccountRateLimitExceededException(wait.get());
102         }
103     }
104 }
