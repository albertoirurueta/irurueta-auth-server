34  /**
35   * Wires the TOTP second factor into the login/UI chain. The filters are built here and handed to the customizer,
36   * never declared as beans: Spring Boot would register every {@code Filter} bean in the servlet container, outside
37   * the security chain.
38   */
39  @Configuration(proxyBeanMethods = false)
40  public class TotpConfiguration {
41  
42      /**
43       * Installs the TOTP second factor in the login/UI chain.
44       *
45       * @param totpCodeVerifier           verifies a posted code with the use case of its kind.
46       * @param createTrustedDeviceUseCase remembers a device.
47       * @param findTrustedDeviceUseCase   finds a remembered device.
48       * @param userResolver               resolves the identifier of the account of the signed-in user.
49       * @param factorGrantor              adds the factor authority to an authentication.
50       * @param cookieWriter               writes the cookie of a remembered device.
51       * @param pendingRecoveryCodes       keeps the recovery codes of an enrolment until they are shown.
52       * @return the customizer.
53       */
54      @Bean
55      @Order(1)
56      public LoginChainCustomizer totpChainCustomizer(final TotpCodeVerifier totpCodeVerifier,
57              final CreateTrustedDeviceUseCase createTrustedDeviceUseCase,
58              final FindTrustedDeviceUseCase findTrustedDeviceUseCase, final SecondFactorUserResolver userResolver,
59              final FactorGrantor factorGrantor, final TrustedDeviceCookieWriter cookieWriter,
60              final PendingRecoveryCodes pendingRecoveryCodes) {
61          final var provider = new TotpAuthenticationProvider(totpCodeVerifier, userResolver, factorGrantor);
62          final var totpFilter = new TotpAuthenticationFilter(new ProviderManager(provider),
63                  createTrustedDeviceUseCase, userResolver, cookieWriter, pendingRecoveryCodes);
64          final var trustedDeviceFilter = new TrustedDeviceFilter(findTrustedDeviceUseCase, factorGrantor,
65                  new HttpSessionSecurityContextRepository(),
66                  new SavedRequestAwareAuthenticationSuccessHandler(), TrustedDeviceFactor.AUTHORITY);
67          return new TotpChainCustomizer(trustedDeviceFilter, totpFilter);
68      }
69  }
