17  package com.irurueta.authserver.authorization.api.web.pages;
18  
19  import com.irurueta.authserver.authorization.api.web.login.BrandingProvider;
20  import jakarta.servlet.http.HttpServletRequest;
21  import jakarta.servlet.http.HttpServletResponse;
22  import java.nio.charset.Charset;
23  import java.nio.charset.StandardCharsets;
24  import java.util.LinkedHashMap;
25  import java.util.List;
26  import java.util.Locale;
27  import java.util.Map;
28  import java.util.Objects;
29  import lombok.RequiredArgsConstructor;
30  import org.springframework.http.MediaType;
31  import org.springframework.http.ResponseEntity;
32  import org.springframework.stereotype.Component;
33  import org.springframework.web.servlet.View;
34  import org.springframework.web.servlet.ViewResolver;
35  import org.springframework.web.servlet.support.RequestContextUtils;
36  import org.springframework.web.util.ContentCachingResponseWrapper;
37  
38  /**
39   * Renders a server-rendered page (a Thymeleaf template) to the {@code text/html} answer of an operation generated from
40   * the pages contract. The view is resolved through the application's view resolvers, in their order, with the locale
41   * of the request, and rendered against the current request into a wrapper that captures what it writes, so everything
42   * a template relies on keeps working: the CSRF hidden field that {@code th:action} adds, {@code @{...}} links and the
43   * messages. Every page gets the branding as the {@value BrandingProvider#MODEL_ATTRIBUTE} model attribute.
44   */
45  @Component
46  @RequiredArgsConstructor
47  public class PageRenderer {
48  
49      /** Media type of the rendered pages. */
50      static final MediaType TEXT_HTML_UTF8 = new MediaType(MediaType.TEXT_HTML, StandardCharsets.UTF_8);
51  
52      /** View resolvers of the application, in their order. */
53      private final List<ViewResolver> viewResolvers;
54  
55      /** Supplies the branding of the pages. */
56      private final BrandingProvider brandingProvider;
57  
58      /** The current request, injected as a thread-bound proxy. */
59      private final HttpServletRequest request;
60  
61      /** The current response, injected as a thread-bound proxy. */
62      private final HttpServletResponse response;
63  
64      // tag::page-renderer[]
65      /**
66       * Renders a page.
67       *
68       * @param viewName name of the view (the Thymeleaf template).
69       * @param model    model attributes of the page; the branding is added to them.
70       * @return a {@code 200} with the page as {@code text/html}.
71       * @throws IllegalStateException if no view resolver knows the view, or the view cannot be rendered.
72       */
73      public ResponseEntity<String> render(final String viewName, final Map<String, ?> model) {
74          final var locale = RequestContextUtils.getLocale(request);
75          final var view = resolve(viewName, locale);
76          final var attributes = new LinkedHashMap<String, Object>(model);
77          attributes.put(BrandingProvider.MODEL_ATTRIBUTE, brandingProvider.branding());
78          final var capture = new ContentCachingResponseWrapper(response);
79          try {
80              view.render(attributes, request, capture);
81          } catch (final Exception e) {
82              throw new IllegalStateException("The page " + viewName + " cannot be rendered", e);
83          }
84          final var html = new String(capture.getContentAsByteArray(), Charset.forName(capture.getCharacterEncoding()));
85          return ResponseEntity.ok().contentType(TEXT_HTML_UTF8).body(html);
86      }
87      // end::page-renderer[]
88  
89      /**
90       * Resolves a view through the view resolvers, in their order.
91       *
92       * @param viewName name of the view.
93       * @param locale   locale of the request.
94       * @return the view of the first resolver that knows it.
95       * @throws IllegalStateException if no view resolver knows the view.
96       */
97      private View resolve(final String viewName, final Locale locale) {
98          return viewResolvers.stream()
99                  .map(resolver -> resolveWith(resolver, viewName, locale))
100                 .filter(Objects::nonNull)
101                 .findFirst()
102                 .orElseThrow(() -> new IllegalStateException("No view named " + viewName));
103     }
104 
105     /**
106      * Resolves a view with one resolver.
107      *
108      * @param resolver the view resolver.
109      * @param viewName name of the view.
110      * @param locale   locale of the request.
111      * @return the view, or {@code null} when the resolver does not know it.
112      * @throws IllegalStateException if the resolver fails.
113      */
114     private static View resolveWith(final ViewResolver resolver, final String viewName, final Locale locale) {
115         try {
116             return resolver.resolveViewName(viewName, locale);
117         } catch (final Exception e) {
118             throw new IllegalStateException("The view " + viewName + " cannot be resolved", e);
119         }
120     }
121 }
