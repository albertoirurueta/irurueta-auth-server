23  import org.springframework.security.authentication.ott.OneTimeTokenService;

25  import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
26  import org.springframework.security.web.access.intercept.AuthorizationFilter;
27  import org.springframework.security.web.authentication.AuthenticationConverter;
28  import org.springframework.security.web.authentication.AuthenticationFailureHandler;
29  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
30  import org.springframework.security.web.authentication.ott.GenerateOneTimeTokenFilter;
31  import org.springframework.security.web.authentication.ott.GenerateOneTimeTokenRequestResolver;
32  import org.springframework.security.web.authentication.ott.OneTimeTokenGenerationSuccessHandler;
33  import org.springframework.security.web.authentication.session.SessionAuthenticationStrategy;

35  // tag::ott-chain[]
36  /**
37   * Adds the one-time token methods to the login/UI chain through {@code http.oneTimeTokenLogin(...)}: the request of a
38   * token ({@code POST /ott/generate}), the submission of a magic link token or a numeric code ({@code POST /login/ott})
39   * and, as a second factor, the submission of a code ({@code POST /login/second-factor/ott}). The pages the user lands
40   * on (where the code is typed, where a magic link is confirmed) are open to everybody and the second-factor post to
41   * signed-in users only. The rate limits are applied before anything else of the one-time token flow. CSRF protection
42   * stays on: every form carries the token. Once the chain is configured, the second-factor filter gets the chain's
43   * session authentication strategy, the one the form login uses (a new session id and a new CSRF token on success).
44   * Spring's own default submit page is turned off: the pages are this server's own.
45   */
46  @RequiredArgsConstructor
47  public class OttChainCustomizer implements LoginChainCustomizer {
48  
49      /** Order of this customizer in the login chain, after the TOTP ({@code 1}) and passkey ({@code 2}) ones. */
50      public static final int ORDER = 3;
51  
52      /** Applies the rate limits to the endpoints. */
53      private final OttRateLimitFilter rateLimitFilter;
54  
55      /** Resolves the token Spring's filter generates. */
56      private final GenerateOneTimeTokenRequestResolver requestResolver;
57  
58      /** Delivers the token that was generated. */
59      private final OneTimeTokenGenerationSuccessHandler generationSuccessHandler;
60  
61      /** Store of the magic link tokens. */
62      private final OneTimeTokenService tokenService;
63  
64      /** Reads what is posted to the sign-in endpoint. */
65      private final AuthenticationConverter authenticationConverter;
66  
67      /** Confirms a magic link. */
68      private final AuthenticationProvider linkProvider;
69  
70      /** Signs a user in with a numeric code. */
71      private final AuthenticationProvider codeProvider;
72  
73      /** Resumes the interrupted request after a sign-in. */
74      private final AuthenticationSuccessHandler successHandler;
75  
76      /** Sends the browser back after a failed sign-in. */
77      private final AuthenticationFailureHandler failureHandler;
78  
79      /** Processes the post of the second-factor code form. */
80      private final OttSecondFactorFilter secondFactorFilter;
81  
82      /**
83       * Customizes the login/UI chain.
84       *
85       * @param http security builder of the chain.
86       */
87      @Override
88      public void customize(final HttpSecurity http) {
89          http.authorizeHttpRequests(authorize -> authorize
90                  .requestMatchers(OttRateLimitFilter.GENERATE_PATH, OneTimeTokenPageApi.PATH_OTT_CODE_PAGE,
91                          OneTimeTokenPageApi.PATH_OTT_CODE_PAGE + "/**").permitAll()
92                  .requestMatchers(OttSecondFactorFilter.SECOND_FACTOR_OTT_PATH).authenticated());
93          http.addFilterBefore(rateLimitFilter, GenerateOneTimeTokenFilter.class);
94          http.addFilterBefore(secondFactorFilter, AuthorizationFilter.class);
95          http.authenticationProvider(codeProvider);
96          http.oneTimeTokenLogin(ott -> ott
97                  .tokenGeneratingUrl(OttRateLimitFilter.GENERATE_PATH)
98                  .loginProcessingUrl(OneTimeTokenPageApi.PATH_OTT_CODE_PAGE)
99                  .loginPage(LoginController.LOGIN_PATH)
100                 .showDefaultSubmitPage(false)
101                 .tokenService(tokenService)
102                 .generateRequestResolver(requestResolver)
103                 .tokenGenerationSuccessHandler(generationSuccessHandler)
104                 .authenticationConverter(authenticationConverter)
105                 .authenticationProvider(linkProvider)
106                 .successHandler(successHandler)
107                 .failureHandler(failureHandler));
108         http.with(new SessionStrategyConfigurer(secondFactorFilter));
109     }
110 
111     /**
112      * Hands the session authentication strategy of the chain to the second-factor filter. The strategy is a shared
113      * object of the chain that only exists once every configurer has been initialised (the CSRF configurer adds its
114      * {@code CsrfAuthenticationStrategy} to the session management one), so it is read in {@link #configure}, as the
115      * form login does for its own filter. Package-private so its test can drive it.
116      */
117     @RequiredArgsConstructor
118     static final class SessionStrategyConfigurer
119             extends AbstractHttpConfigurer<SessionStrategyConfigurer, HttpSecurity> {
120 
121         /** The filter that gets the strategy. */
122         private final OttSecondFactorFilter filter;
123 
124         /**
125          * Sets the shared strategy, when the chain has one, on the filter.
126          *
127          * @param http security builder of the chain.
128          */
129         @Override
130         public void configure(final HttpSecurity http) {
131             final var strategy = http.getSharedObject(SessionAuthenticationStrategy.class);
132             if (strategy != null) {
133                 filter.setSessionAuthenticationStrategy(strategy);
134             }
135         }
136     }
137 }
