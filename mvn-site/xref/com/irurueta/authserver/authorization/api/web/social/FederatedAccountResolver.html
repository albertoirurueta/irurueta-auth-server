33  /**
34   * Finds the local account of a sign-in with an external provider and refuses it unless that account may sign in.
35   *
36   * <p>The use case finds, links or provisions the account and refuses a disabled one before it links anything; the
37   * account is then loaded through the status-checking user service as well (the rules of
38   * {@code AccountStatusUserDetailsService}: a disabled, locked or expired account is refused, exactly as the password
39   * and passkey sign-ins refuse it). Every refusal is the same {@code access_denied} error, which the login chain turns
40   * into the generic failure of the login page, so the answer never says whether an account exists or why it was
41   * refused.
42   *
43   * <p>When the signed-in user asked to link this provider ({@link PendingIdentityLinks}), the identity is linked to
44   * that account through {@link LinkIdentityUseCase} instead: nothing is resolved by e-mail address or provisioned, and
45   * an identity that belongs to another account is refused with {@link #LINK_REFUSED}, which the failure handler sends
46   * back to the identities page.
47   */
48  @RequiredArgsConstructor
49  public class FederatedAccountResolver {
50  
51      /** Error description of every refusal. */
52      static final String REFUSED = "The sign-in with the external provider was refused";
53  
54      /** Error code of a refused link: the identity belongs to another account. */
55      public static final String LINK_REFUSED = "identity_link_refused";
56  
57      /** Error description of a refused link. */
58      static final String LINK_REFUSED_DESCRIPTION = "The identity is linked to another account";
59  
60      /** Use case that finds, links or provisions the account. */
61      private final ResolveFederatedLoginUseCase resolveFederatedLoginUseCase;
62  
63      /** Loads the account, refusing one that may not sign in. */
64      private final UserDetailsService accountStatusUserDetailsService;
65  
66      /** Use case that links an identity to the signed-in account. */
67      private final LinkIdentityUseCase linkIdentityUseCase;
68  
69      /** The link markers of the explicit link flow. */
70      private final PendingIdentityLinks pendingIdentityLinks;
71  
72      /**
73       * Resolves a sign-in.
74       *
75       * @param provider      the provider that authenticated the user.
76       * @param subject       the provider's stable identifier of the user.
77       * @param email         e-mail address the provider reported, or {@code null}.
78       * @param emailVerified whether the provider vouches for the address.
79       * @return the local account, which may sign in; for a link, the signed-in account.
80       * @throws OAuth2AuthenticationException if the sign-in is refused: the provider did not verify the address, the
81       *                                       address belongs to an unverified account, or the account is disabled,
82       *                                       locked or expired; or if a link is refused because the identity belongs to
83       *                                       another account.
84       */
85      public UserDetails resolve(
86              final IdentityProvider provider, final String subject, final String email, final boolean emailVerified) {
87          final var link = pendingIdentityLinks.consume(provider);
88          if (link.isPresent()) {
89              return link(link.get(), provider, subject, email);
90          }
91          try {
92              final var login = resolveFederatedLoginUseCase.execute(provider, subject, email, emailVerified);
93              return accountStatusUserDetailsService.loadUserByUsername(login.account().username());
94          } catch (final FederatedLoginRefusedException | AuthenticationException e) {
95              throw new OAuth2AuthenticationException(new OAuth2Error(OAuth2ErrorCodes.ACCESS_DENIED, REFUSED, null), e);
96          }
97      }
98  
99      /**
100      * Links an identity to the signed-in account that asked for it.
101      *
102      * @param link     the link marker.
103      * @param provider the provider that authenticated the user.
104      * @param subject  the provider's stable identifier of the user.
105      * @param email    e-mail address the provider reported, or {@code null}.
106      * @return the signed-in account.
107      * @throws OAuth2AuthenticationException if the identity belongs to another account, or the account may no longer
108      *                                       sign in.
109      */
110     private UserDetails link(final PendingIdentityLink link, final IdentityProvider provider, final String subject,
111                              final String email) {
112         try {
113             linkIdentityUseCase.execute(link.userId(), link.username(), provider, subject, email);
114             return accountStatusUserDetailsService.loadUserByUsername(link.username());
115         } catch (final FederatedIdentityAlreadyLinkedException | AuthenticationException e) {
116             throw new OAuth2AuthenticationException(
117                     new OAuth2Error(LINK_REFUSED, LINK_REFUSED_DESCRIPTION, null), e);
118         }
119     }
120 }
