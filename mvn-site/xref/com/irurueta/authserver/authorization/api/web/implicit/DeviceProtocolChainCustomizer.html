16  package com.irurueta.authserver.authorization.api.web.implicit;

18  import com.irurueta.authserver.authorization.api.web.ProtocolChainCustomizer;
19  import com.irurueta.authserver.authorization.api.web.consent.ConsentController;
20  import lombok.RequiredArgsConstructor;
21  import org.springframework.security.authentication.AuthenticationProvider;
22  import org.springframework.security.config.Customizer;
23  import org.springframework.security.config.annotation.web.configurers.oauth2.server.authorization.OAuth2AuthorizationServerConfigurer;
24  import org.springframework.security.web.authentication.AuthenticationConverter;
25  import org.springframework.security.web.authentication.AuthenticationFailureHandler;
26  import org.springframework.security.web.authentication.AuthenticationSuccessHandler;
27  
28  // tag::device-protocol-chain[]
29  /**
30   * Enables the OAuth 2.0 Device Authorization Grant on the protocol chain. The device authorization endpoint
31   * ({@code POST /oauth2/device_authorization}) is where the device, authenticated as a public client by its identifier
32   * alone, obtains a device code and a user code. The verification endpoint ({@code /oauth2/device_verification}) is
33   * where the signed-in user approves the device; its verification URI is the framework's default, the issuer plus the
34   * endpoint, whose {@code GET} without a code is the page of {@link DeviceVerificationController}. The verification
35   * endpoint is customized to guard its requests ({@link DeviceVerificationRequestGuard}), to hand the consent decision
36   * to the existing consent page, to answer its failures with {@link DeviceVerificationErrorHandler} and to go back to
37   * the verification page when the device was approved.
38   */
39  @RequiredArgsConstructor
40  public class DeviceProtocolChainCustomizer implements ProtocolChainCustomizer {
41  
42      /** Order of the customizer: after the multi-factor ({@code 0}) and step-up ({@code 1}) ones. */
43      public static final int ORDER = 2;
44  
45      /** Reads the identifier of a public client of the device flow. */
46      private final AuthenticationConverter clientConverter;
47  
48      /** Authenticates a public client of the device flow. */
49      private final AuthenticationProvider clientProvider;
50  
51      /** Rate limit and multi-factor policy of the verification requests. */
52      private final AuthenticationConverter guard;
53  
54      /** Answers the failures of the verification endpoint. */
55      private final AuthenticationFailureHandler errorHandler;
56  
57      /** Answers a verification that did not need consent, or whose consent was given. */
58      private final AuthenticationSuccessHandler successHandler;
59  
60      /**
61       * Customizes the device endpoints.
62       *
63       * @param server configurer of the authorization server.
64       */
65      @Override
66      public void customize(final OAuth2AuthorizationServerConfigurer server) {
67          server.clientAuthentication(client -> client
68                  .authenticationConverter(clientConverter)
69                  .authenticationProvider(clientProvider));
70          server.deviceAuthorizationEndpoint(Customizer.withDefaults());
71          server.deviceVerificationEndpoint(endpoint -> endpoint
72                  .consentPage(ConsentController.CONSENT_PATH)
73                  .deviceVerificationRequestConverters(converters -> converters.addFirst(guard))
74                  .errorResponseHandler(errorHandler)
75                  .deviceVerificationResponseHandler(successHandler));
76      }
77  }
78  // end::device-protocol-chain[]
