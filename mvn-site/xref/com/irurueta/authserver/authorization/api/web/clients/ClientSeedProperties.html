17  package com.irurueta.authserver.authorization.api.web.clients;
18  
19  import java.util.List;
20  import org.springframework.boot.context.properties.ConfigurationProperties;
21  
22  // tag::clients-seeding[]
23  /**
24   * Configuration of the seeding of the default clients ({@code authserver.clients.*}).
25   *
26   * <p>Which clients are seeded, and where they redirect, is a property of the deployment: a client is seeded only when
27   * its redirect URIs are configured, and nothing is configured by default (the {@code dev} profile sets the local
28   * ones). Secrets come from the environment only and are never committed: the property resolves to an empty value
29   * when the environment variable is not set, and a confidential client without a secret is then not seeded.
30   *
31   * @param bffSecret clear-text secret of the {@code bff} client; empty or {@code null} when not configured.
32   * @param bff       seeding of the {@code bff} client; never {@code null} once constructed.
33   * @param spaPublic seeding of the {@code spa-public} client; never {@code null} once constructed.
34   */
35  @ConfigurationProperties(prefix = "authserver.clients")
36  public record ClientSeedProperties(String bffSecret, SeededClient bff, SeededClient spaPublic) {
37  
38      /**
39       * Replaces a missing client section by an unconfigured one.
40       */
41      public ClientSeedProperties {
42          bff = bff == null ? SeededClient.unconfigured() : bff;
43          spaPublic = spaPublic == null ? SeededClient.unconfigured() : spaPublic;
44      }
45  
46      @Override
47      public String toString() {
48          return "ClientSeedProperties[bffSecret=***, bff=" + bff + ", spaPublic=" + spaPublic + "]";
49      }
50  
51      /**
52       * Seeding of one default client.
53       *
54       * @param redirectUris           redirect URIs of the client in this deployment; empty when the client is not
55       *                               seeded.
56       * @param postLogoutRedirectUris URIs the client may send the user to after an OpenID Connect RP-initiated logout
57       *                               ({@code post-logout-redirect-uris}); empty when none is configured, in which
58       *                               case no {@code post_logout_redirect_uri} is honoured.
59       */
60      public record SeededClient(List<String> redirectUris, List<String> postLogoutRedirectUris) {
61  
62          /**
63           * Copies the URIs, treating a missing list as an empty one.
64           */
65          public SeededClient {
66              redirectUris = redirectUris == null ? List.of() : List.copyOf(redirectUris);
67              postLogoutRedirectUris = postLogoutRedirectUris == null ? List.of() : List.copyOf(postLogoutRedirectUris);
68          }
69  
70          /**
71           * Creates the section of a client that is not seeded.
72           *
73           * @return a section without redirect URIs.
74           */
75          public static SeededClient unconfigured() {
76              return new SeededClient(List.of(), List.of());
77          }
78  
79          /**
80           * Tells whether the client is to be seeded.
81           *
82           * @return {@code true} when at least one redirect URI is configured.
83           */
84          public boolean isConfigured() {
85              return !redirectUris.isEmpty();
86          }
87      }
88  }
89  // end::clients-seeding[]
