16  package com.irurueta.authserver.authorization.api.web.login;
17  
18  import org.springframework.security.authentication.AccountExpiredException;
19  import org.springframework.security.authentication.CredentialsExpiredException;
20  import org.springframework.security.authentication.DisabledException;
21  import org.springframework.security.authentication.LockedException;
22  import org.springframework.security.core.AuthenticationException;
23  import org.springframework.stereotype.Component;
24  
25  /**
26   * Maps the exception of a failed sign-in, which Spring Security keeps in the
27   * {@code SPRING_SECURITY_LAST_EXCEPTION} session attribute, to the key of the message the login page shows.
28   *
29   * <p>An unknown user and a wrong password (a
30   * {@link org.springframework.security.authentication.BadCredentialsException} or a
31   * {@link org.springframework.security.core.userdetails.UsernameNotFoundException}) deliberately share
32   * the generic key, so the two responses are identical and the page never reveals whether an account exists. The
33   * account status messages (locked, disabled, expired) cannot reveal it either: the authentication provider of the
34   * {@code password} feature ({@code PasswordAuthenticationConfiguration}) only checks the status after a correct
35   * password, so a wrong password for a disabled account is reported as bad credentials too.
36   */
37  @Component
38  public class LoginErrorMessages {
39  
40      /** Key of the message shown for bad credentials, an unknown user and any unexpected failure. */
41      public static final String GENERIC = "login.error.generic";
42  
43      /** Key of the message shown for a locked account. */
44      public static final String LOCKED = "login.error.locked";
45  
46      /** Key of the message shown for an expired account. */
47      public static final String ACCOUNT_EXPIRED = "login.error.accountExpired";
48  
49      /** Key of the message shown for expired credentials. */
50      public static final String CREDENTIALS_EXPIRED = "login.error.credentialsExpired";
51  
52      /** Key of the message shown for a disabled account. */
53      public static final String DISABLED = "login.error.disabled";
54  
55      /**
56       * Gets the key of the message that describes a failed sign-in.
57       *
58       * @param exception the exception of the failed sign-in; {@code null} when the session holds none.
59       * @return the message key; the generic one for bad credentials, an unknown user and anything unexpected.
60       */
61      public String keyOf(final AuthenticationException exception) {
62          return switch (exception) {
63              case LockedException locked -> LOCKED;
64              case AccountExpiredException expired -> ACCOUNT_EXPIRED;
65              case CredentialsExpiredException expired -> CREDENTIALS_EXPIRED;
66              case DisabledException disabled -> DISABLED;
67              case null, default -> GENERIC;
68          };
69      }
70  }
