17  package com.irurueta.authserver.authorization.api.web.passkeys;
18  
19  import com.irurueta.authserver.authorization.api.web.login.LoginChainCustomizer;
20  import com.irurueta.authserver.authorization.api.web.mfa.MfaAuthorizationManager;
21  import com.irurueta.authserver.authorization.api.web.mfa.SecondFactorUserResolver;
22  import com.irurueta.authserver.authorization.application.mfa.CheckSecondFactorLockoutUseCase;
23  import com.irurueta.authserver.authorization.application.passkeys.PasskeyProperties;
24  import org.springframework.context.annotation.Bean;
25  import org.springframework.context.annotation.Configuration;
26  import org.springframework.core.annotation.Order;
27  import org.springframework.security.core.userdetails.UserDetailsService;
28  
29  /**
30   * Installs the passkeys (WebAuthn) in the login/UI chain. Nothing outside this feature is edited: the chain applies
31   * every {@link LoginChainCustomizer} bean, in order, before its catch-all rule.
32   */
33  @Configuration(value = "passkeyWebConfiguration", proxyBeanMethods = false)
34  public class PasskeyConfiguration {
35  
36      // tag::passkeys-chain-customizer[]
37      /**
38       * Declares the customizer that enables {@code http.webAuthn(...)}, permits its public endpoints, keeps the
39       * registration endpoints behind the multi-factor policy and removes the ones that would bypass the step-up. It is
40       * applied after the TOTP customizer.
41       *
42       * @param properties           relying party settings.
43       * @param authorizationManager applies the multi-factor policy.
44       * @param userDetailsService   loads the account of the user a passkey belongs to.
45       * @param userResolver         resolves the account of the signed-in user, for the lockout check.
46       * @param lockoutUseCase       tells whether the second factor of a user is locked.
47       * @return the customizer.
48       */
49      @Bean
50      @Order(PasskeyChainCustomizer.ORDER)
51      public LoginChainCustomizer passkeyChainCustomizer(final PasskeyProperties properties,
52              final MfaAuthorizationManager authorizationManager, final UserDetailsService userDetailsService,
53              final SecondFactorUserResolver userResolver, final CheckSecondFactorLockoutUseCase lockoutUseCase) {
54          return new PasskeyChainCustomizer(properties, authorizationManager, userDetailsService, userResolver,
55                  lockoutUseCase);
56      }
57      // end::passkeys-chain-customizer[]
58  }
