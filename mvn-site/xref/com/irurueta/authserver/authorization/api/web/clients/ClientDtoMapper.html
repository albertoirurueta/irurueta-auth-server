17  package com.irurueta.authserver.authorization.api.web.clients;
18  
19  import com.irurueta.authserver.authorization.api.web.v1.generated.model.ClientRequest;
20  import com.irurueta.authserver.authorization.api.web.v1.generated.model.ClientResponse;
21  import com.irurueta.authserver.authorization.api.web.v1.generated.model.ClientSecretResponse;
22  import com.irurueta.authserver.authorization.api.web.v1.generated.model.CreatedClientResponse;
23  import com.irurueta.authserver.authorization.application.clients.CreatedClient;
24  import com.irurueta.authserver.authorization.domain.clients.ClientRegistration;
25  import java.util.List;
26  import org.springframework.stereotype.Component;
27  import org.springframework.util.StringUtils;
28  
29  /**
30   * Maps between the DTOs generated from the contract and the client registrations of the domain, so a domain model is
31   * never returned from {@link ClientAdminController}.
32   */
33  @Component
34  public class ClientDtoMapper {
35  
36      /**
37       * Maps a create or update request to a client registration. An absent flag is {@code false}, absent audiences
38       * are empty (the client identifier is then used), absent post-logout redirect URIs are empty and an absent or
39       * blank format is {@value ClientRegistration#FORMAT_SELF_CONTAINED}.
40       *
41       * @param request the validated request.
42       * @return the registration.
43       * @throws IllegalArgumentException if the registration is not valid.
44       */
45      public ClientRegistration toRegistration(final ClientRequest request) {
46          final var format = StringUtils.hasText(request.accessTokenFormat())
47                  ? request.accessTokenFormat() : ClientRegistration.FORMAT_SELF_CONTAINED;
48          final var audiences = request.audiences() == null ? List.<String>of() : request.audiences();
49          return new ClientRegistration(request.clientId(), request.name(), Boolean.TRUE.equals(request.confidential()),
50                  request.authMethods(), request.grantTypes(), request.redirectUris(), request.scopes(),
51                  Boolean.TRUE.equals(request.requireProofKey()), audiences, request.accessTokenTtl(),
52                  request.refreshTokenTtl(), format, Boolean.TRUE.equals(request.reuseRefreshTokens()),
53                  request.postLogoutRedirectUris() == null ? List.of() : request.postLogoutRedirectUris(),
54                  Boolean.TRUE.equals(request.requireConsent()), Boolean.TRUE.equals(request.mfaRequired()));
55      }
56  
57      /**
58       * Maps a registration to its public view, which never contains a secret.
59       *
60       * @param registration the registration.
61       * @return its public view.
62       */
63      public ClientResponse toResponse(final ClientRegistration registration) {
64          return new ClientResponse(registration.clientId(), registration.name(), registration.confidential(),
65                  registration.authMethods(), registration.grantTypes(), registration.redirectUris(),
66                  registration.scopes(), registration.requireProofKey(), registration.audiences(),
67                  registration.accessTokenTtl(), registration.refreshTokenTtl(), registration.accessTokenFormat(),
68                  registration.reuseRefreshTokens(), registration.postLogoutRedirectUris(),
69                  registration.requireConsent(), registration.mfaRequired());
70      }
71  
72      /**
73       * Maps several registrations, keeping their order.
74       *
75       * @param registrations the registrations.
76       * @return their public views.
77       */
78      public List<ClientResponse> toResponses(final List<ClientRegistration> registrations) {
79          return registrations.stream().map(this::toResponse).toList();
80      }
81  
82      /**
83       * Maps a created client: its public view and, for a confidential one, its secret (shown only once).
84       *
85       * @param created the created client.
86       * @return the answer to the creation.
87       */
88      public CreatedClientResponse toCreatedResponse(final CreatedClient created) {
89          return new CreatedClientResponse(toResponse(created.registration()), created.clientSecret());
90      }
91  
92      /**
93       * Maps a rotated secret (shown only once).
94       *
95       * @param clientSecret the new secret in clear text.
96       * @return the answer to the rotation.
97       */
98      public ClientSecretResponse toSecretResponse(final String clientSecret) {
99          return new ClientSecretResponse(clientSecret);
100     }
101 }
