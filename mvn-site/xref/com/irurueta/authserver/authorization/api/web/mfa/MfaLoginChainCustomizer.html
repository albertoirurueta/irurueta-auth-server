17  package com.irurueta.authserver.authorization.api.web.mfa;
18  
19  import static com.irurueta.authserver.authorization.api.web.consent.ConsentController.CONSENT_PATH;
20  
21  import com.irurueta.authserver.authorization.api.web.login.LoginChainCustomizer;
22  import com.irurueta.authserver.authorization.domain.mfa.FactorStrength;
23  import lombok.RequiredArgsConstructor;
24  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
25  import org.springframework.security.web.AuthenticationEntryPoint;
26  
27  /**
28   * Applies the multi-factor policy to the login/UI chain: the consent page needs the factors the policy asks for, and a
29   * user who lacks a registered second factor is sent to the second-factor page instead of getting a 403.
30   */
31  @RequiredArgsConstructor
32  public class MfaLoginChainCustomizer implements LoginChainCustomizer {
33  
34      /** Manager that applies the policy. */
35      private final MfaAuthorizationManager authorizationManager;
36  
37      /** Registry of the known factors. */
38      private final AuthenticationFactorRegistry factorRegistry;
39  
40      /** Entry point that sends the user to the second-factor page. */
41      private final AuthenticationEntryPoint secondFactorEntryPoint;
42  
43      /**
44       * Customizes the login/UI chain.
45       *
46       * @param http security builder of the chain.
47       */
48      @Override
49      public void customize(final HttpSecurity http) {
50          http.authorizeHttpRequests(authorize -> authorize.requestMatchers(CONSENT_PATH).access(authorizationManager));
51          http.exceptionHandling(exceptions -> factorRegistry.withStrength(
52                  FactorStrength.SECOND).forEach(factor -> exceptions
53                  .defaultDeniedHandlerForMissingAuthority(secondFactorEntryPoint, factor.authority())));
54      }
55  }
