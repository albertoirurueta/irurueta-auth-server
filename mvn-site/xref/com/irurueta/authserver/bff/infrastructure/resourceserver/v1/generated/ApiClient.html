804     /**
805      * Build cookie header. Keeps a single value per cookie (as per <a href="https://tools.ietf.org/html/rfc6265#section-5.3">
806      * RFC6265 section 5.3</a>).
807      *
808      * @param cookies map all cookies
809      * @return header string for cookies.
810      */
811     protected String buildCookieHeader(MultiValueMap<String, String> cookies) {
812         final StringBuilder cookieValue = new StringBuilder();
813         String delimiter = "";
814         for (final Map.Entry<String, List<String>> entry : cookies.entrySet()) {
815             final String value = entry.getValue().get(entry.getValue().size() - 1);
816             cookieValue.append(String.format(java.util.Locale.ROOT, "%s%s=%s", delimiter, entry.getKey(), value));
817             delimiter = "; ";
818         }
819         return cookieValue.toString();
820     }
