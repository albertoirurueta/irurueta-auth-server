16  package com.irurueta.authserver.bff.infrastructure.resourceserver;
17  
18  import com.irurueta.authserver.bff.domain.documents.DocumentsStepUpRequiredException;
19  import java.time.Duration;
20  import java.util.HashMap;
21  import java.util.Locale;
22  import java.util.Map;
23  import java.util.Optional;
24  import java.util.regex.Pattern;
25  import org.springframework.http.HttpHeaders;
26  import org.springframework.stereotype.Component;
27  
28  /**
29   * Maps the {@code WWW-Authenticate} challenge of a {@code 401} answer of the resource server onto the domain: a bearer
30   * challenge with {@code error="insufficient_user_authentication"} (RFC 9470) becomes a
31   * {@link DocumentsStepUpRequiredException} carrying its {@code acr_values} and {@code max_age}. Any other challenge, or
32   * none, is not a step-up request.
33   */
34  @Component
35  public class StepUpChallengeMapper {
36  
37      /** Error code of a step-up challenge (RFC 9470, section 3). */
38      static final String INSUFFICIENT_USER_AUTHENTICATION = "insufficient_user_authentication";
39  
40      /** Authentication scheme of the challenge. */
41      private static final String BEARER = "bearer";
42  
43      /** Name of the error parameter of the challenge. */
44      private static final String ERROR = "error";
45  
46      /** Name of the parameter carrying the authentication context classes asked for. */
47      private static final String ACR_VALUES = "acr_values";
48  
49      /** Name of the parameter carrying the maximum age of the authentication, in seconds. */
50      private static final String MAX_AGE = "max_age";
51  
52      /** One auth-param of the challenge: a name and a quoted string (with escapes) or a bare token. */
53      private static final Pattern PARAMETER = Pattern.compile(
54              "([A-Za-z0-9_-]+)\\s*=\\s*(?:\"((?:[^\"\\\\]|\\\\.)*)\"|([^\\s,\"]+))");
55  
56      /** Group of {@link #PARAMETER} holding the parameter name. */
57      private static final int NAME_GROUP = 1;
58  
59      /** Group of {@link #PARAMETER} holding a quoted value, without its quotes. */
60      private static final int QUOTED_VALUE_GROUP = 2;
61  
62      /** Group of {@link #PARAMETER} holding a bare token value. */
63      private static final int TOKEN_VALUE_GROUP = 3;
64  
65      /** Escape sequence of a quoted string: a backslash and the character it protects. */
66      private static final Pattern QUOTED_PAIR = Pattern.compile("\\\\(.)");
67  
68      /**
69       * Reads the step-up request of a {@code 401} answer, if it is one.
70       *
71       * @param headers headers of the answer; {@code null} when it had none.
72       * @param cause   the failure to keep as the cause of the domain exception.
73       * @return the step-up request, or empty when the answer carries no step-up challenge.
74       */
75      public Optional<DocumentsStepUpRequiredException> toException(final HttpHeaders headers, final Throwable cause) {
76          if (headers == null) {
77              return Optional.empty();
78          }
79          return headers.getOrEmpty(HttpHeaders.WWW_AUTHENTICATE).stream()
80                  .filter(StepUpChallengeMapper::isBearer)
81                  .map(StepUpChallengeMapper::parameters)
82                  .filter(parameters -> INSUFFICIENT_USER_AUTHENTICATION.equals(parameters.get(ERROR)))
83                  .findFirst()
84                  .map(parameters -> new DocumentsStepUpRequiredException(
85                          "The resource server asks for a stronger or more recent authentication",
86                          parameters.get(ACR_VALUES), maxAgeOf(parameters.get(MAX_AGE)), cause));
87      }
88  
89      /**
90       * Tells whether a challenge is of the bearer scheme.
91       *
92       * @param challenge value of a {@code WWW-Authenticate} header.
93       * @return {@code true} when its scheme is {@code Bearer}, in any case.
94       */
95      private static boolean isBearer(final String challenge) {
96          final var trimmed = challenge.trim();
97          return trimmed.length() > BEARER.length() && trimmed.regionMatches(true, 0, BEARER, 0, BEARER.length())
98                  && Character.isWhitespace(trimmed.charAt(BEARER.length()));
99      }
100 
101     /**
102      * Reads the auth-params of a bearer challenge.
103      *
104      * @param challenge value of a {@code WWW-Authenticate} header of the bearer scheme.
105      * @return the parameters by lower-case name; a repeated parameter keeps its first value.
106      */
107     private static Map<String, String> parameters(final String challenge) {
108         final var parameters = new HashMap<String, String>();
109         final var matcher = PARAMETER.matcher(challenge.trim().substring(BEARER.length()));
110         while (matcher.find()) {
111             final var quoted = matcher.group(QUOTED_VALUE_GROUP);
112             final var value = quoted != null ? QUOTED_PAIR.matcher(quoted).replaceAll("$1")
113                     : matcher.group(TOKEN_VALUE_GROUP);
114             parameters.putIfAbsent(matcher.group(NAME_GROUP).toLowerCase(Locale.ROOT), value);
115         }
116         return parameters;
117     }
118 
119     /**
120      * Reads the {@code max_age} parameter.
121      *
122      * @param value the parameter, in seconds; {@code null} when absent.
123      * @return the maximum age, or {@code null} when absent, not a number or negative.
124      */
125     private static Duration maxAgeOf(final String value) {
126         if (value == null) {
127             return null;
128         }
129         try {
130             final var seconds = Long.parseLong(value);
131             return seconds < 0 ? null : Duration.ofSeconds(seconds);
132         } catch (final NumberFormatException e) {
133             return null;
134         }
135     }
136 }
