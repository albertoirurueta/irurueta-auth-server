17  package com.irurueta.authserver.bff.infrastructure.resourceserver;
18  
19  import com.irurueta.authserver.bff.infrastructure.resourceserver.v1.generated.ApiClient;
20  import com.irurueta.authserver.bff.infrastructure.resourceserver.v1.generated.api.DocumentsApi;
21  import org.springframework.boot.context.properties.EnableConfigurationProperties;
22  import org.springframework.context.annotation.Bean;
23  import org.springframework.context.annotation.Configuration;
24  import org.springframework.http.HttpHeaders;
25  import org.springframework.security.oauth2.client.ClientAuthorizationException;
26  import org.springframework.security.oauth2.client.OAuth2AuthorizedClientManager;
27  import org.springframework.security.oauth2.client.web.client.OAuth2ClientHttpRequestInterceptor;
28  import org.springframework.security.oauth2.core.OAuth2Error;
29  import org.springframework.security.oauth2.core.OAuth2ErrorCodes;
30  import org.springframework.web.client.RestClient;
31  
32  /**
33   * Builds the client of the resource server: the {@link RestClient} that carries the access token of the signed-in
34   * user, and the {@link DocumentsApi} generated from the resource server's contract on top of it. The
35   * {@link OAuth2AuthorizedClientManager} is the one the web adapter's security configuration declares; this module
36   * only depends on its type.
37   */
38  @Configuration(proxyBeanMethods = false)
39  @EnableConfigurationProperties(ResourceServerProperties.class)
40  public class ResourceServerClientConfiguration {
41  
42      /** Registration id of the OAuth2 client whose access token is sent to the resource server. */
43      public static final String REGISTRATION_ID = "bff";
44  
45      // tag::bff-relay[]
46      /**
47       * Builds the REST client of the resource server. Its interceptor asks the authorized client manager for the access
48       * token of the user of the current request (renewing it with the refresh token when it expired) and sets it as the
49       * bearer token of the outgoing request; a call whose session has no token is never sent. When the token cannot be
50       * obtained or renewed it throws an {@link org.springframework.security.oauth2.core.OAuth2AuthorizationException},
51       * which {@link RestDocumentGateway} reports as {@link
52       * com.irurueta.authserver.bff.domain.documents.DocumentsUnauthorizedException}, answered 401 so the SPA signs in
53       * again.
54       *
55       * @param builder the builder of REST clients, the one test slices bind a mock server to.
56       * @param manager the manager of the authorized clients of the session.
57       * @return the REST client of the resource server.
58       */
59      @Bean
60      RestClient resourceServerRestClient(final RestClient.Builder builder, final OAuth2AuthorizedClientManager manager) {
61          final var interceptor = new OAuth2ClientHttpRequestInterceptor(manager);
62          interceptor.setClientRegistrationIdResolver(request -> REGISTRATION_ID);
63          // The OAuth2 interceptor lets the call through unauthenticated when the session has no authorized client;
64          // the BFF must never do that, so the call stops here and the SPA gets a 401.
65          return builder.requestInterceptor(interceptor).requestInterceptor((request, body, execution) -> {
66              if (!request.getHeaders().containsHeader(HttpHeaders.AUTHORIZATION)) {
67                  throw new ClientAuthorizationException(new OAuth2Error(OAuth2ErrorCodes.ACCESS_DENIED,
68                          "The session has no access token", null), REGISTRATION_ID);
69              }
70              return execution.execute(request, body);
71          }).build();
72      }
73      // end::bff-relay[]
74  
75      // tag::bff-documents-client[]
76      /**
77       * Builds the documents client generated from the resource server's contract, on top of the REST client that
78       * carries the user's token (so the generated client's own authentication settings stay unused).
79       *
80       * @param resourceServerRestClient the REST client of the resource server.
81       * @param properties               where the resource server is.
82       * @return the generated documents client.
83       */
84      @Bean
85      DocumentsApi resourceServerDocumentsApi(final RestClient resourceServerRestClient,
86              final ResourceServerProperties properties) {
87          final var apiClient = new ApiClient(resourceServerRestClient)
88                  .setBasePath(properties.baseUri().toString().replaceFirst("/+$", ""));
89          return new DocumentsApi(apiClient);
90      }
91      // end::bff-documents-client[]
92  }
