17  package com.irurueta.authserver.bff.api.web.login;
18  
19  import java.util.regex.Pattern;
20  import lombok.RequiredArgsConstructor;
21  import org.springframework.boot.context.properties.EnableConfigurationProperties;
22  import org.springframework.stereotype.Component;
23  
24  // tag::bff-return-to-validator[]
25  /**
26   * Decides whether a {@code returnTo} value asked by the browser is a safe place to send it back to after the login.
27   * An open redirect would let a crafted link bounce a freshly signed-in user to a hostile site, so only a relative
28   * path, listed in {@code bff.login.allowed-return-paths}, is ever accepted.
29   */
30  @Component
31  @RequiredArgsConstructor
32  @EnableConfigurationProperties(BffLoginProperties.class)
33  public class ReturnToValidator {
34  
35      /** An URL scheme at the start (for example {@code javascript:}), also when hidden in the first segment. */
36      private static final Pattern SCHEME = Pattern.compile("^[a-zA-Z][a-zA-Z0-9+.-]*:");
37  
38      /** Allowed paths. */
39      private final BffLoginProperties properties;
40  
41      /**
42       * Checks a {@code returnTo} value.
43       *
44       * @param returnTo the value to check; may be {@code null}.
45       * @return {@code true} when it is a relative path allowed by the configuration, without scheme, authority,
46       *         backslash, dot segment or control character.
47       */
48      public boolean isAllowed(final String returnTo) {
49          if (returnTo == null || !returnTo.startsWith("/") || returnTo.startsWith("//")
50                  || SCHEME.matcher(returnTo).find() || returnTo.indexOf('\\') >= 0
51                  || returnTo.chars().anyMatch(Character::isISOControl)) {
52              return false;
53          }
54          final var path = pathOf(returnTo);
55          if (path.contains("/../") || path.endsWith("/..") || path.contains("/./")
56                  || path.contains("%")) {
57              return false;
58          }
59          return properties.allowedReturnPaths().stream()
60                  .anyMatch(allowed -> path.equals(allowed) || isPrefix(allowed) && path.startsWith(allowed));
61      }
62  
63      /**
64       * Tells whether an allow-list entry covers the paths below it: it ends with {@code /} but is not the root, which
65       * would allow every path.
66       *
67       * @param allowed an allow-list entry.
68       * @return {@code true} for a prefix entry such as {@code /orders/}.
69       */
70      private static boolean isPrefix(final String allowed) {
71          return allowed.length() > 1 && allowed.endsWith("/");
72      }
73  
74      /**
75       * Strips the query string and the fragment.
76       *
77       * @param returnTo a value starting with {@code /}.
78       * @return its path.
79       */
80      private static String pathOf(final String returnTo) {
81          var end = returnTo.length();
82          final var query = returnTo.indexOf('?');
83          if (query >= 0) {
84              end = query;
85          }
86          final var fragment = returnTo.indexOf('#');
87          if (fragment >= 0 && fragment < end) {
88              end = fragment;
89          }
90          return returnTo.substring(0, end);
91      }
92  }
