44  /**
45   * Signs users in with OAuth2 / OpenID Connect login and out with the RP-initiated logout of the authorization server.
46   * API calls from the SPA get a 401 instead of a redirect, and every state-changing request must carry the CSRF token
47   * the SPA reads from the {@code XSRF-TOKEN} cookie. The tokens of the user stay in the HTTP session (persisted by
48   * Spring Session JDBC), never in the browser.
49   */
50  @Configuration(proxyBeanMethods = false)
51  public class SecurityConfig {
52  
53      /** Content security policy of the BFF: same-origin resources only (so {@code script-src 'self'}), no framing. */
54      static final String CONTENT_SECURITY_POLICY = "default-src 'self'; frame-ancestors 'none'";
55  
56      /** Page the browser is sent to when the sign-in fails (for example, the user denied the consent). */
57      static final String SIGN_IN_FAILURE_URL = "/?error=signin";
58  
59      /** Path of the logout, answered by Spring Security's logout filter (only to {@code POST}). */
60      static final String LOGOUT_PATH = "/bff/logout";
61  
62      // tag::bff-security[]
63      /**
64       * Builds the BFF filter chain.
65       *
66       * @param http                 the HTTP security builder.
67       * @param clientRepository     the registered OAuth2 clients.
68       * @param loginSuccessHandler  sends the browser back to the {@code returnTo} of the login once signed in.
69       * @param logoutSuccessHandler redirects the browser to the RP-initiated logout of the authorization server.
70       * @return the security filter chain.
71       * @throws Exception if the chain cannot be built.
72       */
73      @Bean
74      SecurityFilterChain bffFilterChain(final HttpSecurity http, final ClientRegistrationRepository clientRepository,
75              final ReturnToAuthenticationSuccessHandler loginSuccessHandler,
76              final OidcClientInitiatedLogoutSuccessHandler logoutSuccessHandler) throws Exception {
77          final var defaultResolver = new DefaultOAuth2AuthorizationRequestResolver(clientRepository,
78                  "/oauth2/authorization");
79          // The auth server requires PKCE; Spring only enables it by default for public clients.
80          defaultResolver.setAuthorizationRequestCustomizer(OAuth2AuthorizationRequestCustomizers.withPkce());
81          // A login started for step-up (GET /bff/login with acrValues / maxAge) also asks for acr_values and max_age.
82          final var resolver = new StepUpAuthorizationRequestResolver(defaultResolver);
83  
84          http.authorizeHttpRequests(authorize -> authorize
85                          .requestMatchers("/", "/index.html", "/favicon.ico", "/assets/**", "/static/**", "/error",
86                                  "/bff/login", "/actuator/health/**").permitAll()
87                          .anyRequest().authenticated())
88                  // The XSRF-TOKEN cookie is readable by the script (not HttpOnly) and the raw token is accepted from
89                  // the X-XSRF-TOKEN header or, for the sign-out form, from the _csrf parameter. The plain handler, not
90                  // csrf.spa()'s XOR one: masking against BREACH protects a token rendered in a compressed response body,
91                  // and the BFF never renders it in a body, only in the cookie; the XOR handler would also reject the
92                  // raw cookie value the sign-out form posts.
93                  .csrf(csrf -> csrf
94                          .csrfTokenRepository(CookieCsrfTokenRepository.withHttpOnlyFalse())
95                          .csrfTokenRequestHandler(new CsrfTokenRequestAttributeHandler()))
96                  // The SPA is made of same-origin files only: no inline script, no third-party content, no framing.
97                  .headers(headers -> headers.contentSecurityPolicy(
98                          csp -> csp.policyDirectives(CONTENT_SECURITY_POLICY)))
99                  .addFilterAfter(new CsrfCookieFilter(), CsrfFilter.class)
100                 .exceptionHandling(ex -> ex.defaultAuthenticationEntryPointFor(
101                         new HttpStatusEntryPoint(HttpStatus.UNAUTHORIZED),
102                         new OrRequestMatcher(PathPatternRequestMatcher.withDefaults().matcher("/bff/**"),
103                                 PathPatternRequestMatcher.withDefaults().matcher("/api/**"))))
104                 .oauth2Login(login -> login
105                         .authorizationEndpoint(endpoint -> endpoint.authorizationRequestResolver(resolver))
106                         // A refused consent or any other failed sign-in lands on the SPA, which tells the user.
107                         // A handler rather than failureUrl: that would make the generated login page answer it.
108                         .failureHandler(new SimpleUrlAuthenticationFailureHandler(SIGN_IN_FAILURE_URL))
109                         // Back to the returnTo validated by GET /bff/login, never to a request saved elsewhere.
110                         .successHandler(loginSuccessHandler))
111                 // tag::bff-logout[]
112                 // POST /bff/logout (CSRF protected) runs the logout handlers (the XSRF-TOKEN cookie is cleared, the
113                 // security context too, and the session is invalidated), then the success handler answers 302 to the
114                 // end_session_endpoint of the authorization server: the browser follows it as a top-level navigation,
115                 // so the id_token_hint it carries never reaches a script.
116                 .logout(logout -> logout
117                         .logoutRequestMatcher(PathPatternRequestMatcher.withDefaults()
118                                 .matcher(HttpMethod.POST, LOGOUT_PATH))
119                         .logoutSuccessHandler(logoutSuccessHandler));
120                 // end::bff-logout[]
121         return http.build();
122     }
123     // end::bff-security[]
124 
125     // tag::bff-authorized-client[]
126     /**
127      * Keeps the authorized clients (the tokens of the signed-in user) in the HTTP session, which Spring Session JDBC
128      * persists, so no table of its own is needed and the tokens never reach the browser.
129      *
130      * @return the repository of authorized clients.
131      */
132     @Bean
133     OAuth2AuthorizedClientRepository authorizedClientRepository() {
134         return new HttpSessionOAuth2AuthorizedClientRepository();
135     }
136 
137     /**
138      * Builds the manager that obtains the access token of the signed-in user and renews it with its refresh token when
139      * it expires.
140      *
141      * @param clientRepository           the registered OAuth2 clients.
142      * @param authorizedClientRepository where the authorized clients are kept.
143      * @return the authorized client manager, with the {@code authorization_code} and {@code refresh_token} providers.
144      */
145     @Bean
146     OAuth2AuthorizedClientManager authorizedClientManager(final ClientRegistrationRepository clientRepository,
147             final OAuth2AuthorizedClientRepository authorizedClientRepository) {
148         final var manager = new DefaultOAuth2AuthorizedClientManager(clientRepository, authorizedClientRepository);
149         manager.setAuthorizedClientProvider(OAuth2AuthorizedClientProviderBuilder.builder()
150                 .authorizationCode()
151                 .refreshToken()
152                 .build());
153         return manager;
154     }
155     // end::bff-authorized-client[]
156 }
