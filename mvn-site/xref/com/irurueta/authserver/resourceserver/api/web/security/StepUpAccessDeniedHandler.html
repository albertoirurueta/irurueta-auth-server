17  package com.irurueta.authserver.resourceserver.api.web.security;
18  
19  import jakarta.servlet.ServletException;
20  import jakarta.servlet.http.HttpServletRequest;
21  import jakarta.servlet.http.HttpServletResponse;
22  import java.io.IOException;
23  import lombok.RequiredArgsConstructor;
24  import org.springframework.http.HttpHeaders;
25  import org.springframework.http.HttpStatus;
26  import org.springframework.security.access.AccessDeniedException;
27  import org.springframework.security.authorization.AuthorizationDeniedException;
28  import org.springframework.security.web.access.AccessDeniedHandler;
29  
30  /**
31   * Answers a {@link StepUpRequiredDecision} with {@code 401} and the RFC 9470 challenge
32   * ({@code error="insufficient_user_authentication"} with the required {@code acr_values} and {@code max_age});
33   * every other denial goes to the delegate, the RFC 6750 handler.
34   */
35  // tag::resource-server-step-up-handler[]
36  @RequiredArgsConstructor
37  public class StepUpAccessDeniedHandler implements AccessDeniedHandler {
38  
39      /** Handler of every denial that is not a step-up requirement. */
40      private final AccessDeniedHandler delegate;
41  
42      /** Step-up policy, whose values are advertised in the challenge. */
43      private final StepUpProperties properties;
44  
45      @Override
46      public void handle(final HttpServletRequest request, final HttpServletResponse response,
47              final AccessDeniedException accessDeniedException) throws IOException, ServletException {
48          if (accessDeniedException instanceof AuthorizationDeniedException denied
49                  && denied.getAuthorizationResult() instanceof StepUpRequiredDecision) {
50              response.setStatus(HttpStatus.UNAUTHORIZED.value());
51              response.setHeader(HttpHeaders.WWW_AUTHENTICATE, "Bearer error=\"insufficient_user_authentication\", "
52                      + "acr_values=\"" + properties.acrValues() + "\", max_age=\""
53                      + properties.maxAge().toSeconds() + "\"");
54              return;
55          }
56          delegate.handle(request, response, accessDeniedException);
57      }
58  }
59  // end::resource-server-step-up-handler[]
