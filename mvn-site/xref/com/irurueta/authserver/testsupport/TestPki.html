35  /**
36   * A small public key infrastructure generated at run time for the tests of TLS client authentication: a certificate
37   * authority, and server and client certificates it signs, each kept in a PKCS#12 key store inside one temporary
38   * directory that {@link #close()} deletes. It drives {@code keytool} (see {@link Keytool}); nothing is ever kept in
39   * the repository.
40   */
41  public final class TestPki implements AutoCloseable {
42  
43      /** Alias of the authority's key. */
44      private static final String CA_ALIAS = "ca";
45  
46      /** Alias of the key of an issued entity. */
47      private static final String ENTITY_ALIAS = "entity";
48  
49      /** Type of the key stores. */
50      private static final String KEY_STORE_TYPE = "PKCS12";
51  
52      /** Number of random bytes of the password of the key stores. */
53      private static final int PASSWORD_BYTES = 16;
54  
55      /** Days the issued certificates are valid. */
56      private static final String VALIDITY_DAYS = "2";
57  
58      /** The {@code keytool} option that names a key store. */
59      private static final String KEYSTORE = "-keystore";
60  
61      /** The {@code keytool} option that names the file with a key store password. */
62      private static final String STORE_PASSWORD_FILE = "-storepass:file";
63  
64      /** The {@code keytool} option that names the alias of an entry. */
65      private static final String ALIAS = "-alias";
66  
67      /** The {@code keytool} option that names the input file. */
68      private static final String INPUT = "-file";
69  
70      /** Source of the password. */
71      private static final SecureRandom RANDOM = new SecureRandom();
72  
73      /** Temporary directory that holds every file. */
74      private final Path directory;
75  
76      /** Password of every key store. */
77      private final String password;
78  
79      /** File that holds {@link #password}. */
80      private final Path passwordFile;
81  
82      /** Key store of the authority. */
83      private final Path authorityKeyStore;
84  
85      /** Certificate of the authority. */
86      private final X509Certificate authority;
87  
88      /** Path of the authority's certificate in PEM. */
89      private final Path authorityPem;
90  
91      /** Number of entities issued, to name their files. */
92      private int issued;
93  
94      /**
95       * Describes an infrastructure whose files already exist.
96       *
97       * @param directory temporary directory that holds every file.
98       * @param password  password of every key store.
99       * @param authority certificate of the authority.
100      */
101     private TestPki(final Path directory, final String password, final X509Certificate authority) {
102         this.directory = directory;
103         this.password = password;
104         passwordFile = directory.resolve("password");
105         authorityKeyStore = directory.resolve("ca.p12");
106         authorityPem = directory.resolve("ca.pem");
107         this.authority = authority;
108     }
109 
110     /**
111      * Creates the infrastructure with a new certificate authority.
112      *
113      * @param authorityName common name of the authority.
114      * @return the infrastructure; the caller closes it.
115      */
116     public static TestPki create(final String authorityName) {
117         final var bytes = new byte[PASSWORD_BYTES];
118         RANDOM.nextBytes(bytes);
119         final var password = HexFormat.of().formatHex(bytes);
120         Path directory = null;
121         try {
122             directory = Files.createTempDirectory("test-pki");
123             Files.writeString(directory.resolve("password"), password);
124             final var keyStore = directory.resolve("ca.p12");
125             final var passwordFile = directory.resolve("password").toString();
126             Keytool.run(List.of("-genkeypair", ALIAS, CA_ALIAS, "-keyalg", "EC", "-groupname", "secp256r1",
127                     "-validity", VALIDITY_DAYS, "-dname", "CN=" + authorityName, "-ext", "bc:c",
128                     "-storetype", KEY_STORE_TYPE, KEYSTORE, keyStore.toString(), STORE_PASSWORD_FILE, passwordFile));
129             final var pem = directory.resolve("ca.pem");
130             Keytool.run(List.of("-exportcert", ALIAS, CA_ALIAS, "-rfc", KEYSTORE, keyStore.toString(),
131                     STORE_PASSWORD_FILE, passwordFile, "-file", pem.toString()));
132             return new TestPki(directory, password, readCertificate(pem));
133         } catch (final IOException | GeneralSecurityException e) {
134             Keytool.delete(directory);
135             throw new IllegalStateException("Cannot create the test infrastructure", e);
136         }
137     }

139     /**
140      * Issues the certificate of a TLS server, valid for {@code localhost} and the loopback address.
141      *
142      * @param commonName common name of the subject.
143      * @return the server's key store and certificate.
144      */
145     public Identity issueServer(final String commonName) {
146         return issue(commonName, List.of("-ext", "san=dns:localhost,ip:127.0.0.1", "-ext", "eku=serverAuth"));
147     }

149     /**
150      * Issues the certificate of a client, signed by the authority.
151      *
152      * @param commonName common name of the subject.
153      * @return the client's key store and certificate.
154      */
155     public Identity issueClient(final String commonName) {
156         return issue(commonName, List.of("-ext", "eku=clientAuth"));
157     }

159     /**
160      * Gives the TLS context of a client that presents a certificate and trusts the authority.
161      *
162      * @param client the client's identity, {@code null} for a client with no certificate.
163      * @return the context.
164      */
165     public SSLContext clientContext(final Identity client) {
166         try {
167             final var trust = KeyStore.getInstance(KEY_STORE_TYPE);
168             trust.load(null, null);
169             trust.setCertificateEntry(CA_ALIAS, authority);
170             final var trustManagers = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
171             trustManagers.init(trust);
172             final var context = SSLContext.getInstance("TLS");
173             if (client == null) {
174                 context.init(null, trustManagers.getTrustManagers(), null);
175             } else {
176                 final var keys = KeyStore.getInstance(KEY_STORE_TYPE);
177                 try (InputStream in = Files.newInputStream(client.keyStore())) {
178                     keys.load(in, client.password().toCharArray());
179                 }
180                 final var keyManagers = KeyManagerFactory.getInstance(KeyManagerFactory.getDefaultAlgorithm());
181                 keyManagers.init(keys, client.password().toCharArray());
182                 context.init(keyManagers.getKeyManagers(), trustManagers.getTrustManagers(), null);
183             }
184             return context;
185         } catch (final IOException e) {
186             throw new UncheckedIOException("Cannot read the test key store", e);
187         } catch (final GeneralSecurityException e) {
188             throw new IllegalStateException("Cannot build the test TLS context", e);
189         }
190     }
191 
192     /**
193      * Gives the identity of a client with a self-signed certificate that expired days ago.
194      *
195      * @param commonName common name of the subject.
196      * @return the client's key store and certificate.
197      */
198     public Identity issueExpiredClient(final String commonName) {
199         final var name = nextName();
200         final var keyStore = directory.resolve(name + ".p12");
201         try {
202             Keytool.run(List.of("-genkeypair", ALIAS, ENTITY_ALIAS, "-keyalg", "EC", "-groupname", "secp256r1",
203                     "-startdate", "-5d", "-validity", VALIDITY_DAYS, "-dname", "CN=" + commonName,
204                     "-storetype", KEY_STORE_TYPE, KEYSTORE, keyStore.toString(),
205                     STORE_PASSWORD_FILE, passwordFile.toString()));
206             return identityOf(keyStore);
207         } catch (final IOException | GeneralSecurityException e) {
208             throw new IllegalStateException("Cannot issue the expired test certificate", e);
209         }
210     }
211 
212     @Override
213     public void close() {
214         Keytool.delete(directory);
215     }

217     /**
218      * Issues an entity signed by the authority.
219      *
220      * @param commonName common name of the subject.
221      * @param extensions the {@code keytool} extension options the certificate gets.
222      * @return the entity's key store and certificate.
223      */
224     private Identity issue(final String commonName, final List<String> extensions) {
225         final var name = nextName();
226         final var keyStore = directory.resolve(name + ".p12");
227         final var request = directory.resolve(name + ".csr");
228         final var reply = directory.resolve(name + ".pem");
229         try {
230             Keytool.run(List.of("-genkeypair", ALIAS, ENTITY_ALIAS, "-keyalg", "EC", "-groupname", "secp256r1",
231                     "-validity", VALIDITY_DAYS, "-dname", "CN=" + commonName, "-storetype", KEY_STORE_TYPE,
232                     KEYSTORE, keyStore.toString(), STORE_PASSWORD_FILE, passwordFile.toString()));
233             Keytool.run(List.of("-certreq", ALIAS, ENTITY_ALIAS, KEYSTORE, keyStore.toString(),
234                     STORE_PASSWORD_FILE, passwordFile.toString(), "-file", request.toString()));
235             return identityOf(sign(keyStore, request, reply, extensions));
236         } catch (final IOException | GeneralSecurityException e) {
237             throw new IllegalStateException("Cannot issue the test certificate", e);
238         }
239     }
240 
241     /**
242      * Has the authority sign a request and installs the answer in the entity's key store.
243      *
244      * @param keyStore   key store of the entity.
245      * @param request    certificate request of the entity.
246      * @param reply      file the signed certificate is written to.
247      * @param extensions the {@code keytool} extension options the certificate gets.
248      * @return the entity's key store.
249      * @throws IOException if {@code keytool} fails.
250      */
251     private Path sign(final Path keyStore, final Path request, final Path reply, final List<String> extensions)
252             throws IOException {
253         final var gencert = new ArrayList<>(List.of("-gencert", ALIAS, CA_ALIAS, "-rfc", "-validity", VALIDITY_DAYS,
254                 KEYSTORE, authorityKeyStore.toString(), STORE_PASSWORD_FILE, passwordFile.toString(),
255                 "-infile", request.toString(), "-outfile", reply.toString()));
256         gencert.addAll(extensions);
257         Keytool.run(gencert);
258         Keytool.run(List.of("-importcert", ALIAS, CA_ALIAS, INPUT, authorityPem.toString(), KEYSTORE,
259                 keyStore.toString(), STORE_PASSWORD_FILE, passwordFile.toString(), "-noprompt"));
260         Keytool.run(List.of("-importcert", ALIAS, ENTITY_ALIAS, INPUT, reply.toString(), KEYSTORE,
261                 keyStore.toString(), STORE_PASSWORD_FILE, passwordFile.toString()));
262         return keyStore;
263     }
264 
265     /**
266      * Reads an identity from its key store.
267      *
268      * @param keyStore the key store.
269      * @return the identity.
270      * @throws IOException              if the key store cannot be read.
271      * @throws GeneralSecurityException if the key store cannot be loaded.
272      */
273     private Identity identityOf(final Path keyStore) throws IOException, GeneralSecurityException {
274         final var store = KeyStore.getInstance(KEY_STORE_TYPE);
275         try (InputStream in = Files.newInputStream(keyStore)) {
276             store.load(in, password.toCharArray());
277         }
278         return new Identity(keyStore, password,
279                 TestCertificates.of((X509Certificate) store.getCertificate(ENTITY_ALIAS)));
280     }
281 
282     /**
283      * Names the files of the next entity.
284      *
285      * @return the name.
286      */
287     private String nextName() {
288         issued++;
289         return "entity-" + issued;
290     }
291 
292     /**
293      * Reads a PEM certificate.
294      *
295      * @param pem the file.
296      * @return the certificate.
297      * @throws IOException              if the file cannot be read.
298      * @throws GeneralSecurityException if it is not a certificate.
299      */
300     private static X509Certificate readCertificate(final Path pem) throws IOException, GeneralSecurityException {
301         try (InputStream in = Files.newInputStream(pem)) {
302             return (X509Certificate) CertificateFactory.getInstance("X.509").generateCertificate(in);
303         }
304     }
