117     /**
118      * Verifies that the bearer token is sent to the resource server while the browser's credentials are not, and that
119      * neither the token nor anything the resource server set in its headers reaches the browser.
120      *
121      * @throws Exception if the request cannot be performed.
122      */
123     @Test
124     void listDocuments_whenSignedIn_sendsTheBearerUpstreamAndKeepsItFromTheBrowser() throws Exception {
125         server.expect(requestTo(DOCUMENTS))
126                 .andExpect(method(HttpMethod.GET))
127                 .andExpect(header("Authorization", "Bearer " + ACCESS_TOKEN))
128                 .andExpect(headerDoesNotExist("Cookie"))
129                 .andExpect(headerDoesNotExist("X-XSRF-TOKEN"))
130                 .andRespond(withSuccess("[{\"id\":\"1\",\"owner\":\"user\",\"title\":\"T\",\"content\":\"C\"}]",
131                         MediaType.APPLICATION_JSON)
132                         .header("Set-Cookie", "upstream=secret")
133                         .header("Cache-Control", "max-age=3600"));
134 
135         final var result = mockMvc.perform(get(BFF_DOCUMENTS)
136                         .header("Cookie", "other=1").header("X-XSRF-TOKEN", "browser-csrf")
137                         .header("Authorization", "Bearer browser-supplied")
138                         .with(oidcLogin()).with(signedInClient()))
139                 .andExpect(status().isOk())
140                 .andExpect(content().json("[{\"id\":\"1\",\"owner\":\"user\",\"title\":\"T\",\"content\":\"C\"}]"))
141                 .andReturn().getResponse();
142 
143         server.verify();
144         assertFalse(result.getHeaders("Set-Cookie").stream().anyMatch(value -> value.contains("upstream")));
145         assertEquals(List.of("no-cache, no-store, max-age=0, must-revalidate"), result.getHeaders("Cache-Control"));
146         assertNull(result.getHeader("Authorization"));
147         assertFalse(result.getContentAsString().contains(ACCESS_TOKEN));
148     }

198     /**
199      * Verifies that a step-up challenge of the resource server (RFC 9470) is answered 401 with the values the SPA
200      * passes to the login, and without the challenge itself, to a client that accepts the JSON results and the
201      * problem details, as the SPA does.
202      *
203      * @throws Exception if the request cannot be performed.
204      */
205     @Test
206     void deleteDocument_whenResourceServerAsksForStepUp_answersStepUpRequiredProblem() throws Exception {
207         server.expect(requestTo(DOCUMENT)).andRespond(withStatus(HttpStatus.UNAUTHORIZED).header("WWW-Authenticate",
208                 "Bearer error=\"insufficient_user_authentication\", error_description=\"step-up\", "
209                         + "acr_values=\"" + STEP_UP_ACR + "\", max_age=\"" + STEP_UP_MAX_AGE + "\""));
210 
211         final var result = mockMvc.perform(delete(BFF_DOCUMENT).with(csrf()).with(oidcLogin())
212                         .with(signedInClient()).header("Accept", ACCEPT_JSON_OR_PROBLEM))
213                 .andExpect(status().isUnauthorized())
214                 .andExpect(content().contentType(PROBLEM_JSON))
215                 .andExpect(jsonPath("$.type").value("urn:problem:documents:step-up-required"))
216                 .andExpect(jsonPath("$.acrValues").value(STEP_UP_ACR))
217                 .andExpect(jsonPath("$.maxAge").value(STEP_UP_MAX_AGE))
218                 .andReturn().getResponse();
219 
220         assertNull(result.getHeader("WWW-Authenticate"));
221     }

223     /**
224      * Verifies that a client that only accepts JSON can delete, and gets the step-up problem with its status instead of
225      * {@code 406}: the operation also declares its problems as JSON, and Spring writes a problem detail as
226      * {@code application/problem+json} when the client's types do not include it.
227      *
228      * @throws Exception if a request cannot be performed.
229      */
230     @Test
231     void deleteDocument_whenClientOnlyAcceptsJson_negotiatesInsteadOfAnsweringNotAcceptable() throws Exception {
232         server.expect(requestTo(DOCUMENT)).andRespond(withStatus(HttpStatus.NO_CONTENT));
233         server.expect(requestTo(DOCUMENT)).andRespond(withStatus(HttpStatus.UNAUTHORIZED).header("WWW-Authenticate",
234                 "Bearer error=\"insufficient_user_authentication\", acr_values=\"" + STEP_UP_ACR + "\""));
235 
236         mockMvc.perform(delete(BFF_DOCUMENT).with(csrf()).with(oidcLogin()).with(signedInClient())
237                         .accept(MediaType.APPLICATION_JSON))
238                 .andExpect(status().isNoContent());
239         mockMvc.perform(delete(BFF_DOCUMENT).with(csrf()).with(oidcLogin()).with(signedInClient())
240                         .accept(MediaType.APPLICATION_JSON))
241                 .andExpect(status().isUnauthorized())
242                 .andExpect(content().contentType(PROBLEM_JSON))
243                 .andExpect(jsonPath("$.type").value("urn:problem:documents:step-up-required"))
244                 .andExpect(jsonPath("$.acrValues").value(STEP_UP_ACR));
245     }

356     /**
357      * Builds the post-processor that puts an authorized client with the test access token in the session.
358      *
359      * @return the request post-processor.
360      */
361     private static RequestPostProcessor signedInClient() {
362         final var now = Instant.now();
363         return oauth2Client("bff").accessToken(new OAuth2AccessToken(OAuth2AccessToken.TokenType.BEARER, ACCESS_TOKEN,
364                 now, now.plus(TOKEN_LIFETIME)));
365     }
