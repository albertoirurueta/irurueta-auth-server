57      private static final long EXPIRES_IN = 300L;

62      /** The device code. */
63      private static final String DEVICE_CODE = "device-code";

65      /** The user code. */
66      private static final String USER_CODE = "ABCD-EFGH";

113     /**
114      * Pending answers are retried after the same interval until the tokens arrive.
115      *
116      * @throws InterruptedException never: the sleeper is a mock.
117      */
118     @Test
119     void run_whenApprovedAfterPending_printsLinesAndWaitsTheInterval() throws InterruptedException {
120         stubAuthorization(COMPLETE_URI, START, START.plusSeconds(INTERVAL), START.plusSeconds(2 * INTERVAL));
121         when(client.pollToken(DEVICE_CODE)).thenReturn(new PollOutcome.Pending(), new PollOutcome.Granted(token()));
122 
123         cli.run();
124 
125         final var order = inOrder(output, sleeper);
126         order.verify(output).line("User code: " + USER_CODE);
127         order.verify(output).line("Verification URI: " + VERIFICATION_URI);
128         order.verify(output).line("Verification URI (complete): " + COMPLETE_URI);
129         order.verify(sleeper, times(2)).sleep(Duration.ofSeconds(INTERVAL));
130         order.verify(output).line("Device flow completed: token_type=Bearer expires_in=" + TOKEN_EXPIRES_IN
131                 + " scope=profile documents.read refresh_token=true");
132         verify(output, never()).line("Access token: " + ACCESS_TOKEN);
133     }

135     /**
136      * Every {@code slow_down} makes the interval five seconds longer.
137      *
138      * @throws InterruptedException never: the sleeper is a mock.
139      */
140     @Test
141     void run_whenServerSaysSlowDown_increasesTheInterval() throws InterruptedException {
142         stubAuthorization(COMPLETE_URI, START, START.plusSeconds(INTERVAL), START.plusSeconds(2 * INTERVAL),
143                 START.plusSeconds(POLLS * INTERVAL));
144         when(client.pollToken(DEVICE_CODE)).thenReturn(new PollOutcome.SlowDown(), new PollOutcome.SlowDown(),
145                 new PollOutcome.Granted(token()));
146 
147         cli.run();
148 
149         final var order = inOrder(sleeper);
150         order.verify(sleeper).sleep(Duration.ofSeconds(INTERVAL));
151         order.verify(sleeper).sleep(Duration.ofSeconds(INTERVAL + DeviceFlowCli.SLOW_DOWN_INCREMENT_SECONDS));
152         order.verify(sleeper).sleep(Duration.ofSeconds(INTERVAL + 2 * DeviceFlowCli.SLOW_DOWN_INCREMENT_SECONDS));
153     }

182     /**
183      * Once the deadline passes the token endpoint is not called any more.
184      */
185     @Test
186     void run_whenDeadlinePasses_throwsWithoutPolling() {
187         stubAuthorization(COMPLETE_URI, START, START.plusSeconds(EXPIRES_IN));

191         assertTrue(error.getMessage().contains("expired"));

234     /**
235      * An interruption stops the flow and keeps the interrupt flag.
236      *
237      * @throws InterruptedException never: the sleeper is a mock.
238      */
239     @Test
240     void run_whenInterrupted_throwsAndRestoresTheInterruptFlag() throws InterruptedException {
241         stubAuthorization(COMPLETE_URI, START);
242         doThrow(new InterruptedException("stop")).when(sleeper).sleep(Duration.ofSeconds(INTERVAL));
243 
244         try {
245             assertThrows(DeviceFlowException.class, () -> cli.run());
246             assertTrue(Thread.currentThread().isInterrupted());
247         } finally {
248             assertTrue(Thread.interrupted());
249         }
250         assertFalse(Thread.currentThread().isInterrupted());
251     }

253     /**
254      * A failing device authorization request ends the flow before anything is printed.
255      */
256     @Test
257     void run_whenAuthorizationRequestFails_propagatesTheException() {
258         final var failure = new DeviceFlowException("down");
259         when(client.requestDeviceAuthorization()).thenThrow(failure);
260 
261         final var error = assertThrows(DeviceFlowException.class, () -> cli.run());
262 
263         assertSame(failure, error);
264         verify(output, never()).line(anyString());
265     }

267     /**
268      * Stubs the device authorization and the successive readings of the clock.
269      *
270      * @param completeUri the complete verification URI, or {@code null}.
271      * @param first       the first reading, taken to compute the deadline.
272      * @param others      the following readings, one per poll.
273      */
274     private void stubAuthorization(final String completeUri, final Instant first, final Instant... others) {
275         when(client.requestDeviceAuthorization()).thenReturn(new DeviceAuthorization(DEVICE_CODE, USER_CODE,
276                 VERIFICATION_URI, completeUri, EXPIRES_IN, INTERVAL));
277         when(clock.instant()).thenReturn(first, others);
278     }

280     /**
281      * Builds a token response.
282      *
283      * @return the tokens.
284      */
285     private static TokenResponse token() {
286         return new TokenResponse(ACCESS_TOKEN, "the-refresh-token", "Bearer", TOKEN_EXPIRES_IN,
287                 "profile documents.read");
288     }
