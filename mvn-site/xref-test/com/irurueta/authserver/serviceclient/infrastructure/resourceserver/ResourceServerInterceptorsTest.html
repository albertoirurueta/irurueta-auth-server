54  /**
55   * Unit tests of {@link ResourceServerInterceptors}.
56   */
57  @ExtendWith(MockitoExtension.class)
58  class ResourceServerInterceptorsTest {
59  
60      /** Registration id. */
61      private static final String REGISTRATION_ID = "service-client";
62  
63      /** URI called. */
64      private static final URI URI_CALLED = URI.create("https://rs.example.com/api/documents");
65  
66      /** Lifetime of the token, in seconds. */
67      private static final long LIFETIME_SECONDS = 300L;
68  
69      /** Registration of the client. */
70      private static final ClientRegistration REGISTRATION = ClientRegistration.withRegistrationId(REGISTRATION_ID)
71              .clientId("client")
72              .authorizationGrantType(AuthorizationGrantType.CLIENT_CREDENTIALS)
73              .tokenUri("https://auth.example.com/oauth2/token")
74              .build();
75  
76      /** Obtains the access token. */
77      @Mock
78      private OAuth2AuthorizedClientManager manager;
79  
80      /** Creates the DPoP proofs, when there are any. */
81      @Mock
82      private ObjectProvider<DPoPProofs> dpopProofs;
83  
84      /** The proofs. */
85      @Mock
86      private DPoPProofs proofs;
87  
88      /** Principal the token is cached under. */
89      @Mock
90      private Authentication principal;
91  
92      /** Next element of the chain. */
93      @Mock
94      private ClientHttpRequestExecution execution;
95  
96      /** Answer of the chain. */
97      @Mock
98      private ClientHttpResponse response;
99  
100     /**
101      * Verifies that without DPoP the interceptors send a bearer token for the given registration and principal.
102      *
103      * @throws IOException if an interceptor fails.
104      */
105     @Test
106     void create_whenDPoPIsOff_sendsABearerTokenOfTheRegistrationAndPrincipal() throws IOException {
107         final var asked = ArgumentCaptor.<OAuth2AuthorizeRequest>captor();
108         when(dpopProofs.getIfAvailable()).thenReturn(null);
109         when(manager.authorize(asked.capture())).thenReturn(client());
110         final var request = new MockClientHttpRequest(HttpMethod.GET, URI_CALLED);
111         when(execution.execute(request, new byte[0])).thenReturn(response);
112         when(response.getHeaders()).thenReturn(new HttpHeaders());
113         when(response.getStatusCode()).thenReturn(HttpStatus.OK);
114 
115         final var interceptors = new ResourceServerInterceptors(manager, dpopProofs)
116                 .create(REGISTRATION_ID, principal);
117         final var oauth = interceptors.getFirst();
118         final var guard = interceptors.getLast();
119         oauth.intercept(request, new byte[0], (req, body) -> guard.intercept(req, body, execution));
120 
121         assertEquals(2, interceptors.size());
122         assertInstanceOf(OAuth2ClientHttpRequestInterceptor.class, oauth);
123         assertEquals("Bearer the-token", request.getHeaders().getFirst("Authorization"));
124         assertEquals(REGISTRATION_ID, asked.getValue().getClientRegistrationId());
125         assertEquals(principal, asked.getValue().getPrincipal());
126     }
127 
128     /**
129      * Verifies that a call is refused when the manager returned no authorized client.
130      *
131      * @throws IOException if the execution fails.
132      */
133     @Test
134     void create_whenNoTokenIsAvailable_refusesTheCall() throws IOException {
135         when(dpopProofs.getIfAvailable()).thenReturn(null);
136         final var request = new MockClientHttpRequest(HttpMethod.GET, URI_CALLED);
137         final var interceptors = new ResourceServerInterceptors(manager, dpopProofs)
138                 .create(REGISTRATION_ID, principal);
139         final var guard = interceptors.getLast();
140         final var body = new byte[0];
141 
142         assertThrows(ClientAuthorizationException.class, () -> guard.intercept(request, body, execution));
143 
144         verify(execution, never()).execute(any(), any());
145     }
146 
147     /**
148      * Verifies that with DPoP the only interceptor is the DPoP one.
149      */
150     @Test
151     void create_whenDPoPIsOn_returnsTheDPoPInterceptorAlone() {
152         when(dpopProofs.getIfAvailable()).thenReturn(proofs);
153 
154         final List<?> interceptors = new ResourceServerInterceptors(manager, dpopProofs)
155                 .create(REGISTRATION_ID, principal);
156 
157         assertEquals(1, interceptors.size());
158         assertInstanceOf(DPoPClientHttpRequestInterceptor.class, interceptors.getFirst());
159     }
160 
161     /**
162      * Builds an authorized client with a bearer token.
163      *
164      * @return the authorized client.
165      */
166     private static OAuth2AuthorizedClient client() {
167         final var now = Instant.now();
168         return new OAuth2AuthorizedClient(REGISTRATION, "name", new OAuth2AccessToken(
169                 OAuth2AccessToken.TokenType.BEARER, "the-token", now, now.plusSeconds(LIFETIME_SECONDS)));
170     }
171 }
