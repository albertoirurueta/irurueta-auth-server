18  package com.irurueta.authserver.serviceclient.infrastructure.resourceserver;
19  
20  import static org.junit.jupiter.api.Assertions.assertEquals;
21  import static org.junit.jupiter.api.Assertions.assertInstanceOf;
22  import static org.junit.jupiter.api.Assertions.assertNotEquals;
23  import static org.junit.jupiter.api.Assertions.assertTrue;
24  
25  import com.nimbusds.jose.JWSAlgorithm;
26  import com.nimbusds.jose.JWSHeader;
27  import com.nimbusds.jose.crypto.RSASSASigner;
28  import com.nimbusds.jwt.JWTClaimsSet;
29  import com.nimbusds.jwt.SignedJWT;
30  import java.security.interfaces.RSAPrivateKey;
31  import org.junit.jupiter.api.Test;
32  import org.springframework.security.oauth2.core.OAuth2AccessToken;
33  
34  /**
35   * Unit tests of {@link PrincipalMapper}.
36   */
37  class PrincipalMapperTest {
38  
39      /** Mapper under test. */
40      private final PrincipalMapper mapper = new PrincipalMapper();
41  
42      /**
43       * Verifies that the principal of the service client is named after the client.
44       */
45      @Test
46      void serviceClient_whenCalled_isNamedAfterTheClient() {
47          final var principal = mapper.serviceClient("my-client");
48  
49          assertEquals("my-client", principal.getName());
50          assertTrue(principal.isAuthenticated());
51      }
52  
53      /**
54       * Verifies that a JWT's subject, plus a digest, names the principal, and that the token is the principal object.
55       *
56       * @throws Exception if the test JWT cannot be built.
57       */
58      @Test
59      void user_whenTokenIsAJwt_isNamedAfterItsSubjectAndDigest() throws Exception {
60          final var token = jwt("alice");
61  
62          final var principal = mapper.user(token);
63  
64          assertTrue(principal.getName().startsWith("alice#"));
65          final var accessToken = assertInstanceOf(OAuth2AccessToken.class, principal.getPrincipal());
66          assertEquals(token, accessToken.getTokenValue());
67      }
68  
69      /**
70       * Verifies that two different tokens of the same subject never share a name.
71       *
72       * @throws Exception if the test JWTs cannot be built.
73       */
74      @Test
75      void user_whenTwoTokensShareTheSubject_namesDiffer() throws Exception {
76          assertNotEquals(mapper.user(jwt("alice")).getName(), mapper.user(jwt("alice")).getName());
77      }
78  
79      /**
80       * Verifies that an opaque token still gets a name that depends on the token.
81       */
82      @Test
83      void user_whenTokenIsOpaque_isNamedUserAndDigest() {
84          final var principal = mapper.user("opaque-reference-token");
85  
86          assertTrue(principal.getName().startsWith("user#"));
87          assertNotEquals(principal.getName(), mapper.user("another-opaque-token").getName());
88      }
89  
90      /**
91       * Builds a signed JWT.
92       *
93       * @param subject its {@code sub} claim.
94       * @return the compact serialization.
95       * @throws Exception if signing fails.
96       */
97      private static String jwt(final String subject) throws Exception {
98          final var jwt = new SignedJWT(new JWSHeader(JWSAlgorithm.RS256),
99                  new JWTClaimsSet.Builder().subject(subject).build());
100         jwt.sign(new RSASSASigner((RSAPrivateKey) TestKeys.rsaKeyPair().getPrivate()));
101         return jwt.serialize();
102     }
103 }
