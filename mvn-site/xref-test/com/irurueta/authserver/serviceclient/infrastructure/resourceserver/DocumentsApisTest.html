41  /**
42   * Unit tests of {@link DocumentsApis}.
43   */
44  @ExtendWith(MockitoExtension.class)
45  class DocumentsApisTest {
46  
47      /** Registration id. */
48      private static final String REGISTRATION_ID = "service-client";
49  
50      /** Request factory over the SSL bundle, when there is one. */
51      @Mock
52      private SslRequestFactories requestFactories;
53  
54      /** Interceptors that obtain and send the access token. */
55      @Mock
56      private ResourceServerInterceptors interceptors;
57  
58      /** Principal the token is cached under. */
59      @Mock
60      private Authentication principal;
61  
62      /**
63       * Verifies that the client calls the base URI (without a trailing slash) with the interceptors it was given.
64       */
65      @Test
66      void forPrincipal_whenCalled_callsTheBaseUriThroughTheInterceptors() {
67          final var builder = RestClient.builder();
68          final var server = MockRestServiceServer.bindTo(builder).build();
69          final ClientHttpRequestInterceptor tokenSetter = (request, body, execution) -> {
70              request.getHeaders().setBearerAuth("the-token");
71              return execution.execute(request, body);
72          };
73          when(requestFactories.create()).thenReturn(Optional.empty());
74          when(interceptors.create(REGISTRATION_ID, principal)).thenReturn(List.of(tokenSetter));
75          server.expect(requestTo("https://rs.example.com/api/documents"))
76                  .andExpect(method(HttpMethod.GET))
77                  .andExpect(header("Authorization", "Bearer the-token"))
78                  .andRespond(withSuccess("[{\"id\":\"1\",\"owner\":\"alice\",\"title\":\"T\"}]",
79                          MediaType.APPLICATION_JSON));
80          final var apis = new DocumentsApis(builder,
81                  new ResourceServerProperties(URI.create("https://rs.example.com/")), requestFactories, interceptors);
82  
83          final var documents = apis.forPrincipal(REGISTRATION_ID, principal).listDocuments();
84  
85          assertEquals(1, documents.size());
86          assertEquals("alice", documents.getFirst().getOwner());
87          server.verify();
88      }
89  }
