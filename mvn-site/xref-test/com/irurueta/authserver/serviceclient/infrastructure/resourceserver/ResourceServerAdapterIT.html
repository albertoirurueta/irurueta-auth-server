58  /**
59   * Integration test of the resource-server adapter against WireMock playing the token endpoint and the resource server:
60   * the access token is requested with every authentication method, reused while it is valid and renewed inside the
61   * refresh margin, exchanged with a subject and an actor token, and DPoP-bound when asked. The context is the adapter
62   * alone (no Docker, no other module); keys are generated when the test runs.
63   * <p>
64   * Each nested class starts its own context, as each one configures the service client differently.
65   */

117     /**
118      * Points the service client at the fake servers.
119      *
120      * @param registry registry of the properties of the context.
121      * @param method   authentication method.
122      * @param dpop     whether tokens are DPoP-bound.
123      */
124     private static void configure(final DynamicPropertyRegistry registry, final String method, final boolean dpop) {
125         registry.add("service-client.auth.method", () -> method);
126         registry.add("service-client.auth.client-id", () -> "service-client");
127         registry.add("service-client.auth.client-secret", () -> "s3cret");
128         registry.add("service-client.auth.token-uri", () -> WIRE_MOCK.baseUrl() + TOKEN_PATH);
129         registry.add("service-client.auth.dpop", () -> Boolean.toString(dpop));
130         registry.add("service-client.auth.key.private-key", () -> TestKeys.pkcs8Pem(KEY_PAIR));
131         registry.add("service-client.resource-server.base-uri", WIRE_MOCK::baseUrl);
132     }

134     /**
135      * Makes the token endpoint answer a token for a grant.
136      *
137      * @param grantType grant type the request must carry.
138      * @param token     value of the access token.
139      * @param expiresIn lifetime of the token, in seconds.
140      * @param type      token type.
141      */
142     private static void stubToken(final String grantType, final String token, final int expiresIn, final String type) {
143         WIRE_MOCK.stubFor(post(urlEqualTo(TOKEN_PATH)).withRequestBody(containing("grant_type=" + grantType))
144                 .willReturn(aResponse().withHeader("Content-Type", "application/json")
145                         .withBody("{\"access_token\":\"" + token + "\",\"token_type\":\"" + type
146                                 + "\",\"expires_in\":" + expiresIn + ",\"scope\":\"documents.read\"}")));
147     }
148 
149     /**
150      * Makes the documents endpoint answer the documents to a request with an authorization header.
151      *
152      * @param authorization value the {@code Authorization} header must have.
153      */
154     private static void stubDocuments(final String authorization) {
155         WIRE_MOCK.stubFor(get(urlEqualTo(DOCUMENTS_PATH)).withHeader("Authorization", equalTo(authorization))
156                 .willReturn(aResponse().withHeader("Content-Type", "application/json").withBody(DOCUMENTS_BODY)));
157     }
158 
159     /**
160      * Reads a form parameter of a request the token endpoint received.
161      *
162      * @param request the request.
163      * @param name    name of the parameter.
164      * @return the decoded value, or {@code null}.
165      */
166     private static String parameter(final LoggedRequest request, final String name) {
167         for (final var pair : request.getBodyAsString().split("&")) {
168             final var parts = pair.split("=", 2);
169             if (parts[0].equals(name)) {
170                 return URLDecoder.decode(parts[1], StandardCharsets.UTF_8);
171             }
172         }
173         return null;
174     }

184      * Client credentials with a shared secret: the token is reused while valid. The context is discarded after each
185      * test because it caches the access token: a token cached by one test would otherwise be reused by the next one,
186      * which then would not request any.

194         @Autowired
195         private DocumentGateway gateway;

198          * Registers the properties.
199          *
200          * @param registry registry of the properties of the context.
201          */
202         @DynamicPropertySource
203         static void properties(final DynamicPropertyRegistry registry) {
204             configure(registry, "client_secret_basic", false);
205         }

215         /**
216          * Verifies that two calls read the documents with the same token, requested once with Basic authentication.
217          */
218         @Test
219         void listDocuments_whenTokenIsValid_reusesItAcrossCalls() {
220             stubToken("client_credentials", "cc-token", ONE_HOUR_SECONDS, "Bearer");
221             stubDocuments("Bearer cc-token");
222 
223             assertEquals(DOCUMENTS, gateway.listDocuments());
224             assertEquals(DOCUMENTS, gateway.listDocuments());
225 
226             final var tokenRequests = WIRE_MOCK.findAll(postRequestedFor(urlEqualTo(TOKEN_PATH)));
227             assertEquals(1, tokenRequests.size());
228             assertNotNull(tokenRequests.getFirst().getHeader("Authorization"));
229             assertTrue(tokenRequests.getFirst().getHeader("Authorization").startsWith("Basic "));
230             WIRE_MOCK.verify(2, getRequestedFor(urlEqualTo(DOCUMENTS_PATH)));
231         }
232 
233         /**
234          * Verifies that a resource server that refuses the token is an access failure.
235          */
236         @Test
237         void listDocuments_whenResourceServerRefusesTheToken_throwsDocumentAccessException() {
238             stubToken("client_credentials", "cc-token", ONE_HOUR_SECONDS, "Bearer");
239             WIRE_MOCK.stubFor(get(urlEqualTo(DOCUMENTS_PATH)).willReturn(aResponse().withStatus(FORBIDDEN)));
240 
241             assertThrows(DocumentAccessException.class, () -> gateway.listDocuments());
242         }
243 
244         /**
245          * Verifies that a token endpoint that refuses the client is an access failure.
246          */
247         @Test
248         void listDocuments_whenTokenEndpointRefusesTheClient_throwsDocumentAccessException() {
249             WIRE_MOCK.stubFor(post(urlEqualTo(TOKEN_PATH)).willReturn(aResponse().withStatus(UNAUTHORIZED)
250                     .withHeader("Content-Type", "application/json").withBody("{\"error\":\"invalid_client\"}")));
251 
252             assertThrows(DocumentAccessException.class, () -> gateway.listDocuments());
253         }

285         /**
286          * Verifies that the token request carries an assertion signed with the configured key and no secret.
287          *
288          * @throws Exception if the assertion cannot be parsed or verified.
289          */
290         @Test
291         void listDocuments_whenMethodIsPrivateKeyJwt_sendsAnAssertionSignedWithTheConfiguredKey() throws Exception {
292             stubToken("client_credentials", "cc-token", ONE_HOUR_SECONDS, "Bearer");
293             stubDocuments("Bearer cc-token");
294 
295             assertEquals(DOCUMENTS, gateway.listDocuments());
296 
297             final var request = WIRE_MOCK.findAll(postRequestedFor(urlEqualTo(TOKEN_PATH))).getFirst();
298             final var assertion = SignedJWT.parse(parameter(request, "client_assertion"));
299             assertTrue(assertion.verify(new RSASSAVerifier((RSAPublicKey) KEY_PAIR.getPublic())));
300             assertEquals("service-client", assertion.getJWTClaimsSet().getIssuer());
301             assertEquals(null, parameter(request, "client_secret"));
302         }

378         /**
379          * Verifies that the exchange sends the user's token as subject and the client's own token as actor, and that
380          * the documents are read with the exchanged token.
381          */
382         @Test
383         void listDocumentsAs_whenUserTokenIsGiven_exchangesItWithTheClientTokenAsActor() {
384             stubToken("client_credentials", "cc-token", ONE_HOUR_SECONDS, "Bearer");
385             stubToken("urn%3Aietf%3Aparams%3Aoauth%3Agrant-type%3Atoken-exchange", "exchanged", ONE_HOUR_SECONDS,
386                     "Bearer");
387             stubDocuments("Bearer exchanged");
388 
389             assertEquals(DOCUMENTS, gateway.listDocumentsAs("user-token"));
390 
391             final var exchange = WIRE_MOCK.findAll(postRequestedFor(urlEqualTo(TOKEN_PATH))).stream()
392                     .filter(request -> request.getBodyAsString().contains("token-exchange")
393                             || request.getBodyAsString().contains("token%2Dexchange"))
394                     .findFirst().orElseThrow();
395             assertEquals("user-token", parameter(exchange, "subject_token"));
396             assertEquals("cc-token", parameter(exchange, "actor_token"));
397             WIRE_MOCK.verify(1, getRequestedFor(urlEqualTo(DOCUMENTS_PATH))
398                     .withHeader("Authorization", equalTo("Bearer exchanged")));
399         }

431         /**
432          * Verifies that the token request and the call both carry a proof and the call uses the DPoP scheme.
433          *
434          * @throws Exception if a proof cannot be parsed.
435          */
436         @Test
437         void listDocuments_whenDPoPIsOn_sendsProofsAndTheDPoPScheme() throws Exception {
438             stubToken("client_credentials", "dpop-token", ONE_HOUR_SECONDS, "DPoP");
439             stubDocuments("DPoP dpop-token");
440 
441             assertEquals(DOCUMENTS, gateway.listDocuments());
442 
443             final var tokenRequest = WIRE_MOCK.findAll(postRequestedFor(urlEqualTo(TOKEN_PATH))).getFirst();
444             final var tokenProof = SignedJWT.parse(tokenRequest.getHeader("DPoP")).getJWTClaimsSet();
445             assertEquals("POST", tokenProof.getStringClaim("htm"));
446             final var call = WIRE_MOCK.findAll(getRequestedFor(urlEqualTo(DOCUMENTS_PATH))).getFirst();
447             final var callProof = SignedJWT.parse(call.getHeader("DPoP")).getJWTClaimsSet();
448             assertEquals("GET", callProof.getStringClaim("htm"));
449             assertNotNull(callProof.getStringClaim("ath"));
450         }
