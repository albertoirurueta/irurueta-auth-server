16  package com.irurueta.authserver.e2e;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  
20  import com.microsoft.playwright.Page;
21  import com.microsoft.playwright.Playwright;
22  import com.microsoft.playwright.Route;
23  import com.microsoft.playwright.TimeoutError;
24  import java.net.URI;
25  import java.net.URLEncoder;
26  import java.net.http.HttpClient;
27  import java.net.http.HttpRequest;
28  import java.net.http.HttpResponse;
29  import java.nio.charset.StandardCharsets;
30  import java.security.MessageDigest;
31  import java.security.NoSuchAlgorithmException;
32  import java.security.SecureRandom;
33  import java.util.Base64;
34  import tools.jackson.databind.json.JsonMapper;

36  /**
37   * Obtains the access token of the demo user the way a client application does: the authorization code flow with
38   * PKCE, with the user signing in in headless Chromium. The redirect to the client is intercepted, so no client
39   * application has to listen at the redirect URI.
40   */
41  final class UserTokenFlow {
42  
43      /** Number of random bytes of a PKCE code verifier. */
44      private static final int VERIFIER_BYTES = 32;
45  
46      /** Longest wait for a page transition, in milliseconds. */
47      private static final double PAGE_TIMEOUT_MILLIS = 30_000;
48  
49      /** Longest wait for the consent page, in milliseconds, before assuming the consent was granted earlier. */
50      private static final double CONSENT_WAIT_MILLIS = 5_000;
51  
52      /** HTTP status of a successful response. */
53      private static final int HTTP_OK = 200;
54  
55      /** Mapper of the token endpoint's answer. */
56      private static final JsonMapper JSON = JsonMapper.builder().build();
57  
58      /**
59       * Not instantiable.
60       */
61      private UserTokenFlow() {
62      }
63  
64      /**
65       * Runs the flow for the demo user with the seeded {@code bff} client and returns the access token.
66       *
67       * @param authServer  the running server.
68       * @param redirectUri the redirect URI registered for the {@code bff} client (nothing listens there).
69       * @param scope       space separated scopes to request.
70       * @return the access token of the user.
71       * @throws Exception if a step of the flow fails.
72       */
73      static String bffAccessToken(final RunningAuthServer authServer, final String redirectUri, final String scope)
74              throws Exception {
75          final var clientId = "bff";
76          final var verifier = randomVerifier();
77          final var authorizeUrl = authServer.issuer() + "/oauth2/authorize?response_type=code&client_id=" + clientId
78                  + "&redirect_uri=" + encode(redirectUri) + "&scope=" + encode(scope)
79                  + "&state=xyz&code_challenge=" + s256(verifier) + "&code_challenge_method=S256";
80          final String code;
81          try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
82              code = signInAndCaptureCode(browser.newPage(), authServer, redirectUri, authorizeUrl);
83          }
84          final var credentials = Base64.getEncoder().encodeToString(
85                  (clientId + ":" + authServer.clientSecret()).getBytes(StandardCharsets.UTF_8));
86          final var request = HttpRequest.newBuilder(URI.create(authServer.issuer() + "/oauth2/token"))
87                  .header("Authorization", "Basic " + credentials)
88                  .header("Content-Type", "application/x-www-form-urlencoded")
89                  .POST(HttpRequest.BodyPublishers.ofString("grant_type=authorization_code&code=" + encode(code)
90                          + "&redirect_uri=" + encode(redirectUri) + "&code_verifier=" + verifier))
91                  .build();
92          final var response = HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString());
93          assertEquals(HTTP_OK, response.statusCode(), "token endpoint answer to the code exchange");
94          return JSON.readTree(response.body()).get("access_token").asString();
95      }
96  
97      /**
98       * Signs the demo user in and returns the authorization code the server sends to the redirect URI.
99       *
100      * @param page         the browser page.
101      * @param authServer   the running server.
102      * @param redirectUri  the redirect URI whose request is intercepted.
103      * @param authorizeUrl authorization request URL.
104      * @return the authorization code.
105      */
106     private static String signInAndCaptureCode(final Page page, final RunningAuthServer authServer,
107                                                final String redirectUri, final String authorizeUrl) {
108         page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
109         page.route(redirectUri + "**", Route::abort);
110         page.navigate(authorizeUrl);
111         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
112         page.fill("#username", "demo");
113         page.fill("#password", authServer.demoPassword());
114         final var redirect = page.waitForRequest(redirectUri + "**", () -> {
115             page.click("button[type=submit]");
116             allowWhenAsked(page, authServer);
117         });
118         for (final var pair : URI.create(redirect.url()).getRawQuery().split("&")) {
119             if (pair.startsWith("code=")) {
120                 return pair.substring("code=".length());
121             }
122         }
123         throw new IllegalStateException("The redirect carries no authorization code: " + redirect.url());
124     }
125 
126     /**
127      * Allows the requested scopes when the consent page is shown.
128      *
129      * @param page       the page of the browser, after the credentials were submitted.
130      * @param authServer the running server.
131      */
132     private static void allowWhenAsked(final Page page, final RunningAuthServer authServer) {
133         try {
134             page.waitForURL(url -> url.startsWith(authServer.issuer() + "/oauth2/consent"),
135                     new Page.WaitForURLOptions().setTimeout(CONSENT_WAIT_MILLIS));
136         } catch (final TimeoutError notAsked) {
137             return;
138         }
139         page.click("#allow");
140     }
141 
142     /**
143      * Generates a PKCE code verifier.
144      *
145      * @return the verifier.
146      */
147     private static String randomVerifier() {
148         final var bytes = new byte[VERIFIER_BYTES];
149         new SecureRandom().nextBytes(bytes);
150         return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
151     }
152 
153     /**
154      * Computes the S256 PKCE code challenge of a verifier.
155      *
156      * @param verifier code verifier.
157      * @return the code challenge.
158      * @throws NoSuchAlgorithmException if SHA-256 is not available.
159      */
160     private static String s256(final String verifier) throws NoSuchAlgorithmException {
161         final var digest = MessageDigest.getInstance("SHA-256").digest(verifier.getBytes(StandardCharsets.US_ASCII));
162         return Base64.getUrlEncoder().withoutPadding().encodeToString(digest);
163     }
164 
165     /**
166      * URL-encodes a value.
167      *
168      * @param value the value.
169      * @return the encoded value.
170      */
171     private static String encode(final String value) {
172         return URLEncoder.encode(value, StandardCharsets.UTF_8);
173     }
174 }
