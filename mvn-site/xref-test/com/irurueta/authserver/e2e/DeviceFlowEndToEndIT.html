35  /**
36   * End-to-end test of the OAuth 2.0 device authorization grant (RFC 8628) with the packaged applications: the
37   * {@code auth-server} (dev profile, which seeds the {@code device-client}) runs on a random port against PostgreSQL,
38   * and the polling command line tool of the {@code service-client} runs from its own jar as a child process. The tool
39   * prints a user code and a verification address; headless Chromium opens the complete address, signs the demo user in,
40   * approves (or denies) the consent and the tool, which has been polling, ends with the tokens (or with the refusal).
41   */
42  class DeviceFlowEndToEndIT {
43  
44      /** Longest wait for a page transition, in milliseconds. */
45      private static final double PAGE_TIMEOUT_MILLIS = 30_000;
46  
47      /** Longest wait for the tool to print something that depends on the browser or on a polling round. */
48      private static final Duration TOOL_TIMEOUT = Duration.ofSeconds(90);
49  
50      /** Longest wait for the tool to end once it has what it needs. */
51      private static final Duration EXIT_TIMEOUT = Duration.ofSeconds(30);
52  
53      /** Matches the user code the tool prints. */
54      private static final Pattern USER_CODE = Pattern.compile("^User code: (\\S+)");
55  
56      /** Matches the complete verification address the tool prints. */
57      private static final Pattern COMPLETE_URI = Pattern.compile("^Verification URI \\(complete\\): (\\S+)");
58  
59      /** Matches the summary the tool prints when it obtained the tokens. */
60      private static final Pattern COMPLETED = Pattern.compile("^Device flow completed: (.*)");
61  
62      /** Matches the access token the tool prints when asked to. */
63      private static final Pattern ACCESS_TOKEN = Pattern.compile("^Access token: (\\S+)");
64  
65      /** Selector of the button that approves the consent. */
66      private static final String ALLOW = "#allow";
67  
68      /** Selector of the button that denies the consent. */
69      private static final String DENY = "#deny";
70  
71      /** Exit code of a tool that ended without error. */
72      private static final int SUCCESS = 0;
73  
74      /** Position of the payload in a compact JWT. */
75      private static final int PAYLOAD = 1;
76  
77      /** Mapper of the JSON of the token. */
78      private static final JsonMapper JSON = JsonMapper.builder().build();
79  
80      /** The packaged authorization server and its database. */
81      private static RunningAuthServer authServer;
82  
83      /**
84       * Starts the authorization server.
85       *
86       * @throws Exception if the database or the server cannot be started.
87       */
88      @BeforeAll
89      static void startAuthServer() throws Exception {
90          authServer = RunningAuthServer.start();
91      }
92  
93      /** Stops the server and its database. */
94      @AfterAll
95      static void stopAuthServer() {
96          if (authServer != null) {
97              authServer.close();
98          }
99      }
100 
101     /**
102      * Runs the whole flow: the tool asks for a device code, the user approves it in the browser and the tool ends with
103      * a bearer access token for the user, a refresh token and the scopes of the client.
104      *
105      * @throws Exception if the tool or the browser fails.
106      */
107     @Test
108     void deviceFlow_whenTheUserApproves_theToolObtainsTheTokens() throws Exception {
109         final var tool = startTool();
110         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
111             final var userCode = tool.await(USER_CODE, TOOL_TIMEOUT);
112             final var verification = tool.await(COMPLETE_URI, TOOL_TIMEOUT);
113             assertTrue(verification.contains(userCode), "The complete address carries the user code: " + verification);
114             final var page = browser.newPage();
115             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
116 
117             signIn(page, verification);
118             page.waitForURL(url -> url.startsWith(authServer.issuer() + "/oauth2/consent"));
119             page.click(ALLOW);
120             page.waitForURL(url -> url.contains("success"));
121 
122             final var summary = tool.await(COMPLETED, TOOL_TIMEOUT);
123             assertTrue(summary.contains("token_type=Bearer"), summary);
124             assertTrue(summary.contains("refresh_token=true"), summary);
125             final var claims = claimsOf(tool.await(ACCESS_TOKEN, EXIT_TIMEOUT));
126             assertEquals("demo", claims.get("sub").asString());
127             assertTrue(claims.get("scope").toString().contains("documents.read"), claims.toString());
128             assertEquals(SUCCESS, tool.awaitExit(EXIT_TIMEOUT));
129         } finally {
130             tool.stop();
131         }
132     }
133 
134     /**
135      * Verifies that a user who denies the consent leaves the tool without tokens: it ends with an error.
136      *
137      * @throws Exception if the tool or the browser fails.
138      */
139     @Test
140     void deviceFlow_whenTheUserDenies_theToolEndsWithoutTokens() throws Exception {
141         final var tool = startTool();
142         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch()) {
143             tool.await(USER_CODE, TOOL_TIMEOUT);
144             final var verification = tool.await(COMPLETE_URI, TOOL_TIMEOUT);
145             final var page = browser.newPage();
146             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
147 
148             signIn(page, verification);
149             page.waitForURL(url -> url.startsWith(authServer.issuer() + "/oauth2/consent"));
150             page.click(DENY);
151 
152             assertNotEquals(SUCCESS, tool.awaitExit(TOOL_TIMEOUT));
153         } finally {
154             tool.stop();
155         }
156     }
157 
158     /**
159      * Starts the command line tool against the server and has it print the access token it obtains.
160      *
161      * @return the running tool.
162      * @throws Exception if the process cannot be started.
163      */
164     private static ManagedApp startTool() throws Exception {
165         return ManagedApp.start(RunningAuthServer.jar("serviceclient.jar"), List.of(
166                 "--spring.profiles.active=device-flow",
167                 "--device-flow.issuer=" + authServer.issuer(),
168                 "--device-flow.print-access-token=true"), List.of());
169     }
170 
171     /**
172      * Opens the verification address, which sends the browser to the login page, and signs the demo user in.
173      *
174      * @param page         the browser page.
175      * @param verification the complete verification address.
176      */
177     private static void signIn(final Page page, final String verification) {
178         page.navigate(verification);
179         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
180         page.fill("#username", "demo");
181         page.fill("#password", authServer.demoPassword());
182         page.click("button[type=submit]");
183     }
184 
185     /**
186      * Reads the claims of a JWT without verifying it: the test only inspects what the server issued.
187      *
188      * @param jwt the compact serialization.
189      * @return the claims.
190      */
191     private static JsonNode claimsOf(final String jwt) {
192         return JSON.readTree(new String(Base64.getUrlDecoder().decode(jwt.split("\\.")[PAYLOAD]),
193                 StandardCharsets.UTF_8));
194     }
195 }
