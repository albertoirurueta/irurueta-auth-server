16  package com.irurueta.authserver.e2e;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  
20  import java.net.URI;
21  import java.net.URLEncoder;
22  import java.net.http.HttpClient;
23  import java.net.http.HttpRequest;
24  import java.net.http.HttpResponse;
25  import java.nio.charset.StandardCharsets;
26  import java.util.Base64;
27  import java.util.List;
28  import tools.jackson.databind.JsonNode;
29  import tools.jackson.databind.json.JsonMapper;
30  import tools.jackson.databind.node.ObjectNode;
31  
32  /**
33   * Registers clients in a running {@code auth-server} through its admin API ({@code POST /api/admin/clients}), the way
34   * an
35   * operator would. The API needs a token with the {@code admin} scope, and the only clients a fresh {@code dev} server
36   * has are the seeded ones, so one confidential admin client is inserted by SQL as a copy of the seeded {@code bff}
37   * client (it shares its secret) limited to the {@code client_credentials} grant and the {@code admin} scope.
38   */
39  final class ClientAdminApi {
40  
41      /** Scope that lets a client read documents. */
42      static final String DOCUMENTS_READ = "documents.read";
43  
44      /** Audience of the resource server. */
45      static final String RESOURCE_SERVER_AUDIENCE = "resource-server";
46  
47      /** Identifier of the administration client. */
48      private static final String ADMIN_CLIENT_ID = "e2e-admin";
49  
50      /** Scope that opens the admin API. */
51      private static final String ADMIN_SCOPE = "admin";
52  
53      /** Seeded client the administration client is copied from. */
54      private static final String SEEDED_CLIENT = "bff";
55  
56      /** HTTP status of a created resource. */
57      private static final int HTTP_CREATED = 201;
58  
59      /** HTTP status of a successful response. */
60      private static final int HTTP_OK = 200;
61  
62      /** Lifetime of the access tokens of the clients that are created. */
63      private static final String ACCESS_TOKEN_TTL = "PT10M";
64  
65      /** Lifetime of the refresh tokens of the clients that are created. */
66      private static final String REFRESH_TOKEN_TTL = "PT1H";
67  
68      /** Mapper of the JSON of the API. */
69      private static final JsonMapper JSON = JsonMapper.builder().build();
70  
71      /** The server whose clients are administered. */
72      private final RunningAuthServer authServer;
73  
74      /** Access token with the admin scope. */
75      private final String adminToken;
76  
77      /**
78       * Creates the administration client of the server and obtains its token.
79       *
80       * @param authServer the running server.
81       * @throws Exception if the client cannot be inserted or the token cannot be obtained.
82       */
83      ClientAdminApi(final RunningAuthServer authServer) throws Exception {
84          this.authServer = authServer;
85          authServer.sql("INSERT INTO oauth2_registered_client (id, client_id, client_secret, client_name, "
86                  + "client_authentication_methods, authorization_grant_types, redirect_uris, scopes, "
87                  + "client_settings, token_settings) SELECT gen_random_uuid()::text, '" + ADMIN_CLIENT_ID
88                  + "', client_secret, client_name, client_authentication_methods, 'client_credentials', '', '"
89                  + ADMIN_SCOPE + "', client_settings, token_settings FROM oauth2_registered_client "
90                  + "WHERE client_id = '" + SEEDED_CLIENT + "'");
91          final var credentials = Base64.getEncoder().encodeToString(
92                  (ADMIN_CLIENT_ID + ":" + authServer.clientSecret()).getBytes(StandardCharsets.UTF_8));
93          final var request = HttpRequest.newBuilder(URI.create(authServer.issuer() + "/oauth2/token"))
94                  .header("Authorization", "Basic " + credentials)
95                  .header("Content-Type", "application/x-www-form-urlencoded")
96                  .POST(HttpRequest.BodyPublishers.ofString("grant_type=client_credentials&scope="
97                          + URLEncoder.encode(ADMIN_SCOPE, StandardCharsets.UTF_8)))
98                  .build();
99          final var response = HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString());
100         assertEquals(HTTP_OK, response.statusCode(), "token of the admin client: " + response.body());
101         adminToken = JSON.readTree(response.body()).get("access_token").asString();
102     }
103 
104     /**
105      * Starts the definition of a confidential client that reads documents for the resource server, with the
106      * given authentication methods and grant types; the caller adds what the methods need.
107      *
108      * @param clientId    identifier of the client.
109      * @param authMethods client authentication methods.
110      * @param grantTypes  grant types.
111      * @return the definition, ready to be completed and passed to {@link #create(ObjectNode)}.
112      */
113     static ObjectNode confidentialClient(final String clientId, final List<String> authMethods,
114                                          final List<String> grantTypes) {
115         final var definition = JSON.createObjectNode();
116         definition.put("clientId", clientId);
117         definition.put("name", clientId);
118         definition.put("confidential", true);
119         final var methods = definition.putArray("authMethods");
120         authMethods.forEach(methods::add);
121         final var grants = definition.putArray("grantTypes");
122         grantTypes.forEach(grants::add);
123         definition.putArray("redirectUris");
124         definition.putArray("scopes").add(DOCUMENTS_READ);
125         definition.putArray("audiences").add(RESOURCE_SERVER_AUDIENCE);
126         definition.put("accessTokenTtl", ACCESS_TOKEN_TTL);
127         definition.put("refreshTokenTtl", REFRESH_TOKEN_TTL);
128         return definition;
129     }
130 
131     /**
132      * Creates a client.
133      *
134      * @param definition the client definition, as the admin API's {@code ClientRequest}.
135      * @return the generated secret of the client, or {@code null} when it has none.
136      * @throws Exception if the request fails or the server refuses the definition.
137      */
138     String create(final ObjectNode definition) throws Exception {
139         final var request = HttpRequest.newBuilder(URI.create(authServer.issuer() + "/api/admin/clients"))
140                 .header("Authorization", "Bearer " + adminToken)
141                 .header("Content-Type", "application/json")
142                 .POST(HttpRequest.BodyPublishers.ofString(JSON.writeValueAsString(definition)))
143                 .build();
144         final var response = HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString());
145         assertEquals(HTTP_CREATED, response.statusCode(), response.body());
146         final JsonNode secret = JSON.readTree(response.body()).get("clientSecret");
147         return secret == null || secret.isNull() ? null : secret.asString();
148     }
149 }
