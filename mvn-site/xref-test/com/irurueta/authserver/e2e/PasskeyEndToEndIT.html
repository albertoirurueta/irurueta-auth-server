38  /**
39   * End-to-end test of the passkeys with a real browser: the packaged {@code auth-server} and {@code bff} run on random
40   * ports against a PostgreSQL container, and headless Chromium gets a CDP virtual authenticator (CTAP2, internal,
41   * resident key, user verified) so that the script of the pages runs the real WebAuthn ceremonies. The first scenario
42   * registers a passkey from the account page and signs in through the BFF with it alone, clicking its button; the
43   * second signs in with the password and completes the second-factor page with the passkey; the third lets the browser
44   * offer the passkey in the username field. Each scenario has a user of its own, copied from the demo user. The
45   * claims are read from the access token the authorization server stored, so the BFF never has to expose it.
46   */
47  class PasskeyEndToEndIT {
48  
49      /** Matches the log line printed once the BFF is up. */
50      private static final Pattern BFF_STARTED = Pattern.compile("Started BffApplication in");
51  
52      /** Query that returns the most recently issued access token. */
53      private static final String LATEST_ACCESS_TOKEN = "SELECT access_token_value FROM oauth2_authorization"
54              + " WHERE access_token_value IS NOT NULL ORDER BY access_token_issued_at DESC LIMIT 1";

57      private static final double PAGE_TIMEOUT_MILLIS = 30_000;

59      /** Number of segments of a JSON Web Token. */
60      private static final int JWT_SEGMENTS = 3;
61  
62      /** Index of the payload among the segments of a JSON Web Token. */
63      private static final int PAYLOAD_INDEX = 1;

65      /** Name of the claim with the authentication methods. */
66      private static final String AMR = "amr";

68      /** Path of the passkeys account page. */

71      /** Path of the second-factor page. */
72      private static final String SECOND_FACTOR_PATH = "/login/second-factor";
73  
74      /** Path of the consent page. */
75      private static final String CONSENT_PATH = "/oauth2/consent";
76  
77      /** Name of the user of the first scenario. */
78      private static final String PRIMARY_USER = "passkey-primary";
79  
80      /** Name of the user of the second scenario. */
81      private static final String SECOND_FACTOR_USER = "passkey-second";
82  
83      /** Name of the user of the autofill scenario. */
84      private static final String AUTOFILL_USER = "passkey-autofill";

86      /** Selector of what the account page shows once it lists a passkey. */
87      private static final String LISTED_PASSKEY = ".passkey-list li";

89      /** Selector of the button that signs in with a passkey. */

93      private static RunningAuthServer authServer;

101     /**
102      * Starts the database, the authorization server and the BFF, registering the BFF's real address in the
103      * {@code bff} client.
104      *
105      * @throws Exception if an application or the container cannot be started.
106      */
107     @BeforeAll
108     static void startApplications() throws Exception {
109         final var bffPort = RunningAuthServer.freePort();
110         bffUrl = "http://localhost:" + bffPort;
111         authServer = RunningAuthServer.start(List.of(
112                 "--authserver.clients.bff.redirect-uris=" + bffUrl + "/login/oauth2/code/bff",
113                 "--authserver.clients.bff.post-logout-redirect-uris=" + bffUrl + "/"));
114         final var bffArguments = new ArrayList<>(authServer.bffDatabaseArguments());
115         bffArguments.add("--server.port=" + bffPort);
116         bff = ManagedApp.start(RunningAuthServer.jar("bff.jar"), bffArguments,
117                 List.of("AUTH_ISSUER=" + authServer.issuer(), "BFF_CLIENT_SECRET=" + authServer.clientSecret()));
118         bff.await(BFF_STARTED, RunningAuthServer.STARTUP_TIMEOUT);
119     }
120 
121     /** Stops the applications and the database. */
122     @AfterAll
123     static void stopApplications() {
124         if (bff != null) {
125             bff.stop();
126         }
127         if (authServer != null) {
128             authServer.close();
129         }
130     }

132     /**
133      * Registers a passkey from the account page, signs out, and signs in through the BFF with the passkey alone: the
134      * access token reports the hardware-key method and no password.
135      *
136      * @throws Exception if a step of the flow fails.
137      */
138     @Test
139     void signIn_whenUserRegisteredAPasskey_usesItAsTheOnlyFactor() throws Exception {
140         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch();
141              var context = browser.newContext()) {
142             final var page = context.newPage();
143             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
144             addVirtualAuthenticator(context, page);
145             withoutPasskeyAutofill(page);
146             authServer.createUserLikeDemo(PRIMARY_USER);
147 
148             registerAPasskey(page, PRIMARY_USER, "Primary key");
149             context.clearCookies();
150             signInThroughTheBff(page);
151             page.click(PASSKEY_BUTTON);
152             completeAuthorization(page, PRIMARY_USER);
153 
154             final var methods = latestAmr();
155             assertTrue(methods.contains("hwk"), methods.toString());
156             assertFalse(methods.contains("pwd"), methods.toString());
157         }
158     }

160     /**
161      * Registers a passkey, signs out, signs in with the password, and completes the second-factor page with the
162      * passkey: the access token reports the password, the hardware key and the multi-factor method.
163      *
164      * @throws Exception if a step of the flow fails.
165      */
166     @Test
167     void signIn_whenUserHasAPasskey_passwordThenPasskeyReportsBothFactors() throws Exception {
168         try (var playwright = Playwright.create(); var browser = playwright.chromium().launch();
169              var context = browser.newContext()) {
170             final var page = context.newPage();
171             page.setDefaultTimeout(PAGE_TIMEOUT_MILLIS);
172             addVirtualAuthenticator(context, page);
173             withoutPasskeyAutofill(page);
174             authServer.createUserLikeDemo(SECOND_FACTOR_USER);
175 
176             registerAPasskey(page, SECOND_FACTOR_USER, "Second factor key");
177             context.clearCookies();
178             signInThroughTheBff(page);
179             page.fill("#username", SECOND_FACTOR_USER);
180             page.fill("#password", authServer.demoPassword());
181             page.click("button[type=submit]");
182             page.waitForURL(url -> url.startsWith(authServer.issuer() + SECOND_FACTOR_PATH));
183             page.click(PASSKEY_BUTTON);
184             completeAuthorization(page, SECOND_FACTOR_USER);
185 
186             final var methods = latestAmr();
187             assertTrue(methods.containsAll(List.of("pwd", "hwk", "mfa")), methods.toString());
188         }

218     /**
219      * Makes the page behave like a browser without conditional mediation, so that the login page does not start a
220      * passkey sign-in by itself and the tests drive the passkey button or the password form.
221      *
222      * @param page the page of the browser.
223      */
224     private static void withoutPasskeyAutofill(final Page page) {
225         page.addInitScript("window.PublicKeyCredential.isConditionalMediationAvailable = undefined;");
226     }
227 
228     /**
229      * Gives the page a virtual authenticator that behaves like a platform passkey: CTAP2, internal transport, able
230      * to store discoverable credentials, and user verified.
231      *
232      * @param context the browser context.
233      * @param page    the page that gets the authenticator.
234      */
235     private static void addVirtualAuthenticator(final BrowserContext context, final Page page) {
236         final var session = context.newCDPSession(page);
237         session.send("WebAuthn.enable");
238         final var options = new JsonObject();
239         options.addProperty("protocol", "ctap2");
240         options.addProperty("transport", "internal");
241         options.addProperty("hasResidentKey", true);
242         options.addProperty("hasUserVerification", true);
243         options.addProperty("isUserVerified", true);
244         options.addProperty("automaticPresenceSimulation", true);
245         final var parameters = new JsonObject();
246         parameters.add("options", options);
247         session.send("WebAuthn.addVirtualAuthenticator", parameters);
248     }

250     /**
251      * Signs a user in on the login page of the authorization server and registers a passkey from the account page.
252      * The page reloads after the registration and, as the account now has a passkey that this session has not
253      * presented, the server asks for the second factor before showing the list: the passkey just registered
254      * completes it, and the page then lists the passkey.
255      *
256      * @param page     the page of the browser.
257      * @param username the user.
258      * @param label    name given to the passkey.
259      */
260     private static void registerAPasskey(final Page page, final String username, final String label) {
261         page.navigate(authServer.issuer() + "/login");
262         page.fill("#username", username);
263         page.fill("#password", authServer.demoPassword());
264         page.click("button[type=submit]");
265         page.waitForURL(url -> !url.contains("/login"));
266         page.navigate(authServer.issuer() + ACCOUNT_PATH);
267         page.fill("#passkey-label", label);
268         page.click("[data-passkey-register-button]");
269         page.locator(LISTED_PASSKEY + ", " + PASSKEY_BUTTON).first().waitFor();
270         if (page.locator(PASSKEY_BUTTON).count() > 0) {
271             page.click(PASSKEY_BUTTON);
272         }
273         page.waitForURL(url -> url.startsWith(authServer.issuer() + ACCOUNT_PATH));
274         assertEquals(label, page.locator(LISTED_PASSKEY + " strong").first().textContent());
275     }

277     /**
278      * Opens the SPA, clicks <em>Sign in</em> and waits for the login page of the authorization server.
279      *
280      * @param page the page of the browser.
281      */
282     private static void signInThroughTheBff(final Page page) {
283         page.navigate(bffUrl + "/");
284         page.waitForSelector("#signed-out:not([hidden])");
285         page.click("#sign-in");
286         page.waitForURL(url -> url.startsWith(authServer.issuer() + "/login"));
287     }

289     /**
290      * Waits until the browser leaves the sign-in pages, answers the consent page when it is shown (it is skipped when
291      * an earlier sign-in already granted it), and waits for the SPA to show the profile.
292      *
293      * @param page     the page of the browser.
294      * @param username the user the profile must show.
295      */
296     private static void completeAuthorization(final Page page, final String username) {
297         page.waitForURL(url -> url.startsWith(authServer.issuer() + CONSENT_PATH) || url.startsWith(bffUrl));
298         if (page.url().startsWith(authServer.issuer() + CONSENT_PATH)) {
299             page.click("#allow");
300             page.waitForURL(url -> url.startsWith(bffUrl));
301         }
302         page.waitForSelector("#profile:not([hidden])");
303         assertEquals(username, page.textContent("#profile-subject"));
304     }

306     /**
307      * Reads the {@code amr} claim of the access token issued last.
308      *
309      * @return the authentication methods.
310      * @throws Exception if the database cannot be queried.
311      */
312     private static List<String> latestAmr() throws Exception {
313         final var token = authServer.query(LATEST_ACCESS_TOKEN).getFirst();
314         final var segments = token.split("\\.");
315         assertEquals(JWT_SEGMENTS, segments.length, "the access token is a JWT");
316         final var claims = JsonParser.parseString(
317                 new String(Base64.getUrlDecoder().decode(segments[PAYLOAD_INDEX]), StandardCharsets.UTF_8))
318                 .getAsJsonObject();
319         return claims.getAsJsonArray(AMR).asList().stream().map(JsonElement::getAsString).toList();
320     }
