17  package com.irurueta.authserver.bff.api.web.login;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertFalse;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  import java.util.List;
24  import org.junit.jupiter.api.Test;
25  import org.junit.jupiter.params.ParameterizedTest;
26  import org.junit.jupiter.params.provider.NullSource;
27  import org.junit.jupiter.params.provider.ValueSource;
28  
29  /**
30   * Unit tests of {@link ReturnToValidator}, mostly open-redirect attempts.
31   */
32  class ReturnToValidatorTest {
33  
34      /** Validator with an exact path and a prefix allowed. */
35      private final ReturnToValidator validator = new ReturnToValidator(
36              new BffLoginProperties(List.of("/", "/index.html", "/orders/")));
37  
38      /**
39       * Verifies that allowed relative paths, with or without query string and fragment, are accepted.
40       *
41       * @param returnTo the value to check.
42       */
43      @ParameterizedTest
44      @ValueSource(strings = {"/", "/index.html", "/index.html?tab=1", "/orders/", "/orders/42", "/orders/42?x=y#top"})
45      void isAllowed_whenPathIsAllowed_returnsTrue(final String returnTo) {
46          assertTrue(validator.isAllowed(returnTo), returnTo);
47      }
48  
49      /**
50       * Verifies that open-redirect payloads and paths not listed are rejected.
51       *
52       * @param returnTo the value to check.
53       */
54      @ParameterizedTest
55      @NullSource
56      @ValueSource(strings = {"", "orders", "https://evil.example/", "http://evil.example", "//evil.example",
57              "///evil.example", "/\\evil.example", "\\\\evil.example", "/orders/\\..\\x", "javascript:alert(1)",
58              "/orders/../admin", "/orders/..", "/orders/./x", "/%2e%2e/admin", "/orders/%2F%2Fevil", "/other",
59              "/ordersx", "/orders/\r\nSet-Cookie: a=b", "/orders/\u0000", "/orders/\t", " /orders/1"})
60      void isAllowed_whenValueIsNotSafe_returnsFalse(final String returnTo) {
61          assertFalse(validator.isAllowed(returnTo), String.valueOf(returnTo));
62      }
63  
64      /**
65       * Verifies that a query string cannot smuggle a path past the allow list.
66       */
67      @Test
68      void isAllowed_whenQueryLooksLikeAllowedPath_checksThePathOnly() {
69          assertFalse(validator.isAllowed("/other?next=/orders/1"));
70          assertFalse(validator.isAllowed("/other#/orders/1"));
71      }
72  
73      /**
74       * Verifies the defaults used when nothing is configured.
75       */
76      @Test
77      void constructor_whenNothingIsConfigured_allowsRootAndIndexOnly() {
78          final var defaults = new ReturnToValidator(new BffLoginProperties(null));
79  
80          assertTrue(defaults.isAllowed("/"));
81          assertTrue(defaults.isAllowed("/index.html"));
82          assertFalse(defaults.isAllowed("/anything"));
83          assertEquals(List.of("/", "/index.html"), new BffLoginProperties(List.of()).allowedReturnPaths());
84      }
85  }
