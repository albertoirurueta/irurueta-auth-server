17  package com.irurueta.authserver.bff.api.web.security;
18  
19  import static org.junit.jupiter.api.Assertions.assertInstanceOf;
20  import static org.junit.jupiter.api.Assertions.assertNotNull;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  import static org.mockito.Mockito.mock;
23  
24  import org.junit.jupiter.api.Test;
25  import org.springframework.security.oauth2.client.registration.ClientRegistrationRepository;
26  import org.springframework.security.oauth2.client.web.HttpSessionOAuth2AuthorizedClientRepository;
27  
28  /**
29   * Unit tests of the OAuth2 client beans of {@link SecurityConfig}; the filter chain itself is covered by the
30   * integration tests.
31   */
32  class SecurityConfigTest {
33  
34      /** Configuration under test. */
35      private final SecurityConfig config = new SecurityConfig();
36  
37      /**
38       * Verifies that the authorized clients are kept in the HTTP session.
39       */
40      @Test
41      void authorizedClientRepository_whenCreated_keepsClientsInTheHttpSession() {
42          assertInstanceOf(HttpSessionOAuth2AuthorizedClientRepository.class, config.authorizedClientRepository());
43      }
44  
45      /**
46       * Verifies that the manager is built over the registrations and the session repository.
47       */
48      @Test
49      void authorizedClientManager_whenCreated_isBuilt() {
50          assertNotNull(config.authorizedClientManager(mock(ClientRegistrationRepository.class),
51                  config.authorizedClientRepository()));
52      }
53  
54      /**
55       * Verifies that the content security policy only allows same-origin scripts and forbids framing.
56       */
57      @Test
58      void contentSecurityPolicy_whenRead_isSameOriginAndForbidsFraming() {
59          assertTrue(SecurityConfig.CONTENT_SECURITY_POLICY.contains("default-src 'self'"));
60          assertTrue(SecurityConfig.CONTENT_SECURITY_POLICY.contains("frame-ancestors 'none'"));
61      }
62  }
