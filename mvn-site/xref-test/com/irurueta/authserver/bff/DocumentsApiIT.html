58  /**
59   * Integration tests of the documents API ({@code /api/documents}) through the whole filter chain, the controller, the
60   * use cases and the generated resource-server client. The resource server is a {@link MockRestServiceServer} bound to
61   * the {@code RestClient.Builder} the client is built from.
62   */

64  @AutoConfigureMockMvc
65  @AutoConfigureMockRestServiceServer

68      /** Access token of the session; it must reach the resource server and nobody else. */
69      private static final String ACCESS_TOKEN = "documents-test-access-token";
70  
71      /** Minutes the access token of the session lives. */
72      private static final long TOKEN_MINUTES = 5;
73  
74      /** Lifetime of the access token of the session. */
75      private static final Duration TOKEN_LIFETIME = Duration.ofMinutes(TOKEN_MINUTES);
76  
77      /** Documents of the resource server (the default of {@code bff.resource-server.base-uri}). */
78      private static final String DOCUMENTS = "http://localhost:8081/api/documents";
79  
80      /** One document of the resource server. */
81      private static final String DOCUMENT = DOCUMENTS + "/7";
82  
83      /** Path of the documents on the BFF. */
84      private static final String BFF_DOCUMENTS = "/api/documents";
85  
86      /** Path of one document on the BFF. */
87      private static final String BFF_DOCUMENT = BFF_DOCUMENTS + "/7";
88  
89      /** Media type of the problem details. */
90      private static final String PROBLEM_JSON = MediaType.APPLICATION_PROBLEM_JSON_VALUE;

92      /** Entry point to the web layer. */
93      @Autowired
94      private MockMvc mockMvc;

97      @Autowired
98      private MockRestServiceServer server;

108     /**
109      * Verifies that the bearer token is sent to the resource server while the browser's credentials are not, and that
110      * neither the token nor anything the resource server set in its headers reaches the browser.
111      *
112      * @throws Exception if the request cannot be performed.
113      */
114     @Test
115     void listDocuments_whenSignedIn_sendsTheBearerUpstreamAndKeepsItFromTheBrowser() throws Exception {
116         server.expect(requestTo(DOCUMENTS))
117                 .andExpect(method(HttpMethod.GET))
118                 .andExpect(header("Authorization", "Bearer " + ACCESS_TOKEN))
119                 .andExpect(headerDoesNotExist("Cookie"))
120                 .andExpect(headerDoesNotExist("X-XSRF-TOKEN"))
121                 .andRespond(withSuccess("[{\"id\":\"1\",\"owner\":\"user\",\"title\":\"T\",\"content\":\"C\"}]",
122                         MediaType.APPLICATION_JSON)
123                         .header("Set-Cookie", "upstream=secret")
124                         .header("Cache-Control", "max-age=3600"));
125 
126         final var result = mockMvc.perform(get(BFF_DOCUMENTS)
127                         .header("Cookie", "other=1").header("X-XSRF-TOKEN", "browser-csrf")
128                         .header("Authorization", "Bearer browser-supplied")
129                         .with(oidcLogin()).with(signedInClient()))
130                 .andExpect(status().isOk())
131                 .andExpect(content().json("[{\"id\":\"1\",\"owner\":\"user\",\"title\":\"T\",\"content\":\"C\"}]"))
132                 .andReturn().getResponse();
133 
134         server.verify();
135         assertFalse(result.getHeaders("Set-Cookie").stream().anyMatch(value -> value.contains("upstream")));
136         assertEquals(List.of("no-cache, no-store, max-age=0, must-revalidate"), result.getHeaders("Cache-Control"));
137         assertNull(result.getHeader("Authorization"));
138         assertFalse(result.getContentAsString().contains(ACCESS_TOKEN));
139     }

141     /**
142      * Verifies that a deletion carrying the CSRF token reaches the resource server with the token and answers 204.
143      *
144      * @throws Exception if the request cannot be performed.
145      */
146     @Test
147     void deleteDocument_whenRequestHasCsrfToken_deletesUpstream() throws Exception {
148         server.expect(requestTo(DOCUMENT))
149                 .andExpect(method(HttpMethod.DELETE))
150                 .andExpect(header("Authorization", "Bearer " + ACCESS_TOKEN))
151                 .andRespond(withStatus(HttpStatus.NO_CONTENT));
152 
153         mockMvc.perform(delete(BFF_DOCUMENT).with(csrf()).with(oidcLogin()).with(signedInClient()))
154                 .andExpect(status().isNoContent());
155 
156         server.verify();
157     }
158 
159     /**
160      * Verifies that a deletion without the CSRF token never reaches the resource server.
161      *
162      * @throws Exception if the request cannot be performed.
163      */
164     @Test
165     void deleteDocument_whenCsrfTokenIsMissing_isForbidden() throws Exception {
166         server.expect(never(), requestTo(DOCUMENT)).andRespond(withStatus(HttpStatus.OK));
167 
168         mockMvc.perform(delete(BFF_DOCUMENT).with(oidcLogin()).with(signedInClient()))
169                 .andExpect(status().isForbidden());
170 
171         server.verify();
172     }

174     /**
175      * Verifies that a missing document of the resource server is answered 404 as problem details.
176      *
177      * @throws Exception if the request cannot be performed.
178      */
179     @Test
180     void deleteDocument_whenResourceServerAnswersNotFound_answersNotFoundProblem() throws Exception {
181         server.expect(requestTo(DOCUMENT)).andRespond(withStatus(HttpStatus.NOT_FOUND));
182 
183         mockMvc.perform(delete(BFF_DOCUMENT).with(csrf()).with(oidcLogin()).with(signedInClient()))
184                 .andExpect(status().isNotFound())
185                 .andExpect(content().contentType(PROBLEM_JSON))
186                 .andExpect(jsonPath("$.type").value("urn:problem:documents:not-found"));
187     }
188 
189     /**
190      * Verifies that a refusal of the resource server is answered 403 as problem details, without its challenge or
191      * its body.
192      *
193      * @throws Exception if the request cannot be performed.
194      */
195     @Test
196     void listDocuments_whenResourceServerForbids_answersForbiddenProblem() throws Exception {
197         server.expect(requestTo(DOCUMENTS)).andRespond(withStatus(HttpStatus.FORBIDDEN)
198                 .header("WWW-Authenticate", "Bearer error=\"insufficient_scope\"").body("denied"));
199 
200         final var result = mockMvc.perform(get(BFF_DOCUMENTS).with(oidcLogin()).with(signedInClient()))
201                 .andExpect(status().isForbidden())
202                 .andExpect(content().contentType(PROBLEM_JSON))
203                 .andExpect(jsonPath("$.type").value("urn:problem:documents:forbidden"))
204                 .andReturn().getResponse();
205 
206         assertNull(result.getHeader("WWW-Authenticate"));
207         assertFalse(result.getContentAsString().contains("denied"));
208     }

282     /**
283      * Builds the post-processor that puts an authorized client with the test access token in the session.
284      *
285      * @return the request post-processor.
286      */
287     private static RequestPostProcessor signedInClient() {
288         final var now = Instant.now();
289         return oauth2Client("bff").accessToken(new OAuth2AccessToken(OAuth2AccessToken.TokenType.BEARER, ACCESS_TOKEN,
290                 now, now.plus(TOKEN_LIFETIME)));
291     }
