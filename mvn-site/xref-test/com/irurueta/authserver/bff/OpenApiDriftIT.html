34  /**
35   * Fails when the JSON API the running BFF serves drifts from its contract {@code apis/bff/rest-server/bff-api-v1.yaml}:
36   * an endpoint mapped by hand outside the generated interfaces, a contract operation no controller implements, or a
37   * different status code, parameter, body, security requirement or schema. The OpenAPI document springdoc serves (to a
38   * signed-in user, like every BFF page) is built from the running application's mappings and the annotations
39   * generated from the contract, and both are compared through {@link OpenApiContract}.
40   */
41  @PostgresIT
42  @AutoConfigureMockMvc
43  class OpenApiDriftIT {
44  
45      /** The contract, relative to the directory of this module. */
46      private static final Path CONTRACT = Path.of("..", "..", "apis", "bff", "rest-server", "bff-api-v1.yaml");
47  
48      /**
49       * Prefix of the paths compared: every path, since the JSON API spans {@code /bff/**} and {@code /api/**}. A path
50       * the application serves outside the contract (mapped by hand) is then reported too.
51       */
52      private static final String API_PREFIX = "/";
53  
54      /** MockMvc bound to the application context. */
55      @Autowired
56      private MockMvc mockMvc;
57  
58      /**
59       * Verifies that the served OpenAPI document describes exactly the API of the contract.
60       *
61       * @throws Exception if the document cannot be fetched or the contract cannot be read.
62       */
63      @Test
64      void apiDocs_whenApplicationRuns_matchContract() throws Exception {
65          final var body = mockMvc.perform(get(OpenApiConfig.API_DOCS_PATH).with(oidcLogin()))
66                  .andExpect(status().isOk())
67                  .andReturn().getResponse().getContentAsString(StandardCharsets.UTF_8);
68  
69          final var expected = OpenApiContract.summarize(OpenApiContract.read(CONTRACT), API_PREFIX);
70          final var actual = OpenApiContract.summarize(OpenApiContract.parse(body), API_PREFIX);
71  
72          assertEquals(expected, OpenApiContract.restrictTo(actual, expected),
73                  "The served API differs from " + CONTRACT + ". Change the contract (the interface and DTOs are "
74                          + "generated from it) instead of mapping an endpoint by hand.");
75      }
76  }
