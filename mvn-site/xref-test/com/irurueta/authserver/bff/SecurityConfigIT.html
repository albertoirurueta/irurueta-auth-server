64      /** Authorization request parameter naming the maximum age of the authentication. */

67      /** Authentication context class a step-up login asks for. */

70      /** Maximum age, in seconds, a step-up login asks for. */

74      private static final int MAX_RETURN_TO_LENGTH = 2048;

76      /** Query that counts the stored sessions with a given id. */
77      private static final String SESSIONS_WITH_ID = "SELECT COUNT(*) FROM spring_session WHERE session_id = ?";

87      /**
88       * Verifies that the authorization request uses PKCE, asks for the OpenID scopes and for both document scopes (the
89       * resource server needs {@code documents.delete} to delete), and carries no step-up parameter unless asked.
90       *
91       * @throws Exception if the request cannot be performed.
92       */
93      @Test
94      void bffFilterChain_whenAuthorizationIsRequested_usesPkce() throws Exception {
95          mockMvc.perform(get(AUTHORIZATION))
96                  .andExpect(status().is3xxRedirection())
97                  .andExpect(header().string("Location", containsString("code_challenge=")))
98                  .andExpect(header().string("Location", containsString("code_challenge_method=S256")))
99                  .andExpect(header().string("Location",
100                         containsString("scope=openid%20profile%20documents.read%20documents.delete")))
101                 .andExpect(header().string("Location", not(containsString(ACR_VALUES_PARAMETER))))
102                 .andExpect(header().string("Location", not(containsString(MAX_AGE_PARAMETER))));
103     }

105     /**
106      * Verifies that a login started for step-up keeps its values in the session and that the next authorization
107      * request, and only that one, carries them as {@code acr_values} and {@code max_age}.
108      *
109      * @throws Exception if a request cannot be performed.
110      */
111     @Test
112     void bffFilterChain_whenLoginAsksForStepUp_addsAcrValuesAndMaxAgeToNextAuthorizationRequestOnly()
113             throws Exception {
114         final var session = mockMvc.perform(get("/bff/login").param("returnTo", "/")
115                         .param("acrValues", STEP_UP_ACR).param("maxAge", STEP_UP_MAX_AGE))
116                 .andExpect(status().isFound())
117                 .andReturn().getResponse().getCookie(SESSION_COOKIE);
118 
119         mockMvc.perform(get(AUTHORIZATION).cookie(session))
120                 .andExpect(status().is3xxRedirection())
121                 .andExpect(header().string("Location", containsString(ACR_VALUES_PARAMETER + "=" + STEP_UP_ACR)))
122                 .andExpect(header().string("Location", containsString(MAX_AGE_PARAMETER + "=" + STEP_UP_MAX_AGE)))
123                 .andExpect(header().string("Location", containsString("code_challenge=")));
124         mockMvc.perform(get(AUTHORIZATION).cookie(session))
125                 .andExpect(status().is3xxRedirection())
126                 .andExpect(header().string("Location", not(containsString(ACR_VALUES_PARAMETER))));
127     }

178     /**
179      * Verifies the sign-out form: the raw value of the {@code XSRF-TOKEN} cookie posted as the {@code _csrf}
180      * parameter is accepted (a BREACH-masking handler would reject it), and Spring Security's logout answers a
181      * redirect (to the root here: the test provider advertises no end session endpoint), clears the CSRF cookie and
182      * deletes the stored session and its cookie. No security test post-processor is used for the CSRF token: those
183      * replace the CSRF token repository.
184      *
185      * @throws Exception if a request cannot be performed.
186      */
187     @Test
188     void bffFilterChain_whenLogoutFormCarriesRawCsrfParameter_endsSessionAndRedirects() throws Exception {
189         final var started = startSession();
190         final var session = started.getCookie(SESSION_COOKIE);
191         final var xsrf = started.getCookie(XSRF_COOKIE);
192         assertEquals(1, storedSessions(session));
193 
194         final var logout = mockMvc.perform(post(LOGOUT).cookie(session, xsrf).param("_csrf", xsrf.getValue())
195                         .with(oidcLogin()))
196                 .andExpect(status().isFound())
197                 .andExpect(redirectedUrl("/"))
198                 .andReturn().getResponse();
199 
200         assertEquals(0, logout.getCookie(XSRF_COOKIE).getMaxAge());
201         assertEquals(0, logout.getCookie(SESSION_COOKIE).getMaxAge());
202         assertEquals(0, storedSessions(session));
203     }

205     /**
206      * Verifies that the raw value of the {@code XSRF-TOKEN} cookie is also accepted from the {@code X-XSRF-TOKEN}
207      * header, the way the SPA sends it with {@code fetch}.
208      *
209      * @throws Exception if a request cannot be performed.
210      */
211     @Test
212     void bffFilterChain_whenHeaderCarriesRawCsrfToken_isAccepted() throws Exception {
213         final var xsrf = startSession().getCookie(XSRF_COOKIE);
214 
215         mockMvc.perform(post(LOGOUT).cookie(xsrf).header("X-XSRF-TOKEN", xsrf.getValue()).with(oidcLogin()))
216                 .andExpect(status().isFound());

277     /**
278      * Verifies that the CSRF token reaches the SPA in a cookie it can read (not HttpOnly). No security test
279      * post-processor is used: those replace the CSRF token repository.
280      *
281      * @throws Exception if the request cannot be performed.
282      */
283     @Test
284     void bffFilterChain_whenAnyRequestIsServed_setsReadableXsrfCookie() throws Exception {
285         final var cookies = mockMvc.perform(get("/bff/login"))
286                 .andExpect(status().isFound())
287                 .andReturn().getResponse().getHeaders("Set-Cookie");
288 
289         final var xsrf = cookies.stream().filter(header -> header.startsWith("XSRF-TOKEN=")).findFirst().orElseThrow();
290         assertFalse(xsrf.contains("HttpOnly"), xsrf);
291     }

293     /**
294      * Starts a session the way the SPA does, through the login endpoint, which also hands out the CSRF cookie.
295      *
296      * @return the response, carrying the session and CSRF cookies.
297      * @throws Exception if the request cannot be performed.
298      */
299     private MockHttpServletResponse startSession() throws Exception {
300         return mockMvc.perform(get("/bff/login")).andExpect(status().isFound()).andReturn().getResponse();
301     }
302 
303     /**
304      * Counts the stored sessions a session cookie points to.
305      *
306      * @param cookie the session cookie, whose value is the Base64 encoded session id.
307      * @return the number of stored sessions with that id.
308      */
309     private long storedSessions(final Cookie cookie) {
310         final var id = new String(Base64.getDecoder().decode(cookie.getValue()), StandardCharsets.UTF_8);
311         return jdbcTemplate.queryForObject(SESSIONS_WITH_ID, Long.class, id);
