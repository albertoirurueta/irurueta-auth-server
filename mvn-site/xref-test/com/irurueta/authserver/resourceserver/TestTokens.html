16  package com.irurueta.authserver.resourceserver;
17  
18  import com.nimbusds.jose.jwk.JWKSet;
19  import com.nimbusds.jose.jwk.RSAKey;
20  import com.nimbusds.jose.jwk.source.ImmutableJWKSet;
21  import java.io.IOException;
22  import java.nio.charset.StandardCharsets;
23  import java.nio.file.Files;
24  import java.security.KeyPair;
25  import java.security.KeyPairGenerator;
26  import java.security.NoSuchAlgorithmException;
27  import java.security.interfaces.RSAPrivateKey;
28  import java.security.interfaces.RSAPublicKey;
29  import java.time.Duration;
30  import java.time.Instant;
31  import java.util.Base64;
32  import java.util.List;
33  import java.util.Map;
34  import org.springframework.security.oauth2.jwt.JwtClaimsSet;
35  import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
36  import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;
37  import org.springframework.test.context.DynamicPropertyRegistry;
38  
39  /**
40   * Signs the JWTs the resource server integration tests present, with an RSA key generated when the class loads. The
41   * public half is handed to the application through
42   * {@code spring.security.oauth2.resourceserver.jwt.public-key-location}; Spring Boot only builds a decoder from that
43   * key when {@code issuer-uri} is blank (otherwise it resolves the issuer's metadata over HTTP), so the tests blank it.
44   * No key is ever committed.
45   */
46  final class TestTokens {
47  
48      /** Lifetime of the test tokens. */
49      static final Duration TOKEN_TTL = Duration.ofMinutes(5);
50  
51      /** Size in bits of the generated RSA key. */
52      private static final int KEY_SIZE = 2048;
53  
54      /** Number of base64 characters per line of a PEM document. */
55      private static final int PEM_LINE_LENGTH = 64;
56  
57      /** Key pair that signs the test tokens. */
58      private static final KeyPair KEY_PAIR = generateKeyPair();
59  
60      /** Utility class, not meant to be instantiated. */
61      private TestTokens() {
62      }
63  
64      /**
65       * Points the resource server's decoder at the public half of the generated key.
66       *
67       * @param registry registry of the test properties.
68       * @throws IOException if the PEM file cannot be written.
69       */
70      static void publicKeyLocation(final DynamicPropertyRegistry registry) throws IOException {
71          final var pem = Files.createTempFile("resource-server-test-key", ".pem");
72          pem.toFile().deleteOnExit();
73          Files.writeString(pem, toPem((RSAPublicKey) KEY_PAIR.getPublic()), StandardCharsets.US_ASCII);
74          registry.add("spring.security.oauth2.resourceserver.jwt.public-key-location", () -> pem.toUri().toString());
75          registry.add("spring.security.oauth2.resourceserver.jwt.issuer-uri", () -> "");
76      }
77  
78      /**
79       * Builds an {@code Authorization} header value carrying a token signed with the generated key.
80       *
81       * @param issuer    issuer claim of the token.
82       * @param subject   subject claim of the token.
83       * @param audiences audience claim of the token; empty to leave the claim out.
84       * @param scope     space-separated scopes of the token; empty to leave the claim out.
85       * @param roles     roles of the caller; empty to leave the claim out.
86       * @param extra     further claims of the token.
87       * @return the header value.
88       */
89      static String bearer(final String issuer, final String subject, final List<String> audiences, final String scope,
90                           final List<String> roles, final Map<String, Object> extra) {
91          final var rsaKey = new RSAKey.Builder((RSAPublicKey) KEY_PAIR.getPublic())
92                  .privateKey((RSAPrivateKey) KEY_PAIR.getPrivate())
93                  .build();
94          final var encoder = new NimbusJwtEncoder(new ImmutableJWKSet<>(new JWKSet(rsaKey)));
95          final var now = Instant.now();
96          final var claims = JwtClaimsSet.builder()
97                  .issuer(issuer)
98                  .subject(subject)
99                  .issuedAt(now)
100                 .expiresAt(now.plus(TOKEN_TTL));
101         if (!audiences.isEmpty()) {
102             claims.audience(audiences);
103         }
104         if (!scope.isEmpty()) {
105             claims.claim("scope", scope);
106         }
107         if (!roles.isEmpty()) {
108             claims.claim("roles", roles);
109         }
110         extra.forEach(claims::claim);
111         return "Bearer " + encoder.encode(JwtEncoderParameters.from(claims.build())).getTokenValue();
112     }
113 
114     /**
115      * Encodes a public key as a PEM document with {@code \n} line endings.
116      *
117      * @param key the public key.
118      * @return the PEM document.
119      */
120     private static String toPem(final RSAPublicKey key) {
121         final var encoder = Base64.getMimeEncoder(PEM_LINE_LENGTH, "\n".getBytes(StandardCharsets.US_ASCII));
122         return "-----BEGIN PUBLIC KEY-----\n" + encoder.encodeToString(key.getEncoded())
123                 + "\n-----END PUBLIC KEY-----\n";
124     }
125 
126     /**
127      * Generates the RSA key pair that signs the test tokens.
128      *
129      * @return the key pair.
130      */
131     private static KeyPair generateKeyPair() {
132         try {
133             final var generator = KeyPairGenerator.getInstance("RSA");
134             generator.initialize(KEY_SIZE);
135             return generator.generateKeyPair();
136         } catch (final NoSuchAlgorithmException e) {
137             throw new IllegalStateException("RSA is not available", e);
138         }
139     }
140 }
