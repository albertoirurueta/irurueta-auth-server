58  /**
59   * Checks the sign-in with external providers end to end, as the whole application wires it, against fake providers
60   * that answer over HTTP (Google as the OpenID Connect provider that does not vouch for multi-factor authentication,
61   * GitHub as the plain OAuth 2.0 one and Apple with the client secret the application signs): the first sign-in
62   * provisions an account, a verified e-mail address links an existing account and an unverified one never does, the
63   * tokens carry the local {@code sub}, the {@code idp} and the {@code amr} of the sign-in, a provider that is not
64   * trusted for multi-factor authentication still leads to the second-factor step, linked identities are removed
65   * through the account API under step-up, a disabled account is never linked, and a signed-in user links an identity
66   * explicitly from the identities page without the session changing its user. Every client id, secret, signing key and
67   * Apple key is generated at run time.

333     /**
334      * Verifies the shape of the client secret the application sends Apple: an ES256 token signed by the registered
335      * key, issued by the team for the client, for Apple's audience and short lived; and that the sign-in with the
336      * posted answer of Apple provisions an account.
337      *
338      * @throws Exception if a request cannot be performed or the secret cannot be read.
339      */
340     @Test
341     void signIn_whenApple_sendsAShortLivedEs256ClientSecretSignedWithTheTeamKey() throws Exception {
342         final var email = "apple-" + UUID.randomUUID() + "@example.com";
343         final var session = new MockHttpSession();
344         final var claims = FakeIdentityProviders.userClaims("apple-subject-" + UUID.randomUUID(), email, true);
345         claims.put("email_verified", "true");
346 
347         signInWithOidc(session, "apple", APPLE_ISSUER, APPLE_CLIENT_ID, claims);
348 
349         assertEquals(email, signedInName(session).orElseThrow());
350         final var requests = FakeIdentityProviders.tokenRequests("apple");
351         assertEquals(1, requests.size());
352         assertEquals(APPLE_CLIENT_ID, requests.getFirst().get("client_id"));
353         final var secret = SignedJWT.parse(requests.getFirst().get("client_secret"));
354         assertEquals(JWSAlgorithm.ES256, secret.getHeader().getAlgorithm());
355         assertEquals(KEY_ID, secret.getHeader().getKeyID());
356         assertEquals(TEAM_ID, secret.getJWTClaimsSet().getIssuer());
357         assertEquals(APPLE_CLIENT_ID, secret.getJWTClaimsSet().getSubject());
358         assertEquals(List.of(APPLE_AUDIENCE), secret.getJWTClaimsSet().getAudience());
359         final var lifetime = Duration.between(secret.getJWTClaimsSet().getIssueTime().toInstant(),
360                 secret.getJWTClaimsSet().getExpirationTime().toInstant());
361         assertTrue(lifetime.compareTo(MAX_SECRET_LIFETIME) <= 0 && !lifetime.isNegative());
362         assertTrue(secret.verify(new ECDSAVerifier((ECPublicKey) APPLE_KEY.getPublic())));
363     }

497                 .andExpect(header().string(LOCATION, AUTHORIZATION + "google"));
498         final var result = signInWithOidc(session, "google", GOOGLE_ISSUER, GOOGLE_CLIENT_ID,
