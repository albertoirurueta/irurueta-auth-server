19  import static org.junit.jupiter.api.Assertions.assertFalse;
20  import static org.junit.jupiter.api.Assertions.assertTrue;
21  import static org.springframework.security.test.web.servlet.request.SecurityMockMvcRequestPostProcessors.csrf;
22  import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
23  import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.redirectedUrl;
24  import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

30  import org.junit.jupiter.api.Test;
31  import org.springframework.beans.factory.annotation.Autowired;
32  import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
33  import org.springframework.jdbc.core.simple.JdbcClient;
34  import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
35  import org.springframework.security.crypto.password.PasswordEncoder;

37  import org.springframework.test.web.servlet.MockMvc;

39  /**
40   * Checks, through the real login form and a PostgreSQL container, that a hash stored with an old algorithm is
41   * replaced by the current Argon2 one the first time its owner signs in.
42   */
43  @PostgresIT
44  @AutoConfigureMockMvc
45  class PasswordUpgradeIT {
46  
47      /** Prefix of an Argon2 hash in the delegating encoder's format. */
48      private static final String ARGON2_PREFIX = "{" + PasswordConfiguration.DEFAULT_ENCODING_ID + "}";
49  
50      /** Prefix of a BCrypt hash in the delegating encoder's format. */
51      private static final String BCRYPT_PREFIX = "{" + PasswordConfiguration.BCRYPT_ENCODING_ID + "}";
52  
53      /** Query that reads the stored hash of a user. */
54      private static final String STORED_HASH = """
55              SELECT p.encoded_password FROM user_passwords p JOIN users u ON u.id = p.user_id WHERE u.username = ?
56              """;
57  
58      /** MockMvc bound to the application context. */
59      @Autowired
60      private MockMvc mockMvc;
61  
62      /** JDBC client that inserts the users and reads their hashes. */
63      @Autowired
64      private JdbcClient jdbcClient;
65  
66      /** Encoder of the application. */
67      @Autowired
68      private PasswordEncoder passwordEncoder;
69  
70      /** The compromised-password check never reaches the network in tests. */
71      @MockitoBean
72      private CompromisedPasswordCheck compromisedPasswordCheck;
73  
74      /**
75       * Verifies that a successful login replaces a BCrypt hash with an Argon2 one that still matches.
76       *
77       * @throws Exception if a request cannot be performed.
78       */
79      @Test
80      void login_whenStoredHashIsBcrypt_storesAnArgon2HashThatStillMatches() throws Exception {
81          final var username = "user-" + UUID.randomUUID();
82          final var credential = AuthServerFixtures.randomValue();
83          insertUser(username, BCRYPT_PREFIX + new BCryptPasswordEncoder().encode(credential));
84          assertTrue(storedHash(username).startsWith(BCRYPT_PREFIX));
85  
86          mockMvc.perform(post("/login").with(csrf()).param("username", username).param("password", credential))
87                  .andExpect(status().is3xxRedirection())
88                  .andExpect(redirectedUrl("/"));
89  
90          final var stored = storedHash(username);
91          assertTrue(stored.startsWith(ARGON2_PREFIX), "the stored hash is upgraded");
92          assertTrue(passwordEncoder.matches(credential, stored));
93          assertFalse(passwordEncoder.upgradeEncoding(stored));
94      }
95  
96      /**
97       * Verifies that a failed login leaves the stored hash untouched.
98       *
99       * @throws Exception if a request cannot be performed.
100      */
101     @Test
102     void login_whenCredentialIsWrong_keepsTheStoredHash() throws Exception {
103         final var username = "user-" + UUID.randomUUID();
104         final var hash = BCRYPT_PREFIX + new BCryptPasswordEncoder().encode(AuthServerFixtures.randomValue());
105         insertUser(username, hash);
106 
107         mockMvc.perform(post("/login").with(csrf()).param("username", username)
108                         .param("password", AuthServerFixtures.randomValue()))
109                 .andExpect(redirectedUrl("/login?error"));
110 
111         assertEquals(hash, storedHash(username));
112     }
113 
114     /**
115      * Verifies that a hash that is already current is not rewritten.
116      *
117      * @throws Exception if a request cannot be performed.
118      */
119     @Test
120     void login_whenStoredHashIsCurrent_keepsTheStoredHash() throws Exception {
121         final var username = "user-" + UUID.randomUUID();
122         final var credential = AuthServerFixtures.randomValue();
123         final var hash = passwordEncoder.encode(credential);
124         insertUser(username, hash);
125 
126         mockMvc.perform(post("/login").with(csrf()).param("username", username).param("password", credential))
127                 .andExpect(redirectedUrl("/"));
128 
129         assertEquals(hash, storedHash(username));
130     }
131 
132     /**
133      * Inserts an enabled user with the given stored hash.
134      *
135      * @param username name of the user.
136      * @param hash     stored hash, including its algorithm prefix.
137      */
138     private void insertUser(final String username, final String hash) {
139         final var id = UUID.randomUUID();
140         jdbcClient.sql("INSERT INTO users (id, username, enabled) VALUES (?, ?, true)").params(id, username).update();
141         jdbcClient.sql("INSERT INTO user_passwords (user_id, encoded_password) VALUES (?, ?)").params(id, hash)
142                 .update();
143         jdbcClient.sql("INSERT INTO user_authorities (user_id, authority) VALUES (?, 'ROLE_USER')").param(id).update();
144     }
145 
146     /**
147      * Reads the stored hash of a user.
148      *
149      * @param username name of the user.
150      * @return the stored hash.
151      */
152     private String storedHash(final String username) {
153         return jdbcClient.sql(STORED_HASH).param(username).query(String.class).single();
154     }
155 }
