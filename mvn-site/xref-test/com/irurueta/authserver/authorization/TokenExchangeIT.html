52  /**
53   * Checks the token exchange (RFC 8693) of the real application: a user token exchanged by a confidential client
54   * together with an actor token, the audience honoured by
55   * {@link com.irurueta.authserver.authorization.api.web.exchange.TokenExchangeAudienceContributor}, and the refusals
56   * (client without the grant, token not issued by this server, {@code may_act} that excludes the actor).
57   */

104     /**
105      * The exchange of a user token with an actor token answers an access token of the user, acted by the client, for
106      * the requested audience only.
107      *
108      * @throws Exception if a request cannot be performed.
109      */
110     @Test
111     void tokenExchange_whenSubjectAndActorTokensAreValid_issuesTokenOfTheUserActedByTheClient() throws Exception {
112         final var user = fixtures.createUser();
113         final var subject = fixtures.authorizationCodeTokens(user, fixtures.createCodeClient())
114                 .get(ACCESS_TOKEN).asString();
115         final var client = createExchangeClient(true);
116         final var actor = fixtures.clientCredentialsToken(client, "profile");
117 
118         final var answer = AuthServerFixtures.parse(exchange(client, subject, actor,
119                 AuthServerFixtures.RESOURCE_SERVER_AUDIENCE).andExpect(status().isOk()).andReturn().getResponse()
120                 .getContentAsString());
121 
122         assertEquals(ACCESS_TOKEN_TYPE, answer.get("issued_token_type").asString());
123         final var claims = AuthServerFixtures.claimsOf(answer.get(ACCESS_TOKEN).asString());
124         assertEquals(user.username(), claims.get(OAuth2TokenClaimNames.SUB).asString());
125         assertEquals(client.clientId(), claims.get("act").get(OAuth2TokenClaimNames.SUB).asString());
126         assertEquals(List.of(AuthServerFixtures.RESOURCE_SERVER_AUDIENCE),
127                 audiencesOf(claims.get(OAuth2TokenClaimNames.AUD)));
128     }

183     /**
184      * When the subject token names who may act and the actor is somebody else, the exchange is refused.
185      *
186      * @throws Exception if a request cannot be performed.
187      */
188     @Test
189     void tokenExchange_whenMayActExcludesTheActor_isRefused() throws Exception {
190         final var user = fixtures.createUser();
191         final var codeClient = fixtures.createCodeClient();
192         mayActClaims.put(codeClient.clientId(), Map.of(OAuth2TokenClaimNames.SUB, "somebody-else"));
193         final var subject = fixtures.authorizationCodeTokens(user, codeClient).get(ACCESS_TOKEN).asString();
194         assertTrue(AuthServerFixtures.claimsOf(subject).has("may_act"));
195         final var client = createExchangeClient(true);
196         final var actor = fixtures.clientCredentialsToken(client, "profile");
197 
198         final var answer = refused(client, subject, actor);
199 
200         assertEquals("invalid_grant", answer.get(ERROR).asString());

203     /**
204      * When the subject token names the actor that exchanges it, the exchange succeeds.
205      *
206      * @throws Exception if a request cannot be performed.
207      */
208     @Test
209     void tokenExchange_whenMayActNamesTheActor_issuesToken() throws Exception {
210         final var user = fixtures.createUser();
211         final var codeClient = fixtures.createCodeClient();
212         final var client = createExchangeClient(true);
213         final var actor = fixtures.clientCredentialsToken(client, "profile");
214         final var actorClaims = AuthServerFixtures.claimsOf(actor);
215         final var mayAct = new HashMap<String, Object>();
216         mayAct.put(OAuth2TokenClaimNames.ISS, issuerOf(actorClaims));
217         mayAct.put(OAuth2TokenClaimNames.SUB, client.clientId());
218         mayActClaims.put(codeClient.clientId(), mayAct);
219         final var subject = fixtures.authorizationCodeTokens(user, codeClient).get(ACCESS_TOKEN).asString();
220 
221         exchange(client, subject, actor, AuthServerFixtures.RESOURCE_SERVER_AUDIENCE).andExpect(status().isOk());
222     }

224     /**
225      * Reads the issuer of a token as the framework compares it: the stored claims of an issued token hold {@code iss}
226      * as a {@link URL}, and a {@code may_act} claim must carry the same type to match.
227      *
228      * @param claims the claims of the token.
229      * @return the issuer.
230      * @throws MalformedURLException if the issuer is not a URL.
231      */
232     private static URL issuerOf(final JsonNode claims) throws MalformedURLException {
233         return URI.create(claims.get(OAuth2TokenClaimNames.ISS).asString()).toURL();
234     }
235 
236     /**
237      * Reads an {@code aud} claim, which a JWT serializes as a string when it has one value and as an array otherwise.
238      *
239      * @param aud the claim.
240      * @return its values.
241      */
242     private static List<String> audiencesOf(final JsonNode aud) {
243         if (aud.isArray()) {
244             return aud.values().stream().map(JsonNode::asString).toList();
245         }
246         return List.of(aud.asString());
247     }

249     /**
250      * Creates a confidential client for the exchange.
251      *
252      * @param withExchangeGrant whether it has the token exchange grant.
253      * @return the client.
254      */
255     private TestClient createExchangeClient(final boolean withExchangeGrant) {
256         final var grants = withExchangeGrant
257                 ? List.of("client_credentials", DefaultClientDefinitions.TOKEN_EXCHANGE_GRANT)
258                 : List.of("client_credentials");
259         return fixtures.createClient(grants, EXCHANGE_SCOPES,
260                 List.of(AuthServerFixtures.RESOURCE_SERVER_AUDIENCE, AuthServerFixtures.API_AUDIENCE));
261     }

263     /**
264      * Performs a token exchange request.
265      *
266      * @param client   the exchanging client.
267      * @param subject  the subject token.
268      * @param actor    the actor token.
269      * @param audience the requested audience.
270      * @return the result actions.
271      * @throws Exception if the request cannot be performed.
272      */
273     private ResultActions exchange(final TestClient client, final String subject, final String actor,
274                                    final String audience)
275             throws Exception {
276         return mockMvc.perform(post(AuthServerFixtures.TOKEN_PATH)
277                 .header(AuthServerFixtures.AUTHORIZATION,
278                         AuthServerFixtures.basic(client.clientId(), client.credential()))
279                 .param(AuthServerFixtures.GRANT_TYPE, DefaultClientDefinitions.TOKEN_EXCHANGE_GRANT)
280                 .param("subject_token", subject).param("subject_token_type", ACCESS_TOKEN_TYPE)
281                 .param("actor_token", actor).param("actor_token_type", ACCESS_TOKEN_TYPE)
282                 .param("audience", audience));
283     }

285     /**
286      * Performs a token exchange request that must be answered with a {@code 400}.
287      *
288      * @param client  the exchanging client.
289      * @param subject the subject token.
290      * @param actor   the actor token.
291      * @return the error answer.
292      * @throws Exception if the request cannot be performed.
293      */
294     private JsonNode refused(final TestClient client, final String subject, final String actor) throws Exception {
295         return AuthServerFixtures.parse(exchange(client, subject, actor, AuthServerFixtures.RESOURCE_SERVER_AUDIENCE)
296                 .andExpect(status().isBadRequest()).andReturn().getResponse().getContentAsString());
297     }

299     /**
300      * Adds a {@code may_act} claim to the access tokens of the clients registered in the test. It stays
301      * package-private: the test class's own {@code @Import} cannot name a private nested class.
302      */
303     @TestConfiguration(proxyBeanMethods = false)
304     static class MayActConfiguration {
305 
306         /**
307          * Claims by client identifier.
308          *
309          * @return the registry.
310          */
311         @Bean
312         Map<String, Map<String, Object>> mayActClaims() {
313             return new ConcurrentHashMap<>();
314         }
315 
316         /**
317          * Contributes the registered claim to the access tokens of its client.
318          *
319          * @param registry the claims by client identifier.
320          * @return the contributor.
321          */
322         @Bean
323         @Order(130)
324         JwtClaimsContributor mayActContributor(final Map<String, Map<String, Object>> registry) {
325             return (context, claims) -> {
326                 final var mayAct = registry.get(context.getRegisteredClient().getClientId());
327                 if (mayAct != null && OAuth2TokenType.ACCESS_TOKEN.equals(context.getTokenType())) {
328                     claims.put("may_act", new HashMap<>(mayAct));
329                 }
330             };
331         }
332     }
