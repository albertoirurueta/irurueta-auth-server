41  /**
42   * Checks the development data seeding against a PostgreSQL container, with the {@code dev} profile active.
43   */
44  @PostgresIT
45  @ActiveProfiles("dev")
46  class DevDataInitializerIT {
47  
48      /** Initializer under test. */
49      @Autowired
50      private DevDataInitializer initializer;
51  
52      /** JDBC client used to count the seeded rows. */
53      @Autowired
54      private JdbcClient jdbcClient;
55  
56      /** Repository of registered clients. */
57      @Autowired
58      private RegisteredClientRepository clients;
59  
60      /** Service that loads the seeded user. */
61      @Autowired
62      private UserDetailsService userDetailsService;
63  
64      /** Encoder of the application. */
65      @Autowired
66      private PasswordEncoder passwordEncoder;
67  
68      /**
69       * Verifies that the data is created once, even if the initializer runs again.
70       */
71      @Test
72      void run_whenCalledRepeatedly_createsDataOnceAndIsIdempotent() {
73          // the runner already executed once at context start; running it again must not duplicate anything
74          initializer.run(new DefaultApplicationArguments());
75          initializer.run(new DefaultApplicationArguments());
76  
77          assertEquals(1, jdbcClient.sql("SELECT count(*) FROM users WHERE username = 'demo'")
78                  .query(Integer.class).single());
79          assertEquals(1, jdbcClient.sql("SELECT count(*) FROM oauth2_registered_client WHERE client_id = 'bff'")
80                  .query(Integer.class).single());
81  
82          final var user = userDetailsService.loadUserByUsername("demo");
83          assertEquals(List.of("ROLE_USER"), user.getAuthorities().stream().map(GrantedAuthority::getAuthority).toList());
84          assertTrue(user.getPassword().startsWith("{" + PasswordConfiguration.DEFAULT_ENCODING_ID + "}"));
85  
86          final var client = clients.findByClientId("bff");
87          assertNotNull(client);
88          assertTrue(client.getClientSettings().isRequireProofKey());
89          assertEquals(Set.of("http://localhost:8080/login/oauth2/code/bff"), client.getRedirectUris(),
90                  "the dev profile supplies the local redirect URI");
91          assertNotNull(clients.findByClientId("spa-public"), "the dev profile configures the public client too");
92          assertFalse(passwordEncoder.upgradeEncoding(client.getClientSecret()));
93  
94          final var device = clients.findByClientId("device-client");
95          assertNotNull(device, "the dev profile seeds the public client of the device flow");
96          assertTrue(device.getAuthorizationGrantTypes().contains(AuthorizationGrantType.DEVICE_CODE));
97          assertTrue(device.getClientAuthenticationMethods().contains(ClientAuthenticationMethod.NONE));
98          assertTrue(device.getRedirectUris().isEmpty());
99      }
