56  /**
57   * Checks the sign-in with a client certificate over TLS client authentication on the dedicated port, with a real
58   * handshake: the server presents a certificate issued by an in-test certificate authority and asks the client for one;
59   * the browser opens the login page on that port and presses the certificate button, which posts to
60   * {@code POST /login/x509} with the CSRF token; a certificate bound to an account signs it in (the tokens report
61   * {@code sc}), and a certificate bound to nobody, one whose binding was removed, one that expired and a client with no
62   * certificate at all are sent back to the login page.
63   * A {@code GET} of the path never signs anybody in, and the login page served on the main port posts to the TLS port.
64   * The authority, the server and the client certificates are generated at run time and live in a temporary directory
65   * that is removed afterwards; nothing is kept in the repository.
66   */

156     /**
157      * Verifies that a bound certificate signs its account in over TLS, resumes the authorization request and shows
158      * as {@code sc} in the tokens, and that the session that results works on the TLS port.
159      *
160      * @throws Exception if a request cannot be performed.
161      */
162     @Test
163     void x509Login_whenTheClientPresentsABoundCertificate_signsInAndTheTokensReportSc() throws Exception {
164         final var fixtures = fixtures();
165         final var user = fixtures.createUser();
166         final var identity = PKI.issueClient("client-" + UUID.randomUUID());
167         bind(user.username(), identity);
168         final var client = fixtures.createCodeClient();
169         final var verifier = AuthServerFixtures.randomValue() + AuthServerFixtures.randomValue();
170         final var browser = browser(identity);
171 
172         final var asked = send(browser, UriComponentsBuilder
173                 .fromUriString(base() + AuthServerFixtures.authorizeUri(client, verifier, Map.of())).build().encode()
174                 .toUri());
175         assertEquals(FOUND, asked.statusCode());
176         final var resumed = signIn(browser);
177         assertEquals(FOUND, resumed.statusCode());
178         final var authorizeUri = asked.uri().resolve(location(resumed));
179         assertTrue(authorizeUri.getPath().endsWith("/oauth2/authorize"),
180                 "Expected to resume the request: " + authorizeUri);
181         final var authorized = send(browser, authorizeUri);
182         assertEquals(FOUND, authorized.statusCode());
183         final var code = UriComponentsBuilder.fromUri(authorizeUri.resolve(location(authorized))).build()
184                 .getQueryParams().getFirst("code");
185         assertNotNull(code);
186         final var tokens = fixtures.exchangeCode(client, code, verifier);
187 
188         final var claims = AuthServerFixtures.claimsOf(tokens.get("access_token").asString());
189         assertEquals(List.of("sc"), claims.get("amr").valueStream().map(node -> node.asString()).toList());
190         assertEquals(user.username(), claims.get("sub").asString());
191         assertEquals(OK, send(browser, URI.create(base() + PROTECTED_PAGE)).statusCode());
192     }

300      * Binds the certificate of an identity to an account.
301      *
302      * @param username name of the account.
303      * @param identity the identity.
304      * @return the binding.

306     private CertificateBinding bind(final String username, final TestPki.Identity identity) {
307         final var account = userAccounts.findByUsername(username).orElseThrow();
308         return bindCertificateUseCase.execute(account.id(), username, identity.certificate().fingerprintSha256(),

312     /**
313      * Builds a browser that talks TLS with the given identity and keeps its cookies.
314      *
315      * @param identity the certificate it presents, {@code null} for none.
316      * @return the browser.
317      */
318     private static HttpClient browser(final TestPki.Identity identity) {
319         return HttpClient.newBuilder().sslContext(PKI.clientContext(identity)).cookieHandler(new CookieManager())
320                 .followRedirects(HttpClient.Redirect.NEVER).build();

323     /**
324      * Performs a request without following redirects.
325      *
326      * @param browser the browser.
327      * @param uri     the address.
328      * @return the answer.
329      * @throws IOException          if the request fails.
330      * @throws InterruptedException if the thread is interrupted.
331      */
332     private static HttpResponse<String> send(final HttpClient browser, final URI uri)
333             throws IOException, InterruptedException {
334         return browser.send(HttpRequest.newBuilder(uri).GET().build(), HttpResponse.BodyHandlers.ofString());
335     }

337     /**
338      * Signs in the way a user does: opens the login page on the TLS port and submits the form of the certificate
339      * sign-in, with its CSRF token.
340      *
341      * @param browser the browser.
342      * @return the answer of the sign-in.
343      * @throws IOException          if a request fails.
344      * @throws InterruptedException if the thread is interrupted.
345      */
346     private static HttpResponse<String> signIn(final HttpClient browser) throws IOException, InterruptedException {
347         final var page = send(browser, URI.create(base() + LOGIN_PAGE));
348         assertEquals(OK, page.statusCode());
349         final var form = X509_FORM.matcher(page.body());
350         assertTrue(form.find(), "The login page must offer the certificate sign-in");
351         final var csrf = CSRF_FIELD.matcher(form.group(2));
352         assertTrue(csrf.find(), "The form must carry the CSRF token");
353         final var request = HttpRequest.newBuilder(URI.create(base()).resolve(formAction(page.body())))
354                 .header("Content-Type", "application/x-www-form-urlencoded")
355                 .POST(HttpRequest.BodyPublishers.ofString(
356                         "_csrf=" + URLEncoder.encode(csrf.group(1), StandardCharsets.UTF_8)))
357                 .build();
358         return browser.send(request, HttpResponse.BodyHandlers.ofString());
359     }

361     /**
362      * Reads the address the certificate sign-in form of a login page posts to.
363      *
364      * @param page the login page.
365      * @return the address, unescaped.
366      */
367     private static String formAction(final String page) {
368         final var form = X509_FORM.matcher(page);
369         assertTrue(form.find(), "The login page must offer the certificate sign-in");
370         final var action = ACTION.matcher(form.group(1));
371         assertTrue(action.find(), "The form must have an action");
372         return HtmlUtils.htmlUnescape(action.group(1));
373     }

375     /**
376      * Reads the location of a redirect.
377      *
378      * @param response the answer.
379      * @return the location.
380      */
381     private static String location(final HttpResponse<String> response) {
382         return response.headers().firstValue(LOCATION).orElse("");
383     }

385     /**
386      * Checks that an answer sends the browser back to the login page with an error.
387      *
388      * @param response the answer.
389      */
390     private static void assertFailed(final HttpResponse<String> response) {
391         assertEquals(FOUND, response.statusCode());
392         assertTrue(location(response).endsWith(FAILED), "Expected a failed sign-in but got " + location(response));
393     }

404     /**
405      * Finds a free port.
406      *
407      * @return the port.
408      */
409     private static int freePort() {
410         try (var socket = new ServerSocket(0)) {
411             return socket.getLocalPort();
412         } catch (final IOException e) {
413             throw new IllegalStateException("No free port", e);
414         }
