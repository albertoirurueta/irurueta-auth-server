17  package com.irurueta.authserver.authorization;
18  
19  import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
20  import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
21  import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;
22  
23  import com.irurueta.authserver.testsupport.PostgresIT;
24  import org.junit.jupiter.api.Test;
25  import org.springframework.beans.factory.annotation.Autowired;
26  import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
27  import org.springframework.test.web.servlet.MockMvc;
28  
29  /**
30   * Checks the security filter chains of the whole application against a PostgreSQL container.
31   */
32  @PostgresIT
33  @AutoConfigureMockMvc
34  class SecurityIT {
35  
36      /** MockMvc bound to the application context. */
37      @Autowired
38      private MockMvc mockMvc;
39  
40      /**
41       * Verifies that the discovery document and the JWK set are public.
42       *
43       * @throws Exception if a request cannot be performed.
44       */
45      @Test
46      void discoveryAndJwks_whenRequestedAnonymously_arePublic() throws Exception {
47          mockMvc.perform(get("/.well-known/openid-configuration"))
48                  .andExpect(status().isOk())
49                  .andExpect(jsonPath("$.issuer").value("http://localhost:9000"));
50          mockMvc.perform(get("/oauth2/jwks"))
51                  .andExpect(status().isOk())
52                  .andExpect(jsonPath("$.keys[0].kty").value("RSA"));
53      }
54  
55      /**
56       * Verifies that the login page is public and that the rest of the pages require a login.
57       *
58       * @throws Exception if a request cannot be performed.
59       */
60      @Test
61      void loginAndProtectedPages_whenRequestedAnonymously_renderLoginAndRedirect() throws Exception {
62          mockMvc.perform(get("/login")).andExpect(status().isOk());
63          mockMvc.perform(get("/account")).andExpect(status().is3xxRedirection());
64      }
65  }
