19  import static org.mockito.ArgumentMatchers.any;
20  import static org.mockito.Mockito.inOrder;
21  import static org.mockito.Mockito.never;
22  import static org.mockito.Mockito.verify;
23  import static org.mockito.Mockito.verifyNoInteractions;
24  import static org.mockito.Mockito.when;

26  import com.irurueta.authserver.authorization.application.mfa.RecordSecurityEventUseCase;
27  import com.irurueta.authserver.authorization.application.social.LockSignInMethodsUseCase;
28  import com.irurueta.authserver.authorization.domain.mfa.SecurityEventType;
29  import com.irurueta.authserver.authorization.domain.social.FederatedLinkingPolicy;
30  import com.irurueta.authserver.authorization.domain.social.LastSignInMethodException;
31  import com.irurueta.authserver.authorization.domain.social.SignInMethods;
32  import com.irurueta.authserver.authorization.domain.x509.CertificateBinding;
33  import com.irurueta.authserver.authorization.domain.x509.CertificateBindingRepository;
34  import com.irurueta.authserver.authorization.domain.x509.CertificateNotFoundException;
35  import java.time.Instant;
36  import java.util.List;
37  import java.util.UUID;
38  import org.junit.jupiter.api.Test;
39  import org.junit.jupiter.api.extension.ExtendWith;
40  import org.mockito.InjectMocks;
41  import org.mockito.Mock;
42  import org.mockito.Spy;
43  import org.mockito.junit.jupiter.MockitoExtension;
44  
45  /**
46   * Unit tests of {@link UnbindCertificateUseCase}.
47   */
48  @ExtendWith(MockitoExtension.class)
49  class UnbindCertificateUseCaseTest {
50  
51      /** Fingerprint used by the tests. */
52      private static final String FINGERPRINT = "a".repeat(CertificateBinding.FINGERPRINT_LENGTH);
53  
54      /** Instant the certificate was bound. */
55      private static final Instant NOW = Instant.parse("2026-01-01T10:00:00Z");
56  
57      /** Identifier of the account. */
58      private final UUID userId = UUID.randomUUID();
59  
60      /** Identifier of the binding. */
61      private final UUID id = UUID.randomUUID();
62  
63      /** Port that stores the bindings. */
64      @Mock
65      private CertificateBindingRepository certificateBindingRepository;
66  
67      /** Locks and counts the sign-in methods of the account. */
68      @Mock
69      private LockSignInMethodsUseCase lockSignInMethodsUseCase;
70  
71      /** Rules of removing a sign-in method; the real behaviour is wanted. */
72      @Spy
73      private FederatedLinkingPolicy federatedLinkingPolicy = new FederatedLinkingPolicy();
74  
75      /** Use case that records the audit events. */
76      @Mock
77      private RecordSecurityEventUseCase recordSecurityEventUseCase;
78  
79      /** Use case under test. */
80      @InjectMocks
81      private UnbindCertificateUseCase useCase;
82  
83      /**
84       * Verifies that a binding of the account is deleted and audited, after the lock is taken, while a password
85       * remains.
86       */
87      @Test
88      void execute_whenAPasswordRemains_locksDeletesAndAudits() {
89          givenBinding(new SignInMethods(true, 0, false, 1));
90          when(certificateBindingRepository.delete(id, userId)).thenReturn(true);
91  
92          useCase.execute(userId, "alice", id);
93  
94          final var order = inOrder(lockSignInMethodsUseCase, certificateBindingRepository);
95          order.verify(lockSignInMethodsUseCase).execute(userId);
96          order.verify(certificateBindingRepository).delete(id, userId);
97          verify(recordSecurityEventUseCase).execute(SecurityEventType.CERTIFICATE_UNBOUND, userId, "alice",
98                  id.toString());
99      }
100 
101     /**
102      * Verifies that a certificate is unbound while another certificate remains.
103      */
104     @Test
105     void execute_whenAnotherCertificateRemains_deletesTheBinding() {
106         givenBinding(new SignInMethods(false, 0, false, 2));
107         when(certificateBindingRepository.delete(id, userId)).thenReturn(true);
108 
109         useCase.execute(userId, "alice", id);
110 
111         verify(certificateBindingRepository).delete(id, userId);
112     }
113 
114     /**
115      * Verifies that a certificate is unbound while a linked identity remains.
116      */
117     @Test
118     void execute_whenAnIdentityRemains_deletesTheBinding() {
119         givenBinding(new SignInMethods(false, 1, false, 1));
120         when(certificateBindingRepository.delete(id, userId)).thenReturn(true);
121 
122         useCase.execute(userId, "alice", id);
123 
124         verify(certificateBindingRepository).delete(id, userId);
125     }
126 
127     /**
128      * Verifies that the last way to sign in is never removed, and nothing is audited.
129      */
130     @Test
131     void execute_whenCertificateIsTheLastMethod_throwsLastSignInMethodException() {
132         givenBinding(new SignInMethods(false, 0, false, 1));
133 
134         assertThrows(LastSignInMethodException.class, () -> useCase.execute(userId, "alice", id));
135 
136         verify(certificateBindingRepository, never()).delete(any(), any());
137         verifyNoInteractions(recordSecurityEventUseCase);
138     }
139 
140     /**
141      * Verifies that a binding the account does not have is reported, never deleted and not audited.
142      */
143     @Test
144     void execute_whenBindingIsNotTheAccounts_throwsCertificateNotFoundException() {
145         when(lockSignInMethodsUseCase.execute(userId)).thenReturn(new SignInMethods(true, 0, false, 1));
146         when(certificateBindingRepository.findByUser(userId)).thenReturn(List.of(
147                 new CertificateBinding(UUID.randomUUID(), userId, FINGERPRINT, "CN=other", NOW)));
148 
149         assertThrows(CertificateNotFoundException.class, () -> useCase.execute(userId, "alice", id));
150 
151         verify(certificateBindingRepository, never()).delete(any(), any());
152         verifyNoInteractions(recordSecurityEventUseCase);
153     }
154 
155     /**
156      * Verifies that a binding that disappears before it is deleted is reported and not audited.
157      */
158     @Test
159     void execute_whenDeleteFindsNothing_throwsCertificateNotFoundException() {
160         givenBinding(new SignInMethods(true, 0, false, 1));
161         when(certificateBindingRepository.delete(id, userId)).thenReturn(false);
162 
163         assertThrows(CertificateNotFoundException.class, () -> useCase.execute(userId, "alice", id));
164 
165         verifyNoInteractions(recordSecurityEventUseCase);
166     }
167 
168     /**
169      * Arranges the account whose certificate is being unbound.
170      *
171      * @param methods the sign-in methods of the account now, the certificate being removed included.
172      */
173     private void givenBinding(final SignInMethods methods) {
174         when(lockSignInMethodsUseCase.execute(userId)).thenReturn(methods);
175         when(certificateBindingRepository.findByUser(userId)).thenReturn(List.of(
176                 new CertificateBinding(id, userId, FINGERPRINT, "CN=alice", NOW)));
177     }
178 }
