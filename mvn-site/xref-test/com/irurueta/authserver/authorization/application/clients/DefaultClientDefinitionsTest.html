17  package com.irurueta.authserver.authorization.application.clients;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertFalse;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  import java.util.List;
24  import org.junit.jupiter.api.Test;
25  
26  /**
27   * Unit tests of {@link DefaultClientDefinitions}.
28   */
29  class DefaultClientDefinitionsTest {
30  
31      /** Redirect URI given to the definitions. */
32      private static final String REDIRECT_URI = "https://app.example.test/callback";
33  
34      /** Audience of the authorization server's API given to the definitions. */
35      private static final String API_AUDIENCE = "custom-api";
36  
37      /** Scopes both default clients must carry. */
38      private static final List<String> SCOPES = List.of("openid", "profile", "documents.read", "documents.delete");
39  
40      /**
41       * Verifies that the BFF is a confidential client with rotation on, the given redirect URIs and both audiences.
42       */
43      @Test
44      void bff_whenCalled_isConfidentialWithPkceAndRotation() {
45          final var bff = DefaultClientDefinitions.bff(List.of(REDIRECT_URI), API_AUDIENCE);
46  
47          assertEquals(DefaultClientDefinitions.BFF_CLIENT_ID, bff.clientId());
48          assertTrue(bff.confidential());
49          assertEquals(List.of("client_secret_basic"), bff.authMethods());
50          assertEquals(List.of("authorization_code", "refresh_token"), bff.grantTypes());
51          assertTrue(bff.requireProofKey());
52          assertFalse(bff.reuseRefreshTokens());
53          assertTrue(bff.requireConsent());
54          assertEquals(List.of(REDIRECT_URI), bff.redirectUris());
55          assertEquals(List.of(DefaultClientDefinitions.RESOURCE_SERVER_AUDIENCE, API_AUDIENCE), bff.audiences());
56      }
57  
58      /**
59       * Verifies that the SPA is a public client without client authentication, the given redirect URIs and both
60       * audiences.
61       */
62      @Test
63      void spaPublic_whenCalled_isPublicWithPkce() {
64          final var spa = DefaultClientDefinitions.spaPublic(List.of(REDIRECT_URI), API_AUDIENCE);
65  
66          assertEquals(DefaultClientDefinitions.SPA_PUBLIC_CLIENT_ID, spa.clientId());
67          assertFalse(spa.confidential());
68          assertEquals(List.of("none"), spa.authMethods());
69          assertTrue(spa.requireProofKey());
70          assertFalse(spa.requireConsent());
71          assertEquals(List.of(REDIRECT_URI), spa.redirectUris());
72          assertEquals(List.of(DefaultClientDefinitions.RESOURCE_SERVER_AUDIENCE, API_AUDIENCE), spa.audiences());
73      }
74  
75      /**
76       * Verifies that both clients may read and delete documents and never get the {@code admin} scope.
77       */
78      @Test
79      void scopes_whenDefined_includeTheDocumentScopesButNotAdmin() {
80          final var bff = DefaultClientDefinitions.bff(List.of(REDIRECT_URI), API_AUDIENCE);
81          final var spa = DefaultClientDefinitions.spaPublic(List.of(REDIRECT_URI), API_AUDIENCE);
82  
83          assertEquals(SCOPES, bff.scopes());
84          assertEquals(SCOPES, spa.scopes());
85          assertFalse(bff.scopes().contains("admin"));
86          assertFalse(spa.scopes().contains("admin"));
87      }
88  
89      /**
90       * Verifies that the post-logout redirect URIs reach both clients, and that they are empty by default.
91       */
92      @Test
93      void definitions_whenPostLogoutRedirectUrisAreGiven_carryThem() {
94          final var uris = List.of("https://app.example.test/");
95  
96          assertEquals(uris, DefaultClientDefinitions.bff(List.of(REDIRECT_URI), API_AUDIENCE, uris)
97                  .postLogoutRedirectUris());
98          assertEquals(uris, DefaultClientDefinitions.spaPublic(List.of(REDIRECT_URI), API_AUDIENCE, uris)
99                  .postLogoutRedirectUris());
100         assertEquals(List.of(), DefaultClientDefinitions.bff(List.of(REDIRECT_URI), API_AUDIENCE)
101                 .postLogoutRedirectUris());
102         assertEquals(List.of(), DefaultClientDefinitions.spaPublic(List.of(REDIRECT_URI), API_AUDIENCE)
103                 .postLogoutRedirectUris());
104     }
105 }
