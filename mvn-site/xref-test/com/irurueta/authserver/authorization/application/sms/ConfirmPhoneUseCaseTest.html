49  /**
50   * Unit tests of {@link ConfirmPhoneUseCase}.
51   */
52  @ExtendWith(MockitoExtension.class)
53  class ConfirmPhoneUseCaseTest {
54  
55      /** Current instant used by the tests. */
56      private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
57  
58      /** Code typed by the user. */
59      private static final String CODE = "123456";
60  
61      /** Stored hash. */
62      private static final String HASH = "stored-hash";
63  
64      /** Attempts allowed. */
65      private static final int MAX_ATTEMPTS = 3;
66  
67      /** Phone being verified. */
68      private static final PhoneNumber PHONE = new PhoneNumber("+34600000000");
69  
70      /** Another phone, verified before. */
71      private static final PhoneNumber PREVIOUS_PHONE = new PhoneNumber("+34600000001");
72  
73      /** Identifier of the user. */
74      private static final UUID USER_ID = UUID.randomUUID();
75  
76      /** Name of the user. */
77      private static final String USERNAME = "alice";
78  
79      /** Number of trusted devices revoked by a replacement. */
80      private static final int REVOKED = 2;
81  
82      /** Port that stores the verified phones. */
83      @Mock
84      private UserPhoneRepository userPhoneRepository;
85  
86      /** Port that stores the pending verification. */
87      @Mock
88      private PhoneVerificationRepository phoneVerificationRepository;
89  
90      /** Hasher of the code. */
91      @Mock
92      private OneTimeCodeHasher oneTimeCodeHasher;
93  
94      /** Port that stores the trusted devices. */
95      @Mock
96      private TrustedDeviceRepository trustedDeviceRepository;
97  
98      /** Use case that records the audit events. */
99      @Mock
100     private RecordSecurityEventUseCase recordSecurityEventUseCase;
101 
102     /** Real settings with three attempts. */
103     @Spy
104     private SmsProperties smsProperties = new SmsProperties(null, null, null, 0L, null, 0L, 0L, 0, null, MAX_ATTEMPTS,
105             null);
106 
107     /** Source of the current time. */
108     @Mock
109     private Clock clock;
110 
111     /** Use case under test. */
112     @InjectMocks
113     private ConfirmPhoneUseCase useCase;
114 
115     /**
116      * Verifies that a correct code stores the verified phone and clears the pending verification.
117      */
118     @Test
119     void execute_whenCodeIsCorrect_savesPhoneAndDeletesVerification() {
120         stubPending(NOW.plusSeconds(1));
121         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(1);
122         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(true);
123         when(userPhoneRepository.save(any(UserPhone.class))).thenReturn(true);
124         final var saved = ArgumentCaptor.<UserPhone>captor();
125 
126         assertTrue(useCase.execute(USER_ID, USERNAME, CODE));
127 
128         verify(userPhoneRepository).save(saved.capture());
129         assertEquals(new UserPhone(USER_ID, PHONE, NOW), saved.getValue());
130         verify(phoneVerificationRepository).deleteByUserId(USER_ID);
131         verifyNoInteractions(trustedDeviceRepository, recordSecurityEventUseCase);
132     }
133 
134     /**
135      * Verifies that confirming a number that replaces another verified one revokes the trusted devices and audits
136      * the removal of the old phone.
137      */
138     @Test
139     void execute_whenNumberReplacesAnotherVerifiedOne_revokesTrustedDevicesAndAudits() {
140         stubPending(NOW.plusSeconds(1));
141         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(1);
142         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(true);
143         when(userPhoneRepository.findByUserId(USER_ID))
144                 .thenReturn(Optional.of(new UserPhone(USER_ID, PREVIOUS_PHONE, NOW.minusSeconds(1))));
145         when(userPhoneRepository.save(any(UserPhone.class))).thenReturn(true);
146         when(trustedDeviceRepository.deleteByUser(USER_ID)).thenReturn(REVOKED);
147 
148         assertTrue(useCase.execute(USER_ID, USERNAME, CODE));
149 
150         verify(recordSecurityEventUseCase).execute(SecurityEventType.SECOND_FACTOR_REMOVED, USER_ID, USERNAME,
151                 ConfirmPhoneUseCase.REPLACED_DETAIL + REVOKED);
152     }
153 
154     /**
155      * Verifies that confirming the number the user already has verified is not a replacement.
156      */
157     @Test
158     void execute_whenNumberIsTheVerifiedOne_doesNotRevokeAnything() {
159         stubPending(NOW.plusSeconds(1));
160         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(1);
161         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(true);
162         when(userPhoneRepository.findByUserId(USER_ID))
163                 .thenReturn(Optional.of(new UserPhone(USER_ID, PHONE, NOW.minusSeconds(1))));
164         when(userPhoneRepository.save(any(UserPhone.class))).thenReturn(true);
165 
166         assertTrue(useCase.execute(USER_ID, USERNAME, CODE));
167 
168         verifyNoInteractions(trustedDeviceRepository, recordSecurityEventUseCase);
169     }
170 
171     /**
172      * Verifies that a number verified by another user meanwhile fails like a wrong code.
173      */
174     @Test
175     void execute_whenNumberIsTakenMeanwhile_returnsFalse() {
176         stubPending(NOW.plusSeconds(1));
177         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(1);
178         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(true);
179         when(userPhoneRepository.save(any(UserPhone.class))).thenReturn(false);
180 
181         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
182 
183         verify(phoneVerificationRepository).deleteByUserId(USER_ID);
184         verifyNoInteractions(trustedDeviceRepository, recordSecurityEventUseCase);
185     }
186 
187     /**
188      * Verifies that a missing verification fails.
189      */
190     @Test
191     void execute_whenNothingIsPending_returnsFalse() {
192         when(clock.instant()).thenReturn(NOW);
193         when(phoneVerificationRepository.findByUserId(USER_ID)).thenReturn(Optional.empty());
194 
195         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
196 
197         verifyNoInteractions(userPhoneRepository, oneTimeCodeHasher);
198     }
199 
200     /**
201      * Verifies that an expired verification is deleted and fails.
202      */
203     @Test
204     void execute_whenVerificationIsExpired_deletesItAndReturnsFalse() {
205         stubPending(NOW);
206 
207         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
208 
209         verify(phoneVerificationRepository).deleteByUserId(USER_ID);
210         verifyNoInteractions(oneTimeCodeHasher);
211     }
212 
213     /**
214      * Verifies that a wrong code below the limit fails and keeps the verification.
215      */
216     @Test
217     void execute_whenCodeIsWrongBelowLimit_returnsFalseAndKeepsVerification() {
218         stubPending(NOW.plusSeconds(1));
219         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(1);
220         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(false);
221 
222         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
223 
224         verify(phoneVerificationRepository, never()).deleteByUserId(USER_ID);
225         verifyNoInteractions(userPhoneRepository);
226     }
227 
228     /**
229      * Verifies that the last wrong attempt deletes the verification.
230      */
231     @Test
232     void execute_whenLastAttemptIsWrong_deletesVerification() {
233         stubPending(NOW.plusSeconds(1));
234         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(MAX_ATTEMPTS);
235         when(oneTimeCodeHasher.matches(CODE, HASH)).thenReturn(false);
236 
237         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
238 
239         verify(phoneVerificationRepository).deleteByUserId(USER_ID);
240     }
241 
242     /**
243      * Verifies that a counter past the limit deletes the verification without comparing the code.
244      */
245     @Test
246     void execute_whenAttemptsExceedLimit_deletesWithoutComparing() {
247         stubPending(NOW.plusSeconds(1));
248         when(phoneVerificationRepository.incrementAttempts(USER_ID)).thenReturn(MAX_ATTEMPTS + 1);
249 
250         assertFalse(useCase.execute(USER_ID, USERNAME, CODE));
251 
252         verify(phoneVerificationRepository).deleteByUserId(USER_ID);
253         verifyNoInteractions(oneTimeCodeHasher, userPhoneRepository);
254     }
255 
256     /**
257      * Stubs a pending verification.
258      *
259      * @param expiresAt when it expires.
260      */
261     private void stubPending(final Instant expiresAt) {
262         when(clock.instant()).thenReturn(NOW);
263         when(phoneVerificationRepository.findByUserId(USER_ID))
264                 .thenReturn(Optional.of(new PhoneVerification(USER_ID, PHONE, HASH, 0, expiresAt)));
265     }
