52  /**
53   * Unit tests of {@link StartPhoneEnrolmentUseCase}.
54   */
55  @ExtendWith(MockitoExtension.class)
56  class StartPhoneEnrolmentUseCaseTest {
57  
58      /** Current instant used by the tests. */
59      private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
60  
61      /** Number as typed. */
62      private static final String TYPED = "600 00 00 00";
63  
64      /** Normalised number. */
65      private static final PhoneNumber PHONE = new PhoneNumber("+34600000000");
66  
67      /** Clear code from the mocked generator. */
68      private static final String CODE = "123456";
69  
70      /** Hash from the mocked hasher. */
71      private static final String HASH = "hash-of-code";
72  
73      /** Digits of the code. */
74      private static final int LENGTH = 6;
75  
76      /** Validity of the code. */
77      private static final Duration TTL = Duration.ofMinutes(10);
78  
79      /** Identifier of the user. */
80      private static final UUID USER_ID = UUID.randomUUID();
81  
82      /** Port that normalises the number. */
83      @Mock
84      private PhoneNormalizer phoneNormalizer;
85  
86      /** Port that finds the country. */
87      @Mock
88      private PhoneCountryResolver phoneCountryResolver;
89  
90      /** Port that reads the verified phones. */
91      @Mock
92      private UserPhoneRepository userPhoneRepository;
93  
94      /** Port that stores the pending verification. */
95      @Mock
96      private PhoneVerificationRepository phoneVerificationRepository;
97  
98      /** Generator of the code. */
99      @Mock
100     private OneTimeCodeGenerator oneTimeCodeGenerator;
101 
102     /** Hasher of the code. */
103     @Mock
104     private OneTimeCodeHasher oneTimeCodeHasher;
105 
106     /** Use case that sends the message. */
107     @Mock
108     private SendSmsOneTimeCodeUseCase sendSmsOneTimeCodeUseCase;
109 
110     /** Real settings: Spain is served. */
111     @Spy
112     private SmsProperties smsProperties = new SmsProperties("ES", Set.of("ES"), null, 0L, null, 0L, 0L, LENGTH, TTL,
113             0, null);
114 
115     /** Source of the current time. */
116     @Mock
117     private Clock clock;
118 
119     /** Use case under test. */
120     @InjectMocks
121     private StartPhoneEnrolmentUseCase useCase;
122 
123     /**
124      * Verifies that a free number gets its hash stored and the clear code sent.
125      */
126     @Test
127     void execute_whenNumberIsFree_storesHashAndSendsCode() {
128         stubNumber();
129         stubCode();
130         when(clock.instant()).thenReturn(NOW);
131         when(userPhoneRepository.findByPhone(PHONE)).thenReturn(Optional.empty());
132         final var saved = ArgumentCaptor.<PhoneVerification>captor();
133         final var text = ArgumentCaptor.<String>captor();
134 
135         final var result = useCase.execute(USER_ID, TYPED);
136 
137         assertEquals(new PhoneEnrolmentStarted(TTL), result);
138         verify(phoneVerificationRepository).save(saved.capture());
139         assertEquals(HASH, saved.getValue().codeHash());
140         assertEquals(PHONE, saved.getValue().phone());
141         assertEquals(USER_ID, saved.getValue().userId());
142         assertEquals(0, saved.getValue().attempts());
143         assertEquals(NOW.plus(TTL), saved.getValue().expiresAt());
144         verify(sendSmsOneTimeCodeUseCase).execute(eqPhone(), text.capture());
145         assertEquals(SmsMessages.verificationCode(CODE, TTL), text.getValue());
146     }
147 
148     /**
149      * Verifies that re-enrolling the user's own verified number is allowed.
150      */
151     @Test
152     void execute_whenNumberBelongsToSameUser_sendsCode() {
153         stubNumber();
154         stubCode();
155         when(clock.instant()).thenReturn(NOW);
156         when(userPhoneRepository.findByPhone(PHONE))
157                 .thenReturn(Optional.of(new UserPhone(USER_ID, PHONE, Instant.EPOCH)));
158 
159         useCase.execute(USER_ID, TYPED);
160 
161         verify(sendSmsOneTimeCodeUseCase).execute(eqPhone(), anyString());
162     }
163 
164     /**
165      * Verifies that a number owned by another user looks like a success but nothing is stored or sent.
166      */
167     @Test
168     void execute_whenNumberBelongsToAnotherUser_returnsSameResultAndSendsNothing() {
169         stubNumber();
170         stubCode();
171         when(userPhoneRepository.findByPhone(PHONE))
172                 .thenReturn(Optional.of(new UserPhone(UUID.randomUUID(), PHONE, Instant.EPOCH)));
173 
174         final var result = useCase.execute(USER_ID, TYPED);
175 
176         assertEquals(new PhoneEnrolmentStarted(TTL), result);
177         verifyNoInteractions(phoneVerificationRepository, sendSmsOneTimeCodeUseCase);
178     }
179 
180     /**
181      * Verifies that an invalid number fails visibly.
182      */
183     @Test
184     void execute_whenNumberIsInvalid_throwsInvalidPhoneNumberException() {
185         when(phoneNormalizer.normalize(TYPED, "ES")).thenReturn(Optional.empty());
186 
187         assertThrows(InvalidPhoneNumberException.class, () -> useCase.execute(USER_ID, TYPED));
188 
189         verifyNoInteractions(phoneVerificationRepository, sendSmsOneTimeCodeUseCase);
190     }
191 
192     /**
193      * Verifies that an unknown or disallowed country fails visibly.
194      */
195     @Test
196     void execute_whenCountryIsNotAllowed_throwsSmsCountryNotAllowedException() {
197         when(phoneNormalizer.normalize(TYPED, "ES")).thenReturn(Optional.of(PHONE));
198         when(phoneCountryResolver.countryOf(PHONE)).thenReturn(Optional.of("FR"), Optional.empty());
199 
200         assertThrows(SmsCountryNotAllowedException.class, () -> useCase.execute(USER_ID, TYPED));
201         assertThrows(SmsCountryNotAllowedException.class, () -> useCase.execute(USER_ID, TYPED));
202 
203         verifyNoInteractions(phoneVerificationRepository, sendSmsOneTimeCodeUseCase);
204     }
205 
206     /**
207      * Verifies that an exhausted budget is not reported. (A provider failure cannot reach the use case: the message is
208      * sent after the transaction commits.)
209      */
210     @Test
211     void execute_whenSendingIsRefused_returnsSameResult() {
212         stubNumber();
213         stubCode();
214         when(clock.instant()).thenReturn(NOW);
215         when(userPhoneRepository.findByPhone(PHONE)).thenReturn(Optional.empty());
216         doThrow(new SmsBudgetExceededException())
217                 .when(sendSmsOneTimeCodeUseCase).execute(any(PhoneNumber.class), anyString());
218 
219         assertEquals(new PhoneEnrolmentStarted(TTL), useCase.execute(USER_ID, TYPED));
220     }
221 
222     /**
223      * Matches the normalised phone.
224      *
225      * @return {@code null}, as Mockito matchers do.
226      */
227     private static PhoneNumber eqPhone() {
228         return eq(PHONE);
229     }
230 
231     /**
232      * Stubs the normaliser and the country.
233      */
234     private void stubNumber() {
235         when(phoneNormalizer.normalize(TYPED, "ES")).thenReturn(Optional.of(PHONE));
236         when(phoneCountryResolver.countryOf(PHONE)).thenReturn(Optional.of("ES"));
237     }
238 
239     /**
240      * Stubs the generator and the hasher.
241      */
242     private void stubCode() {
243         when(oneTimeCodeGenerator.generate(LENGTH)).thenReturn(CODE);
244         when(oneTimeCodeHasher.hash(CODE)).thenReturn(HASH);
245     }
