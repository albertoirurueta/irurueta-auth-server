16  package com.irurueta.authserver.authorization;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertFalse;
20  import static org.junit.jupiter.api.Assertions.assertTrue;
21  
22  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
23  import java.util.Map;
24  import java.util.UUID;
25  import org.junit.jupiter.api.Test;
26  import org.springframework.mock.web.MockHttpSession;
27  import org.springframework.test.context.DynamicPropertyRegistry;
28  import org.springframework.test.context.DynamicPropertySource;
29  
30  /**
31   * Checks the sign-in with Microsoft Entra ID as a multi-tenant application restricted to a list of tenants, against a
32   * fake provider: the issuer of an Entra ID token is the issuer of the tenant the token comes from, so it is checked
33   * against the token's own tenant, and the tenant has to be one of the allowed ones. A token of a tenant that is not
34   * allowed, a token whose issuer is another tenant's and an address Entra does not vouch for are all refused, and an
35   * address Entra marks as verified by either of its claims is accepted. Every client id, secret and tenant is
36   * generated at run time.
37   */
38  class SocialEntraMultiTenantIT extends AbstractSocialFlow {
39  
40      /** Client id of the application at Microsoft, generated for the run. */
41      private static final String CLIENT_ID = "microsoft-" + UUID.randomUUID();
42  
43      /** The tenant that is allowed to sign in, generated for the run. */
44      private static final String ALLOWED_TENANT = UUID.randomUUID().toString();
45  
46      /** A tenant that is not allowed, generated for the run. */
47      private static final String OTHER_TENANT = UUID.randomUUID().toString();
48  
49      /** Beginning of the issuer of a tenant. */
50      private static final String ISSUER_BASE = "https://login.microsoftonline.com/";
51  
52      /** End of the issuer of a tenant. */
53      private static final String ISSUER_END = "/v2.0";
54  
55      /**
56       * Configures Microsoft as a multi-tenant application restricted to one tenant.
57       *
58       * @param registry registry of the properties of the context.
59       */
60      @DynamicPropertySource
61      static void providers(final DynamicPropertyRegistry registry) {
62          registry.add("authserver.social.providers.microsoft.client-id", () -> CLIENT_ID);
63          registry.add("authserver.social.providers.microsoft.client-secret", () -> UUID.randomUUID().toString());
64          registry.add("authserver.social.providers.microsoft.tenant", () -> "organizations");
65          registry.add("authserver.social.providers.microsoft.allowed-tenants", () -> ALLOWED_TENANT);
66      }
67  
68      /**
69       * Verifies that a token of the allowed tenant, issued by that tenant, provisions an account when Entra marks the
70       * address as verified with its own claim.
71       *
72       * @throws Exception if a request cannot be performed.
73       */
74      @Test
75      void signIn_whenTheTokenIsOfAnAllowedTenant_provisionsAndLinksAnAccount() throws Exception {
76          final var email = "entra-" + UUID.randomUUID() + "@example.com";
77          final var session = new MockHttpSession();
78  
79          signInWithOidc(session, "microsoft", issuerOf(ALLOWED_TENANT), CLIENT_ID,
80                  entraClaims("entra-subject-" + UUID.randomUUID(), email, ALLOWED_TENANT, Map.of("xms_edov", true)));
81  
82          assertEquals(email, signedInName(session).orElseThrow());
83          assertEquals(IdentityProvider.MICROSOFT, identitiesOf(accountOf(email).orElseThrow()).getFirst().provider());
84      }
85  
86      /**
87       * Verifies that the standard verified-address claim is accepted as well.
88       *
89       * @throws Exception if a request cannot be performed.
90       */
91      @Test
92      void signIn_whenTheStandardVerifiedClaimIsPresent_provisionsAnAccount() throws Exception {
93          final var email = "entra-standard-" + UUID.randomUUID() + "@example.com";
94          final var session = new MockHttpSession();
95  
96          signInWithOidc(session, "microsoft", issuerOf(ALLOWED_TENANT), CLIENT_ID,
97                  entraClaims("entra-subject-" + UUID.randomUUID(), email, ALLOWED_TENANT,
98                          Map.of("email_verified", true)));
99  
100         assertEquals(email, signedInName(session).orElseThrow());
101     }
102 
103     /**
104      * Verifies that a token of a tenant that is not on the list is refused, though its issuer is that tenant's own.
105      *
106      * @throws Exception if a request cannot be performed.
107      */
108     @Test
109     void signIn_whenTheTenantIsNotAllowed_isRefused() throws Exception {
110         final var email = "other-tenant-" + UUID.randomUUID() + "@example.com";
111         final var session = new MockHttpSession();
112 
113         final var result = signInWithOidc(session, "microsoft", issuerOf(OTHER_TENANT), CLIENT_ID,
114                 entraClaims("entra-subject-" + UUID.randomUUID(), email, OTHER_TENANT, Map.of("xms_edov", true)));
115 
116         assertRefused(result, session);
117         assertTrue(accountOf(email).isEmpty());
118     }
119 
120     /**
121      * Verifies that a token whose issuer is the issuer of another tenant than the one it claims to come from is
122      * refused: an allowed tenant id cannot be borrowed.
123      *
124      * @throws Exception if a request cannot be performed.
125      */
126     @Test
127     void signIn_whenTheIssuerIsNotTheIssuerOfTheTokensTenant_isRefused() throws Exception {
128         final var email = "mixed-" + UUID.randomUUID() + "@example.com";
129         final var session = new MockHttpSession();
130 
131         final var result = signInWithOidc(session, "microsoft", issuerOf(OTHER_TENANT), CLIENT_ID,
132                 entraClaims("entra-subject-" + UUID.randomUUID(), email, ALLOWED_TENANT, Map.of("xms_edov", true)));
133 
134         assertRefused(result, session);
135         assertTrue(accountOf(email).isEmpty());
136     }
137 
138     /**
139      * Verifies that an address Entra does not vouch for, which any tenant admin can set to anything, never links to
140      * the account that owns it nor provisions an account.
141      *
142      * @throws Exception if a request cannot be performed.
143      */
144     @Test
145     void signIn_whenEntraDoesNotVouchForTheEmail_isRefusedAndLinksNothing() throws Exception {
146         final var user = getFixtures().createUser();
147         final var session = new MockHttpSession();
148 
149         final var result = signInWithOidc(session, "microsoft", issuerOf(ALLOWED_TENANT), CLIENT_ID,
150                 entraClaims("entra-subject-" + UUID.randomUUID(), user.email(), ALLOWED_TENANT, Map.of()));
151 
152         assertRefused(result, session);
153         assertFalse(identitiesOf(accountOf(user.email()).orElseThrow()).stream()
154                 .anyMatch(identity -> identity.provider() == IdentityProvider.MICROSOFT));
155     }
156 
157     /**
158      * Gives the issuer of a tenant.
159      *
160      * @param tenant the tenant id.
161      * @return its issuer.
162      */
163     private static String issuerOf(final String tenant) {
164         return ISSUER_BASE + tenant + ISSUER_END;
165     }
166 }
