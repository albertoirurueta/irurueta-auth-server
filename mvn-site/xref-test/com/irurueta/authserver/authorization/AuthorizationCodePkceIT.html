54  /**
55   * Full authorization code + PKCE round trip against the real application and a PostgreSQL container:
56   * authorize, log in, obtain the code, exchange it with the code verifier and validate the issued tokens
57   * against the published JWK set.
58   */
59  @PostgresIT(webEnvironment = WebEnvironment.RANDOM_PORT)
60  @AutoConfigureMockMvc
61  class AuthorizationCodePkceIT {
62  
63      /** Redirect URI of the test client. */
64      private static final String REDIRECT_URI = "http://127.0.0.1:8080/callback";
65  
66      /** Number of random bytes of generated values. */
67      private static final int RANDOM_BYTES = 32;
68  
69      /** Name of the request header that carries the client credentials. */
70      private static final String AUTHORIZATION = "Authorization";
71  
72      /** Name of the token request parameter that carries the grant type. */
73      private static final String GRANT_TYPE = "grant_type";
74  
75      /** Name of the token request parameter that carries the authorization code. */
76      private static final String CODE = "code";
77  
78      /** Name of the token request parameter that carries the redirect URI. */
79      private static final String REDIRECT_URI_PARAM = "redirect_uri";
80  
81      /** Name of the token request parameter that carries the code verifier. */
82      private static final String CODE_VERIFIER = "code_verifier";
83  
84      /** Path of the token endpoint. */
85      private static final String TOKEN_PATH = "/oauth2/token";
86  
87      /** MockMvc bound to the application context. */
88      @Autowired
89      private MockMvc mockMvc;
90  
91      /** Repository where the test client is stored. */
92      @Autowired
93      private RegisteredClientRepository clients;
94  
95      /** JDBC template used to insert the test user. */
96      @Autowired
97      private JdbcTemplate jdbcTemplate;
98  
99      /** Encoder of the application, used for the generated credentials. */
100     @Autowired
101     private PasswordEncoder passwordEncoder;
102 
103     /** Port the application listens on. */
104     @LocalServerPort
105     private int port;
106 
107     /**
108      * Verifies that the whole flow issues an access token and an ID token that validate against the JWK set.
109      *
110      * @throws Exception if a request cannot be performed.
111      */
112     @Test
113     void authorizationCodeFlowWithPkce_whenCredentialsAreValid_issuesTokensValidAgainstJwks() throws Exception {
114         // all test credentials are generated here and never stored anywhere else
115         final var username = "user-" + UUID.randomUUID();
116         final var password = randomValue();
117         final var clientId = "client-" + UUID.randomUUID();
118         final var clientSecret = randomValue();
119         final var verifier = randomValue() + randomValue();
120         final var challenge = s256(verifier);
121 
122         final var userId = UUID.randomUUID();
123         jdbcTemplate.update("INSERT INTO users (id, username, enabled) VALUES (?, ?, true)", userId, username);
124         jdbcTemplate.update("INSERT INTO user_passwords (user_id, encoded_password) VALUES (?, ?)",
125                 userId, passwordEncoder.encode(password));
126         jdbcTemplate.update("INSERT INTO user_authorities (user_id, authority) VALUES (?, 'ROLE_USER')", userId);
127 
128         clients.save(RegisteredClient.withId(UUID.randomUUID().toString())
129                 .clientId(clientId)
130                 .clientSecret(passwordEncoder.encode(clientSecret))
131                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_BASIC)
132                 .authorizationGrantType(AuthorizationGrantType.AUTHORIZATION_CODE)
133                 .redirectUri(REDIRECT_URI)
134                 .scope(OidcScopes.OPENID)
135                 .clientSettings(ClientSettings.builder()
136                         .requireProofKey(true).requireAuthorizationConsent(false).build())
137                 .build());
138 
139         final var authorizeUri = UriComponentsBuilder.fromPath("/oauth2/authorize")
140                 .queryParam("response_type", "code")
141                 .queryParam("client_id", clientId)
142                 .queryParam(REDIRECT_URI_PARAM, REDIRECT_URI)
143                 .queryParam("scope", "openid")
144                 .queryParam("state", "xyz")
145                 .queryParam("nonce", "n-0S6_WzA2Mj")
146                 .queryParam("code_challenge", challenge)
147                 .queryParam("code_challenge_method", "S256")
148                 .build().toUriString();
149 
150         // 1. unauthenticated authorization request is sent to the login page
151         final var session = new MockHttpSession();
152         mockMvc.perform(get(authorizeUri).session(session))
153                 .andExpect(status().is3xxRedirection());
154 
155         // 2. log in with the generated user
156         mockMvc.perform(post("/login").session(session).with(csrf())
157                         .param("username", username).param("password", password))
158                 .andExpect(status().is3xxRedirection());
159 
160         // 3. the authorization request is now answered with a redirect carrying the code
161         final var authorized = mockMvc.perform(get(authorizeUri).session(session))
162                 .andExpect(status().is3xxRedirection()).andReturn();
163         final var location = UriComponentsBuilder.fromUriString(
164                 authorized.getResponse().getRedirectedUrl()).build();
165         assertTrue(location.toUriString().startsWith(REDIRECT_URI));
166         assertEquals("xyz", location.getQueryParams().getFirst("state"));
167         final var code = location.getQueryParams().getFirst(CODE);
168         assertNotNull(code);
169 
170         // 4. exchange the code with the code verifier
171         final var tokenResponse = mockMvc.perform(post(TOKEN_PATH)
172                         .header(AUTHORIZATION, basic(clientId, clientSecret))
173                         .param(GRANT_TYPE, "authorization_code")
174                         .param(CODE, code)
175                         .param(REDIRECT_URI_PARAM, REDIRECT_URI)
176                         .param(CODE_VERIFIER, verifier))
177                 .andExpect(status().isOk()).andReturn().getResponse().getContentAsString();
178         final var tokens = JsonMapper.builder().build().readTree(tokenResponse);
179 
180         // 5. both tokens validate against the published JWK set
181         final var decoder = NimbusJwtDecoder.withJwkSetUri("http://localhost:" + port + "/oauth2/jwks").build();
182         final var accessToken = decoder.decode(tokens.get("access_token").asString());
183         final var idToken = decoder.decode(tokens.get("id_token").asString());
184         assertEquals(username, accessToken.getSubject());
185         assertEquals(username, idToken.getSubject());
186         assertEquals(List.of(clientId), idToken.getAudience());
187         assertEquals("n-0S6_WzA2Mj", idToken.getClaimAsString("nonce"));
188         assertEquals("http://localhost:9000", idToken.getIssuer().toString());
189     }

191     /**
192      * Verifies that a token request with an unknown authorization code is rejected.
193      *
194      * @throws Exception if a request cannot be performed.
195      */
196     @Test
197     void tokenRequest_whenCodeIsUnknown_isRejected() throws Exception {
198         final var clientId = "client-" + UUID.randomUUID();
199         final var clientSecret = randomValue();
200         clients.save(RegisteredClient.withId(UUID.randomUUID().toString())
201                 .clientId(clientId)
202                 .clientSecret(passwordEncoder.encode(clientSecret))
203                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_BASIC)
204                 .authorizationGrantType(AuthorizationGrantType.AUTHORIZATION_CODE)
205                 .redirectUri(REDIRECT_URI)
206                 .scope(OidcScopes.OPENID)
207                 .build());
208 
209         mockMvc.perform(post(TOKEN_PATH)
210                         .header(AUTHORIZATION, basic(clientId, clientSecret))
211                         .param(GRANT_TYPE, "authorization_code")
212                         .param(CODE, "not-a-real-code")
213                         .param(REDIRECT_URI_PARAM, REDIRECT_URI)
214                         .param(CODE_VERIFIER, randomValue()))
215                 .andExpect(status().isBadRequest());
216     }

218     /**
219      * Builds the value of a basic authentication header.
220      *
221      * @param clientId     identifier of the client.
222      * @param clientSecret secret of the client.
223      * @return the header value.
224      */
225     private static String basic(final String clientId, final String clientSecret) {
226         return "Basic " + Base64.getEncoder().encodeToString(
227                 (clientId + ":" + clientSecret).getBytes(StandardCharsets.UTF_8));
228     }
229 
230     /**
231      * Generates a random URL-safe value.
232      *
233      * @return the value.
234      */
235     private static String randomValue() {
236         final var bytes = new byte[RANDOM_BYTES];
237         new SecureRandom().nextBytes(bytes);
238         return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
239     }
240 
241     /**
242      * Computes the S256 PKCE code challenge of a verifier.
243      *
244      * @param verifier code verifier.
245      * @return the code challenge.
246      * @throws NoSuchAlgorithmException if SHA-256 is not available.
247      */
248     private static String s256(final String verifier) throws NoSuchAlgorithmException {
249         final var digest = MessageDigest.getInstance("SHA-256")
250                 .digest(verifier.getBytes(StandardCharsets.US_ASCII));
251         return Base64.getUrlEncoder().withoutPadding().encodeToString(digest);
252     }
