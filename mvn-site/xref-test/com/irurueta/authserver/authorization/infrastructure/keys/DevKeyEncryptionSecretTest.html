19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertNotEquals;
21  import static org.junit.jupiter.api.Assertions.assertThrows;
22  import static org.junit.jupiter.api.Assertions.assertTrue;
23  
24  import java.io.IOException;
25  import java.io.UncheckedIOException;
26  import java.nio.file.Files;
27  import java.nio.file.Path;
28  import java.util.UUID;
29  import org.junit.jupiter.api.Test;
30  import org.junit.jupiter.api.io.TempDir;
31  
32  /**
33   * Unit tests of {@link DevKeyEncryptionSecret}.
34   */
35  class DevKeyEncryptionSecretTest {
36  
37      /** Temporary directory standing in for the user's home. */
38      @TempDir
39      private Path home;
40  
41      /**
42       * Verifies that the first load creates the file and the second reuses it.
43       *
44       * @throws IOException if the file cannot be read.
45       */
46      @Test
47      void load_whenCalledTwice_createsOnceAndReuses() throws IOException {
48          final var directory = home.resolve(DevKeyEncryptionSecret.DIRECTORY);
49  
50          final var first = DevKeyEncryptionSecret.load(directory);
51          final var second = DevKeyEncryptionSecret.load(directory);
52  
53          assertEquals(first, second);
54          assertTrue(Files.exists(directory.resolve(DevKeyEncryptionSecret.FILE_NAME)));
55          assertNotEquals(first.password(), first.salt());
56          assertEquals(2, Files.readAllLines(directory.resolve(DevKeyEncryptionSecret.FILE_NAME)).size());
57      }
58  
59      /**
60       * Verifies that two machines get different secrets.
61       */
62      @Test
63      void load_whenDirectoriesDiffer_generatesDifferentSecrets() {
64          final var first = DevKeyEncryptionSecret.load(home.resolve("one"));
65          final var second = DevKeyEncryptionSecret.load(home.resolve("two"));
66  
67          assertNotEquals(first, second);
68      }
69  
70      /**
71       * Verifies that the generated inputs can drive the cipher.
72       */
73      @Test
74      void load_whenGenerated_workWithTheCipher() {
75          final var cipher = new AesGcmKeyCipher(DevKeyEncryptionSecret.load(home));
76          final var plain = UUID.randomUUID().toString();
77  
78          assertEquals(plain, cipher.decrypt(cipher.encrypt(plain)));
79      }
80  
81      /**
82       * Verifies that explicitly configured values are used and no file is touched.
83       */
84      @Test
85      void keyEncryptionProperties_whenValuesAreConfigured_usesThem() {
86          final var first = UUID.randomUUID().toString();
87          final var second = UUID.randomUUID().toString();
88  
89          final var properties = new DevKeyEncryptionSecret().keyEncryptionProperties(first, second);
90  
91          assertEquals(new KeyEncryptionProperties(first, second), properties);
92      }
93  
94      /**
95       * Verifies that a path that cannot be a directory is reported.
96       *
97       * @throws IOException if the blocking file cannot be created.
98       */
99      @Test
100     void load_whenDirectoryCannotBeCreated_throwsUncheckedIOException() throws IOException {
101         final var blocker = Files.createFile(home.resolve("blocker"));
102 
103         assertThrows(UncheckedIOException.class, () -> DevKeyEncryptionSecret.load(blocker.resolve("sub")));
104     }
105 }
