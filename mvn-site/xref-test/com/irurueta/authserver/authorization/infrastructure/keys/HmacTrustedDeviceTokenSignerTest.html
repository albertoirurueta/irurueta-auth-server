17  package com.irurueta.authserver.authorization.infrastructure.keys;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertNotEquals;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  import java.util.Base64;
24  import java.util.UUID;
25  import org.junit.jupiter.api.Test;
26  import org.junit.jupiter.params.ParameterizedTest;
27  import org.junit.jupiter.params.provider.NullAndEmptySource;
28  import org.junit.jupiter.params.provider.ValueSource;
29  
30  /**
31   * Unit tests of {@link HmacTrustedDeviceTokenSigner}, with a secret generated at run time.
32   */
33  class HmacTrustedDeviceTokenSignerTest {
34  
35      /** Properties shared by the signer under test. */
36      private final KeyEncryptionProperties properties = KeyTestSupport.randomProperties();
37  
38      /** Signer under test. */
39      private final HmacTrustedDeviceTokenSigner signer =
40              new HmacTrustedDeviceTokenSigner(new PurposeKeyDerivation(properties));
41  
42      /** A raw token. */
43      private final String token = UUID.randomUUID().toString().replace("-", "");
44  
45      /**
46       * Verifies that a signed value has the form token, dot, signature, and verifies back to the token.
47       */
48      @Test
49      void verify_whenValueWasSigned_returnsTheToken() {
50          final var signed = signer.sign(token);
51  
52          assertTrue(signed.startsWith(token + "."));
53          assertEquals(token, signer.verify(signed).orElseThrow());
54      }
55  
56      /**
57       * Verifies that an unsigned token is rejected.
58       */
59      @Test
60      void verify_whenValueIsUnsigned_returnsEmpty() {
61          assertTrue(signer.verify(token).isEmpty());
62      }
63  
64      /**
65       * Verifies that a value whose token was changed is rejected.
66       */
67      @Test
68      void verify_whenTokenIsAltered_returnsEmpty() {
69          final var signed = signer.sign(token);
70          final var signature = signed.substring(signed.lastIndexOf('.'));
71  
72          assertTrue(signer.verify("x" + token + signature).isEmpty());
73      }
74  
75      /**
76       * Verifies that a forged signature of the right length is rejected.
77       */
78      @Test
79      void verify_whenSignatureIsForged_returnsEmpty() {
80          final var signed = signer.sign(token);
81          final var length = Base64.getUrlDecoder().decode(signed.substring(signed.lastIndexOf('.') + 1)).length;
82          final var forged = token + "." + Base64.getUrlEncoder().withoutPadding().encodeToString(new byte[length]);
83  
84          assertTrue(signer.verify(forged).isEmpty());
85      }
86  
87      /**
88       * Verifies that a value signed with another secret is rejected.
89       */
90      @Test
91      void verify_whenSignedWithAnotherSecret_returnsEmpty() {
92          final var other = new HmacTrustedDeviceTokenSigner(new PurposeKeyDerivation(KeyTestSupport.randomProperties()));
93  
94          assertNotEquals(signer.sign(token), other.sign(token));
95          assertTrue(signer.verify(other.sign(token)).isEmpty());
96      }
97  
98      /**
99       * Verifies that missing, blank and malformed values are rejected without failing.
100      *
101      * @param value the value presented.
102      */
103     @ParameterizedTest
104     @NullAndEmptySource
105     @ValueSource(strings = {" ", ".", ".signature", "token.", "token.!!not-base64!!"})
106     void verify_whenValueIsMalformed_returnsEmpty(final String value) {
107         assertTrue(signer.verify(value).isEmpty());
108     }
109 }
