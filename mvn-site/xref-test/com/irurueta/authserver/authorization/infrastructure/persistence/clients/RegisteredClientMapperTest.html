67      /**
68       * Verifies that a registration is mapped with its identifiers, secret, collections, audiences as a custom
69       * setting and token settings.
70       */
71      @Test
72      void toRegisteredClient_whenRegistrationIsConfidential_mapsEveryField() {
73          final var secret = encoded();
74  
75          final var client = mapper.toRegisteredClient(confidential(), ROW_ID, ISSUED_AT, secret);
76  
77          assertEquals(ROW_ID, client.getId());
78          assertEquals(CLIENT_ID, client.getClientId());
79          assertEquals(ISSUED_AT, client.getClientIdIssuedAt());
80          assertEquals(secret, client.getClientSecret());
81          assertEquals("Backend for Frontend", client.getClientName());
82          assertEquals(Set.of(ClientAuthenticationMethod.CLIENT_SECRET_BASIC), client.getClientAuthenticationMethods());
83          assertEquals(Set.of(AuthorizationGrantType.AUTHORIZATION_CODE, AuthorizationGrantType.REFRESH_TOKEN),
84                  client.getAuthorizationGrantTypes());
85          assertEquals(Set.of("openid", "profile"), client.getScopes());
86          assertEquals(List.of(AUDIENCE),
87                  client.getClientSettings().getSetting(RegisteredClientMapper.AUDIENCES_SETTING));
88          assertTrue(client.getClientSettings().isRequireProofKey());
89          assertFalse(client.getClientSettings().isRequireAuthorizationConsent());
90          assertEquals(ACCESS_TTL, client.getTokenSettings().getAccessTokenTimeToLive());
91          assertEquals(REFRESH_TTL, client.getTokenSettings().getRefreshTokenTimeToLive());
92          assertEquals(ClientRegistration.FORMAT_SELF_CONTAINED,
93                  client.getTokenSettings().getAccessTokenFormat().getValue());
94          assertFalse(client.getTokenSettings().isReuseRefreshTokens());
95      }

127     @Test

129         final var client = RegisteredClient.withId(ROW_ID).clientId(CLIENT_ID)
130                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_POST)
131                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_BASIC)

137 
138         final var registration = mapper.toRegistration(client);
139 
140         assertEquals(List.of("client_secret_basic", "client_secret_post"), registration.authMethods());
141         assertEquals(List.of("authorization_code", "refresh_token"), registration.grantTypes());
142         assertEquals(List.of("http://localhost:8080/a", "http://localhost:8080/b"), registration.redirectUris());
143         assertEquals(List.of("openid", "profile"), registration.scopes());

276     /**
277      * Verifies that the key, algorithm, certificate subject and certificate-bound settings are mapped both ways.
278      */
279     @Test
280     void toRegisteredClient_whenAuthenticationSettingsAreGiven_mapsThemBothWays() {
281         final var registration = withAuthSettings("https://client.example.test/jwks.json", JWKS, "ES256", "CN=client");
282 
283         final var client = mapper.toRegisteredClient(registration, ROW_ID, ISSUED_AT, null);
284 
285         assertEquals("https://client.example.test/jwks.json", client.getClientSettings().getJwkSetUrl());
286         assertEquals(SignatureAlgorithm.ES256,
287                 client.getClientSettings().getTokenEndpointAuthenticationSigningAlgorithm());
288         assertEquals("CN=client", client.getClientSettings().getX509CertificateSubjectDN());
289         assertEquals(JWKS, client.getClientSettings().getSetting(RegisteredClientMapper.JWKS_SETTING));
290         assertTrue(client.getTokenSettings().isX509CertificateBoundAccessTokens());
291         assertEquals(registration.jwkSetUrl(), mapper.toRegistration(client).jwkSetUrl());
292         assertEquals(registration.jwks(), mapper.toRegistration(client).jwks());
293         assertEquals("ES256", mapper.toRegistration(client).tokenEndpointSigningAlgorithm());
294         assertEquals("CN=client", mapper.toRegistration(client).tlsClientAuthSubjectDn());
295         assertTrue(mapper.toRegistration(client).certificateBoundAccessTokens());
296     }

336     /**
337      * Builds a {@code private_key_jwt} registration with the given key settings.
338      *
339      * @param jwkSetUrl the JWK set URL, or {@code null}.
340      * @param jwks      the JWK set, or {@code null}.
341      * @param algorithm the signing algorithm, or {@code null}.
342      * @param subjectDn the TLS subject DN, or {@code null}.
343      * @return the registration.
344      */
345     private static ClientRegistration withAuthSettings(
346             final String jwkSetUrl, final String jwks, final String algorithm, final String subjectDn) {
347         return new ClientRegistration("service", "Service", true, List.of("private_key_jwt"),
348                 List.of("client_credentials"), List.of(), List.of("documents.read"), false, List.of(AUDIENCE),
349                 ACCESS_TTL, REFRESH_TTL, ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false, false,
350                 jwkSetUrl, jwks, algorithm, subjectDn, true);
351     }
