130     /**
131      * Verifies that the configured default clients exist after start with the configured redirect URIs and both
132      * audiences, that seeding again creates nothing and that the stored secret is encoded, never the raw value.
133      */
134     @Test
135     void seeding_whenRepeated_isIdempotentAndStoresEncodedSecrets() {
136         final var again = seedUseCase.execute(List.of(
137                 DefaultClientDefinitions.bff(List.of(BFF_REDIRECT_URI), DefaultClientDefinitions.DEFAULT_API_AUDIENCE),
138                 DefaultClientDefinitions.spaPublic(List.of(SPA_REDIRECT_URI),
139                         DefaultClientDefinitions.DEFAULT_API_AUDIENCE)), Map.of("bff", BFF_SEED_SECRET));
140         final var bff = clientRegistrations.findByClientId("bff").orElseThrow();
141         assertEquals(List.of(BFF_REDIRECT_URI), bff.redirectUris());
142         assertEquals(List.of(DefaultClientDefinitions.RESOURCE_SERVER_AUDIENCE,
143                 DefaultClientDefinitions.DEFAULT_API_AUDIENCE), bff.audiences());
144         assertFalse(bff.scopes().contains("admin"));
145         assertEquals(List.of(SPA_REDIRECT_URI),
146                 clientRegistrations.findByClientId("spa-public").orElseThrow().redirectUris());
147 
148         assertEquals(List.of(), again.created());
149         assertEquals(List.of(), again.skipped());
150         assertEquals(1, count("bff"));
151         assertEquals(1, count("spa-public"));
152         final var stored = secretOf("bff");
153         assertTrue(stored.startsWith(ARGON2_PREFIX));
154         assertNotEquals(BFF_SEED_SECRET, stored);
155         assertFalse(stored.contains(BFF_SEED_SECRET));
156         assertTrue(passwordEncoder.matches(BFF_SEED_SECRET, stored));
157         assertNull(secretOf("spa-public"), "a public client has no secret");
158     }

160     /**
161      * Verifies the whole life of a client through the administration API.
162      *
163      * @throws Exception if a request cannot be performed.
164      */
165     @Test
166     void adminApi_whenClientIsCreatedUpdatedAndDeleted_followsTheCrudContract() throws Exception {
167         final var admin = fixtures.adminToken();
168         final var clientId = "managed-" + UUID.randomUUID();
169 
170         final var created = body(send(post(CLIENTS_PATH), admin, clientJson(clientId, "Managed"))
171                 .andExpect(status().isCreated()));
172         final var issued = created.get("clientSecret").asString();
173         assertEquals(clientId, created.get("client").get("clientId").asString());
174         assertTrue(passwordEncoder.matches(issued, secretOf(clientId)));
175         assertFalse(secretOf(clientId).contains(issued));
176 
177         final var fetched = body(mockMvc.perform(get(CLIENTS_PATH + "/" + clientId).header(
178                         AuthServerFixtures.AUTHORIZATION, "Bearer " + admin))
179                 .andExpect(status().isOk()));
180         assertEquals("Managed", fetched.get("name").asString());
181         assertFalse(fetched.has("clientSecret"), "the secret is only shown on creation");
182         mockMvc.perform(get(CLIENTS_PATH).header(AuthServerFixtures.AUTHORIZATION, "Bearer " + admin))
183                 .andExpect(status().isOk())
184                 .andExpect(content().string(containsString(clientId)));
185 
186         send(post(CLIENTS_PATH), admin, clientJson(clientId, "Managed")).andExpect(status().isConflict());
187 
188         final var updated = body(send(put(CLIENTS_PATH + "/" + clientId), admin, clientJson(clientId, "Renamed"))
189                 .andExpect(status().isOk()));
190         assertEquals("Renamed", updated.get("name").asString());
191         assertTrue(passwordEncoder.matches(issued, secretOf(clientId)), "updating keeps the secret");
192         send(put(CLIENTS_PATH + "/other"), admin, clientJson(clientId, "Renamed"))
193                 .andExpect(status().isBadRequest());
194 
195         mockMvc.perform(delete(CLIENTS_PATH + "/" + clientId).header(AuthServerFixtures.AUTHORIZATION,
196                 "Bearer " + admin)).andExpect(status().isNoContent());
197         assertEquals(0, count(clientId));
198         mockMvc.perform(get(CLIENTS_PATH + "/" + clientId).header(AuthServerFixtures.AUTHORIZATION,
199                 "Bearer " + admin)).andExpect(status().isNotFound());
200     }

203     /**
204      * Verifies that a {@code private_key_jwt} client is created with its public JWK set and no secret, that the
205      * authentication settings round trip through the API and survive an update.
206      *
207      * @throws Exception if a request cannot be performed.
208      */
209     @Test
210     void adminApi_whenClientUsesPrivateKeyJwt_storesKeysAndGeneratesNoSecret() throws Exception {
211         final var admin = fixtures.adminToken();
212         final var clientId = "keyed-" + UUID.randomUUID();
213         final var publicSet = new JWKSet(new RSAKeyGenerator(RSA_KEY_SIZE).keyID("k1").generate().toPublicJWK());
214 
215         final var created = body(send(post(CLIENTS_PATH), admin, keyClientJson(clientId, publicSet.toString()))
216                 .andExpect(status().isCreated()));
217 
218         assertTrue(created.get("clientSecret") == null || created.get("clientSecret").isNull());
219         assertNull(secretOf(clientId));
220         final var client = created.get("client");
221         assertEquals("PS256", client.get("tokenEndpointAuthSigningAlgorithm").asString());
222         assertEquals("k1", client.get("jwks").get("keys").get(0).get("kid").asString());
223         assertTrue(client.get("certificateBoundAccessTokens").asBoolean());
224         assertEquals("CN=" + clientId, client.get("tlsClientAuthSubjectDn").asString());
225         assertEquals("https://client.example.test/jwks.json", client.get("jwkSetUrl").asString());
226 
227         final var updated = body(send(put(CLIENTS_PATH + "/" + clientId), admin,
228                 keyClientJson(clientId, publicSet.toString())).andExpect(status().isOk()));
229         assertEquals(client, updated);
230         assertNull(secretOf(clientId));
231         send(post(CLIENTS_PATH + "/" + clientId + "/secret"), admin, "{}").andExpect(status().isBadRequest());
232     }

275     /**
276      * Verifies that rotating the secret returns a new one, stores it encoded and stops the old one working.
277      *
278      * @throws Exception if a request cannot be performed.
279      */
280     @Test
281     void rotateSecret_whenCalled_replacesTheSecret() throws Exception {
282         final var admin = fixtures.adminToken();
283         final var clientId = "rotating-" + UUID.randomUUID();
284         final var issued = body(send(post(CLIENTS_PATH), admin, clientJson(clientId, "Rotating"))
285                 .andExpect(status().isCreated())).get("clientSecret").asString();
286         tokenRequest(clientId, issued).andExpect(status().isOk());
287 
288         final var replacement = body(mockMvc.perform(post(CLIENTS_PATH + "/" + clientId + "/secret")
289                         .header(AuthServerFixtures.AUTHORIZATION, "Bearer " + admin))
290                 .andExpect(status().isOk())).get("clientSecret").asString();
291 
292         assertNotEquals(issued, replacement);
293         assertTrue(passwordEncoder.matches(replacement, secretOf(clientId)));
294         tokenRequest(clientId, issued).andExpect(status().isUnauthorized());
295         tokenRequest(clientId, replacement).andExpect(status().isOk());
296     }

328     /**
329      * Verifies that deleting a client also deletes its authorizations and consents, so a token it was issued is
330      * later introspected as inactive instead of failing with a server error.
331      *
332      * @throws Exception if a request cannot be performed.
333      */
334     @Test
335     void delete_whenClientHasAuthorizations_removesThemAndItsTokensBecomeInactive() throws Exception {
336         final var admin = fixtures.adminToken();
337         final var doomed = fixtures.createClient(List.of("client_credentials"), List.of("read"),
338                 List.of(AuthServerFixtures.API_AUDIENCE));
339         final var token = fixtures.clientCredentialsToken(doomed, "read");
340         final var internalId = jdbcClient.sql("SELECT id FROM oauth2_registered_client WHERE client_id = ?")
341                 .param(doomed.clientId()).query(String.class).single();
342         jdbcClient.sql("""
343                 INSERT INTO oauth2_authorization_consent (registered_client_id, principal_name, authorities)
344                 VALUES (?, 'alice', 'SCOPE_read')
345                 """).param(internalId).update();
346 
347         mockMvc.perform(delete(CLIENTS_PATH + "/" + doomed.clientId()).header(AuthServerFixtures.AUTHORIZATION,
348                 "Bearer " + admin)).andExpect(status().isNoContent());
349 
350         assertEquals(0, jdbcClient.sql("SELECT count(*) FROM oauth2_authorization WHERE registered_client_id = ?")
351                 .param(internalId).query(Integer.class).single());
352         assertEquals(0, jdbcClient.sql(
353                         "SELECT count(*) FROM oauth2_authorization_consent WHERE registered_client_id = ?")
354                 .param(internalId).query(Integer.class).single());
355         final var introspector = fixtures.createClient(List.of("client_credentials"), List.of("read"),
356                 List.of(AuthServerFixtures.API_AUDIENCE));
357         mockMvc.perform(post("/oauth2/introspect")
358                         .header(AuthServerFixtures.AUTHORIZATION,
359                                 AuthServerFixtures.basic(introspector.clientId(), introspector.credential()))
360                         .param("token", token))
361                 .andExpect(status().isOk())
362                 .andExpect(content().string(containsString("\"active\":false")));
363     }
