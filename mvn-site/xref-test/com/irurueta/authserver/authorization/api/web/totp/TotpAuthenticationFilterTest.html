133     /**
134      * Verifies that the code is trimmed and handed to the manager as a TOTP request.
135      */
136     @Test
137     void attemptAuthentication_whenCodeIsPosted_asksTheManagerToVerifyIt() {
138         request.setParameter(TotpAuthenticationFilter.CODE_PARAMETER, " 123456 ");
139         final var verified = mock(Authentication.class);
140         when(authenticationManager.authenticate(any())).thenReturn(verified);
141 
142         final var result = filter.attemptAuthentication(request, response);
143 
144         final var captor = ArgumentCaptor.<Authentication>captor();
145         verify(authenticationManager).authenticate(captor.capture());
146         final var sent = (TotpAuthenticationToken) captor.getValue();
147         assertSame(verified, result);
148         assertSame(primary, sent.primary());
149         assertEquals("123456", sent.getCredentials());
150         assertEquals(TotpCodeKind.TOTP, sent.kind());
151         assertFalse(pendingRecoveryCodes.isPending(request));
152     }

184         filter.attemptAuthentication(request, response);

186         final var captor = ArgumentCaptor.<Authentication>captor();
187         verify(authenticationManager).authenticate(captor.capture());
188         assertEquals(TotpCodeKind.ENROLMENT, ((TotpAuthenticationToken) captor.getValue()).kind());
189         assertEquals(List.of("AAAAAAAAAA"), pendingRecoveryCodes.take(request));
190     }

281     /**
282      * Verifies that a device is remembered and its cookie written when the user asks for it, and that the browser
283      * agent is cut to the size that is stored.
284      */
285     @Test
286     void doFilter_whenRememberIsChecked_remembersTheDevice() throws Exception {
287         final var userId = UUID.randomUUID();
288         request.setParameter(TotpAuthenticationFilter.CODE_PARAMETER, "123456");
289         request.setParameter(TotpAuthenticationFilter.REMEMBER_PARAMETER, "on");
290         request.addHeader("User-Agent", "a".repeat(LONG_USER_AGENT));
291         final var verified = UsernamePasswordAuthenticationToken.authenticated(USERNAME, "secret",
292                 AuthorityUtils.createAuthorityList(TotpFactor.AUTHORITY));
293         final var created = createdDevice();
294         when(authenticationManager.authenticate(any())).thenReturn(verified);
295         when(userResolver.userIdOf(USERNAME)).thenReturn(userId);
296         when(createTrustedDeviceUseCase.execute(any(), any(), any())).thenReturn(created);
297 
298         filter.doFilter(request, response, chain);
299 
300         final var agent = ArgumentCaptor.<String>captor();
301         verify(createTrustedDeviceUseCase).execute(any(), any(), agent.capture());
302         assertEquals(TotpAuthenticationFilter.MAX_USER_AGENT_LENGTH, agent.getValue().length());
303         verify(cookieWriter).write(response, created);
304         assertEquals("/", response.getRedirectedUrl());
305     }

372     /**
373      * Builds a remembered device.
374      *
375      * @return the device and its raw token.
376      */
377     private static CreatedTrustedDevice createdDevice() {
378         final var now = Instant.now();
379         return new CreatedTrustedDevice(

381                 "token");
