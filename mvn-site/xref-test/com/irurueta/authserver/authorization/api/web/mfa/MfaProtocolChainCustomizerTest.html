54  /**
55   * Unit tests of {@link MfaProtocolChainCustomizer}.
56   */
57  @ExtendWith(MockitoExtension.class)
58  class MfaProtocolChainCustomizerTest {
59  
60      /** Redirect URI of the client. */
61      private static final String REDIRECT_URI = "https://client.example/cb";
62  
63      /** Identifier of the client. */
64      private static final String CLIENT_ID = "client";
65  
66      /** Scope requested. */
67      private static final String SCOPE = "openid";
68  
69      /** Manager that applies the policy. */
70      @Mock
71      private MfaAuthorizationManager authorizationManager;
72  
73      /** Handler of the endpoint's errors. */
74      @Mock
75      private AuthenticationFailureHandler errorResponseHandler;
76  
77      /** Provider of the authorization code request. */
78      @Mock
79      private OAuth2AuthorizationCodeRequestAuthenticationProvider codeProvider;
80  
81      /** Validator installed in the provider, once the customizer ran. */
82      private Consumer<OAuth2AuthorizationCodeRequestAuthenticationContext> validator;
83  
84      /**
85       * Runs the customizer against mocked configurers and captures the validator it installs.
86       */
87      @BeforeEach
88      @SuppressWarnings("unchecked")
89      void setUp() {
90          final var customizer = new MfaProtocolChainCustomizer(authorizationManager, errorResponseHandler);
91          final var server = mock(OAuth2AuthorizationServerConfigurer.class);
92          final var endpoint = mock(OAuth2AuthorizationEndpointConfigurer.class);
93          final var serverCaptor = ArgumentCaptor.<Customizer<OAuth2AuthorizationEndpointConfigurer>>captor();
94          final var providersCaptor = ArgumentCaptor.<Consumer<List<AuthenticationProvider>>>captor();
95          when(server.authorizationEndpoint(serverCaptor.capture())).thenReturn(server);
96          when(endpoint.authenticationProviders(providersCaptor.capture())).thenReturn(endpoint);
97          when(endpoint.errorResponseHandler(any())).thenReturn(endpoint);
98  
99          customizer.customize(server);
100         serverCaptor.getValue().customize(endpoint);
101         final var providers = new ArrayList<AuthenticationProvider>();
102         providers.add(mock(AuthenticationProvider.class));
103         providers.add(codeProvider);
104         providersCaptor.getValue().accept(providers);
105 
106         verify(endpoint).errorResponseHandler(errorResponseHandler);
107         final var validatorCaptor = ArgumentCaptor
108                 .<Consumer<OAuth2AuthorizationCodeRequestAuthenticationContext>>captor();
109         verify(codeProvider).setAuthenticationValidator(validatorCaptor.capture());
110         validator = validatorCaptor.getValue();
111     }
112 
113     /**
114      * Verifies that a user who owes a second factor is stopped by a {@link SecondFactorRequiredException}.
115      */
116     @Test
117     void validator_whenPolicyIsNotSatisfied_throwsSecondFactorRequiredException() {
118         final var user = user();
119         when(authorizationManager.clientRequiresMfa(CLIENT_ID)).thenReturn(true);
120         when(authorizationManager.evaluate(user, true)).thenReturn(new AuthorizationDecision(false));
121         final var context = context(user);
122 
123         assertThrows(SecondFactorRequiredException.class, () -> validator.accept(context));
124     }
125 
126     /**
127      * Verifies that a user who satisfies the policy goes on.
128      */
129     @Test
130     void validator_whenPolicyIsSatisfied_doesNotThrow() {
131         final var user = user();
132         when(authorizationManager.clientRequiresMfa(CLIENT_ID)).thenReturn(false);
133         when(authorizationManager.evaluate(user, false)).thenReturn(new AuthorizationDecision(true));
134         final var context = context(user);
135 
136         assertDoesNotThrow(() -> validator.accept(context));
137     }
138 
139     /**
140      * Verifies that a principal that is not signed in is left to the framework.
141      */
142     @Test
143     void validator_whenPrincipalIsNotSignedIn_doesNotConsultThePolicy() {
144         final var context = context(UsernamePasswordAuthenticationToken.unauthenticated("alice", "x"));
145 
146         assertDoesNotThrow(() -> validator.accept(context));
147         verifyNoInteractions(authorizationManager);
148     }
149 
150     /**
151      * Builds a signed-in user.
152      *
153      * @return the authentication.
154      */
155     private static Authentication user() {
156         return UsernamePasswordAuthenticationToken.authenticated("alice", null,
157                 AuthorityUtils.createAuthorityList("FACTOR_PASSWORD"));
158     }
159 
160     /**
161      * Builds the context of an authorization request.
162      *
163      * @param principal the user making the request.
164      * @return the context.
165      */
166     private static OAuth2AuthorizationCodeRequestAuthenticationContext context(final Authentication principal) {
167         final var client = RegisteredClient.withId("id").clientId(CLIENT_ID)
168                 .authorizationGrantType(AuthorizationGrantType.AUTHORIZATION_CODE).redirectUri(REDIRECT_URI)
169                 .scope(SCOPE)
170                 .build();
171         final var token = new OAuth2AuthorizationCodeRequestAuthenticationToken("https://as.example/oauth2/authorize",
172                 CLIENT_ID, principal, REDIRECT_URI, "state", Set.of(SCOPE), Map.of());
173         return OAuth2AuthorizationCodeRequestAuthenticationContext.with(token).registeredClient(client)
174                 .authorizationRequest(OAuth2AuthorizationRequest.authorizationCode()
175                         .authorizationUri("https://as.example/oauth2/authorize").clientId(CLIENT_ID)
176                         .redirectUri(REDIRECT_URI).scopes(Set.of(SCOPE)).state("state").build())
177                 .build();
178     }
