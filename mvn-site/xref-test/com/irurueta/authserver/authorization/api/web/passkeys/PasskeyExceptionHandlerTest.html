17  package com.irurueta.authserver.authorization.api.web.passkeys;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertNotNull;
21  import static org.junit.jupiter.api.Assertions.assertSame;
22  
23  import com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException;
24  import com.irurueta.authserver.authorization.api.web.mfa.MfaExceptionHandler;
25  import com.irurueta.authserver.authorization.domain.mfa.AuthenticationContextClass;
26  import com.irurueta.authserver.authorization.domain.passkeys.InvalidPasskeyLabelException;
27  import com.irurueta.authserver.authorization.domain.passkeys.PasskeyNotFoundException;
28  import jakarta.validation.ConstraintViolationException;
29  import java.net.URI;
30  import java.util.Set;
31  import java.time.Duration;
32  import org.junit.jupiter.api.Test;
33  import org.springframework.http.HttpStatus;
34  import org.springframework.http.MediaType;
35  
36  /**
37   * Unit tests of {@link PasskeyExceptionHandler}.
38   */
39  class PasskeyExceptionHandlerTest {
40  
41      /** Maximum age of a step-up, in minutes. */
42      private static final int STEP_UP_MINUTES = 5;
43  
44      /** Builds the step-up challenge. */
45      private final MfaExceptionHandler mfaExceptionHandler = new MfaExceptionHandler();
46  
47      /** Handler under test. */
48      private final PasskeyExceptionHandler handler = new PasskeyExceptionHandler(mfaExceptionHandler);
49  
50      /**
51       * Checks the answer to a passkey the account does not have.
52       */
53      @Test
54      void handleNotFound_whenCalled_answersNotFoundAsAProblem() {
55          final var response = handler.handleNotFound(new PasskeyNotFoundException("none"));
56  
57          assertEquals(HttpStatus.NOT_FOUND, response.getStatusCode());
58          assertEquals(MediaType.APPLICATION_PROBLEM_JSON, response.getHeaders().getContentType());
59          final var problem = response.getBody();
60          assertNotNull(problem);
61          assertEquals(URI.create("urn:problem:passkeys:passkey-not-found"), problem.getType());
62          assertEquals("none", problem.getDetail());
63      }
64  
65      /**
66       * Checks the answer to a label the use case refuses.
67       */
68      @Test
69      void handleInvalidLabel_whenCalled_answersBadRequestAsAProblem() {
70          final var response = handler.handleInvalidLabel(new InvalidPasskeyLabelException("too long"));
71  
72          assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
73          final var problem = response.getBody();
74          assertNotNull(problem);
75          assertEquals(URI.create("urn:problem:passkeys:invalid-label"), problem.getType());
76          assertEquals("too long", problem.getDetail());
77      }
78  
79      /**
80       * Checks the answer to a path parameter that breaks the constraints of the contract.
81       */
82      @Test
83      void handleInvalidParameter_whenCalled_answersBadRequestAsAProblem() {
84          final var response = handler.handleInvalidParameter(new ConstraintViolationException("bad id", Set.of()));
85  
86          assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
87          assertEquals(MediaType.APPLICATION_PROBLEM_JSON, response.getHeaders().getContentType());
88          final var problem = response.getBody();
89          assertNotNull(problem);
90          assertEquals(URI.create("urn:problem:passkeys:invalid-parameter"), problem.getType());
91          assertEquals("bad id", problem.getDetail());
92      }
93  
94      /**
95       * Checks that the step-up challenge is the one of the rest of the account API.
96       */
97      @Test
98      void handleInsufficientUserAuthentication_whenCalled_answersTheSameChallengeAsTheMfaHandler() {
99          final var exception = new InsufficientUserAuthenticationException(AuthenticationContextClass.MULTI_FACTOR,
100                 Duration.ofMinutes(STEP_UP_MINUTES));
101 
102         final var response = handler.handleInsufficientUserAuthentication(exception);
103         final var expected = mfaExceptionHandler.handleInsufficientUserAuthentication(exception);
104 
105         assertEquals(HttpStatus.UNAUTHORIZED, response.getStatusCode());
106         assertEquals(expected.getHeaders(), response.getHeaders());
107         assertSame(exception.getAcrValues(), exception.getAcrValues());
108     }
109 }
