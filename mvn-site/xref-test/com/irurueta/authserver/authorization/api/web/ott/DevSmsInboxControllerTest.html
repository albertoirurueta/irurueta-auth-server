19  import static org.mockito.ArgumentMatchers.anyMap;
20  import static org.mockito.ArgumentMatchers.eq;
21  import static org.mockito.Mockito.mock;
22  import static org.mockito.Mockito.verify;
23  import static org.mockito.Mockito.when;
24  
25  import com.irurueta.authserver.authorization.api.web.pages.PageRenderer;
26  import com.irurueta.authserver.authorization.domain.sms.PhoneNumber;
27  import com.irurueta.authserver.authorization.domain.sms.SmsInbox;
28  import com.irurueta.authserver.authorization.domain.sms.SmsInboxEntry;
29  import java.time.Instant;
30  import java.util.List;
31  import java.util.Map;
32  import org.junit.jupiter.api.Test;
33  import org.junit.jupiter.api.extension.ExtendWith;
34  import org.mockito.ArgumentCaptor;
35  import org.mockito.InjectMocks;
36  import org.mockito.Mock;
37  import org.mockito.junit.jupiter.MockitoExtension;
38  import org.springframework.beans.factory.ObjectProvider;
39  import org.springframework.context.annotation.Profile;
40  import org.springframework.http.ResponseEntity;
41  
42  /** Unit tests of {@link DevSmsInboxController}. */
43  @ExtendWith(MockitoExtension.class)
44  class DevSmsInboxControllerTest {
45  
46      /** The inbox of the logging sender, when there is one. */
47      @Mock
48      private ObjectProvider<SmsInbox> smsInbox;
49  
50      /** Renders the page. */
51      @Mock
52      private PageRenderer pageRenderer;
53  
54      /** Controller under test. */
55      @InjectMocks
56      private DevSmsInboxController controller;
57  
58      /** The recorded messages are rendered and the page is never cached. */
59      @Test
60      void devSmsInboxPage_whenTheLoggingSenderIsInUse_rendersItsMessages() {
61          final var entry = new SmsInboxEntry(new PhoneNumber("+34600111222"), "Your code is 123456", Instant.EPOCH);
62          final var inbox = mock(SmsInbox.class);
63          when(inbox.entries()).thenReturn(List.of(entry));
64          when(smsInbox.getIfAvailable()).thenReturn(inbox);
65          when(pageRenderer.render(eq("dev-sms-inbox"), anyMap())).thenReturn(ResponseEntity.ok("<html></html>"));
66  
67          final var response = controller.devSmsInboxPage();
68  
69          assertEquals("<html></html>", response.getBody());
70          assertEquals("no-store", response.getHeaders().getCacheControl());
71          final var model = ArgumentCaptor.<Map<String, ?>>captor();
72          verify(pageRenderer).render(eq("dev-sms-inbox"), model.capture());
73          assertEquals(Map.of("messages", List.of(entry)), model.getValue());
74      }
75  
76      /** With a real gateway there is no inbox: the page is empty. */
77      @Test
78      void devSmsInboxPage_whenARealGatewayIsConfigured_rendersNoMessages() {
79          when(smsInbox.getIfAvailable()).thenReturn(null);
80          when(pageRenderer.render(eq("dev-sms-inbox"), anyMap())).thenReturn(ResponseEntity.ok("<html></html>"));
81  
82          controller.devSmsInboxPage();
83  
84          final var model = ArgumentCaptor.<Map<String, ?>>captor();
85          verify(pageRenderer).render(eq("dev-sms-inbox"), model.capture());
86          assertEquals(Map.of("messages", List.of()), model.getValue());
87      }
88  
89      /** The controller only exists in the dev profile. */
90      @Test
91      void controller_whenInspected_isDevProfileOnly() {
92          assertEquals("dev", DevSmsInboxController.class.getAnnotation(Profile.class).value()[0]);
93          assertEquals("/dev/sms-inbox", DevSmsInboxController.DEV_SMS_INBOX_PATH);
94      }
95  }
