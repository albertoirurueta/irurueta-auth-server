18  package com.irurueta.authserver.authorization.api.web.mfa;
19  
20  import static org.junit.jupiter.api.Assertions.assertEquals;
21  import static org.mockito.Mockito.mock;
22  import static org.mockito.Mockito.verify;
23  import static org.mockito.Mockito.when;
24  
25  import java.util.ArrayList;
26  import java.util.List;
27  import java.util.function.Consumer;
28  import org.junit.jupiter.api.Test;
29  import org.junit.jupiter.api.extension.ExtendWith;
30  import org.mockito.ArgumentCaptor;
31  import org.mockito.Mock;
32  import org.mockito.junit.jupiter.MockitoExtension;
33  import org.springframework.security.config.Customizer;
34  import org.springframework.security.config.annotation.web.configurers.oauth2.server.authorization.OAuth2AuthorizationEndpointConfigurer;
35  import org.springframework.security.config.annotation.web.configurers.oauth2.server.authorization.OAuth2AuthorizationServerConfigurer;
36  import org.springframework.security.web.authentication.AuthenticationConverter;
37  
38  /**
39   * Unit tests of {@link StepUpProtocolChainCustomizer}.
40   */
41  @ExtendWith(MockitoExtension.class)
42  class StepUpProtocolChainCustomizerTest {
43  
44      /** Converter to install. */
45      @Mock
46      private AuthenticationConverter stepUpConverter;
47  
48      /**
49       * Verifies that the converter is installed before the framework's ones.
50       */
51      @Test
52      @SuppressWarnings("unchecked")
53      void customize_whenApplied_installsConverterFirst() {
54          final var server = mock(OAuth2AuthorizationServerConfigurer.class);
55          final var endpoint = mock(OAuth2AuthorizationEndpointConfigurer.class);
56          final var serverCaptor = ArgumentCaptor.<Customizer<OAuth2AuthorizationEndpointConfigurer>>captor();
57          final var convertersCaptor = ArgumentCaptor.<Consumer<List<AuthenticationConverter>>>captor();
58          when(server.authorizationEndpoint(serverCaptor.capture())).thenReturn(server);
59          when(endpoint.authorizationRequestConverters(convertersCaptor.capture())).thenReturn(endpoint);
60  
61          new StepUpProtocolChainCustomizer(stepUpConverter).customize(server);
62          serverCaptor.getValue().customize(endpoint);
63          final var converters = new ArrayList<AuthenticationConverter>();
64          converters.add(mock(AuthenticationConverter.class));
65          convertersCaptor.getValue().accept(converters);
66  
67          verify(server).authorizationEndpoint(serverCaptor.getValue());
68          assertEquals(stepUpConverter, converters.getFirst());
69          assertEquals(2, converters.size());
70      }
71  }
