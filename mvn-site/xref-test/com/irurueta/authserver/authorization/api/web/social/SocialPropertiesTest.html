31  /**
32   * Unit tests of {@link SocialProperties} and its {@link SocialProperties.Provider}.
33   */
34  class SocialPropertiesTest {
35  
36      /** Tenant id used by the tests. */
37      private static final String TENANT_ID = "9188040d-6c67-4c5b-b112-36a304b66dad";
38  
39      /** Name of the settings of the tests. */
40      private static final String CLIENT = "client";
41  
42      /** Secret of the tests, never a real one. */
43      private static final String SECRET = "not-a-secret";
44  
45      /**
46       * Verifies that a provider with a client id and a secret is enabled and the others are not.
47       */
48      @Test
49      void enabledProviders_whenOnlySomeHaveAClientId_listsOnlyThoseInEnumOrder() {
50          final var properties = new SocialProperties(Map.of(
51                  "github", provider(CLIENT, SECRET),
52                  "google", provider(CLIENT, SECRET),
53                  "microsoft", provider("", "")));
54  
55          assertEquals(List.of(IdentityProvider.GOOGLE, IdentityProvider.GITHUB),
56                  List.copyOf(properties.enabledProviders().keySet()));
57          assertTrue(properties.find(IdentityProvider.GOOGLE).isPresent());
58          assertTrue(properties.find(IdentityProvider.MICROSOFT).isEmpty());
59      }
60  
61      /**
62       * Verifies that no settings at all is valid and enables nothing.
63       */
64      @Test
65      void constructor_whenProvidersAreMissing_enablesNothing() {
66          assertTrue(new SocialProperties(null).enabledProviders().isEmpty());
67      }
68  
69      /**
70       * Verifies that a provider setting that is {@code null} is ignored.
71       */
72      @Test
73      void constructor_whenAProviderIsNull_ignoresIt() {
74          final var providers = new HashMap<String, SocialProperties.Provider>();
75          providers.put("google", null);
76  
77          assertTrue(new SocialProperties(providers).providers().isEmpty());
78      }
79  
80      /**
81       * Verifies that an unknown provider id is refused.
82       */
83      @Test
84      void constructor_whenProviderIsUnknown_throwsIllegalArgumentException() {
85          final var providers = Map.of("facebook", provider(CLIENT, SECRET));
86  
87          assertThrows(IllegalArgumentException.class, () -> new SocialProperties(providers));
88      }
89  
90      /**
91       * Verifies that an enabled provider without a secret is refused.
92       */
93      @Test
94      void constructor_whenEnabledProviderHasNoSecret_throwsIllegalArgumentException() {
95          final var providers = Map.of("google", provider(CLIENT, " "));
96  
97          assertThrows(IllegalArgumentException.class, () -> new SocialProperties(providers));
98      }
99  
100     /**
101      * Verifies that an enabled Apple provider needs its key settings, not a secret.
102      */
103     @Test
104     void constructor_whenAppleLacksItsKeySettings_throwsIllegalArgumentException() {
105         final var withoutTeam = Map.of("apple", apple("", "key", "pem", Duration.ofMinutes(1)));
106         final var withoutKeyId = Map.of("apple", apple("team", "", "pem", Duration.ofMinutes(1)));
107         final var withoutKey = Map.of("apple", apple("team", "key", "", Duration.ofMinutes(1)));
108 
109         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(withoutTeam));
110         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(withoutKeyId));
111         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(withoutKey));
112     }
113 
114     /**
115      * Verifies that the lifetime of the Apple client secret must be positive and at most six months.
116      */
117     @Test
118     void constructor_whenAppleSecretLifetimeIsOutOfRange_throwsIllegalArgumentException() {
119         final var zero = Map.of("apple", apple("team", "key", "pem", Duration.ZERO));
120         final var negative = Map.of("apple", apple("team", "key", "pem", Duration.ofSeconds(-1)));
121         final var tooLong = Map.of("apple", apple("team", "key", "pem",
122                 SocialProperties.MAX_APPLE_CLIENT_SECRET_TTL.plusSeconds(1)));
123 
124         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(zero));
125         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(negative));
126         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(tooLong));
127         assertTrue(new SocialProperties(Map.of("apple", apple("team", "key", "pem",
128                 SocialProperties.MAX_APPLE_CLIENT_SECRET_TTL))).find(IdentityProvider.APPLE).isPresent());
129     }
130 
131     /**
132      * Verifies that a tenant that cannot be part of a URL is refused.
133      */
134     @Test
135     void constructor_whenMicrosoftTenantIsInvalid_throwsIllegalArgumentException() {
136         final var badTenant = Map.of("microsoft", microsoft("a/b", List.of()));
137         final var badAllowed = Map.of("microsoft", microsoft("common", List.of("ok", "../x")));
138 
139         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(badTenant));
140         assertThrows(IllegalArgumentException.class, () -> new SocialProperties(badAllowed));
141     }
142 
143     /**
144      * Verifies that a disabled provider is not validated: its other settings may be empty.
145      */
146     @Test
147     void constructor_whenProviderIsDisabled_doesNotValidateIt() {
148         final var disabledMicrosoft = new SocialProperties.Provider("", "", false, "a/b", List.of(), null, null, null,
149                 null);
150         final var disabledApple = new SocialProperties.Provider("", "", false, null, null, "", "", "", Duration.ZERO);
151 
152         assertTrue(new SocialProperties(Map.of("microsoft", disabledMicrosoft, "apple", disabledApple))
153                 .enabledProviders().isEmpty());
154     }
155 
156     /**
157      * Verifies the defaults the provider record fills in.
158      */
159     @Test
160     void providerConstructor_whenSettingsAreMissing_fillsTheDefaults() {
161         final var provider = new SocialProperties.Provider(CLIENT, SECRET, false, " ", null, null, null, null, null);
162 
163         assertEquals(SocialProperties.COMMON_TENANT, provider.tenant());
164         assertTrue(provider.allowedTenants().isEmpty());
165         assertEquals(SocialProperties.DEFAULT_APPLE_CLIENT_SECRET_TTL, provider.clientSecretTtl());
166     }
167 
168     /**
169      * Verifies that the permitted tenants are the configured ones, else the tenant id, else none.
170      */
171     @Test
172     void permittedTenants_whenAsked_prefersTheListThenTheTenantIdThenNone() {
173         assertEquals(List.of("a"), microsoft("common", List.of("a")).permittedTenants());
174         assertEquals(List.of(TENANT_ID), microsoft(TENANT_ID, List.of()).permittedTenants());
175         assertTrue(microsoft("common", List.of()).permittedTenants().isEmpty());
176         assertTrue(microsoft("contoso.com", List.of()).permittedTenants().isEmpty());
177     }
178 
179     /**
180      * Verifies the trust of a provider: off by default and when the provider is not enabled.
181      */
182     @Test
183     void trust_whenAsked_isOnlyTrueForAnEnabledTrustedProvider() {
184         final var trusted = new SocialProperties.Provider(CLIENT, SECRET, true, null, null, null, null, null, null);
185         final var properties = new SocialProperties(Map.of("google", trusted, "github", provider(CLIENT, SECRET)));
186 
187         assertTrue(properties.trust(IdentityProvider.GOOGLE).trustMfa());
188         assertFalse(properties.trust(IdentityProvider.GITHUB).trustMfa());
189         assertFalse(properties.trust(IdentityProvider.APPLE).trustMfa());
190         assertEquals(IdentityProvider.APPLE, properties.trust(IdentityProvider.APPLE).provider());
191     }

193     /**
194      * Builds a provider that needs a client secret.
195      *
196      * @param clientId client id.
197      * @param secret   client secret.
198      * @return the provider.
199      */
200     private static SocialProperties.Provider provider(final String clientId, final String secret) {
201         return new SocialProperties.Provider(clientId, secret, false, null, null, null, null, null, null);
202     }
203 
204     /**
205      * Builds an enabled Apple provider.
206      *
207      * @param teamId team id.
208      * @param keyId  key id.
209      * @param key    private key.
210      * @param ttl    lifetime of the secret.
211      * @return the provider.
212      */
213     private static SocialProperties.Provider apple(
214             final String teamId, final String keyId, final String key, final Duration ttl) {
215         return new SocialProperties.Provider(CLIENT, null, false, null, null, teamId, keyId, key, ttl);
216     }
217 
218     /**
219      * Builds an enabled Microsoft provider.
220      *
221      * @param tenant  tenant.
222      * @param allowed allowed tenants.
223      * @return the provider.
224      */
225     private static SocialProperties.Provider microsoft(final String tenant, final List<String> allowed) {
226         return new SocialProperties.Provider(CLIENT, SECRET, false, tenant, allowed, null, null, null, null);
227     }
