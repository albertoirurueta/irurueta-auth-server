16  package com.irurueta.authserver.authorization.api.web.ott;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertNotNull;
20  
21  import com.irurueta.authserver.authorization.api.web.mfa.InsufficientUserAuthenticationException;
22  import com.irurueta.authserver.authorization.api.web.mfa.MfaExceptionHandler;
23  import com.irurueta.authserver.authorization.application.sms.InvalidPhoneNumberException;
24  import com.irurueta.authserver.authorization.application.sms.SmsCountryNotAllowedException;
25  import com.irurueta.authserver.authorization.domain.mfa.AuthenticationContextClass;
26  import java.net.URI;
27  import java.time.Duration;
28  import org.junit.jupiter.api.Test;
29  import org.springframework.http.HttpStatus;
30  import org.springframework.http.MediaType;
31  
32  /** Unit tests of {@link OttExceptionHandler}. */
33  class OttExceptionHandlerTest {
34  
35      /** Maximum age of a step-up, in minutes. */
36      private static final int STEP_UP_MINUTES = 5;
37  
38      /** Seconds to wait when rate limited. */
39      private static final long WAIT_SECONDS = 30;
40  
41      /** Handler under test. */
42      private final OttExceptionHandler handler = new OttExceptionHandler(new MfaExceptionHandler());
43  
44      /** A missing phone is answered not found as a problem. */
45      @Test
46      void handleNotFound_whenCalled_answersNotFoundAsAProblem() {
47          final var response = handler.handleNotFound(new PhoneNotFoundException());
48  
49          assertEquals(HttpStatus.NOT_FOUND, response.getStatusCode());
50          assertEquals(MediaType.APPLICATION_PROBLEM_JSON, response.getHeaders().getContentType());
51          assertEquals(URI.create("urn:problem:phone:phone-not-found"), response.getBody().getType());
52      }
53  
54      /** An invalid number is answered bad request. */
55      @Test
56      void handleInvalidPhone_whenCalled_answersBadRequestAsAProblem() {
57          final var response = handler.handleInvalidPhone(new InvalidPhoneNumberException());
58  
59          assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
60          assertEquals(URI.create("urn:problem:phone:invalid-phone"), response.getBody().getType());
61      }
62  
63      /** A country that gets no SMS is answered bad request. */
64      @Test
65      void handleCountryNotAllowed_whenCalled_answersBadRequestAsAProblem() {
66          final var response = handler.handleCountryNotAllowed(new SmsCountryNotAllowedException());
67  
68          assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
69          assertEquals(URI.create("urn:problem:phone:country-not-allowed"), response.getBody().getType());
70      }
71  
72      /** A wrong code is answered bad request. */
73      @Test
74      void handleInvalidCode_whenCalled_answersBadRequestAsAProblem() {
75          final var response = handler.handleInvalidCode(new InvalidPhoneCodeException());
76  
77          assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
78          assertEquals(URI.create("urn:problem:phone:invalid-code"), response.getBody().getType());
79      }
80  
81      /** A rate-limited client is told to wait at least one second. */
82      @Test
83      void handleRateLimited_whenCalled_answersTooManyRequestsWithRetryAfter() {
84          final var response = handler.handleRateLimited(
85                  new OttRateLimitExceededException(Duration.ofSeconds(WAIT_SECONDS)));
86          final var shortWait = handler.handleRateLimited(new OttRateLimitExceededException(Duration.ofMillis(1)));
87  
88          assertEquals(HttpStatus.TOO_MANY_REQUESTS, response.getStatusCode());
89          assertEquals("30", response.getHeaders().getFirst("Retry-After"));
90          assertEquals("1", shortWait.getHeaders().getFirst("Retry-After"));
91      }
92  
93      /** A missing step-up is answered as the multi-factor handler does. */
94      @Test
95      void handleInsufficientUserAuthentication_whenCalled_answersTheStepUpChallenge() {
96          final var response = handler.handleInsufficientUserAuthentication(new InsufficientUserAuthenticationException(
97                  AuthenticationContextClass.MULTI_FACTOR, Duration.ofMinutes(STEP_UP_MINUTES)));
98  
99          assertEquals(HttpStatus.UNAUTHORIZED, response.getStatusCode());
100         assertNotNull(response.getHeaders().getFirst("WWW-Authenticate"));
101     }
102 }
