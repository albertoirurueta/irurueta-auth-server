17  package com.irurueta.authserver.authorization.api.web.password;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertFalse;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  import com.irurueta.authserver.authorization.domain.password.UserAccount;
24  import java.util.List;
25  import java.util.UUID;
26  import org.junit.jupiter.api.Test;
27  import org.springframework.security.core.GrantedAuthority;
28  
29  /**
30   * Unit tests of {@link UserDetailsMapper}.
31   */
32  class UserDetailsMapperTest {
33  
34      /** Username used by the tests. */
35      private static final String USERNAME = "alice";
36  
37      /** Encoded password stored on the account; not a real credential. */
38      private static final String ENCODED_PASSWORD = "{noop}encoded";
39  
40      /** Encoded password passed to the mapper; not a real credential. */
41      private static final String OTHER_ENCODED_PASSWORD = "{argon2}upgraded";
42  
43      /** Authority used by the tests. */
44      private static final String ROLE_USER = "ROLE_USER";
45  
46      /** Mapper under test. */
47      private final UserDetailsMapper mapper = new UserDetailsMapper();
48  
49      /**
50       * Verifies that an enabled account is mapped to enabled details with its authorities.
51       */
52      @Test
53      void toUserDetails_whenAccountIsEnabled_returnsEnabledUserWithAuthorities() {
54          final var account = new UserAccount(UUID.randomUUID(), USERNAME, ENCODED_PASSWORD, true, List.of(ROLE_USER));
55  
56          final var user = mapper.toUserDetails(account, ENCODED_PASSWORD);
57  
58          assertEquals(USERNAME, user.getUsername());
59          assertEquals(ENCODED_PASSWORD, user.getPassword());
60          assertTrue(user.isEnabled());
61          assertEquals(List.of(ROLE_USER), user.getAuthorities().stream().map(GrantedAuthority::getAuthority).toList());
62      }
63  
64      /**
65       * Verifies that a disabled account is mapped to disabled details.
66       */
67      @Test
68      void toUserDetails_whenAccountIsDisabled_returnsDisabledUser() {
69          final var account = new UserAccount(UUID.randomUUID(), USERNAME, ENCODED_PASSWORD, false, List.of());
70  
71          final var user = mapper.toUserDetails(account, ENCODED_PASSWORD);
72  
73          assertFalse(user.isEnabled());
74          assertTrue(user.getAuthorities().isEmpty());
75      }
76  
77      /**
78       * Verifies that the details carry the given password rather than the stored one.
79       */
80      @Test
81      void toUserDetails_whenPasswordDiffersFromStoredOne_carriesGivenPassword() {
82          final var account = new UserAccount(UUID.randomUUID(), USERNAME, ENCODED_PASSWORD, true, List.of(ROLE_USER));
83  
84          final var user = mapper.toUserDetails(account, OTHER_ENCODED_PASSWORD);
85  
86          assertEquals(OTHER_ENCODED_PASSWORD, user.getPassword());
87      }
88  }
