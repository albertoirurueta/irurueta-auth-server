17  package com.irurueta.authserver.authorization.api.web.social;
18  
19  import static org.junit.jupiter.api.Assertions.assertEquals;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  
22  import com.irurueta.authserver.authorization.domain.social.FederatedIdentity;
23  import com.irurueta.authserver.authorization.domain.social.IdentityProvider;
24  import java.time.Instant;
25  import java.util.List;
26  import java.util.UUID;
27  import org.junit.jupiter.api.Test;
28  
29  /**
30   * Unit tests of {@link IdentityDtoMapper}.
31   */
32  class IdentityDtoMapperTest {
33  
34      /** Link instant. */
35      private static final Instant LINKED = Instant.parse("2026-01-01T00:00:00Z");
36  
37      /** Last sign-in instant. */
38      private static final Instant LAST_LOGIN = Instant.parse("2026-02-01T00:00:00Z");
39  
40      /** Mapper under test. */
41      private final IdentityDtoMapper mapper = new IdentityDtoMapper();
42  
43      /**
44       * Verifies that every public field is copied, the provider as the contract's value, and that the subject and the
45       * account are not part of the answer.
46       */
47      @Test
48      void toResponse_whenIdentityWasUsed_copiesItsPublicFields() {
49          final var id = UUID.randomUUID();
50  
51          final var response = mapper.toResponse(new FederatedIdentity(id, UUID.randomUUID(), IdentityProvider.GITHUB,
52                  "subject", "alice@example.org", LINKED, LAST_LOGIN));
53  
54          assertEquals(id, response.id());
55          assertEquals("github", response.provider().getValue());
56          assertEquals("alice@example.org", response.email());
57          assertEquals(LINKED, response.linkedAt());
58          assertEquals(LAST_LOGIN, response.lastLoginAt());
59      }
60  
61      /**
62       * Verifies that an identity without an address or a sign-in has neither.
63       */
64      @Test
65      void toResponse_whenIdentityHasNoEmailAndWasNeverUsed_hasNeither() {
66          final var response = mapper.toResponse(new FederatedIdentity(UUID.randomUUID(), UUID.randomUUID(),
67                  IdentityProvider.GOOGLE, "subject", null, LINKED, null));
68  
69          assertNull(response.email());
70          assertNull(response.lastLoginAt());
71      }
72  
73      /**
74       * Verifies that several identities are mapped in order.
75       */
76      @Test
77      void toResponses_whenSeveralIdentities_keepsTheirOrder() {
78          final var responses = mapper.toResponses(List.of(
79                  new FederatedIdentity(UUID.randomUUID(), UUID.randomUUID(), IdentityProvider.GOOGLE, "a", null,
80                          LINKED, null),
81                  new FederatedIdentity(UUID.randomUUID(), UUID.randomUUID(), IdentityProvider.APPLE, "b", null,
82                          LINKED, null)));
83  
84          assertEquals(List.of("google", "apple"), responses.stream().map(r -> r.provider().getValue()).toList());
85      }
86  }
