58  @ExtendWith(MockitoExtension.class)

61      /** Name of the signed-in account. */
62      private static final String USERNAME = "alice";
63  
64      /** Account of the requests. */
65      private final CurrentAccount account = new CurrentAccount(UUID.randomUUID(), USERNAME);

92      @Mock

96      @InjectMocks
97      private IdentitiesPageController controller;

99      /**
100      * Resolves the account of every request.
101      */
102     @BeforeEach
103     void setUp() {
104         when(currentAccountResolver.resolve()).thenReturn(account);
105     }

107     /**
108      * The page lists the linked identities and the configured providers, tells a known status and is never cached.
109      */
110     @Test
111     @SuppressWarnings("unchecked")
112     void identitiesPage_whenProvidersAreConfigured_rendersTheIdentitiesAndTheProviders() {
113         givenProviders(IdentityProvider.GOOGLE, IdentityProvider.GITHUB);
114         when(listIdentitiesUseCase.execute(account.id())).thenReturn(List.of(new FederatedIdentity(UUID.randomUUID(),
115                 account.id(), IdentityProvider.GOOGLE, "subject", null, Instant.EPOCH, null)));
116         when(pageRenderer.render(eq(IdentitiesPageController.VIEW_NAME), anyMap()))
117                 .thenReturn(ResponseEntity.ok("<html></html>"));
118 
119         final var page = controller.identitiesPage(IdentitiesPageController.LINKED);
120 
121         final var model = ArgumentCaptor.<Map<String, Object>>captor();
122         verify(pageRenderer).render(eq(IdentitiesPageController.VIEW_NAME), model.capture());
123         assertEquals(List.of(new IdentityView("google", null, Instant.EPOCH)),
124                 model.getValue().get(IdentitiesPageController.IDENTITIES_ATTRIBUTE));
125         assertEquals(List.of("google", "github"), model.getValue().get(IdentitiesPageController.PROVIDERS_ATTRIBUTE));
126         assertEquals(IdentitiesPageController.LINKED, model.getValue().get(IdentitiesPageController.STATUS_ATTRIBUTE));
127         assertEquals("no-store", page.getHeaders().getCacheControl());
128     }
129 
130     /**
131      * Without a configured provider the page lists none, and an unknown status is ignored.
132      */
133     @Test
134     @SuppressWarnings("unchecked")
135     void identitiesPage_whenNoProviderIsConfigured_rendersNoProviderAndIgnoresAnUnknownStatus() {
136         when(clientRegistrations.getIfAvailable()).thenReturn(null);
137         when(listIdentitiesUseCase.execute(account.id())).thenReturn(List.of());
138         when(pageRenderer.render(eq(IdentitiesPageController.VIEW_NAME), anyMap()))
139                 .thenReturn(ResponseEntity.ok("<html></html>"));
140 
141         controller.identitiesPage("<script>");
142 
143         final var model = ArgumentCaptor.<Map<String, Object>>captor();
144         verify(pageRenderer).render(eq(IdentitiesPageController.VIEW_NAME), model.capture());
145         assertEquals(List.of(), model.getValue().get(IdentitiesPageController.PROVIDERS_ATTRIBUTE));
146         assertFalse(model.getValue().containsKey(IdentitiesPageController.STATUS_ATTRIBUTE));
147     }
148 
149     /**
150      * A configured provider records the marker and redirects to the provider's sign-in.
151      */
152     @Test
153     void linkIdentityPage_whenTheProviderIsConfigured_recordsTheMarkerAndRedirectsToTheProvider() {
154         givenProviders(IdentityProvider.GOOGLE);
155         final var session = mock(HttpSession.class);
156         when(request.getSession()).thenReturn(session);
157         when(request.getContextPath()).thenReturn("");
158 
159         final var response = controller.linkIdentityPage("google");
160 
161         assertEquals(HttpStatus.SEE_OTHER, response.getStatusCode());
162         assertEquals("/oauth2/authorization/google", response.getHeaders().getFirst(HttpHeaders.LOCATION));
163         verify(pendingIdentityLinks).begin(session, IdentityProvider.GOOGLE, account.id(), USERNAME);
164     }
165 
166     /**
167      * A provider that is not configured, or not a provider at all, records nothing.
168      */
169     @Test
170     void linkIdentityPage_whenTheProviderIsNotConfigured_redirectsToThePageWithUnknownProvider() {
171         givenProviders(IdentityProvider.GOOGLE);
172         when(request.getContextPath()).thenReturn("");
173 
174         final var notConfigured = controller.linkIdentityPage("github");
175         final var unknown = controller.linkIdentityPage("myspace");
176 
177         assertEquals("/account/identities?status=unknown-provider",
178                 notConfigured.getHeaders().getFirst(HttpHeaders.LOCATION));
179         assertEquals("/account/identities?status=unknown-provider",
180                 unknown.getHeaders().getFirst(HttpHeaders.LOCATION));
181         verifyNoInteractions(pendingIdentityLinks);
182     }
183 
184     /**
185      * Configures the given providers.
186      *
187      * @param providers the configured providers.
188      */
189     private void givenProviders(final IdentityProvider... providers) {
190         final var configured = Arrays.stream(providers).map(IdentityProvider::id).toList();
191         final var repository = mock(ClientRegistrationRepository.class);
192         when(repository.findByRegistrationId(anyString())).thenAnswer(invocation -> {
193             final String id = invocation.getArgument(0);
194             return configured.contains(id) ? registration(id) : null;
195         });
196         when(clientRegistrations.getIfAvailable()).thenReturn(repository);
197     }
198 
199     /**
200      * Builds the registration of a provider.
201      *
202      * @param id registration id of the provider.
203      * @return its registration.
204      */
205     private static ClientRegistration registration(final String id) {
206         return ClientRegistration.withRegistrationId(id).clientId("client")
207                 .authorizationGrantType(AuthorizationGrantType.AUTHORIZATION_CODE)
208                 .redirectUri("{baseUrl}/login/oauth2/code/{registrationId}")
209                 .authorizationUri("https://idp.example/authorize").tokenUri("https://idp.example/token").build();
210     }
