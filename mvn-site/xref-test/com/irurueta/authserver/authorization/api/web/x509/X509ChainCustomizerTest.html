103     /**
104      * Verifies that only {@code POST /login/x509} is public: a {@code GET} of the path is not a sign-in.
105      */
106     @Test
107     @SuppressWarnings({"unchecked", "rawtypes"})
108     void customize_whenApplied_permitsOnlyTheSignInRequest() {
109         final var http = mock(HttpSecurity.class, RETURNS_SELF);
110         final var captor = ArgumentCaptor.<Customizer<
111                 AuthorizeHttpRequestsConfigurer<HttpSecurity>.AuthorizationManagerRequestMatcherRegistry>>captor();
112         when(http.authorizeHttpRequests(captor.capture())).thenReturn(http);
113         final var registry = mock(AuthorizeHttpRequestsConfigurer.AuthorizationManagerRequestMatcherRegistry.class);
114         final var open = mock(AuthorizeHttpRequestsConfigurer.AuthorizedUrl.class);
115         final var matcherCaptor = ArgumentCaptor.<RequestMatcher>captor();
116         when(registry.requestMatchers(matcherCaptor.capture())).thenReturn(open);
117 
118         customizer(forwardedFilter, new X509LoginFilter()).customize(http);
119         captor.getValue().customize(registry);
120 
121         verify(open).permitAll();
122         final var matcher = matcherCaptor.getValue();
123         assertTrue(matcher.matches(new MockHttpServletRequest("POST", "/login/x509")));
124         assertFalse(matcher.matches(new MockHttpServletRequest("GET", "/login/x509")));
125         assertFalse(matcher.matches(new MockHttpServletRequest("POST", "/login")));
126     }

128     /**
129      * Verifies that the filter, the extractor and the user service are handed to the {@code x509} configurer.
130      */
131     @Test
132     @SuppressWarnings("unchecked")
133     void customize_whenApplied_configuresTheX509Login() {
134         final var http = mock(HttpSecurity.class, RETURNS_SELF);
135         final var captor = ArgumentCaptor.<Customizer<X509Configurer<HttpSecurity>>>captor();
136         when(http.x509(captor.capture())).thenReturn(http);
137         final var x509 = mock(X509Configurer.class, RETURNS_SELF);
138         final var filter = new X509LoginFilter();
139 
140         customizer(forwardedFilter, filter).customize(http);
141         captor.getValue().customize(x509);
142 
143         verify(x509).x509AuthenticationFilter(filter);
144         verify(x509).x509PrincipalExtractor(principalExtractor);
145         verify(x509).authenticationUserDetailsService(userDetailsService);
146     }

172     /**
173      * Verifies that a certificate on the sign-in path is authenticated through the chain's manager, with the
174      * multi-factor merge on, and handed to the success handler.
175      *
176      * @throws Exception if the filter fails.
177      */
178     @Test
179     void customize_whenACertificateArrivesAtTheSignInPath_authenticatesAndCallsTheSuccessHandler()
180             throws Exception {
181         final var http = mock(HttpSecurity.class, RETURNS_SELF);
182         when(http.getSharedObject(AuthenticationManager.class)).thenReturn(authenticationManager);
183         final var result = mock(Authentication.class);
184         when(authenticationManager.authenticate(any())).thenReturn(result);
185         when(principalExtractor.extractPrincipal(any())).thenReturn(FINGERPRINT);
186         final var filter = new X509LoginFilter();
187         customizer(null, filter).customize(http);
188         final var request = signInRequest();
189         final var response = new MockHttpServletResponse();
190 
191         filter.doFilter(request, response, new MockFilterChain());
192 
193         final var captor = ArgumentCaptor.<Authentication>captor();
194         verify(authenticationManager).authenticate(captor.capture());
195         assertEquals(FINGERPRINT, captor.getValue().getPrincipal());
196         verify(successHandler).onAuthenticationSuccess(request, response, result);
197     }

218     /**
219      * Verifies that a {@code GET} of the sign-in path, even with a certificate, neither authenticates nor is answered
220      * by the filter: it goes on along the chain, whose catch-all rule handles it.
221      *
222      * @throws Exception if the filter fails.
223      */
224     @Test
225     void customize_whenTheSignInPathIsReadWithGet_doesNotAuthenticate() throws Exception {
226         final var http = mock(HttpSecurity.class, RETURNS_SELF);
227         final var filter = new X509LoginFilter();
228         customizer(null, filter).customize(http);
229         final var request = signInRequest();
230         request.setMethod("GET");
231         final var chain = new MockFilterChain();
232 
233         filter.doFilter(request, new MockHttpServletResponse(), chain);
234 
235         verify(authenticationManager, never()).authenticate(any());
236         assertSame(request, chain.getRequest());
237     }

239     /**
240      * Verifies that a certificate nobody owns is sent back to the login page.
241      *
242      * @throws Exception if the filter fails.
243      */
244     @Test
245     void customize_whenTheCertificateIsRefused_redirectsToTheLoginPageWithError() throws Exception {
246         final var http = mock(HttpSecurity.class, RETURNS_SELF);
247         when(http.getSharedObject(AuthenticationManager.class)).thenReturn(authenticationManager);
248         when(authenticationManager.authenticate(any())).thenThrow(new BadCredentialsException("unknown"));
249         when(principalExtractor.extractPrincipal(any())).thenReturn(FINGERPRINT);
250         final var filter = new X509LoginFilter();
251         customizer(null, filter).customize(http);
252         final var response = new MockHttpServletResponse();
253         final var chain = new MockFilterChain();
254 
255         filter.doFilter(signInRequest(), response, chain);
256 
257         assertEquals("/login?error", response.getRedirectedUrl());
258         assertEquals(null, chain.getRequest(), "the chain stops after the redirect");
259     }

287     /**
288      * Builds the customizer.
289      *
290      * @param forwarded the forwarded filter, or {@code null}.
291      * @param filter    the X.509 filter.
292      * @return the customizer.
293      */
294     private X509ChainCustomizer customizer(final ForwardedCertificateFilter forwarded, final X509LoginFilter filter) {
295         return new X509ChainCustomizer(filter, principalExtractor, userDetailsService, forwarded, successHandler);
296     }
297 
298     /**
299      * Builds a sign-in request that carries a client certificate.
300      *
301      * @return the request.
302      */
303     private static MockHttpServletRequest signInRequest() {
304         final var request = new MockHttpServletRequest("POST", "/login/x509");
305         request.setAttribute(ForwardedCertificateFilter.CERTIFICATE_ATTRIBUTE,
306                 new X509Certificate[] {mock(X509Certificate.class)});
307         return request;
308     }
