43  /**
44   * Checks the certificate operations of the account API as the whole application wires them: a PEM certificate is
45   * bound, listed and unbound through the bearer-protected API; a certificate of another account and a value that is not
46   * a certificate are refused; the last way to sign in to an account is never unbound; and, while the account has a
47   * confirmed second factor, binding needs a token of a recent multi-factor sign-in. Every certificate is generated at
48   * run time.
49   */
50  class CertificateApiIT extends AbstractMfaFlow {
51  
52      /** Path of the certificates of the account API. */
53      private static final String CERTIFICATES = "/api/account/certificates";
54  
55      /** The challenge of an operation that needs a recent multi-factor sign-in. */
56      private static final String STEP_UP_CHALLENGE = "Bearer error=\"insufficient_user_authentication\", "
57              + "acr_values=\"urn:irurueta:acr:mfa\", max_age=\"300\"";
58  
59      /** Writes the request bodies. */
60      private static final JsonMapper JSON = JsonMapper.builder().build();
61  
62      /** Port that stores the users. */
63      @Autowired
64      private UserAccountRepository userAccounts;
65  
66      /** Port that remembers the accounts that only hold a generated password. */
67      @Autowired
68      private GeneratedPasswordRepository generatedPasswords;
69  
70      /** Port that stores the bindings. */
71      @Autowired
72      private CertificateBindingRepository certificateBindings;
73  
74  
75      /**
76       * Verifies that a certificate is bound, listed with the fingerprint the sign-in looks up, and unbound while the
77       * account keeps its password.
78       *
79       * @throws Exception if a request cannot be performed.
80       */
81      @Test
82      void certificates_whenBoundListedAndUnbound_followTheContract() throws Exception {
83          final var user = getFixtures().createUser();
84          final var certificate = TestCertificates.generate("bind-" + UUID.randomUUID());
85  
86          final var bound = getMockMvc().perform(bind(user.username(), certificate.pem()))
87                  .andExpect(status().isCreated())
88                  .andExpect(jsonPath("$.fingerprintSha256").value(certificate.fingerprintSha256()))
89                  .andReturn().getResponse().getContentAsString();
90          final var id = AuthServerFixtures.parse(bound).get("id").asString();
91          getMockMvc().perform(bind(user.username(), certificate.pem()))
92                  .andExpect(status().isCreated())
93                  .andExpect(jsonPath("$.id").value(id));
94          getMockMvc().perform(get(CERTIFICATES).with(bearer(user.username())))
95                  .andExpect(status().isOk())
96                  .andExpect(jsonPath("$", hasSize(1)))
97                  .andExpect(jsonPath("$[0].id").value(id));
98          getMockMvc().perform(delete(CERTIFICATES + "/" + id).with(bearer(user.username())))
99                  .andExpect(status().isNoContent());
100 
101         assertEquals(0, certificateBindings.findByUser(accountId(user.username())).size());
102     }
103 
104     /**
105      * Verifies that a certificate bound to another account, an expired certificate and a value that is not a
106      * certificate are refused.
107      *
108      * @throws Exception if a request cannot be performed.
109      */
110     @Test
111     void bindCertificate_whenTheCertificateCannotBeBound_isRefused() throws Exception {
112         final var owner = getFixtures().createUser();
113         final var other = getFixtures().createUser();
114         final var certificate = TestCertificates.generate("owned-" + UUID.randomUUID());
115         getMockMvc().perform(bind(owner.username(), certificate.pem())).andExpect(status().isCreated());
116 
117         getMockMvc().perform(bind(other.username(), certificate.pem()))
118                 .andExpect(status().isConflict())
119                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:certificate-already-bound"));
120         getMockMvc().perform(bind(other.username(),
121                         TestCertificates.generateExpired("expired-" + UUID.randomUUID()).pem()))
122                 .andExpect(status().isBadRequest())
123                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:invalid-certificate"));
124         getMockMvc().perform(bind(other.username(), "not a certificate"))
125                 .andExpect(status().isBadRequest())
126                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:invalid-certificate"));
127     }
128 
129     /**
130      * Verifies that the only way to sign in to an account without a password of its own, its certificate, is never
131      * unbound, that it can be once another certificate is bound, and that a certificate of another account is answered
132      * as not found.
133      *
134      * @throws Exception if a request cannot be performed.
135      */
136     @Test
137     void unbindCertificate_whenItIsTheLastWayToSignIn_isRefused() throws Exception {
138         final var user = getFixtures().createUser();
139         generatedPasswords.markGenerated(accountId(user.username()));
140         final var first = boundId(user.username(), TestCertificates.generate("first-" + UUID.randomUUID()).pem());
141 
142         getMockMvc().perform(delete(CERTIFICATES + "/" + first).with(bearer(user.username())))
143                 .andExpect(status().isConflict())
144                 .andExpect(jsonPath("$.type").value("urn:problem:certificate:last-sign-in-method"));
145         boundId(user.username(), TestCertificates.generate("second-" + UUID.randomUUID()).pem());
146         getMockMvc().perform(delete(CERTIFICATES + "/" + first).with(bearer(getFixtures().createUser().username())))
147                 .andExpect(status().isNotFound());
148         getMockMvc().perform(delete(CERTIFICATES + "/" + first).with(bearer(user.username())))
149                 .andExpect(status().isNoContent());
150 
151         assertEquals(1, certificateBindings.findByUser(accountId(user.username())).size());
152     }
153 
154     /**
155      * Verifies that, while the account has a confirmed second factor, a token without a recent multi-factor sign-in is
156      * answered with the step-up challenge and binds nothing, and a token of a recent multi-factor sign-in binds.
157      *
158      * @throws Exception if a request cannot be performed.
159      */
160     @Test
161     void bindCertificate_whenTheAccountHasASecondFactor_needsAStepUpToken() throws Exception {
162         final var enrolled = getFixtures().createEnrolledUser(getTools());
163         final var username = enrolled.user().username();
164         final var pem = TestCertificates.generate("step-up-" + UUID.randomUUID()).pem();
165 
166         getMockMvc().perform(bind(username, pem))
167                 .andExpect(status().isUnauthorized())
168                 .andExpect(header().string("WWW-Authenticate", STEP_UP_CHALLENGE));
169         assertEquals(0, certificateBindings.findByUser(enrolled.id()).size());
170         getMockMvc().perform(post(CERTIFICATES).with(bearer(username).jwt(token -> token.subject(username)
171                                 .claim(ACR, AuthenticationContextClass.MULTI_FACTOR)
172                                 .claim(AUTH_TIME, Instant.now())))
173                         .contentType(MediaType.APPLICATION_JSON).content(body(pem)))
174                 .andExpect(status().isCreated());
175 
176         assertEquals(1, certificateBindings.findByUser(enrolled.id()).size());
177     }

179     /**
180      * Builds a bind request.
181      *
182      * @param username the account of the token.
183      * @param pem      the certificate.
184      * @return the request.
185      */
186     private static MockHttpServletRequestBuilder bind(final String username, final String pem) {
187         return post(CERTIFICATES).with(bearer(username)).contentType(MediaType.APPLICATION_JSON).content(body(pem));
188     }
189 
190     /**
191      * Binds a certificate and answers the identifier of the binding.
192      *
193      * @param username the account.
194      * @param pem      the certificate.
195      * @return the identifier of the binding.
196      * @throws Exception if the request cannot be performed.
197      */
198     private String boundId(final String username, final String pem) throws Exception {
199         final var bound = getMockMvc().perform(bind(username, pem)).andExpect(status().isCreated())
200                 .andReturn().getResponse().getContentAsString();
201         return AuthServerFixtures.parse(bound).get("id").asString();
202     }
203 
204     /**
205      * Builds the body of a bind request.
206      *
207      * @param pem the certificate.
208      * @return the JSON body.
209      */
210     private static String body(final String pem) {
211         return JSON.writeValueAsString(Map.of("certificate", pem));
212     }
213 
214     /**
215      * Builds the bearer token of an account, without any multi-factor claim.
216      *
217      * @param username the account.
218      * @return the token.
219      */
220     private static JwtRequestPostProcessor bearer(final String username) {
221         return jwt().jwt(token -> token.subject(username));
222     }
223 
224     /**
225      * Finds the identifier of an account.
226      *
227      * @param username the account.
228      * @return its identifier.
229      */
230     private UUID accountId(final String username) {
231         return userAccounts.findByUsername(username).orElseThrow().id();
232     }
