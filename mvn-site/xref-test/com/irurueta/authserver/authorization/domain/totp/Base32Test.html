16  package com.irurueta.authserver.authorization.domain.totp;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  
20  import java.nio.charset.StandardCharsets;
21  import org.junit.jupiter.api.Test;
22  
23  /**
24   * Unit tests of {@link Base32}, with the vectors of RFC 4648 section 10 without their padding.
25   */
26  class Base32Test {
27  
28      /**
29       * Verifies that nothing encodes to nothing.
30       */
31      @Test
32      void encode_whenInputIsEmpty_returnsEmptyText() {
33          assertEquals("", Base32.encode(new byte[0]));
34      }
35  
36      /**
37       * Verifies the RFC 4648 vectors for every remainder length.
38       */
39      @Test
40      void encode_whenRfcVector_returnsExpectedText() {
41          assertEquals("MY", encode("f"));
42          assertEquals("MZXQ", encode("fo"));
43          assertEquals("MZXW6", encode("foo"));
44          assertEquals("MZXW6YQ", encode("foob"));
45          assertEquals("MZXW6YTB", encode("fooba"));
46          assertEquals("MZXW6YTBOI", encode("foobar"));
47      }
48  
49      /**
50       * Verifies that high bytes are treated as unsigned.
51       */
52      @Test
53      void encode_whenBytesAreNegative_returnsExpectedText() {
54          assertEquals("74", Base32.encode(new byte[] {-1}));
55      }
56  
57      /**
58       * Encodes the ASCII bytes of a text.
59       *
60       * @param text the text.
61       * @return the Base32 form.
62       */
63      private static String encode(final String text) {
64          return Base32.encode(text.getBytes(StandardCharsets.US_ASCII));
65      }
66  }
