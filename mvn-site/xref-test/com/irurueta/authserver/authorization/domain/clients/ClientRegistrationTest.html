34      /** Client id used by the tests. */
35      private static final String CLIENT_ID = "bff";

37      /** Positive lifetime used by the tests. */
38      private static final Duration TTL = Duration.ofMinutes(1);

40      /**
41       * Verifies that valid values are kept.
42       */
43      @Test
44      void constructor_whenValuesAreValid_keepsThem() {
45          final var client = create(CLIENT_ID, "Name", true, true, List.of("api"), TTL, TTL);
46  
47          assertEquals(CLIENT_ID, client.clientId());
48          assertEquals(List.of("api"), client.audiences());
49          assertEquals(List.of("openid"), client.scopes());
50      }

55      @Test
56      void constructor_whenAudiencesAreEmpty_defaultsToClientId() {
57          final var client = create(CLIENT_ID, "Name", true, true, List.of(), TTL, TTL);
58  
59          assertEquals(List.of(CLIENT_ID), client.audiences());
60      }

62      /**
63       * Verifies that a blank id, name or format is rejected.
64       */
65      @Test
66      void constructor_whenIdOrNameIsBlank_throwsIllegalArgumentException() {
67          assertThrows(IllegalArgumentException.class, () -> create(" ", "Name", true, true, List.of(), TTL, TTL));
68          assertThrows(IllegalArgumentException.class, () -> create(CLIENT_ID, null, true, true, List.of(), TTL, TTL));
69          assertThrows(IllegalArgumentException.class, () -> new ClientRegistration(CLIENT_ID, "Name", true,
70                  List.of(), List.of(), List.of(), List.of(), true, List.of(), TTL, TTL, "", false, List.of(), false,
71                  false, null, null, null, null, false));
72      }

75       * Verifies that null collections and null elements are rejected.
76       */
77      @Test
78      void constructor_whenCollectionIsInvalid_throwsIllegalArgumentException() {

88                  ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false, false,
89                  null, null, null, null, false));

92      /**
93       * Verifies that missing, zero and negative lifetimes are rejected.
94       */
95      @Test
96      void constructor_whenLifetimeIsNotPositive_throwsIllegalArgumentException() {
97          assertThrows(IllegalArgumentException.class,

225     /**
226      * Creates a registration.
227      *
228      * @param id           client id.
229      * @param name         name.
230      * @param confidential whether confidential.
231      * @param pkce         whether PKCE is required.
232      * @param audiences    audiences.
233      * @param access       access token lifetime.
234      * @param refresh      refresh token lifetime.
235      * @return the registration.
236      */
237     private static ClientRegistration create(
238             final String id, final String name, final boolean confidential, final boolean pkce,
239             final List<String> audiences, final Duration access, final Duration refresh) {
240         return new ClientRegistration(id, name, confidential, List.of("none"), List.of("authorization_code"),
241                 List.of("http://localhost/cb"), List.of("openid"), pkce, audiences, access, refresh,
242                 ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false, false,
243                 null, null, null, null, false);
244     }

246     /**
247      * Verifies that post-logout redirect URIs are copied and that a null list or a null element is rejected.
248      */
249     @Test
250     void constructor_whenPostLogoutRedirectUrisAreGiven_copiesThemAndRejectsNulls() {
251         final var uris = new ArrayList<>(List.of("https://app.example.test/"));
252         final var registration = new ClientRegistration(CLIENT_ID, "Name", true, List.of(), List.of(), List.of(),
253                 List.of(), true, List.of(), TTL, TTL, ClientRegistration.FORMAT_SELF_CONTAINED, false, uris, false,
254                 false, null, null, null, null, false);
255         uris.clear();
256         final var withNull = new ArrayList<String>();
257         withNull.add(null);
258 
259         assertEquals(List.of("https://app.example.test/"), registration.postLogoutRedirectUris());
260         assertThrows(IllegalArgumentException.class, () -> new ClientRegistration(CLIENT_ID, "Name", true,
261                 List.of(), List.of(), List.of(), List.of(), true, List.of(), TTL, TTL,
262                 ClientRegistration.FORMAT_SELF_CONTAINED, false, null, false, false, null, null, null, null, false));
263         assertThrows(IllegalArgumentException.class, () -> new ClientRegistration(CLIENT_ID, "Name", true,
264                 List.of(), List.of(), List.of(), List.of(), true, List.of(), TTL, TTL,
265                 ClientRegistration.FORMAT_SELF_CONTAINED, false, withNull, false, false,
266                 null, null, null, null, false));
267     }

276                 List.of("authorization_code"), List.of("http://localhost/cb"), List.of("openid"), true, List.of(), TTL,

284     /**
285      * Creates an implicit-flow registration.
286      *
287      * @param consent      whether consent is required.
288      * @param confidential whether the client is confidential.
289      * @param grantTypes   the grant types.
290      * @param noPkce       {@code true} to not require PKCE.
291      * @return the registration.
292      */
293     private static ClientRegistration implicit(
294             final boolean consent, final boolean confidential, final List<String> grantTypes, final boolean noPkce) {
295         return new ClientRegistration(CLIENT_ID, "Name", confidential,
296                 List.of(confidential ? "client_secret_basic" : "none"), grantTypes, List.of("http://localhost/cb"), List.of("openid"), !noPkce, List.of(), TTL, TTL,
297                 ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), consent, false,
298                 null, null, null, null, false);
299     }

301     /**
302      * Creates a registration with one authentication method and the given key settings.
303      *
304      * @param method     the authentication method.
305      * @param jwkSetUrl  the JWK set URL, or {@code null}.
306      * @param jwks       the JWK set, or {@code null}.
307      * @param subjectDn  the TLS subject DN, or {@code null}.
308      * @return the registration.
309      */
310     private static ClientRegistration authClient(
311             final String method, final String jwkSetUrl, final String jwks, final String subjectDn) {
312         return new ClientRegistration(CLIENT_ID, "Name", true, List.of(method), List.of("client_credentials"),
313                 List.of(), List.of("api"), true, List.of(), TTL, TTL, ClientRegistration.FORMAT_SELF_CONTAINED, false,
314                 List.of(), false, false, jwkSetUrl, jwks, null, subjectDn, false);
