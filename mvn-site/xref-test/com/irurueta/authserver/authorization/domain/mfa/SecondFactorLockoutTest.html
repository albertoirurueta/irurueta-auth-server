16  package com.irurueta.authserver.authorization.domain.mfa;
17  
18  import static org.junit.jupiter.api.Assertions.assertEquals;
19  import static org.junit.jupiter.api.Assertions.assertFalse;
20  import static org.junit.jupiter.api.Assertions.assertNull;
21  import static org.junit.jupiter.api.Assertions.assertThrows;
22  import static org.junit.jupiter.api.Assertions.assertTrue;
23  
24  import java.time.Instant;
25  import org.junit.jupiter.api.Test;
26  
27  /**
28   * Unit tests of {@link SecondFactorLockout}.
29   */
30  class SecondFactorLockoutTest {
31  
32      /** Instant used as the end of a lock. */
33      private static final Instant UNTIL = Instant.parse("2026-01-01T00:15:00Z");
34  
35      /** Failed attempts used by the tests. */
36      private static final int FAILED = 5;
37  
38      /**
39       * Verifies that a negative counter is rejected.
40       */
41      @Test
42      void constructor_whenCounterIsNegative_throwsIllegalArgumentException() {
43          assertThrows(IllegalArgumentException.class, () -> new SecondFactorLockout(-1, null));
44      }
45  
46      /**
47       * Verifies that the clean state has no failures and no lock.
48       */
49      @Test
50      void none_whenCalled_returnsCleanState() {
51          final var state = SecondFactorLockout.none();
52  
53          assertEquals(0, state.failedCount());
54          assertNull(state.lockedUntil());
55          assertFalse(state.isLocked(UNTIL));
56      }
57  
58      /**
59       * Verifies that the lock holds before its end.
60       */
61      @Test
62      void isLocked_whenBeforeLockEnd_returnsTrue() {
63          assertTrue(new SecondFactorLockout(FAILED, UNTIL).isLocked(UNTIL.minusSeconds(1)));
64      }
65  
66      /**
67       * Verifies that the lock is over at its end.
68       */
69      @Test
70      void isLocked_whenAtLockEnd_returnsFalse() {
71          assertFalse(new SecondFactorLockout(FAILED, UNTIL).isLocked(UNTIL));
72      }
73  }
