77  /**
78   * Checks the client authentication methods of the token endpoint against the running application: the shared secret,
79   * {@code private_key_jwt} with registered keys and with a {@code jwkSetUrl} served by WireMock (and the refusal of a
80   * replayed or over-long assertion), {@code tls_client_auth} with a certificate issued by the configured authority (and
81   * the refusal of a self-issued one with the same DN), {@code self_signed_tls_client_auth}, the certificate-bound access
82   * token ({@code cnf.x5t#S256}) and the DPoP proof at the token endpoint ({@code token_type=DPoP}, {@code cnf.jkt}).
83   * The mutual-TLS calls are real handshakes against the dedicated TLS port. The authorities, certificates and keys are
84   * generated when the test runs and live in temporary directories removed afterwards; nothing is kept in the repository.

414     /**
415      * Sends a client credentials request authenticated with a signed assertion.
416      *
417      * @param clientId the client.
418      * @param key      the key that signs.
419      * @param jti      the identifier of the assertion.
420      * @param lifetime the lifetime of the assertion.
421      * @return the answer.
422      * @throws IOException          if the request fails.
423      * @throws InterruptedException if the request is interrupted.
424      * @throws JOSEException        if the assertion cannot be signed.
425      */
426     private HttpResponse<String> assertionRequest(final String clientId, final ECKey key, final String jti,
427             final Duration lifetime) throws IOException, InterruptedException, JOSEException {
428         final var now = Instant.now();
429         final var claims = new JWTClaimsSet.Builder().issuer(clientId).subject(clientId)
430                 .audience(issuer + TOKEN_PATH).jwtID(jti).issueTime(Date.from(now))
431                 .expirationTime(Date.from(now.plus(lifetime))).build();
432         final var assertion = new SignedJWT(new JWSHeader.Builder(JWSAlgorithm.ES256).keyID(key.getKeyID()).build(),
433                 claims);
434         assertion.sign(new ECDSASigner(key));
435         return postForm(plainClient(), plainTokenUri(), Map.of("grant_type", "client_credentials", "scope", SCOPE,
436                 "client_id", clientId, "client_assertion_type", ASSERTION_TYPE,
437                 "client_assertion", assertion.serialize()), Map.of());
438     }

440     /**
441      * Sends a client credentials request over the TLS port presenting a certificate.
442      *
443      * @param identity the certificate to present.
444      * @param clientId the client that authenticates.
445      * @param headers  extra headers.
446      * @return the answer.
447      * @throws IOException          if the request fails.
448      * @throws InterruptedException if the request is interrupted.
449      */
450     private HttpResponse<String> tlsRequest(final TestPki.Identity identity, final String clientId,
451             final Map<String, String> headers) throws IOException, InterruptedException {
452         final var client = HttpClient.newBuilder().sslContext(PKI.clientContext(identity)).build();
453         return postForm(client, URI.create("https://localhost:" + TLS_PORT + TOKEN_PATH),
454                 Map.of("grant_type", "client_credentials", "scope", SCOPE, "client_id", clientId), headers);
455     }

457     /**
458      * Builds a DPoP proof for the token endpoint.
459      *
460      * @param key the proof key.
461      * @param uri the address of the token endpoint.
462      * @return the serialized proof.
463      * @throws JOSEException if the proof cannot be signed.
464      */
465     private static String dpopProof(final ECKey key, final URI uri) throws JOSEException {
466         final var claims = new JWTClaimsSet.Builder().jwtID(UUID.randomUUID().toString()).claim("htm", "POST")
467                 .claim("htu", uri.toString()).issueTime(new Date()).build();
468         final var proof = new SignedJWT(new JWSHeader.Builder(JWSAlgorithm.ES256).type(new JOSEObjectType("dpop+jwt"))
469                 .jwk(key.toPublicJWK()).build(), claims);
470         proof.sign(new ECDSASigner(key));
471         return proof.serialize();
472     }

474     /**
475      * Posts a form.
476      *
477      * @param client  the HTTP client.
478      * @param uri     the address.
479      * @param form    the form fields.
480      * @param headers the extra headers.
481      * @return the answer.
482      * @throws IOException          if the request fails.
483      * @throws InterruptedException if the request is interrupted.
484      */
485     private static HttpResponse<String> postForm(final HttpClient client, final URI uri,
486             final Map<String, String> form, final Map<String, String> headers)
487             throws IOException, InterruptedException {
488         final var body = form.entrySet().stream().map(entry -> URLEncoder.encode(entry.getKey(), StandardCharsets.UTF_8)
489                 + "=" + URLEncoder.encode(entry.getValue(), StandardCharsets.UTF_8)).collect(Collectors.joining("&"));
490         final var request = HttpRequest.newBuilder(uri).header("Content-Type", "application/x-www-form-urlencoded");
491         new LinkedHashMap<>(headers).forEach(request::header);
492         return client.send(request.POST(HttpRequest.BodyPublishers.ofString(body)).build(),
493                 HttpResponse.BodyHandlers.ofString());
494     }

496     /**
497      * Registers a client credentials client.
498      *
499      * @param authMethods   the client authentication methods.
500      * @param jwkSetUrl     the URL of the client's JWK set, or {@code null}.
501      * @param jwks          the registered JWK set, or {@code null}.
502      * @param algorithm     the assertion signing algorithm, or {@code null}.
503      * @param subjectDn     the registered certificate subject DN, or {@code null}.
504      * @param certificateBound whether the access tokens are bound to the certificate.
505      * @param secret        the client secret, or {@code null} for none.
506      * @return the identifier of the client.
507      */
508     private String register(final List<String> authMethods, final String jwkSetUrl, final String jwks,
509             final String algorithm, final String subjectDn, final boolean certificateBound, final String secret) {
510         final var clientId = "client-" + UUID.randomUUID();
511         clientRegistrations.save(new ClientRegistration(clientId, "Client authentication test", true, authMethods,
512                 List.of("client_credentials"), List.of(), List.of(SCOPE), true, List.of("resource-server"),
513                 TOKEN_TTL, REFRESH_TTL, ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false, false,
514                 jwkSetUrl, jwks, algorithm, subjectDn, certificateBound),
515                 secret == null ? null : passwordEncoder.encode(secret));
516         return clientId;
517     }

529     /**
530      * Converts the public key of a certificate to an EC JWK.
531      *
532      * @param certificate the certificate.
533      * @return the public JWK.
534      */
535     private static ECKey ecKeyOf(final X509Certificate certificate) {
536         final var key = (ECPublicKey) certificate.getPublicKey();
537         return new ECKey.Builder(Curve.forECParameterSpec(key.getParams()), key).build();
538     }

550     /**
551      * Builds the identity of a client that signs its own certificate: the self-signed certificate and key of the
552      * authority of a throw-away infrastructure.
553      *
554      * @param pki the infrastructure whose authority is used.
555      * @return the identity.
556      */
557     private static TestPki.Identity selfSigned(final TestPki pki) {
558         final var directory = pki.issueClient("unused").keyStore().getParent();
559         try {
560             final var pem = Files.readString(directory.resolve("ca.pem"));
561             final var certificate = (X509Certificate) CertificateFactory.getInstance("X.509")
562                     .generateCertificate(new ByteArrayInputStream(pem.getBytes(StandardCharsets.UTF_8)));
563             return new TestPki.Identity(directory.resolve("ca.p12"), Files.readString(directory.resolve("password")),
564                     TestCertificates.of(certificate));
565         } catch (IOException | GeneralSecurityException e) {
566             throw new IllegalStateException("Cannot read the self-signed certificate", e);
567         }
568     }
