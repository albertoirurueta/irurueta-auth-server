104     /**
105      * Verifies that the access token carries the audience of the client, the roles and the tenant of the user.
106      *
107      * @throws Exception if a request cannot be performed.
108      */
109     @Test
110     void accessToken_whenIssuedToAUser_carriesAudienceRolesAndTenant() throws Exception {
111         final var user = fixtures.createUser();
112         final var tokens = fixtures.authorizationCodeTokens(user, fixtures.createCodeClient());
113 
114         final var claims = NimbusJwtDecoder.withJwkSetUri("http://localhost:" + port + "/oauth2/jwks").build()
115                 .decode(tokens.get(ACCESS_TOKEN).asString());
116 
117         assertEquals(user.username(), claims.getSubject());
118         assertEquals(List.of(AuthServerFixtures.RESOURCE_SERVER_AUDIENCE, AuthServerFixtures.API_AUDIENCE),
119                 claims.getAudience());
120         assertEquals(List.of("ROLE_USER"), claims.getClaimAsStringList("roles"));
121         assertEquals(AuthServerFixtures.TENANT, claims.getClaimAsString("tenant"));
122         assertTrue(claims.getClaimAsStringList("scope").containsAll(List.of("openid", "profile", "email")));
123     }

139         assertFalse(claims.hasClaim("roles"));

143     /**
144      * Verifies that refreshing issues a new access token and a new refresh token, and keeps the authorization.
145      *
146      * @throws Exception if a request cannot be performed.
147      */
148     @Test
149     void refresh_whenTokenIsValid_rotatesTheRefreshToken() throws Exception {
150         final var user = fixtures.createUser();
151         final var client = fixtures.createCodeClient();
152         final var tokens = fixtures.authorizationCodeTokens(user, client);
153 
154         final var refreshed = refresh(client, tokens.get(REFRESH_TOKEN).asString())
155                 .andExpectStatusOk();
156 
157         assertNotEquals(tokens.get(REFRESH_TOKEN).asString(), refreshed.get(REFRESH_TOKEN).asString());
158         assertNotEquals(tokens.get(ACCESS_TOKEN).asString(), refreshed.get(ACCESS_TOKEN).asString());
159         assertEquals(1, authorizationCount(user.username()));
160     }

162     // tag::tokens-reuse-it[]
163     /**
164      * Verifies that presenting a rotated refresh token again fails and revokes the whole authorization, so the
165      * legitimate new refresh token stops working too.
166      *
167      * @throws Exception if a request cannot be performed.
168      */
169     @Test
170     void refresh_whenRotatedTokenIsReused_revokesTheAuthorization() throws Exception {
171         final var user = fixtures.createUser();
172         final var client = fixtures.createCodeClient();
173         final var tokens = fixtures.authorizationCodeTokens(user, client);
174         final var rotated = tokens.get(REFRESH_TOKEN).asString();
175         final var current = refresh(client, rotated).andExpectStatusOk().get(REFRESH_TOKEN).asString();
176 
177         refresh(client, rotated).andExpectBadRequest();
178 
179         assertEquals(0, authorizationCount(user.username()));
180         refresh(client, current).andExpectBadRequest();
181     }

225     /**
226      * Verifies that introspection describes an active token and reports an unknown one as inactive.
227      *
228      * @throws Exception if a request cannot be performed.
229      */
230     @Test
231     void introspect_whenTokenIsActiveOrUnknown_reportsIt() throws Exception {
232         final var user = fixtures.createUser();
233         final var client = fixtures.createCodeClient();
234         final var tokens = fixtures.authorizationCodeTokens(user, client);
235 
236         final var active = introspect(client, tokens.get(ACCESS_TOKEN).asString());
237         final var unknown = introspect(client, AuthServerFixtures.randomValue());
238 
239         assertTrue(active.get("active").asBoolean());
240         assertEquals(user.username(), active.get("sub").asString());
241         assertEquals(client.clientId(), active.get("client_id").asString());
242         assertFalse(unknown.get("active").asBoolean());
243     }

245     /**
246      * Verifies that a reference (opaque) access token carries the same claims as a JWT one: introspected by another
247      * client (as a resource server does), it lists the client's audiences, the roles and the tenant of the user.
248      *
249      * @throws Exception if a request cannot be performed.
250      */
251     @Test
252     void introspect_whenTokenIsAReferenceToken_returnsAudienceRolesAndTenant() throws Exception {
253         final var user = fixtures.createUser();
254         final var client = fixtures.createClient(List.of("authorization_code", "refresh_token"),
255                 List.of("openid", "profile", "email"), List.of(AuthServerFixtures.RESOURCE_SERVER_AUDIENCE),
256                 ClientRegistration.FORMAT_REFERENCE);
257         final var accessToken = fixtures.authorizationCodeTokens(user, client).get(ACCESS_TOKEN).asString();
258         assertFalse(accessToken.contains("."), "a reference token is not a JWT");
259         final var resourceServer = fixtures.createClient(List.of("client_credentials"), List.of("read"),
260                 List.of(AuthServerFixtures.API_AUDIENCE));
261 
262         final var introspected = introspect(resourceServer, accessToken);
263 
264         assertTrue(introspected.get("active").asBoolean());
265         assertEquals(user.username(), introspected.get("sub").asString());
266         assertEquals(List.of(AuthServerFixtures.RESOURCE_SERVER_AUDIENCE), strings(introspected.get("aud")));
267         assertEquals(List.of("ROLE_USER"), strings(introspected.get("roles")));
268         assertEquals(AuthServerFixtures.TENANT, introspected.get("tenant").asString());
269     }

276     @Test
277     void userInfo_whenScopesAreGranted_containsProfileAndEmailClaims() throws Exception {

280 
281         final var body = mockMvc.perform(get("/userinfo")
282                         .header(AuthServerFixtures.AUTHORIZATION, "Bearer " + tokens.get(ACCESS_TOKEN).asString()))
283                 .andExpect(status().isOk()).andReturn().getResponse().getContentAsString();
284 
285         final var info = AuthServerFixtures.parse(body);
286         assertEquals(user.username(), info.get("sub").asString());
287         assertEquals(user.username(), info.get("preferred_username").asString());
288         assertEquals(AuthServerFixtures.TENANT, info.get("tenant").asString());
289         assertEquals("ROLE_USER", info.get("roles").get(0).asString());
290         assertEquals(user.email(), info.get("email").asString());
291         assertTrue(info.get("email_verified").asBoolean());
292     }

308     /**
309      * Reads a JSON array of strings.
310      *
311      * @param array the array.
312      * @return its values.
313      */
314     private static List<String> strings(final JsonNode array) {
315         final var values = new ArrayList<String>();
316         array.forEach(value -> values.add(value.asString()));
317         return values;
318     }

320     /**
321      * Introspects a token with the credentials of a client.
322      *
323      * @param client the client.
324      * @param token  the token.
325      * @return the answer.
326      * @throws Exception if the request cannot be performed.
327      */
328     private JsonNode introspect(final TestClient client, final String token) throws Exception {
329         final var body = mockMvc.perform(post(INTROSPECT_PATH)
330                         .header(AuthServerFixtures.AUTHORIZATION,
331                                 AuthServerFixtures.basic(client.clientId(), client.credential()))
332                         .param(TOKEN, token))
333                 .andExpect(status().isOk()).andReturn().getResponse().getContentAsString();
334         return AuthServerFixtures.parse(body);
335     }

337     /**
338      * Performs a refresh token request.
339      *
340      * @param client       the client.
341      * @param refreshToken the refresh token to present.
342      * @return the pending answer.
343      * @throws Exception if the request cannot be performed.
344      */
345     private RefreshResult refresh(final TestClient client, final String refreshToken) throws Exception {
346         return new RefreshResult(mockMvc.perform(post(AuthServerFixtures.TOKEN_PATH)
347                         .header(AuthServerFixtures.AUTHORIZATION,
348                                 AuthServerFixtures.basic(client.clientId(), client.credential()))
349                         .param(AuthServerFixtures.GRANT_TYPE, REFRESH_TOKEN)
350                         .param(REFRESH_TOKEN, refreshToken))
351                 .andReturn().getResponse());
352     }

354     /**
355      * Counts the OAuth 2 authorizations of a user.
356      *
357      * @param username name of the user.
358      * @return the number of authorizations.
359      */
360     private int authorizationCount(final String username) {
361         return jdbcClient.sql("SELECT count(*) FROM oauth2_authorization WHERE principal_name = ?").param(username)
362                 .query(Integer.class).single();
363     }
364 
365     /**
366      * The answer to a refresh token request.
367      *
368      * @param response the HTTP response.
369      */
370     private record RefreshResult(MockHttpServletResponse response) {
371 
372         /**
373          * Expects a successful answer.
374          *
375          * @return the token endpoint answer.
376          * @throws UnsupportedEncodingException if the body cannot be decoded.
377          */
378         JsonNode andExpectStatusOk() throws UnsupportedEncodingException {
379             assertEquals(HttpStatus.OK.value(), response.getStatus());
380             return AuthServerFixtures.parse(response.getContentAsString());
381         }
382 
383         /**
384          * Expects the {@code invalid_grant} answer.
385          *
386          * @throws UnsupportedEncodingException if the body cannot be decoded.
387          */
388         void andExpectBadRequest() throws UnsupportedEncodingException {
389             assertEquals(HttpStatus.BAD_REQUEST.value(), response.getStatus());
390             assertEquals("invalid_grant", AuthServerFixtures.parse(response.getContentAsString())
391                     .get("error").asString());
392         }
