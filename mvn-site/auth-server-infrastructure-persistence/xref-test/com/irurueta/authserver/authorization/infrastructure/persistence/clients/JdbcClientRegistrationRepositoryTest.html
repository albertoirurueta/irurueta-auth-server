48  /**
49   * Unit tests of {@link JdbcClientRegistrationRepository}, with the registered client repository, the JDBC client and
50   * the mapper mocked. The mapping itself is covered by {@link RegisteredClientMapperTest}, and the mapping through the
51   * real JDBC repository runs against PostgreSQL in {@link JdbcClientRegistrationRepositoryIT}.
52   */
53  @ExtendWith(MockitoExtension.class)
54  class JdbcClientRegistrationRepositoryTest {
55  
56      /** Number of statements that delete a client: its authorizations, its consents and its row. */
57      private static final int DELETE_STATEMENTS = 3;
58  
59      /** Client id used by the tests. */
60      private static final String CLIENT_ID = "bff";
61  
62      /** Client id listed in the table whose client is no longer found. */
63      private static final String GONE_CLIENT_ID = "gone";
64  
65      /** Internal identifier of the stored client. */
66      private static final String ROW_ID = "row-id";
67  
68      /** Lifetime of access tokens used by the tests. */
69      private static final Duration ACCESS_TTL = Duration.ofMinutes(10);
70  
71      /** Lifetime of refresh tokens used by the tests. */
72      private static final Duration REFRESH_TTL = Duration.ofHours(8);
73  
74      /** Repository that maps registered clients to rows. */
75      @Mock
76      private RegisteredClientRepository registeredClientRepository;
77  
78      /** JDBC client that runs the other statements. */
79      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
80      private JdbcClient client;
81  
82      /** Maps between registrations and registered clients. */
83      @Mock
84      private RegisteredClientMapper registeredClientMapper;
85  
86      /** Repository under test. */
87      @InjectMocks
88      private JdbcClientRegistrationRepository repository;
89  
90      /**
91       * Verifies that an empty result is returned, and nothing is mapped, when the client is not registered.
92       */
93      @Test
94      void findByClientId_whenClientIsMissing_returnsEmpty() {
95          assertTrue(repository.findByClientId(CLIENT_ID).isEmpty());
96  
97          verify(registeredClientMapper, never()).toRegistration(any());
98      }
99  
100     /**
101      * Verifies that a registered client is returned as the registration the mapper builds from it.
102      */
103     @Test
104     void findByClientId_whenClientExists_returnsMappedRegistration() {
105         final var registered = registeredClient(null);
106         final var registration = registration();
107         when(registeredClientRepository.findByClientId(CLIENT_ID)).thenReturn(registered);
108         when(registeredClientMapper.toRegistration(registered)).thenReturn(registration);
109 
110         assertSame(registration, repository.findByClientId(CLIENT_ID).orElseThrow());
111     }
112 
113     /**
114      * Verifies that every client id of the table is loaded through the registered client repository, skipping the
115      * ones that are no longer found.
116      */
117     @Test
118     void findAll_whenClientsExist_returnsAllRegistrations() {
119         final var registered = registeredClient(null);
120         final var registration = registration();
121         when(client.sql(any(String.class)).query(String.class).list()).thenReturn(List.of(CLIENT_ID, GONE_CLIENT_ID));
122         when(registeredClientRepository.findByClientId(CLIENT_ID)).thenReturn(registered);
123         when(registeredClientMapper.toRegistration(registered)).thenReturn(registration);
124 
125         assertEquals(List.of(registration), repository.findAll());
126     }
127 
128     /**
129      * Verifies that a new client is mapped with a generated identifier, an issue instant and the given secret, and
130      * that the mapped client is stored.
131      */
132     @Test
133     void save_whenClientIsNew_storesTheMappedClient() {
134         final var registration = registration();
135         final var secret = encoded();
136         final var mapped = registeredClient(secret);
137         when(registeredClientMapper.toRegisteredClient(eq(registration), any(String.class), any(Instant.class),
138                 eq(secret))).thenReturn(mapped);
139 
140         repository.save(registration, secret);
141 
142         final var id = ArgumentCaptor.<String>captor();
143         verify(registeredClientMapper).toRegisteredClient(eq(registration), id.capture(), any(Instant.class),
144                 eq(secret));
145         assertNotNull(UUID.fromString(id.getValue()));
146         verify(registeredClientRepository).save(mapped);
147     }
148 
149     /**
150      * Verifies that saving an existing client with a null secret keeps its identifier, issue instant and secret.
151      */
152     @Test
153     void save_whenClientExistsAndSecretIsNull_keepsStoredSecretAndIdentifier() {
154         final var issuedAt = Instant.parse("2025-06-01T00:00:00Z");
155         final var storedValue = encoded();
156         final var existing = RegisteredClient.withId(ROW_ID).clientId(CLIENT_ID).clientIdIssuedAt(issuedAt)
157                 .clientSecret(storedValue)
158                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_BASIC)
159                 .authorizationGrantType(AuthorizationGrantType.CLIENT_CREDENTIALS)
160                 .build();
161         final var registration = registration();
162         when(registeredClientRepository.findByClientId(CLIENT_ID)).thenReturn(existing);
163         when(registeredClientMapper.toRegisteredClient(registration, ROW_ID, issuedAt, storedValue))
164                 .thenReturn(existing);
165 
166         repository.save(registration, null);
167 
168         verify(registeredClientRepository).save(existing);
169     }
170 
171     /**
172      * Verifies that saving an existing client with a new secret replaces the secret but keeps the identifier.
173      */
174     @Test
175     void save_whenClientExistsAndSecretIsGiven_replacesTheSecret() {
176         final var existing = registeredClient(encoded());
177         final var registration = registration();
178         final var secret = encoded();
179         final var mapped = registeredClient(secret);
180         when(registeredClientRepository.findByClientId(CLIENT_ID)).thenReturn(existing);
181         when(registeredClientMapper.toRegisteredClient(eq(registration), eq(ROW_ID), any(Instant.class),
182                 eq(secret))).thenReturn(mapped);
183 
184         repository.save(registration, secret);
185 
186         verify(registeredClientRepository).save(mapped);
187     }
188 
189     /**
190      * Verifies that the client is deleted by its client id.
191      */
192     @Test
193     void deleteByClientId_whenCalled_deletesAuthorizationsConsentsAndTheRow() {
194         repository.deleteByClientId(CLIENT_ID);
195 
196         final var statements = ArgumentCaptor.<String>captor();
197         verify(client, times(DELETE_STATEMENTS)).sql(statements.capture());
198         assertEquals(List.of("DELETE FROM oauth2_authorization ", "DELETE FROM oauth2_authorization_consent ",
199                         "DELETE FROM oauth2_registered_client "),
200                 statements.getAllValues().stream().map(sql -> sql.substring(0, sql.indexOf("WHERE"))).toList());
201     }
202 
203     /**
204      * Verifies that the secret is replaced for the client id.
205      */
206     @Test
207     void updateSecret_whenCalled_updatesTheRow() {
208         final var value = encoded();
209 
210         repository.updateSecret(CLIENT_ID, value);
211 
212         verify(client.sql(any(String.class)).params(value, CLIENT_ID)).update();
213     }
214 
215     /**
216      * Builds a runtime-generated placeholder for an encoded value.
217      *
218      * @return the placeholder.
219      */
220     private static String encoded() {
221         return "encoded-" + UUID.randomUUID();
222     }
223 
224     /**
225      * Builds a registered client stored under {@link #ROW_ID}.
226      *
227      * @param secret encoded secret, or {@code null}.
228      * @return the registered client.
229      */
230     private static RegisteredClient registeredClient(final String secret) {
231         return RegisteredClient.withId(ROW_ID).clientId(CLIENT_ID).clientSecret(secret)
232                 .clientIdIssuedAt(Instant.now())
233                 .clientAuthenticationMethod(ClientAuthenticationMethod.CLIENT_SECRET_BASIC)
234                 .authorizationGrantType(AuthorizationGrantType.CLIENT_CREDENTIALS)
235                 .build();
236     }
237 
238     /**
239      * Builds a confidential registration.
240      *
241      * @return the registration.
242      */
243     private static ClientRegistration registration() {
244         return new ClientRegistration(CLIENT_ID, "Backend for Frontend", true, List.of("client_secret_basic"),
245                 List.of("client_credentials"), List.of(), List.of("openid"), true, List.of(), ACCESS_TTL,
246                 REFRESH_TTL, ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false, false,
247                 null, null, null, null, false);
248     }
