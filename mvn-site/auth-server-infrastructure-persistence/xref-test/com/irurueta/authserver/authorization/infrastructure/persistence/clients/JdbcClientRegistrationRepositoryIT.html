34  /**
35   * Integration tests of {@link JdbcClientRegistrationRepository} over the real
36   * {@link JdbcRegisteredClientRepository} and PostgreSQL. No Spring context is needed.
37   */
38  class JdbcClientRegistrationRepositoryIT {
39  
40      /** Client id used by the tests. */
41      private static final String CLIENT_ID = "bff";
42  
43      /** Lifetime of access tokens, in minutes. */
44      private static final long ACCESS_MINUTES = 10;
45  
46      /** Lifetime of refresh tokens, in hours. */
47      private static final long REFRESH_HOURS = 8;
48  
49      /** JDBC client connected to the database. */
50      private static JdbcClient jdbcClient;
51  
52      /** Repository under test. */
53      private static JdbcClientRegistrationRepository repository;
54  
55      /**
56       * Connects to the migrated database and builds the repository.
57       */
58      @BeforeAll
59      static void connect() {
60          final var dataSource = MigratedPostgres.dataSource();
61          jdbcClient = JdbcClient.create(dataSource);
62          repository = new JdbcClientRegistrationRepository(
63                  new JdbcRegisteredClientRepository(new JdbcTemplate(dataSource)), jdbcClient,
64                  new RegisteredClientMapper());
65      }
66  
67      /**
68       * Removes the clients, authorizations and consents of the previous test.
69       */
70      @BeforeEach
71      void cleanUp() {
72          jdbcClient.sql("DELETE FROM oauth2_authorization").update();
73          jdbcClient.sql("DELETE FROM oauth2_authorization_consent").update();
74          jdbcClient.sql("DELETE FROM oauth2_registered_client").update();
75      }
76  
77      /**
78       * Verifies that a saved client is read back as the same registration, with its audiences and token settings.
79       */
80      @Test
81      void findByClientId_whenClientWasSaved_returnsTheSameRegistration() {
82          final var registration = confidential(CLIENT_ID);
83  
84          repository.save(registration, encoded());
85  
86          assertEquals(registration, repository.findByClientId(CLIENT_ID).orElseThrow());
87      }
88  
89      /**
90       * Verifies that saving again replaces the settings and keeps the stored secret when none is given.
91       */
92      @Test
93      void save_whenClientExistsAndSecretIsNull_keepsTheStoredSecret() {
94          repository.save(confidential(CLIENT_ID), encoded());
95          final var before = jdbcClient.sql("SELECT client_secret FROM oauth2_registered_client WHERE client_id = ?")
96                  .param(CLIENT_ID).query(String.class).single();
97          final var updated = new ClientRegistration(CLIENT_ID, "Renamed", true, List.of("client_secret_basic"),
98                  List.of("authorization_code"), List.of("http://localhost:8080/login/oauth2/code/bff"),
99                  List.of("openid"), true, List.of("other"), Duration.ofMinutes(5), Duration.ofHours(1),
100                 ClientRegistration.FORMAT_REFERENCE, true, List.of(), true, false, null, null, null, null, false);
101 
102         repository.save(updated, null);
103 
104         assertEquals(updated, repository.findByClientId(CLIENT_ID).orElseThrow());
105         assertEquals(before, jdbcClient.sql("SELECT client_secret FROM oauth2_registered_client WHERE client_id = ?")
106                 .param(CLIENT_ID).query(String.class).single());
107         assertEquals(1, repository.findAll().size());
108     }
109 
110     /**
111      * Verifies that an update round trip keeps the key, algorithm, certificate subject and certificate-bound
112      * settings, and that a later update without them clears them.
113      */
114     @Test
115     void save_whenAuthenticationSettingsAreGiven_keepsThemOnUpdate() {
116         repository.save(confidential(CLIENT_ID), encoded());
117         final var withSettings = new ClientRegistration(CLIENT_ID, "Service", true,
118                 List.of("private_key_jwt", "tls_client_auth"), List.of("client_credentials"), List.of(),
119                 List.of("documents.read"), false, List.of("resource-server"), Duration.ofMinutes(ACCESS_MINUTES),
120                 Duration.ofHours(REFRESH_HOURS), ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false,
121                 false, "https://client.example.test/jwks.json", "{\"keys\":[]}", "ES256", "CN=client", true);
122 
123         repository.save(withSettings, null);
124 
125         assertEquals(withSettings, repository.findByClientId(CLIENT_ID).orElseThrow());
126 
127         repository.save(confidential(CLIENT_ID), null);
128 
129         assertEquals(confidential(CLIENT_ID), repository.findByClientId(CLIENT_ID).orElseThrow());
130     }
131 
132     /**
133      * Verifies that every client is listed.
134      */
135     @Test
136     void findAll_whenSeveralClientsExist_listsThemAll() {
137         repository.save(confidential("a-client"), encoded());
138         repository.save(confidential("b-client"), encoded());
139 
140         assertEquals(List.of("a-client", "b-client"),
141                 repository.findAll().stream().map(ClientRegistration::clientId).toList());
142     }
143 
144     /**
145      * Verifies that a client is deleted and that its secret can be replaced.
146      */
147     @Test
148     void deleteByClientId_whenClientExists_removesIt() {
149         repository.save(confidential(CLIENT_ID), encoded());
150         final var replacement = encoded();
151 
152         repository.updateSecret(CLIENT_ID, replacement);
153 
154         assertEquals(replacement, jdbcClient.sql("SELECT client_secret FROM oauth2_registered_client")
155                 .query(String.class).single());
156         repository.deleteByClientId(CLIENT_ID);
157         assertTrue(repository.findByClientId(CLIENT_ID).isEmpty());
158     }
159 
160     /**
161      * Verifies that deleting a client also deletes, in the same call, the authorizations and the consents that
162      * reference it, and leaves those of other clients alone.
163      */
164     @Test
165     void deleteByClientId_whenClientHasAuthorizationsAndConsents_deletesThemToo() {
166         repository.save(confidential(CLIENT_ID), encoded());
167         repository.save(confidential("other"), encoded());
168         final var deletedId = internalId(CLIENT_ID);
169         final var keptId = internalId("other");
170         for (final var registeredClientId : List.of(deletedId, keptId)) {
171             jdbcClient.sql("""
172                     INSERT INTO oauth2_authorization (id, registered_client_id, principal_name,
173                                                       authorization_grant_type)
174                     VALUES (?, ?, 'alice', 'authorization_code')
175                     """).params(UUID.randomUUID().toString(), registeredClientId).update();
176             jdbcClient.sql("""
177                     INSERT INTO oauth2_authorization_consent (registered_client_id, principal_name, authorities)
178                     VALUES (?, 'alice', 'SCOPE_openid')
179                     """).param(registeredClientId).update();
180         }
181 
182         repository.deleteByClientId(CLIENT_ID);
183 
184         assertTrue(repository.findByClientId(CLIENT_ID).isEmpty());
185         assertEquals(0, countFor("oauth2_authorization", deletedId));
186         assertEquals(0, countFor("oauth2_authorization_consent", deletedId));
187         assertEquals(1, countFor("oauth2_authorization", keptId));
188         assertEquals(1, countFor("oauth2_authorization_consent", keptId));
189     }
190 
191     /**
192      * Reads the internal id of a stored client.
193      *
194      * @param clientId client id.
195      * @return the internal id.
196      */
197     private static String internalId(final String clientId) {
198         return jdbcClient.sql("SELECT id FROM oauth2_registered_client WHERE client_id = ?").param(clientId)
199                 .query(String.class).single();
200     }
201 
202     /**
203      * Counts the rows of a table that reference a client.
204      *
205      * @param table              table name, one of the two fixed names above.
206      * @param registeredClientId internal id of the client.
207      * @return the number of rows.
208      */
209     private static int countFor(final String table, final String registeredClientId) {
210         final var sql = "oauth2_authorization".equals(table)
211                 ? "SELECT count(*) FROM oauth2_authorization WHERE registered_client_id = ?"
212                 : "SELECT count(*) FROM oauth2_authorization_consent WHERE registered_client_id = ?";
213         return jdbcClient.sql(sql).param(registeredClientId).query(Integer.class).single();
214     }
215 
216     /**
217      * Builds a runtime-generated placeholder for an encoded value.
218      *
219      * @return the placeholder.
220      */
221     private static String encoded() {
222         return "{noop}" + UUID.randomUUID();
223     }
224 
225     /**
226      * Builds a confidential registration.
227      *
228      * @param clientId client id.
229      * @return the registration.
230      */
231     private static ClientRegistration confidential(final String clientId) {
232         return new ClientRegistration(clientId, "Backend for Frontend", true, List.of("client_secret_basic"),
233                 List.of("authorization_code", "refresh_token"), List.of("http://localhost:8080/login/oauth2/code/bff"),
234                 List.of("openid", "profile"), true, List.of("resource-server"), Duration.ofMinutes(ACCESS_MINUTES),
235                 Duration.ofHours(REFRESH_HOURS), ClientRegistration.FORMAT_SELF_CONTAINED, false, List.of(), false,
236                 false, null, null, null, null, false);
237     }
