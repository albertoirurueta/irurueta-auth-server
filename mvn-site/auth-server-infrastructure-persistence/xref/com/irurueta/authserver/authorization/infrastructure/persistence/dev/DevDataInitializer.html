17  package com.irurueta.authserver.authorization.infrastructure.persistence.dev;
18  
19  import com.irurueta.authserver.authorization.application.clients.DefaultClientDefinitions;
20  import com.irurueta.authserver.authorization.domain.clients.ClientRegistrationRepository;
21  import java.security.SecureRandom;
22  import java.util.Base64;
23  import java.util.List;
24  import java.util.UUID;
25  import lombok.RequiredArgsConstructor;
26  import lombok.extern.slf4j.Slf4j;
27  import org.springframework.boot.ApplicationArguments;
28  import org.springframework.beans.factory.annotation.Value;
29  import org.springframework.boot.ApplicationRunner;
30  import org.springframework.context.annotation.Profile;
31  import org.springframework.jdbc.core.simple.JdbcClient;
32  import org.springframework.security.crypto.password.PasswordEncoder;
33  import org.springframework.stereotype.Component;
34  import org.springframework.transaction.annotation.Transactional;
35  
36  /**
37   * Seeds a demo user and a confidential BFF client when the {@code dev} profile is active, so a freshly started
38   * authorization server has someone to sign in and a client to sign in through.
39   *
40   * <p>Purpose: local development and the end-to-end tests. Started with
41   * {@code spring-boot:run -Dspring-boot.run.profiles=dev}, it lets a developer sign in through the BFF by hand. The
42   * {@code e2e-tests} module ({@code SignInSmokeIT}) starts the <em>packaged</em> jar with the {@code dev} profile and
43   * reads the generated credentials from the log lines below.
44   *
45   * <p>Why it lives in {@code src/main} although it only seeds demo data: the end-to-end tests run the real jar, and
46   * a class under {@code src/test} never reaches it. It is inert everywhere else: without the {@code dev} profile the
47   * bean is not even created. Never activate that profile outside local development.
48   *
49   * <p>It lives in the persistence adapter, next to the Flyway migrations, because it writes the tables those
50   * migrations create: a schema change that breaks it is then caught by this module's own tests.
51   *
52   * <p>Secrets are generated at first start and logged once as {@code DEV ONLY} warnings; nothing is committed.
53   * Running it again does not change existing data.
54   */
55  @Slf4j
56  @Component
57  @Profile("dev")
58  @RequiredArgsConstructor
59  public class DevDataInitializer implements ApplicationRunner {
60  
61      /** Username of the demo user. */
62      static final String DEMO_USERNAME = "demo";
63  
64      /** Number of random bytes of generated secrets. */
65      private static final int SECRET_BYTES = 24;
66  
67      /** Source of randomness for generated secrets. */
68      private final SecureRandom random = new SecureRandom();
69  
70      /** JDBC client used to insert the demo user. */
71      private final JdbcClient jdbc;
72  
73      /** Encoder of generated secrets. */
74      private final PasswordEncoder encoder;
75  
76      /** Repository where the BFF client is stored. */
77      private final ClientRegistrationRepository clients;
78  
79      /**
80       * Redirect URIs of the BFF client ({@code authserver.clients.bff.redirect-uris}, which the {@code dev} profile
81       * sets to the local BFF); empty when not configured.
82       */
83      @Value("${authserver.clients.bff.redirect-uris:}")
84      private final List<String> bffRedirectUris;
85  
86      /**
87       * Post-logout redirect URIs of the BFF client ({@code authserver.clients.bff.post-logout-redirect-uris}); empty
88       * when not configured.
89       */
90      @Value("${authserver.clients.bff.post-logout-redirect-uris:}")
91      private final List<String> bffPostLogoutRedirectUris;
92  
93      /** Audience of the authorization server's own API ({@code authserver.api.audience}). */
94      @Value("${authserver.api.audience:" + DefaultClientDefinitions.DEFAULT_API_AUDIENCE + "}")
95      private final String apiAudience;
96  
97      /**
98       * Creates the demo user and the BFF client when they do not exist yet.
99       *
100      * @param args application arguments (unused).
101      */
102     @Override
103     @Transactional
104     public void run(final ApplicationArguments args) {
105         createDemoUser();
106         createBffClient();
107     }
108 
109     /**
110      * Creates the demo user, with a generated password, unless it already exists.
111      */
112     private void createDemoUser() {
113         final var exists = jdbc.sql("SELECT count(*) FROM users WHERE username = ?")
114                 .param(DEMO_USERNAME).query(Integer.class).single() > 0;
115         if (exists) {
116             return;
117         }
118         final var password = generateSecret();
119         final var id = UUID.randomUUID();
120         jdbc.sql("INSERT INTO users (id, username) VALUES (?, ?)").params(id, DEMO_USERNAME).update();
121         jdbc.sql("INSERT INTO user_authorities (user_id, authority) VALUES (?, ?)")
122                 .params(id, "ROLE_USER").update();
123         jdbc.sql("INSERT INTO user_passwords (user_id, encoded_password) VALUES (?, ?)")
124                 .params(id, encoder.encode(password)).update();
125         log.warn("DEV ONLY, never use outside local development: created demo user '{}' with generated password: {}",
126                 DEMO_USERNAME, password);
127     }
128 
129     /**
130      * Creates the BFF client from its default definition and the configured redirect URIs, with a generated secret,
131      * unless it already exists or no redirect URI is configured.
132      */
133     private void createBffClient() {
134         if (clients.findByClientId(DefaultClientDefinitions.BFF_CLIENT_ID).isPresent()) {
135             return;
136         }
137         if (bffRedirectUris.isEmpty()) {
138             log.warn("DEV ONLY: client '{}' not created, authserver.clients.bff.redirect-uris is not configured",
139                     DefaultClientDefinitions.BFF_CLIENT_ID);
140             return;
141         }
142         final var secret = generateSecret();
143         clients.save(DefaultClientDefinitions.bff(bffRedirectUris, apiAudience, bffPostLogoutRedirectUris),
144                 encoder.encode(secret));
145         log.warn("DEV ONLY, never use outside local development: created client '{}' with generated secret: {}",
146                 DefaultClientDefinitions.BFF_CLIENT_ID, secret);
147     }
148 
149     /**
150      * Generates a random URL-safe secret.
151      *
152      * @return the secret.
153      */
154     private String generateSecret() {
155         final var bytes = new byte[SECRET_BYTES];
156         random.nextBytes(bytes);
157         return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
158     }
159 }
