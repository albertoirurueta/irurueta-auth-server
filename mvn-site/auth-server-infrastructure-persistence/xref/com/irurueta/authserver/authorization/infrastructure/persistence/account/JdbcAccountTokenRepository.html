17  package com.irurueta.authserver.authorization.infrastructure.persistence.account;
18  
19  import static com.irurueta.authserver.authorization.infrastructure.persistence.JdbcTimestamps.toTimestamp;
20  
21  import com.irurueta.authserver.authorization.domain.account.AccountToken;
22  import com.irurueta.authserver.authorization.domain.account.AccountTokenRepository;
23  import java.time.Instant;
24  import java.util.Optional;
25  import lombok.RequiredArgsConstructor;
26  import org.springframework.jdbc.core.simple.JdbcClient;
27  import org.springframework.stereotype.Repository;
28  
29  /**
30   * {@link AccountTokenRepository} over the {@code account_tokens} table.
31   */
32  @Repository
33  @RequiredArgsConstructor
34  public class JdbcAccountTokenRepository implements AccountTokenRepository {
35  
36      /** Statement storing a token. */
37      private static final String INSERT_TOKEN = """
38              INSERT INTO account_tokens (id, user_id, purpose, hashed_token, expires_at, used_at)
39              VALUES (?, ?, ?, ?, ?, ?)
40              """;
41  
42      /** Query returning a token only when it is unused and not expired, without changing it. */
43      private static final String FIND_USABLE_TOKEN = """
44              SELECT id, user_id, purpose, hashed_token, expires_at, used_at FROM account_tokens
45              WHERE hashed_token = ? AND purpose = ? AND used_at IS NULL AND expires_at > ?
46              """;
47  
48      // tag::account-token-consume[]
49      /** Statement that marks a token as used and returns it, only when it is unused and not expired. */
50      private static final String CONSUME_TOKEN = """
51              UPDATE account_tokens SET used_at = ?
52              WHERE hashed_token = ? AND purpose = ? AND used_at IS NULL AND expires_at > ?
53              RETURNING id, user_id, purpose, hashed_token, expires_at, used_at
54              """;
55      // end::account-token-consume[]
56  
57      /** JDBC client used to run the statements. */
58      private final JdbcClient jdbcClient;
59  
60      /** Maps a row to a token. */
61      private final AccountTokenRowMapper accountTokenRowMapper;
62  
63      @Override
64      public void store(final AccountToken token) {
65          jdbcClient.sql(INSERT_TOKEN)
66                  .params(token.id(), token.userId(), token.purpose(), token.hashedToken(),
67                          toTimestamp(token.expiresAt()), toTimestamp(token.usedAt()))
68                  .update();
69      }
70  
71      @Override
72      public Optional<AccountToken> findUsable(final String hashedToken, final String purpose, final Instant now) {
73          return jdbcClient.sql(FIND_USABLE_TOKEN).params(hashedToken, purpose, toTimestamp(now))
74                  .query(accountTokenRowMapper)
75                  .optional();
76      }
77  
78      @Override
79      public Optional<AccountToken> consume(final String hashedToken, final String purpose, final Instant now) {
80          final var timestamp = toTimestamp(now);
81          return jdbcClient.sql(CONSUME_TOKEN).params(timestamp, hashedToken, purpose, timestamp)
82                  .query(accountTokenRowMapper)
83                  .optional();
84      }
85  
86      @Override
87      public int deleteExpired(final Instant now) {
88          return jdbcClient.sql("DELETE FROM account_tokens WHERE expires_at <= ?").param(toTimestamp(now)).update();
89      }
90  }
