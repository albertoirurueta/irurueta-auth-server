17  package com.irurueta.authserver.authorization.infrastructure.persistence.sms;
18  
19  import static com.irurueta.authserver.authorization.infrastructure.persistence.JdbcTimestamps.toTimestamp;
20  
21  import com.irurueta.authserver.authorization.domain.sms.PhoneVerification;
22  import com.irurueta.authserver.authorization.domain.sms.PhoneVerificationRepository;
23  import java.time.Instant;
24  import java.util.Optional;
25  import java.util.UUID;
26  import lombok.RequiredArgsConstructor;
27  import org.springframework.jdbc.core.simple.JdbcClient;
28  import org.springframework.stereotype.Repository;
29  
30  /**
31   * {@link PhoneVerificationRepository} over the {@code pending_*} columns of the {@code user_phones} table. It opens
32   * no transaction: clearing a verification and then deleting an empty row runs in the caller's.
33   */
34  @Repository
35  @RequiredArgsConstructor
36  public class JdbcPhoneVerificationRepository implements PhoneVerificationRepository {
37  
38      /** Statement storing the pending verification of a user, creating the row if needed. */
39      private static final String UPSERT = """
40              INSERT INTO user_phones (user_id, pending_phone_e164, pending_code_hash, pending_attempts,
41                                       pending_expires_at)
42              VALUES (?, ?, ?, ?, ?)
43              ON CONFLICT (user_id) DO UPDATE
44              SET pending_phone_e164 = EXCLUDED.pending_phone_e164,
45                  pending_code_hash = EXCLUDED.pending_code_hash,
46                  pending_attempts = EXCLUDED.pending_attempts,
47                  pending_expires_at = EXCLUDED.pending_expires_at
48              """;
49  
50      /** Query returning the pending verification of a user. */
51      private static final String FIND_BY_USER = """
52              SELECT user_id, pending_phone_e164, pending_code_hash, pending_attempts, pending_expires_at
53              FROM user_phones
54              WHERE user_id = ? AND pending_phone_e164 IS NOT NULL
55              """;
56  
57      /** Statement counting a failed attempt atomically and returning the new count. */
58      private static final String INCREMENT_ATTEMPTS = """
59              UPDATE user_phones SET pending_attempts = pending_attempts + 1
60              WHERE user_id = ? AND pending_phone_e164 IS NOT NULL
61              RETURNING pending_attempts
62              """;
63  
64      /** Statement clearing the pending columns of a user. */
65      private static final String CLEAR_PENDING = """
66              UPDATE user_phones
67              SET pending_phone_e164 = NULL, pending_code_hash = NULL, pending_attempts = 0, pending_expires_at = NULL
68              WHERE user_id = ?
69              """;
70  
71      /** Statement deleting the row of a user that holds neither a verified phone nor a pending verification. */
72      private static final String DELETE_IF_EMPTY =
73              "DELETE FROM user_phones WHERE user_id = ? AND phone_e164 IS NULL AND pending_phone_e164 IS NULL";
74  
75      /** Statement clearing every expired pending verification. */
76      private static final String CLEAR_EXPIRED = """
77              UPDATE user_phones
78              SET pending_phone_e164 = NULL, pending_code_hash = NULL, pending_attempts = 0, pending_expires_at = NULL
79              WHERE pending_expires_at <= ?
80              """;
81  
82      /** Statement deleting the rows that hold nothing at all. */
83      private static final String DELETE_EMPTY_ROWS =
84              "DELETE FROM user_phones WHERE phone_e164 IS NULL AND pending_phone_e164 IS NULL";
85  
86      /** JDBC client used to run the statements. */
87      private final JdbcClient jdbcClient;
88  
89      /** Maps a row to a verification. */
90      private final PhoneVerificationRowMapper phoneVerificationRowMapper;
91  
92      @Override
93      public void save(final PhoneVerification verification) {
94          jdbcClient.sql(UPSERT)
95                  .params(verification.userId(), verification.phone().e164(), verification.codeHash(),
96                          verification.attempts(), toTimestamp(verification.expiresAt()))
97                  .update();
98      }
99  
100     @Override
101     public Optional<PhoneVerification> findByUserId(final UUID userId) {
102         return jdbcClient.sql(FIND_BY_USER).param(userId).query(phoneVerificationRowMapper).optional();
103     }
104 
105     @Override
106     public int incrementAttempts(final UUID userId) {
107         return jdbcClient.sql(INCREMENT_ATTEMPTS).param(userId).query(Integer.class).optional().orElse(0);
108     }
109 
110     @Override
111     public void deleteByUserId(final UUID userId) {
112         jdbcClient.sql(CLEAR_PENDING).param(userId).update();
113         jdbcClient.sql(DELETE_IF_EMPTY).param(userId).update();
114     }
115 
116     @Override
117     public int deleteExpired(final Instant now) {
118         final var cleared = jdbcClient.sql(CLEAR_EXPIRED).param(toTimestamp(now)).update();
119         jdbcClient.sql(DELETE_EMPTY_ROWS).update();
120         return cleared;
121     }
122 }
